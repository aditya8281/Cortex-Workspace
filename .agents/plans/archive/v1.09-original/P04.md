### Task 1: Implement Personalization Service

**File:** `backend/app/services/learning/personalization.py`
**Purpose:** Build per-user profiles from learned preferences, patterns, and habits. Personalize content delivery based on these profiles.

```python
# backend/app/services/learning/personalization.py

from typing import Dict, Optional
from sqlalchemy.orm import Session


class PersonalizationService:
    """Builds and maintains per-user personalization profiles.
    
    Profiles aggregate data from:
    - Preferences (response style, UI layout, notifications)
    - Habits (active hours, daily routines)
    - Patterns (temporal, behavioral)
    
    Architecture ref: 1.1 (Local-First) — all profiles stored locally.
    """
    
    CONFIDENCE_THRESHOLD = 0.5
    
    def __init__(self, db: Session):
        self.db = db
    
    async def get_user_profile(self, user_id: int) -> Dict:
        """Build personalized user profile from all learning data."""
        profile = {
            "user_id": user_id,
            "response_style": "standard",
            "preferred_topics": [],
            "active_hours": "9-17",
            "expertise_level": "intermediate",
            "suggestion_tolerance": "normal",
        }
        
        # Load preferences
        from backend.app.models.learning.preference import UserPreference
        prefs = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.confidence >= self.CONFIDENCE_THRESHOLD
        ).all()
        
        for p in prefs:
            if p.category == "response_style" and p.key == "length":
                profile["response_style"] = p.value
            elif p.category == "notification":
                profile["suggestion_tolerance"] = p.value
        
        # Load habits for active hours
        from backend.app.models.learning.habit import Habit
        habits = self.db.query(Habit).filter(
            Habit.user_id == user_id,
            Habit.occurrences >= 5
        ).all()
        
        if habits:
            # Extract active hours from time-based habits
            time_triggers = [h.trigger for h in habits if h.trigger and ":" in h.trigger]
            if time_triggers:
                profile["active_hours"] = ", ".join(time_triggers[:3])
        
        return profile
    
    async def personalize_content(self, user_id: int, content: str,
                                   context: Dict = None) -> str:
        """Personalize content based on user profile."""
        profile = await self.get_user_profile(user_id)
        
        if profile.get("response_style") == "concise":
            sentences = content.split('.')
            if len(sentences) > 5:
                return '. '.join(sentences[:5]) + '.'
        
        return content
    
    async def update_profile(self, user_id: int, updates: Dict) -> Dict:
        """Update profile (stores as preference overrides)."""
        from backend.app.services.learning.preference import PreferenceLearningService
        service = PreferenceLearningService(self.db)
        
        for key, value in updates.items():
            await service.record_preference(
                user_id, "profile", key, str(value), "explicit"
            )
        
        return await self.get_user_profile(user_id)
```

### Task 2: Implement Knowledge Refinement Service

**File:** `backend/app/services/learning/refinement.py`
**Purpose:** Adjust memory confidence based on outcomes. If a memory led to correct behavior → boost confidence. If it led to error → reduce confidence.

```python
# backend/app/services/learning/refinement.py

from typing import Dict, List
from sqlalchemy.orm import Session


class KnowledgeRefinementService:
    """Refines knowledge confidence based on outcomes.
    
    When Cortex uses a memory and the outcome is:
    - Correct/positive → boost confidence (+0.1)
    - Incorrect/negative → reduce confidence (-0.2)
    - With correction → create refined knowledge entry
    
    Architecture ref: 4.3 (Memory Architecture) — confidence adjustment
    based on access and outcome.
    """
    
    BOOST_AMOUNT = 0.1
    REDUCE_AMOUNT = 0.2
    
    def __init__(self, db: Session):
        self.db = db
    
    async def refine_memory(self, memory_id: int, outcome: str,
                             feedback: str = None) -> Dict:
        """Refine a memory based on outcome."""
        adjustments = {}
        
        if outcome == "correct":
            adjustments["confidence_boost"] = self.BOOST_AMOUNT
        elif outcome == "incorrect":
            adjustments["confidence_reduce"] = self.REDUCE_AMOUNT
            if feedback:
                adjustments["correction"] = feedback
        
        return {
            "memory_id": memory_id, "outcome": outcome,
            "adjustments": adjustments, "applied": True
        }
    
    async def get_refinement_stats(self, user_id: int) -> Dict:
        """Get refinement statistics for monitoring."""
        from backend.app.models.learning.learning_event import LearningEvent
        
        events = self.db.query(LearningEvent).filter(
            LearningEvent.user_id == user_id
        ).all()
        
        positive = sum(1 for e in events if e.event_type == "affirmation")
        negative = sum(1 for e in events if e.event_type == "correction")
        
        return {
            "total_refinements": len(events),
            "improvements": positive,
            "degradations": negative,
            "net_improvement": positive - negative,
            "improvement_rate": positive / len(events) if events else 0.5
        }
    
    async def suggest_refinements(self, user_id: int) -> List[Dict]:
        """Suggest knowledge items that need refinement."""
        from backend.app.models.memory.memory_node import MemoryNode
        
        low_confidence = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.confidence < 0.3
        ).all()
        
        return [
            {"node_id": n.id, "type": n.memory_type, "confidence": n.confidence,
             "label": n.label}
            for n in low_confidence[:10]
        ]
```

### Task 3: Implement Pattern Recognition Service

**File:** `backend/app/services/learning/patterns.py`
**Purpose:** Detect temporal, behavioral, and preference patterns. Record patterns with evidence lists. Patterns require ≥3 evidence items and confidence ≥0.5 to be actionable.

```python
# backend/app/services/learning/patterns.py

from datetime import datetime
from typing import Dict, List
from sqlalchemy.orm import Session
from backend.app.models.learning.pattern import Pattern


class PatternRecognitionService:
    """Detects and records behavioral patterns.
    
    Pattern types:
    - 'temporal': Time-based (codes in mornings, checks email at 9am)
    - 'behavioral': Action-based (prefers vim, uses dark mode)
    - 'preference': Choice-based (always chooses concise responses)
    
    Architecture ref: 1.6 (Evidence Over Opinion) — patterns require evidence.
    """
    
    MIN_EVIDENCE_FOR_CONFIDENCE = 3
    
    def __init__(self, db: Session):
        self.db = db
    
    async def detect_pattern(self, user_id: int, pattern_type: str,
                              description: str, evidence: List[str]) -> Pattern:
        """Detect and record a pattern. Strengthens existing or creates new."""
        # Check for similar existing pattern
        existing = self.db.query(Pattern).filter(
            Pattern.user_id == user_id,
            Pattern.description == description,
            Pattern.pattern_type == pattern_type
        ).first()
        
        if existing:
            existing.add_evidence()
            self.db.commit()
            return existing
        
        pattern = Pattern(
            user_id=user_id, pattern_type=pattern_type,
            description=description, evidence=str(evidence),
            confidence=0.3, occurrences=1
        )
        self.db.add(pattern)
        self.db.commit()
        return pattern
    
    async def get_patterns(self, user_id: int,
                            min_confidence: float = 0.5) -> List[Dict]:
        """Get detected patterns above confidence threshold."""
        patterns = self.db.query(Pattern).filter(
            Pattern.user_id == user_id,
            Pattern.confidence >= min_confidence
        ).order_by(Pattern.confidence.desc()).all()
        
        return [
            {"id": p.id, "type": p.pattern_type, "description": p.description,
             "confidence": p.confidence, "occurrences": p.occurrences,
             "evidence": p.evidence}
            for p in patterns
        ]
    
    async def get_patterns_by_type(self, user_id: int,
                                    pattern_type: str) -> List[Dict]:
        """Get patterns filtered by type."""
        patterns = self.db.query(Pattern).filter(
            Pattern.user_id == user_id,
            Pattern.pattern_type == pattern_type
        ).all()
        
        return [
            {"description": p.description, "confidence": p.confidence,
             "occurrences": p.occurrences}
            for p in patterns
        ]
    
    async def get_actionable_patterns(self, user_id: int) -> List[Dict]:
        """Get patterns ready for action (confident + enough evidence)."""
        patterns = self.db.query(Pattern).filter(
            Pattern.user_id == user_id
        ).all()
        
        return [
            {"id": p.id, "type": p.pattern_type, "description": p.description,
             "confidence": p.confidence}
            for p in patterns if p.is_confident
        ]
```

### Task 4: Implement Anomaly Detection Service

**File:** `backend/app/services/learning/anomaly.py`
**Purpose:** Detect unusual behavior by comparing current events against learned baselines. Flags anomalies for user review — never auto-acts.

```python
# backend/app/services/learning/anomaly.py

from typing import Dict, List
from datetime import datetime


class AnomalyDetectionService:
    """Detects anomalies in user behavior by comparing against baselines.
    
    Detection types:
    - Frequency: 3x+ normal activity
    - Time: Activity outside usual hours
    - Volume: 5x+ normal data volume
    
    Architecture ref: 1.7 (Incremental Safety) — anomalies flagged for review,
    never auto-acted.
    """
    
    def __init__(self):
        pass
    
    async def detect_anomaly(self, user_id: int, event_type: str,
                              event_data: Dict, baseline: Dict = None) -> Dict:
        """Detect if an event is anomalous compared to baseline."""
        if not baseline:
            return {"anomalous": False, "confidence": 0.0, "reasons": []}
        
        is_anomalous = False
        reasons = []
        
        # Frequency anomaly
        expected_freq = baseline.get("expected_frequency", 10)
        actual_freq = event_data.get("frequency", 1)
        if actual_freq > expected_freq * 3:
            is_anomalous = True
            reasons.append(f"Frequency {actual_freq}x is {actual_freq/expected_freq:.1f}x above expected {expected_freq}")
        
        # Time anomaly
        current_hour = event_data.get("hour", 12)
        active_hours = baseline.get("active_hours", list(range(9, 18)))
        if active_hours and current_hour not in active_hours:
            is_anomalous = True
            reasons.append(f"Activity at unusual hour: {current_hour}:00 (usual: {min(active_hours)}-{max(active_hours)})")
        
        # Volume anomaly
        expected_volume = baseline.get("expected_volume", 100)
        actual_volume = event_data.get("volume", 0)
        if actual_volume > expected_volume * 5:
            is_anomalous = True
            reasons.append(f"Volume {actual_volume} is {actual_volume/expected_volume:.1f}x above baseline {expected_volume}")
        
        return {
            "anomalous": is_anomalous,
            "reasons": reasons,
            "confidence": 0.8 if is_anomalous else 0.2,
            "event_type": event_type,
            "timestamp": datetime.utcnow().isoformat()
        }
    
    async def get_baselines(self, user_id: int) -> Dict:
        """Get behavioral baselines for anomaly detection."""
        return {
            "active_hours": list(range(9, 18)),
            "expected_frequency": 10,
            "expected_volume": 100,
            "common_actions": [],
            "session_duration_avg_minutes": 30
        }
    
    async def update_baseline(self, user_id: int, event_type: str,
                               event_data: Dict) -> Dict:
        """Update baselines based on new observed data (adaptive baselines)."""
        baselines = await self.get_baselines(user_id)
        # In production, this would compute running averages
        # For v1.09, baselines are static defaults
        return baselines
```

### Task 5: Implement Continuous Improvement Service

**File:** `backend/app/services/learning/improvement.py`
**Purpose:** Track improvement metrics over time. Measure accuracy, response quality, suggestion acceptance rates.

```python
# backend/app/services/learning/improvement.py

from typing import Dict, List
from datetime import datetime


class ContinuousImprovementService:
    """Tracks and drives continuous improvement.
    
    Metrics tracked:
    - Suggestion acceptance rate
    - Response accuracy
    - Preference confidence growth
    - Pattern detection rate
    
    Architecture ref: 3.7 (Incremental Safety) — measurable improvement metrics.
    """
    
    def __init__(self):
        self._metrics: List[Dict] = []
    
    async def record_improvement(self, user_id: int, metric: str,
                                  before: float, after: float) -> Dict:
        """Record an improvement metric."""
        improvement = after - before
        percentage = (improvement / before * 100) if before > 0 else 0
        
        record = {
            "user_id": user_id, "metric": metric,
            "before": before, "after": after,
            "improvement": improvement, "percentage": percentage,
            "timestamp": datetime.utcnow().isoformat()
        }
        self._metrics.append(record)
        return record
    
    async def get_improvement_summary(self, user_id: int) -> Dict:
        """Get improvement summary."""
        user_metrics = [m for m in self._metrics if m["user_id"] == user_id]
        
        improved = sum(1 for m in user_metrics if m["improvement"] > 0)
        degraded = sum(1 for m in user_metrics if m["improvement"] < 0)
        
        return {
            "total_metrics": len(user_metrics),
            "improved": improved,
            "degraded": degraded,
            "overall_trend": "improving" if improved > degraded else "stable"
        }
    
    async def get_improvement_history(self, user_id: int,
                                       metric: str = None) -> List[Dict]:
        """Get improvement history for a metric or all metrics."""
        user_metrics = [m for m in self._metrics if m["user_id"] == user_id]
        if metric:
            user_metrics = [m for m in user_metrics if m["metric"] == metric]
        return user_metrics
    
    async def calculate_improvement_score(self, user_id: int) -> float:
        """Calculate overall improvement score (0.0 to 1.0)."""
        summary = await self.get_improvement_summary(user_id)
        if summary["total_metrics"] == 0:
            return 0.5  # Neutral default
        return summary["improved"] / summary["total_metrics"]
```

## Security Considerations

1. **User isolation** — All queries filter by `user_id`.
2. **Profile data** — Profiles reveal personal information. Local storage only.
3. **Anomaly data** — May reveal sensitive behavior patterns. Never auto-act.
4. **Improvement metrics** — Aggregate only. No individual event exposure.

---

## Performance Considerations

1. **Profile building** — O(p + h + n) where p=prefs, h=habits, n=patterns. Cache for 5 minutes.
2. **Pattern detection** — O(n) per detection. Acceptable for <1000 patterns.
3. **Anomaly detection** — O(1) per event (baseline comparison). Very fast.
4. **Improvement tracking** — In-memory for v1.09. PostgreSQL in production.
