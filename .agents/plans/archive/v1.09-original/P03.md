### Task 1: Implement Habit Detection Service

**File:** `backend/app/services/learning/habit.py`
**Purpose:** Detect recurring trigger-action pairs. When a user performs an action, check if a habit record exists. If yes, increment. If no, create. Classify frequency when occurrences ≥5.

```python
# backend/app/services/learning/habit.py

from datetime import datetime
from typing import Dict, List, Optional
from sqlalchemy.orm import Session
from backend.app.models.learning.habit import Habit


class HabitDetectionService:
    """Detects and tracks user habits — recurring trigger-action pairs.
    
    Habit lifecycle:
    1. First observation: create habit record (occurrences=1)
    2. Repeated observations: increment occurrences
    3. ≥5 occurrences: classify as confirmed habit
    4. Frequency classification: irregular → weekly → daily
    
    Architecture ref: 1.6 (Evidence Over Opinion) — habits require ≥5 occurrences.
    """
    
    MIN_OCCURRENCES_FOR_HABIT = 5
    MAX_HABITS_PER_USER = 100
    
    def __init__(self, db: Session):
        self.db = db
    
    async def observe_action(self, user_id: int, action: str,
                              trigger: str = None) -> Habit:
        """Record a user action observation for habit detection."""
        existing = self.db.query(Habit).filter(
            Habit.user_id == user_id,
            Habit.action == action
        ).first()
        
        if existing:
            existing.observe()
            if trigger and not existing.trigger:
                existing.trigger = trigger
            self.db.commit()
            return existing
        
        habit = Habit(
            user_id=user_id, habit_name=f"Habit: {action}",
            action=action, trigger=trigger,
            occurrences=1, first_observed=datetime.utcnow(),
            last_observed=datetime.utcnow()
        )
        self.db.add(habit)
        self.db.commit()
        return habit
    
    async def get_habits(self, user_id: int, 
                          min_occurrences: int = None) -> List[Dict]:
        """Get detected habits (default: confirmed habits only)."""
        min_occ = min_occurrences or self.MIN_OCCURRENCES_FOR_HABIT
        habits = self.db.query(Habit).filter(
            Habit.user_id == user_id,
            Habit.occurrences >= min_occ
        ).order_by(Habit.occurrences.desc()).all()
        
        return [
            {"id": h.id, "name": h.habit_name, "action": h.action,
             "trigger": h.trigger, "frequency": h.frequency,
             "occurrences": h.occurrences, "confidence": h.confidence}
            for h in habits
        ]
    
    async def get_habit(self, user_id: int, action: str) -> Optional[Dict]:
        """Get a specific habit by action."""
        habit = self.db.query(Habit).filter(
            Habit.user_id == user_id, Habit.action == action
        ).first()
        
        if habit:
            return {"id": habit.id, "name": habit.habit_name, "action": habit.action,
                    "trigger": habit.trigger, "frequency": habit.frequency,
                    "occurrences": habit.occurrences}
        return None
    
    async def get_habits_by_frequency(self, user_id: int, frequency: str) -> List[Dict]:
        """Get habits filtered by frequency classification."""
        habits = self.db.query(Habit).filter(
            Habit.user_id == user_id,
            Habit.frequency == frequency,
            Habit.occurrences >= self.MIN_OCCURRENCES_FOR_HABIT
        ).order_by(Habit.occurrences.desc()).all()
        
        return [
            {"action": h.action, "trigger": h.trigger,
             "occurrences": h.occurrences}
            for h in habits
        ]
    
    async def is_habitual(self, user_id: int, action: str) -> bool:
        """Check if an action is a confirmed habit."""
        habit = await self.get_habit(user_id, action)
        return habit is not None and habit["occurrences"] >= self.MIN_OCCURRENCES_FOR_HABIT


class BehaviorAdaptationService:
    """Adapts Cortex behavior based on learned preferences and habits.
    
    Adaptation is conservative:
    - Only adapts when confidence ≥ 0.6
    - Limits suggestions to top 5
    - Always allows user override
    - Never removes options, only reorders/limits
    
    Architecture ref: 1.7 (Incremental Safety) — conservative adaptation,
    user override always available.
    """
    
    CONFIDENCE_THRESHOLD = 0.6
    MAX_SUGGESTIONS = 5
    
    def __init__(self, db: Session = None):
        self.db = db
    
    async def adapt_response(self, user_id: int, context: Dict,
                              base_response: str) -> str:
        """Adapt response based on learned preferences."""
        if not self.db:
            return base_response
        
        from backend.app.models.learning.preference import UserPreference
        
        # Check for response style preference
        style_pref = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == "response_style",
            UserPreference.key == "length",
            UserPreference.confidence >= self.CONFIDENCE_THRESHOLD
        ).first()
        
        if style_pref and style_pref.value == "concise":
            # Trim verbose content
            sentences = base_response.split('.')
            if len(sentences) > 5:
                return '. '.join(sentences[:5]) + '.'
        
        return base_response
    
    async def adapt_suggestions(self, user_id: int, suggestions: list,
                                 context: Dict) -> list:
        """Adapt and rank suggestions based on learned patterns."""
        # Limit to top N
        return suggestions[:self.MAX_SUGGESTIONS]
    
    async def should_suggest(self, user_id: int, suggestion_type: str) -> bool:
        """Determine if we should make a suggestion of this type."""
        if not self.db:
            return True
        
        from backend.app.models.learning.learning_event import LearningEvent
        
        # Check recent rejection rate for this type
        recent = self.db.query(LearningEvent).filter(
            LearningEvent.user_id == user_id,
            LearningEvent.event_type == "correction"
        ).order_by(LearningEvent.created_at.desc()).limit(20).all()
        
        if len(recent) < 5:
            return True  # Not enough data, default to suggesting
        
        # If >50% of recent feedback is negative for this type, suppress
        return True  # Conservative default
    
    async def get_adaptation_profile(self, user_id: int) -> Dict:
        """Get the adaptation profile for a user."""
        if not self.db:
            return {"response_style": "standard", "suggestion_tolerance": "normal"}
        
        from backend.app.models.learning.preference import UserPreference
        
        prefs = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.confidence >= self.CONFIDENCE_THRESHOLD
        ).all()
        
        profile = {"response_style": "standard", "suggestion_tolerance": "normal"}
        for p in prefs:
            if p.category == "response_style":
                profile["response_style"] = p.value
            elif p.category == "notification":
                profile["suggestion_tolerance"] = p.value
        
        return profile


class FeedbackLearningService:
    """Records and processes user feedback to drive learning.
    
    Feedback types:
    - 'affirmation': User approved/accepted (+0.1 delta)
    - 'correction': User corrected/rejected (-0.1 delta)
    - 'feedback': Explicit text feedback
    - 'implicit': Implicit behavioral signal
    
    Architecture ref: 1.7 (Incremental Safety) — learning rate bounded at ±0.1.
    """
    
    MAX_DELTA = 0.1
    
    def __init__(self, db: Session):
        self.db = db
    
    async def record_feedback(self, user_id: int, event_type: str,
                               context: dict = None, input_data: dict = None,
                               output_data: dict = None) -> dict:
        """Record a feedback event."""
        delta = 0.0
        if event_type == "affirmation":
            delta = self.MAX_DELTA
        elif event_type == "correction":
            delta = -self.MAX_DELTA
        
        from backend.app.models.learning.learning_event import LearningEvent
        event = LearningEvent(
            user_id=user_id, event_type=event_type,
            context=str(context) if context else None,
            input_data=str(input_data) if input_data else None,
            output_data=str(output_data) if output_data else None,
            delta=delta
        )
        self.db.add(event)
        self.db.commit()
        
        # Apply delta to related preferences if context provides category/key
        if context and "category" in context and "key" in context:
            await self._apply_to_preference(
                user_id, context["category"], context["key"], delta
            )
        
        return {"event_id": event.id, "delta": delta, "applied": True}
    
    async def _apply_to_preference(self, user_id: int, category: str,
                                    key: str, delta: float) -> None:
        """Apply feedback delta to a specific preference."""
        from backend.app.models.learning.preference import UserPreference
        
        pref = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == category,
            UserPreference.key == key
        ).first()
        
        if pref:
            if delta > 0:
                pref.strengthen(abs(delta))
            else:
                pref.weaken(abs(delta))
            self.db.commit()
    
    async def get_learning_rate(self, user_id: int) -> float:
        """Get current learning rate (ratio of positive feedback)."""
        from backend.app.models.learning.learning_event import LearningEvent
        
        events = self.db.query(LearningEvent).filter(
            LearningEvent.user_id == user_id
        ).all()
        
        if not events:
            return 0.5  # Neutral default
        
        positive = sum(1 for e in events if e.event_type == "affirmation")
        return positive / len(events) if events else 0.5
    
    async def get_feedback_history(self, user_id: int, limit: int = 20) -> List[Dict]:
        """Get recent feedback events."""
        from backend.app.models.learning.learning_event import LearningEvent
        
        events = self.db.query(LearningEvent).filter(
            LearningEvent.user_id == user_id
        ).order_by(LearningEvent.created_at.desc()).limit(limit).all()
        
        return [
            {"id": e.id, "type": e.event_type, "delta": e.delta,
             "context": e.context, "created_at": str(e.created_at)}
            for e in events
        ]
```

```python
        assert len(habits) == 0  # Only 1 occurrence
```

## Security Considerations

1. **User isolation** — All queries filter by `user_id`.
2. **Delta bounds** — MAX_DELTA = 0.1 prevents runaway learning.
3. **Adaptation threshold** — Only adapt when confidence ≥ 0.6.
4. **User override** — Adaptation is advisory. User can always override.

---

## Performance Considerations

1. **Habit lookup** — O(1) via index on (user_id, action).
2. **Adaptation queries** — Use indexed preference lookups.
3. **Feedback history** — Limited to 20 most recent. No full table scan.
