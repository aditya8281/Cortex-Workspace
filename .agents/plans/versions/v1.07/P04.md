```python
# --- Graph Query Endpoints ---

@router.get("/stats", response_model=GraphStatsResponse)
async def get_graph_stats(
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get memory graph statistics for the current user."""
    nodes = db.query(MemoryNode).filter(MemoryNode.user_id == current_user.id).all()
    edges_source = db.query(MemoryEdge).join(
        MemoryNode, MemoryEdge.source_node_id == MemoryNode.id
    ).filter(MemoryNode.user_id == current_user.id).all()

    active_edges = [e for e in edges_source if e.invalid_at is None]
    nodes_by_type = {}
    for n in nodes:
        nodes_by_type[n.memory_type] = nodes_by_type.get(n.memory_type, 0) + 1
    edges_by_rel = {}
    for e in active_edges:
        edges_by_rel[e.relationship] = edges_by_rel.get(e.relationship, 0) + 1

    confidences = [n.confidence for n in nodes if n.confidence is not None]
    weights = [e.weight for e in active_edges if e.weight is not None]

    return GraphStatsResponse(
        total_nodes=len(nodes),
        total_edges=len(edges_source),
        active_edges=len(active_edges),
        nodes_by_type=nodes_by_type,
        edges_by_relationship=edges_by_rel,
        avg_confidence=sum(confidences) / len(confidences) if confidences else 0.0,
        avg_edge_weight=sum(weights) / len(weights) if weights else 0.0,
    )


@router.get("/nodes", response_model=list[MemoryNodeResponse])
async def get_graph_nodes(
    memory_type: Optional[str] = None,
    min_confidence: float = Query(0.0, ge=0.0, le=1.0),
    limit: int = Query(100, ge=1, le=500),
    offset: int = Query(0, ge=0),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get graph nodes with optional filtering."""
    query = db.query(MemoryNode).filter(
        MemoryNode.user_id == current_user.id,
        MemoryNode.confidence >= min_confidence
    )
    if memory_type:
        query = query.filter(MemoryNode.memory_type == memory_type)
    return query.order_by(MemoryNode.confidence.desc()).offset(offset).limit(limit).all()


@router.get("/edges", response_model=list[MemoryEdgeResponse])
async def get_graph_edges(
    relationship: Optional[str] = None,
    active_only: bool = True,
    limit: int = Query(100, ge=1, le=500),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get graph edges with optional filtering."""
    query = db.query(MemoryEdge).join(
        MemoryNode, MemoryEdge.source_node_id == MemoryNode.id
    ).filter(MemoryNode.user_id == current_user.id)
    if active_only:
        query = query.filter(MemoryEdge.invalid_at.is_(None))
    if relationship:
        query = query.filter(MemoryEdge.relationship == relationship)
    return query.limit(limit).all()


@router.post("/build")
async def build_graph(
    db: Session = Depends(get_db),
```

```typescript
// frontend/features/memory/api.ts

import { apiClient } from '@/shared/api/client';

// Types matching backend Pydantic schemas
export interface MemoryNode {
  id: number;
  user_id: number;
  memory_type: 'episodic' | 'semantic' | 'working';
  memory_id: number;
  label: string | null;
  confidence: number;
  created_at: string;
  last_accessed: string | null;
}

export interface MemoryEdge {
  id: number;
  source_node_id: number;
  target_node_id: number;
  relationship: string;
  weight: number;
  confidence: number;
  created_at: string;
  last_reinforced: string | null;
  valid_at: string | null;
  invalid_at: string | null;
}

export interface GraphStats {
  total_nodes: number;
  total_edges: number;
  active_edges: number;
  nodes_by_type: Record<string, number>;
  edges_by_relationship: Record<string, number>;
  avg_confidence: number;
  avg_edge_weight: number;
}

export interface GraphReflection {
  total_nodes: number;
  total_edges: number;
  orphan_count: number;
  avg_degree: number;
  health_score: number;
  knowledge_hubs: Array<{ node_id: number; degree: number }>;
  insights: string[];
}

export interface TemporalPattern {
  type: string;
  description: string;
  confidence: number;
}

export const memoryGraphApi = {
  // Graph query
  getStats: () => apiClient.get<GraphStats>('/memory/graph/stats'),
  getNodes: (params?: { memory_type?: string; min_confidence?: number; limit?: number }) =>
    apiClient.get<MemoryNode[]>('/memory/graph/nodes', { params }),
  getEdges: (params?: { relationship?: string; active_only?: boolean; limit?: number }) =>
    apiClient.get<MemoryEdge[]>('/memory/graph/edges', { params }),
  buildGraph: () => apiClient.post('/memory/graph/build'),

  // Consolidation
  consolidate: () => apiClient.post('/memory/graph/consolidate'),

  // Evolution
  reinforce: (type: string, id: number) =>
    apiClient.post(`/memory/graph/evolve/${type}/${id}/reinforce`),
  decay: () => apiClient.post('/memory/graph/evolve/decay'),
  invalidate: (type: string, id: number, reason: string = 'contradicted') =>
    apiClient.post(`/memory/graph/evolve/${type}/${id}/invalidate?reason=${reason}`),
  getEvolutionStats: () => apiClient.get('/memory/graph/evolve/stats'),

  // Reflection
  reflect: () => apiClient.get<GraphReflection>('/memory/graph/reflect'),
  getOrphans: () => apiClient.get('/memory/graph/reflect/orphans'),
  getHubs: (topN: number = 10) => apiClient.get(`/memory/graph/reflect/hubs?top_n=${topN}`),

  // Cross-domain
  connectCrossDomain: (sourceType: string, sourceId: number, targetType: string, targetId: number, relationship: string) =>
    apiClient.post(`/memory/graph/cross-domain/connect?source_type=${sourceType}&source_id=${sourceId}&target_type=${targetType}&target_id=${targetId}&relationship=${relationship}`),
  getConnections: (type: string, id: number) =>
    apiClient.get(`/memory/graph/cross-domain/connections/${type}/${id}`),
  getDomainSummary: (type: string) =>
    apiClient.get(`/memory/graph/cross-domain/summary/${type}`),

  // Traversal
  bfs: (type: string, id: number, maxDepth: number = 3) =>
    apiClient.get(`/memory/graph/traverse/bfs/${type}/${id}?max_depth=${maxDepth}`),
  shortestPath: (sourceType: string, sourceId: number, targetType: string, targetId: number) =>
    apiClient.get(`/memory/graph/traverse/shortest-path?source_type=${sourceType}&source_id=${sourceId}&target_type=${targetType}&target_id=${targetId}`),
  getNeighborhood: (type: string, id: number, hops: number = 2) =>
    apiClient.get(`/memory/graph/traverse/neighborhood/${type}/${id}?hops=${hops}`),
  getCommunities: () => apiClient.get('/memory/graph/traverse/communities'),

  // Temporal
  getTemporalContext: (hours: number = 24) =>
    apiClient.get(`/memory/graph/temporal/context?hours=${hours}`),
  getTemporalPatterns: () => apiClient.get('/memory/graph/temporal/patterns'),
  getTimeline: (days: number = 30) =>
    apiClient.get(`/memory/graph/temporal/timeline?days=${days}`),
  getByTimeRange: (start: string, end: string, memoryType?: string) =>
    apiClient.get(`/memory/graph/temporal/range?start=${start}&end=${end}${memoryType ? '&memory_type=' + memoryType : ''}`),
};
```

## Security Considerations

1. **Authentication required** — All endpoints use `Depends(get_current_user)`.
2. **Ownership enforcement** — All queries filter by `current_user.id`. No cross-user data access.
3. **Input validation** — Query parameters have `ge`/`le` constraints. Datetime parameters validated with try/except.
4. **Rate limiting** — Graph build and consolidation endpoints should be rate-limited (1 per minute per user).
5. **Data exposure** — Graph stats don't expose memory content, only structural metadata.

---

## Performance Considerations

1. **Pagination** — All list endpoints support `limit` and `offset` parameters.
2. **Query optimization** — Use indexed columns for all WHERE clauses.
3. **Response size** — Limit default results to 100 nodes/edges. Frontend can request more.
4. **Background jobs** — Graph build and consolidation run as background tasks. API returns immediately with job ID.
5. **Caching** — Graph stats can be cached for 5 minutes. Invalidate on graph mutation.
