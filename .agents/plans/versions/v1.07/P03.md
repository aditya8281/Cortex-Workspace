### Task 1: Implement Cross-Domain Service

**File:** `backend/app/services/memory/cross_domain.py`
**Purpose:** Link memories across different Cortex subsystems (code, email, calendar, documents, awareness) into a unified graph. Enables queries like "show me everything related to this commit" that span multiple domains.

```python
# backend/app/services/memory/cross_domain.py

from typing import List, Dict, Optional
from sqlalchemy.orm import Session
from backend.app.models.memory.memory_node import MemoryNode
from backend.app.models.memory.memory_edge import MemoryEdge


class CrossDomainService:
    """Links memories across different Cortex subsystems.
    
    Enables cross-domain queries like:
    - "What email prompted this code change?"
    - "What calendar events are related to this project?"
    - "What documents discuss this topic?"
    
    Architecture ref: 4.4 (Graph Architecture) — general relationship types
    for non-code content. 4.5 (Retrieval Architecture) — graph results feed
    into RRF for hybrid search.
    """
    
    # Maximum traversal depth to prevent infinite loops
    MAX_TRAVERSAL_DEPTH = 10
    # Maximum results per query
    MAX_RESULTS = 100
    
    def __init__(self, db: Session):
        self.db = db
    
    async def connect_domains(self, user_id: int, source_type: str, 
                               source_id: int, target_type: str, 
                               target_id: int, relationship: str,
                               weight: float = 1.0) -> MemoryEdge:
        """Create a cross-domain connection between two memories."""
        source_node = self._get_or_create_node(user_id, source_type, source_id)
        target_node = self._get_or_create_node(user_id, target_type, target_id)
        
        # Check if edge already exists
        existing = self.db.query(MemoryEdge).filter(
            MemoryEdge.source_node_id == source_node.id,
            MemoryEdge.target_node_id == target_node.id,
            MemoryEdge.relationship == relationship
        ).first()
        
        if existing:
            existing.reinforce(0.1)
            self.db.commit()
            return existing
        
        edge = MemoryEdge(
            source_node_id=source_node.id,
            target_node_id=target_node.id,
            relationship=relationship,
            weight=weight,
            confidence=0.8
        )
        self.db.add(edge)
        self.db.commit()
        return edge
    
    async def get_connections(self, user_id: int, memory_type: str, 
                               memory_id: int) -> List[Dict]:
        """Get all connections for a memory (both directions)."""
        node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == memory_type,
            MemoryNode.memory_id == memory_id
        ).first()
        
        if not node:
            return []
        
        # Get all edges (both directions)
        edges = self.db.query(MemoryEdge).filter(
            ((MemoryEdge.source_node_id == node.id) |
             (MemoryEdge.target_node_id == node.id)),
            MemoryEdge.invalid_at.is_(None)
        ).all()
        
        connections = []
        for edge in edges:
            other_node_id = edge.target_node_id if edge.source_node_id == node.id else edge.source_node_id
            other_node = self.db.query(MemoryNode).filter(MemoryNode.id == other_node_id).first()
            if other_node:
                connections.append({
                    "memory_type": other_node.memory_type,
                    "memory_id": other_node.memory_id,
                    "relationship": edge.relationship,
                    "weight": edge.weight,
                    "label": other_node.label,
                    "direction": "outgoing" if edge.source_node_id == node.id else "incoming"
                })
        
        return sorted(connections, key=lambda x: x["weight"], reverse=True)
    
    async def get_domain_summary(self, user_id: int, memory_type: str) -> Dict:
        """Get summary of a domain's connections."""
        nodes = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == memory_type
        ).all()
        
        total_edges = 0
        cross_domain_edges = 0
        for node in nodes:
            edges = self.db.query(MemoryEdge).filter(
                ((MemoryEdge.source_node_id == node.id) |
                 (MemoryEdge.target_node_id == node.id)),
                MemoryEdge.invalid_at.is_(None)
            ).all()
            total_edges += len(edges)
            for edge in edges:
                other_id = edge.target_node_id if edge.source_node_id == node.id else edge.source_node_id
                other = self.db.query(MemoryNode).filter(MemoryNode.id == other_id).first()
                if other and other.memory_type != memory_type:
                    cross_domain_edges += 1
        
        return {
            "domain": memory_type,
            "node_count": len(nodes),
            "total_edges": total_edges,
            "cross_domain_edges": cross_domain_edges,
            "connectivity": total_edges / len(nodes) if nodes else 0
        }
    
    def _get_or_create_node(self, user_id: int, memory_type: str, memory_id: int) -> MemoryNode:
        """Get or create a memory node."""
        node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == memory_type,
            MemoryNode.memory_id == memory_id
        ).first()
        
        if not node:
            node = MemoryNode(
                user_id=user_id, memory_type=memory_type, memory_id=memory_id,
                label=f"{memory_type}:{memory_id}", confidence=0.5
            )
            self.db.add(node)
            self.db.commit()
        
        return node
```

### Task 2: Implement Graph Traversal Service

**File:** `backend/app/services/memory/traversal.py`
**Purpose:** BFS traversal, shortest path finding, and community detection. These algorithms enable graph-powered queries like "find the shortest path between these two memories" and "identify clusters of related knowledge."

```python
# backend/app/services/memory/traversal.py

from typing import List, Dict, Optional, Set, Tuple
from collections import deque
from sqlalchemy.orm import Session
from backend.app.models.memory.memory_node import MemoryNode
from backend.app.models.memory.memory_edge import MemoryEdge


class GraphTraversalService:
    """Graph traversal algorithms for the memory graph.
    
    Provides:
    - BFS traversal from a node
    - Shortest path between two nodes (BFS for unweighted, Dijkstra for weighted)
    - Community detection (label propagation)
    - Neighborhood queries (nodes within N hops)
    
    Architecture ref: 4.4 (Graph Architecture) — graph traversal feeds
    into retrieval via RRF.
    """
    
    MAX_DEPTH = 10  # Prevent infinite traversal
    MAX_RESULTS = 500
    
    def __init__(self, db: Session):
        self.db = db
    
    async def bfs(self, user_id: int, start_type: str, start_id: int, 
                   max_depth: int = None) -> List[Dict]:
        """BFS traversal from a starting node."""
        max_depth = max_depth or self.MAX_DEPTH
        
        start_node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == start_type,
            MemoryNode.memory_id == start_id
        ).first()
        
        if not start_node:
            return []
        
        visited: Set[int] = {start_node.id}
        queue: deque = deque([(start_node.id, 0)])
        result = []
        
        while queue and len(result) < self.MAX_RESULTS:
            node_id, depth = queue.popleft()
            if depth > max_depth:
                continue
            
            node = self.db.query(MemoryNode).filter(MemoryNode.id == node_id).first()
            if node:
                result.append({
                    "node_id": node.id,
                    "memory_type": node.memory_type,
                    "memory_id": node.memory_id,
                    "label": node.label,
                    "confidence": node.confidence,
                    "depth": depth
                })
            
            # Get connected nodes
            edges = self.db.query(MemoryEdge).filter(
                ((MemoryEdge.source_node_id == node_id) |
                 (MemoryEdge.target_node_id == node_id)),
                MemoryEdge.invalid_at.is_(None)
            ).all()
            
            for edge in edges:
                neighbor_id = edge.target_node_id if edge.source_node_id == node_id else edge.source_node_id
                if neighbor_id not in visited:
                    visited.add(neighbor_id)
                    queue.append((neighbor_id, depth + 1))
        
        return result
    
    async def shortest_path(self, user_id: int, 
                             source_type: str, source_id: int,
                             target_type: str, target_id: int) -> Dict:
        """Find shortest path between two nodes (unweighted BFS)."""
        source_node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == source_type,
            MemoryNode.memory_id == source_id
        ).first()
        
        target_node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == target_type,
            MemoryNode.memory_id == target_id
        ).first()
        
        if not source_node or not target_node:
            return {"path": [], "path_length": 0, "found": False}
        
        if source_node.id == target_node.id:
            return {"path": [source_node.id], "path_length": 0, "found": True}
        
        # BFS with parent tracking
        visited: Set[int] = {source_node.id}
        queue: deque = deque([(source_node.id, [source_node.id])])
        
        while queue:
            current_id, path = queue.popleft()
            if len(path) > self.MAX_DEPTH:
                continue
            
            edges = self.db.query(MemoryEdge).filter(
                ((MemoryEdge.source_node_id == current_id) |
                 (MemoryEdge.target_node_id == current_id)),
                MemoryEdge.invalid_at.is_(None)
            ).all()
            
            for edge in edges:
                neighbor_id = edge.target_node_id if edge.source_node_id == current_id else edge.source_node_id
                if neighbor_id == target_node.id:
                    return {
                        "path": path + [neighbor_id],
                        "path_length": len(path),
                        "found": True
                    }
                if neighbor_id not in visited:
                    visited.add(neighbor_id)
                    queue.append((neighbor_id, path + [neighbor_id]))
        
        return {"path": [], "path_length": 0, "found": False}
    
    async def get_neighborhood(self, user_id: int, memory_type: str, 
                                memory_id: int, hops: int = 2) -> Dict:
        """Get all nodes within N hops of a given node."""
        result = await self.bfs(user_id, memory_type, memory_id, max_depth=hops)
        
        return {
            "center": {"type": memory_type, "id": memory_id},
            "hops": hops,
            "node_count": len(result),
            "nodes": result
        }
    
    async def detect_communities(self, user_id: int) -> List[Dict]:
        """Detect communities using label propagation.
        
        Simple label propagation: each node adopts the most common label
        among its neighbors. Repeat until convergence.
        """
        nodes = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id
        ).all()
        
        if not nodes:
            return []
        
        # Build adjacency list
        adj: Dict[int, List[int]] = {n.id: [] for n in nodes}
        edges = self.db.query(MemoryEdge).join(
            MemoryNode, MemoryEdge.source_node_id == MemoryNode.id
        ).filter(
            MemoryNode.user_id == user_id,
            MemoryEdge.invalid_at.is_(None)
        ).all()
        
        for edge in edges:
            if edge.source_node_id in adj and edge.target_node_id in adj:
                adj[edge.source_node_id].append(edge.target_node_id)
                adj[edge.target_node_id].append(edge.source_node_id)
        
        # Label propagation (10 iterations max)
        labels = {n.id: n.id for n in nodes}
        for _ in range(10):
            changed = False
            for node_id in nodes:
                neighbors = adj.get(node_id.id, [])
                if not neighbors:
                    continue
                # Count neighbor labels
                label_counts = {}
                for nid in neighbors:
                    l = labels[nid]
                    label_counts[l] = label_counts.get(l, 0) + 1
                # Adopt most common label
                new_label = max(label_counts, key=label_counts.get)
                if labels[node_id.id] != new_label:
                    labels[node_id.id] = new_label
                    changed = True
            if not changed:
                break
        
        # Group by community
        communities: Dict[int, List[int]] = {}
        for node_id, label in labels.items():
            communities.setdefault(label, []).append(node_id)
        
        # Format results
        result = []
        for label, member_ids in communities.items():
            if len(member_ids) >= 2:  # Only report communities with 2+ members
                result.append({
                    "community_id": label,
                    "size": len(member_ids),
                    "member_node_ids": member_ids
                })
        
        return sorted(result, key=lambda x: x["size"], reverse=True)
```

### Task 3: Implement Temporal Awareness Service

**File:** `backend/app/services/memory/temporal.py`
**Purpose:** Time-based memory queries, temporal pattern detection, and context window retrieval. Answers questions like "what happened last week?" and "what time of day do I create the most memories?"

```python
# backend/app/services/memory/temporal.py

from datetime import datetime, timedelta
from typing import List, Dict
from sqlalchemy.orm import Session
from sqlalchemy import func, extract
from backend.app.models.memory.memory_node import MemoryNode
from backend.app.models.memory.episodic_memory import EpisodicMemory


class TemporalAwarenessService:
    """Time-based memory queries and temporal pattern detection.
    
    Provides:
    - Time-range queries for memories
    - Recent context retrieval
    - Temporal pattern detection (burst, gaps, cycles)
    - Time-of-day analysis
    
    Architecture ref: 4.3 (Memory Architecture) — bi-temporal tracking
    with valid_at/invalid_at timestamps.
    """
    
    def __init__(self, db: Session):
        self.db = db
    
    async def get_memories_by_time(self, user_id: int, 
                                    start: datetime, end: datetime,
                                    memory_type: str = None) -> List[Dict]:
        """Get memories within a time range."""
        query = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.created_at >= start,
            MemoryNode.created_at <= end
        )
        
        if memory_type:
            query = query.filter(MemoryNode.memory_type == memory_type)
        
        memories = query.order_by(MemoryNode.created_at.desc()).limit(100).all()
        
        return [
            {
                "id": mem.id,
                "memory_type": mem.memory_type,
                "memory_id": mem.memory_id,
                "label": mem.label,
                "confidence": mem.confidence,
                "timestamp": mem.created_at.isoformat()
            }
            for mem in memories
        ]
    
    async def get_recent_context(self, user_id: int, hours: int = 24) -> Dict:
        """Get recent memory context (last N hours)."""
        start = datetime.utcnow() - timedelta(hours=hours)
        memories = await self.get_memories_by_time(user_id, start, datetime.utcnow())
        
        # Group by type
        by_type = {}
        for mem in memories:
            t = mem["memory_type"]
            by_type.setdefault(t, []).append(mem)
        
        return {
            "period_hours": hours,
            "memory_count": len(memories),
            "by_type": {k: len(v) for k, v in by_type.items()},
            "recent_memories": memories[:20]  # Latest 20
        }
    
    async def detect_temporal_patterns(self, user_id: int) -> Dict:
        """Detect temporal patterns in memory creation."""
        memories = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id
        ).order_by(MemoryNode.created_at.desc()).limit(200).all()
        
        if len(memories) < 3:
            return {"patterns": [], "total_memories": len(memories)}
        
        patterns = []
        
        # Gap analysis
        gaps = []
        for i in range(len(memories) - 1):
            gap_hours = (memories[i].created_at - memories[i+1].created_at).total_seconds() / 3600
            gaps.append(gap_hours)
        
        avg_gap = sum(gaps) / len(gaps) if gaps else 0
        
        # Burst detection (many memories in short time)
        burst_threshold = avg_gap * 0.2 if avg_gap > 0 else 1
        burst_count = sum(1 for g in gaps if g < burst_threshold)
        if burst_count > 3:
            patterns.append({
                "type": "burst",
                "description": f"Detected {burst_count} memory bursts",
                "confidence": min(1.0, burst_count / len(gaps))
            })
        
        # Gap detection (long silence between memories)
        gap_threshold = avg_gap * 3 if avg_gap > 0 else 48
        long_gaps = sum(1 for g in gaps if g > gap_threshold)
        if long_gaps > 0:
            patterns.append({
                "type": "gap",
                "description": f"Detected {long_gaps} long gaps in memory creation",
                "confidence": min(1.0, long_gaps / len(gaps))
            })
        
        # Time-of-day analysis
        hour_counts = {}
        for mem in memories:
            hour = mem.created_at.hour
            hour_counts[hour] = hour_counts.get(hour, 0) + 1
        
        peak_hour = max(hour_counts, key=hour_counts.get) if hour_counts else None
        if peak_hour is not None:
            patterns.append({
                "type": "peak_hour",
                "description": f"Peak memory creation at hour {peak_hour}:00",
                "confidence": hour_counts[peak_hour] / len(memories)
            })
        
        return {
            "total_memories": len(memories),
            "average_gap_hours": round(avg_gap, 2),
            "oldest": memories[-1].created_at.isoformat() if memories else None,
            "newest": memories[0].created_at.isoformat() if memories else None,
            "patterns": patterns,
            "hour_distribution": hour_counts
        }
    
    async def get_memory_timeline(self, user_id: int, days: int = 30) -> List[Dict]:
        """Get memory creation timeline (daily buckets)."""
        start = datetime.utcnow() - timedelta(days=days)
        
        memories = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.created_at >= start
        ).all()
        
        # Bucket by day
        daily = {}
        for mem in memories:
            day = mem.created_at.strftime("%Y-%m-%d")
            daily.setdefault(day, {"date": day, "count": 0, "types": {}})
            daily[day]["count"] += 1
            t = mem.memory_type
            daily[day]["types"][t] = daily[day]["types"].get(t, 0) + 1
        
        return sorted(daily.values(), key=lambda x: x["date"])
```

## Security Considerations

1. **User isolation** — All traversal queries filter by `user_id`. Never traverse cross-user graphs.
2. **Traversal depth limit** — `MAX_DEPTH = 10` prevents exponential traversal on dense graphs.
3. **Result limits** — `MAX_RESULTS = 500` prevents memory exhaustion from large traversals.
4. **Community detection** — Label propagation runs in bounded iterations (max 10). No unbounded computation.
5. **No raw graph queries** — All queries go through service layer. No direct SQL graph traversal.

---

## Performance Considerations

1. **BFS complexity** — O(V + E) per traversal. With max 10k nodes and 50k edges, this is acceptable.
2. **Shortest path** — BFS is O(V + E). For weighted paths, Dijkstra would be O((V + E) log V). Use BFS for v1.07.
3. **Community detection** — Label propagation is O(V * E) per iteration, max 10 iterations. Acceptable for <10k nodes.
4. **Temporal queries** — Use composite index on `(user_id, created_at)`. Limit results to 100.
5. **Cross-domain joins** — Use indexed columns. Avoid full table scans.
