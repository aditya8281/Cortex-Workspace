Memory services need REST API endpoints for the frontend to interact with. ```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import Optional
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.memory.episodic import EpisodicMemoryService
from backend.app.schemas.memory.episodic import (
    EpisodicMemoryCreate, EpisodicMemoryUpdate, EpisodicMemoryResponse, EpisodicMemoryList
)

router = APIRouter(prefix="/episodic", tags=["memory-episodic"])


@router.post("", response_model=EpisodicMemoryResponse, status_code=201)
async def create_episodic_memory(
    data: EpisodicMemoryCreate,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Create a new episodic memory.
    
    Stores an experience, event, or conversation with temporal context.
    """
    service = EpisodicMemoryService(db)
    memory = await service.create(current_user.id, data)
    return memory


@router.get("", response_model=EpisodicMemoryList)
async def list_episodic_memories(
    limit: int = Query(10, ge=1, le=100),
    offset: int = Query(0, ge=0),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """List recent episodic memories with pagination."""
    service = EpisodicMemoryService(db)
    memories, total = await service.list_recent(current_user.id, limit, offset)
    return EpisodicMemoryList(
        memories=[EpisodicMemoryResponse.model_validate(m) for m in memories],
        total=total,
        page=(offset // limit) + 1,
        page_size=limit
    )


@router.get("/search")
async def search_episodic_memories(
    query: str = Query(..., min_length=1, max_length=200),
    limit: int = Query(10, ge=1, le=50),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Search episodic memories by content."""
    service = EpisodicMemoryService(db)
    results = await service.search_content(current_user.id, query, limit)
    return {"results": results, "query": query, "count": len(results)}


@router.get("/{memory_id}", response_model=EpisodicMemoryResponse)
async def get_episodic_memory(
    memory_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get a specific episodic memory by ID."""
    service = EpisodicMemoryService(db)
    memory = await service.retrieve(current_user.id, memory_id)
    if not memory:
        raise HTTPException(status_code=404, detail="Memory not found")
    return memory


@router.patch("/{memory_id}", response_model=EpisodicMemoryResponse)
async def update_episodic_memory(
    memory_id: int,
    data: EpisodicMemoryUpdate,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Update an episodic memory (partial update)."""
    service = EpisodicMemoryService(db)
    memory = await service.update(current_user.id, memory_id, data)
    if not memory:
        raise HTTPException(status_code=404, detail="Memory not found")
    return memory


@router.delete("/{memory_id}", status_code=204)
async def delete_episodic_memory(
    memory_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Delete an episodic memory."""
    service = EpisodicMemoryService(db)
    deleted = await service.delete(current_user.id, memory_id)
    if not deleted:
        raise HTTPException(status_code=404, detail="Memory not found")
    return None

import { useState, useEffect, useCallback } from 'react';
import { memoryApi, EpisodicMemory, EpisodicMemoryCreate } from '../api';

export function useEpisodicMemory(limit = 10) {
  const [memories, setMemories] = useState<EpisodicMemory[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const loadMemories = useCallback(async () => {
    try {
      setLoading(true);
      const response = await memoryApi.listEpisodic(limit);
      setMemories(response.data.memories);
      setTotal(response.data.total);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load memories');
    } finally {
      setLoading(false);
    }
  }, [limit]);

  useEffect(() => {
    loadMemories();
  }, [loadMemories]);

  const createMemory = useCallback(async (data: EpisodicMemoryCreate) => {
    const response = await memoryApi.createEpisodic(data);
    setMemories(prev => [response.data, ...prev]);
    setTotal(prev => prev + 1);
    return response.data;
  }, []);

  const deleteMemory = useCallback(async (id: number) => {
    await memoryApi.deleteEpisodic(id);
    setMemories(prev => prev.filter(m => m.id !== id));
    setTotal(prev => prev - 1);
  }, []);

  return { memories, total, loading, error, reload: loadMemories, create: createMemory, remove: deleteMemory };
}

import { useState, useCallback } from 'react';
import { memoryApi, SearchResult } from '../api';

export function useMemorySearch() {
  const [results, setResults] = useState<SearchResult[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const search = useCallback(async (query: string, type?: string, limit = 10) => {
    if (!query.trim()) {
      setResults([]);
      return;
    }
    try {
      setLoading(true);
      setError(null);
      const response = await memoryApi.searchAll(query, type, limit);
      setResults(response.data.results);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Search failed');
    } finally {
      setLoading(false);
    }
  }, []);

  return { results, loading, error, search };
}

- **No secrets in memory:** Schema validators reject content matching secret patterns.
