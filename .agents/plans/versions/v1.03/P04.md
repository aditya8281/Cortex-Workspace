## Purpose

Memories without search are useless — you can't retrieve what you can't find. Memories without forgetting are dangerous — information overload degrades performance. This phase implements the intelligence layer that makes memory useful: search finds relevant memories, forgetting fades irrelevant ones, and temporal scoring ensures recent and frequently-accessed memories surface first.

## Implementation Tasks

### Task 1: Implement Memory Search Service

**Architecture:** The search service provides cross-type retrieval with multi-signal scoring. It searches episodic, semantic, and working memories simultaneously, then ranks results using a composite score that blends:
1. **Text relevance** — ILIKE match quality (how many query terms match)
2. **Recency** — How recently the memory was created or accessed
3. **Importance** — The memory's intrinsic importance score
4. **Access frequency** — How often the memory has been retrieved
5. **Graph centrality** — How many connections the memory has (more connected = more important)

The composite score is a weighted sum:
```
score = 0.3 * text_relevance + 0.25 * recency + 0.2 * importance + 0.15 * access_frequency + 0.1 * graph_centrality
```

**File:** `backend/app/services/memory/search.py`

```python
import math
from datetime import datetime
from typing import List, Optional, Dict, Any
from sqlalchemy.orm import Session
from sqlalchemy import desc, func
from backend.app.models.memory.episodic import EpisodicMemory
from backend.app.models.memory.semantic import SemanticMemory
from backend.app.models.memory.working import WorkingMemory
from backend.app.models.memory.graph import MemoryNode, MemoryEdge
from backend.app.services.memory.temporal import TemporalScoring


# Scoring weights
WEIGHT_TEXT_RELEVANCE = 0.30
WEIGHT_RECENCY = 0.25
WEIGHT_IMPORTANCE = 0.20
WEIGHT_ACCESS_FREQUENCY = 0.15
WEIGHT_GRAPH_CENTRALITY = 0.10


class MemorySearchService:
    """Cross-type memory search with multi-signal scoring."""
    
    def __init__(self, db: Session):
        self.db = db
        self.temporal = TemporalScoring()
    
    async def search(
        self,
        user_id: int,
        query: str,
        memory_type: Optional[str] = None,
        limit: int = 10,
        min_score: float = 0.0
    ) -> List[Dict[str, Any]]:
        """Search across all memory types with multi-signal scoring.
        
        Args:
            user_id: Owner's user ID
            query: Search query string
            memory_type: Optional filter ("episodic", "semantic", "working")
            limit: Maximum results
            min_score: Minimum composite score threshold
            
        Returns:
            List of scored results sorted by composite score descending
        """
        results = []
        query_terms = query.lower().split()
        
        # Search episodic memories
        if memory_type is None or memory_type == "episodic":
            episodic = self.db.query(EpisodicMemory).filter(
                EpisodicMemory.user_id == user_id,
                EpisodicMemory.content.ilike(f"%{query}%")
            ).all()
            
            for mem in episodic:
                score = await self._compute_score(
                    user_id=user_id,
                    content=mem.content,
                    importance=mem.importance,
                    confidence=mem.confidence,
                    access_count=mem.access_count,
                    created_at=mem.created_at,
                    last_accessed=mem.last_accessed,
                    memory_id=mem.id,
                    memory_type="episodic",
                    query_terms=query_terms
                )
                if score >= min_score:
                    results.append({
                        "type": "episodic",
                        "id": mem.id,
                        "content": mem.content,
                        "importance": mem.importance,
                        "confidence": mem.confidence,
                        "emotion": mem.emotion,
                        "created_at": mem.created_at.isoformat(),
                        "score": score
                    })
        
        # Search semantic memories
        if memory_type is None or memory_type == "semantic":
            semantic = self.db.query(SemanticMemory).filter(
                SemanticMemory.user_id == user_id,
                SemanticMemory.content.ilike(f"%{query}%")
            ).all()
            
            for mem in semantic:
                score = await self._compute_score(
                    user_id=user_id,
                    content=mem.content,
                    importance=mem.confidence,  # Use confidence as importance for semantic
                    confidence=mem.confidence,
                    access_count=mem.access_count,
                    created_at=mem.created_at,
                    last_accessed=mem.last_accessed,
                    memory_id=mem.id,
                    memory_type="semantic",
                    query_terms=query_terms
                )
                if score >= min_score:
                    results.append({
                        "type": "semantic",
                        "id": mem.id,
                        "content": mem.content,
                        "category": mem.category,
                        "confidence": mem.confidence,
                        "source": mem.source,
                        "created_at": mem.created_at.isoformat(),
                        "score": score
                    })
        
        # Search working memory
        if memory_type is None or memory_type == "working":
            working = self.db.query(WorkingMemory).filter(
                WorkingMemory.user_id == user_id,
                WorkingMemory.content.ilike(f"%{query}%"),
                WorkingMemory.expires_at > datetime.utcnow()
            ).all()
            
            for mem in working:
                # Working memory gets a recency boost (it's current context)
                score = await self._compute_score(
                    user_id=user_id,
                    content=mem.content,
                    importance=0.5,
                    confidence=1.0,
                    access_count=0,
                    created_at=mem.created_at,
                    last_accessed=None,
                    memory_id=mem.id,
                    memory_type="working",
                    query_terms=query_terms
                )
                # Boost working memory score by 20% (current context is more relevant)
                score *= 1.2
                
                if score >= min_score:
                    results.append({
                        "type": "working",
                        "id": mem.id,
                        "content": mem.content,
                        "slot": mem.slot,
                        "priority": mem.priority,
                        "created_at": mem.created_at.isoformat(),
                        "score": min(1.0, score)
                    })
        
        # Sort by composite score descending
        results.sort(key=lambda x: x["score"], reverse=True)
        return results[:limit]
    
    async def _compute_score(
        self,
        user_id: int,
        content: str,
        importance: float,
        confidence: float,
        access_count: int,
        created_at: datetime,
        last_accessed: Optional[datetime],
        memory_id: int,
        memory_type: str,
        query_terms: List[str]
    ) -> float:
        """Compute composite relevance score."""
        # Text relevance: fraction of query terms found in content
        content_lower = content.lower()
        text_relevance = sum(1 for term in query_terms if term in content_lower) / max(len(query_terms), 1)
        
        # Recency score
        recency = self.temporal.recency_score(created_at, last_accessed)
        
        # Importance weight
        importance_weight = self.temporal.importance_weight(importance, confidence)
        
        # Access frequency weight
        access_weight = self.temporal.access_frequency_weight(access_count)
        
        # Graph centrality
        graph_centrality = await self._get_graph_centrality(user_id, memory_type, memory_id)
        
        # Composite score
        score = (
            WEIGHT_TEXT_RELEVANCE * text_relevance +
            WEIGHT_RECENCY * recency +
            WEIGHT_IMPORTANCE * importance_weight +
            WEIGHT_ACCESS_FREQUENCY * access_weight +
            WEIGHT_GRAPH_CENTRALITY * graph_centrality
        )
        
        return min(1.0, max(0.0, score))
    
    async def _get_graph_centrality(
        self, user_id: int, memory_type: str, memory_id: int
    ) -> float:
        """Get graph centrality score (0.0-1.0) for a memory.
        
        Centrality = number of edges / max_expected_edges (normalized to 0-1).
        """
        node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == memory_type,
            MemoryNode.memory_id == memory_id
        ).first()
        
        if not node:
            return 0.0  # No graph node = no centrality
        
        edge_count = self.db.query(MemoryEdge).filter(
            (MemoryEdge.source_id == node.id) | (MemoryEdge.target_id == node.id)
        ).count()
        
        # Normalize: 0 edges = 0.0, 10+ edges = 1.0
        return min(1.0, edge_count / 10.0)
    
    async def search_by_importance(
        self, user_id: int, min_importance: float = 0.5, limit: int = 10
    ) -> List[Dict[str, Any]]:
        """Search by importance threshold across episodic memories."""
        memories = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id,
            EpisodicMemory.importance >= min_importance
        ).order_by(desc(EpisodicMemory.importance)).limit(limit).all()
        
        return [
            {
                "type": "episodic",
                "id": m.id,
                "content": m.content,
                "importance": m.importance,
                "confidence": m.confidence,
                "created_at": m.created_at.isoformat()
            }
            for m in memories
        ]
    
    async def search_by_recency(
        self, user_id: int, limit: int = 10
    ) -> List[Dict[str, Any]]:
        """Search by most recent across all types."""
        episodic = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id
        ).order_by(desc(EpisodicMemory.created_at)).limit(limit).all()
        
        semantic = self.db.query(SemanticMemory).filter(
            SemanticMemory.user_id == user_id
        ).order_by(desc(SemanticMemory.created_at)).limit(limit).all()
        
        results = []
        for m in episodic:
            results.append({
                "type": "episodic", "id": m.id, "content": m.content,
                "created_at": m.created_at.isoformat()
            })
        for m in semantic:
            results.append({
                "type": "semantic", "id": m.id, "content": m.content,
                "created_at": m.created_at.isoformat()
            })
        
        results.sort(key=lambda x: x["created_at"], reverse=True)
        return results[:limit]
    
    async def get_related_memories(
        self, user_id: int, memory_type: str, memory_id: int, limit: int = 5
    ) -> List[Dict[str, Any]]:
        """Get memories related via graph connections."""
        node = self.db.query(MemoryNode).filter(
            MemoryNode.user_id == user_id,
            MemoryNode.memory_type == memory_type,
            MemoryNode.memory_id == memory_id
        ).first()
        
        if not node:
            return []
        
        edges = self.db.query(MemoryEdge).filter(
            (MemoryEdge.source_id == node.id) | (MemoryEdge.target_id == node.id)
        ).order_by(desc(MemoryEdge.weight)).limit(limit).all()
        
        results = []
        for edge in edges:
            connected_id = edge.target_id if edge.source_id == node.id else edge.source_id
            connected_node = self.db.query(MemoryNode).filter(
                MemoryNode.id == connected_id
            ).first()
            
            if connected_node:
                results.append({
                    "memory_type": connected_node.memory_type,
                    "memory_id": connected_node.memory_id,
                    "label": connected_node.label,
                    "edge_weight": edge.weight,
                    "relationship": edge.relationship
                })
        
        return results
```

**Key Design Decisions:**
- **Multi-signal scoring:** Five signals blended with configurable weights. This prevents any single signal from dominating (e.g., a recently created but unimportant memory won't outrank an important old memory).
- **Working memory boost:** Working memory items get a 20% score boost. Current context is inherently more relevant.
- **Graph centrality:** More-connected memories score higher. This surfaces "hub" memories that are central to the knowledge network.
- **ILIKE fulltext:** Simple but effective for v1.03. Trigram index (`pg_trgm`) can be added later for better performance.
- **`get_related_memories`:** Uses graph edges to find related memories. Useful for "show me related" UI features.

**Risk Notes:**
- Risk: ILIKE fulltext is slow on large datasets (>100K memories). Mitigation: Add `pg_trgm` trigram index in v1.07. For v1.03, acceptable for personal use.
- Risk: Scoring weights may need tuning. Mitigation: Weights are constants at module level. Easy to adjust.
- Risk: Graph centrality query is expensive (joins across tables). Mitigation: Cache centrality scores in Redis.

### Task 2: Implement Forgetting Service

**Architecture:** The forgetting service implements Ebbinghaus-style decay. Memories lose confidence over time based on:
1. **Time since creation:** Older memories decay faster
2. **Time since last access:** Recently accessed memories decay slower
3. **Importance:** Important memories decay slower
4. **Access frequency:** Frequently accessed memories decay slower

The decay formula:
```
decay_factor = max(0.05, 1.0 - (days_since_access * decay_rate))
new_confidence = old_confidence * decay_factor
```

Where `decay_rate` varies by memory type:
- Episodic: 0.01 per day (faster decay — experiences fade)
- Semantic: 0.005 per day (slower decay — facts are stable)

Memories below confidence threshold (0.1) are candidates for garbage collection.

**File:** `backend/app/services/memory/decay.py`

```python
from datetime import datetime, timedelta
from typing import Dict, List, Tuple
from sqlalchemy.orm import Session
from sqlalchemy import func
from backend.app.models.memory.episodic import EpisodicMemory
from backend.app.models.memory.semantic import SemanticMemory
from backend.app.models.memory.graph import MemoryNode, MemoryEdge

# Decay rates per day
EPISODIC_DECAY_RATE = 0.01    # Episodic memories fade faster
SEMANTIC_DECAY_RATE = 0.005   # Semantic memories fade slower

# Confidence floor (memories below this are garbage collected)
CONFIDENCE_FLOOR = 0.05

# Garbage collection threshold
GC_THRESHOLD = 0.1


class ForgettingService:
    """Intelligent forgetting via Ebbinghaus-style decay."""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def apply_decay(self, user_id: int) -> Dict[str, Any]:
        """Apply forgetting decay to all memories for a user.
        
        Returns:
            {
                "episodic_decayed": int,
                "semantic_decayed": int,
                "episodic_gc": int,
                "semantic_gc": int,
                "timestamp": str
            }
        """
        episodic_decayed = self._decay_episodic(user_id)
        semantic_decayed = self._decay_semantic(user_id)
        
        # Garbage collect low-confidence memories
        episodic_gc = await self._garbage_collect_episodic(user_id)
        semantic_gc = await self._garbage_collect_semantic(user_id)
        
        return {
            "episodic_decayed": episodic_decayed,
            "semantic_decayed": semantic_decayed,
            "episodic_gc": episodic_gc,
            "semantic_gc": semantic_gc,
            "timestamp": datetime.utcnow().isoformat()
        }
    
    def _decay_episodic(self, user_id: int) -> int:
        """Apply decay to episodic memories.
        
        Returns:
            Number of memories decayed
        """
        memories = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id
        ).all()
        
        count = 0
        for memory in memories:
            days_since_access = self._days_since(memory.last_accessed or memory.created_at)
            
            # Calculate decay factor
            # More access = slower decay (access_count dampens decay)
            effective_rate = EPISODIC_DECAY_RATE / max(1.0, memory.access_count * 0.1)
            decay_factor = max(CONFIDENCE_FLOOR, 1.0 - (days_since_access * effective_rate))
            
            # Apply importance damping (important memories decay slower)
            importance_damping = 0.5 + (memory.importance * 0.5)  # 0.5-1.0
            decay_factor = max(CONFIDENCE_FLOOR, decay_factor * importance_damping)
            
            new_confidence = memory.confidence * decay_factor
            if new_confidence != memory.confidence:
                memory.confidence = max(CONFIDENCE_FLOOR, new_confidence)
                count += 1
        
        if count > 0:
            self.db.commit()
        return count
    
    def _decay_semantic(self, user_id: int) -> int:
        """Apply decay to semantic memories (slower decay rate)."""
        memories = self.db.query(SemanticMemory).filter(
            SemanticMemory.user_id == user_id
        ).all()
        
        count = 0
        for memory in memories:
            days_since_access = self._days_since(memory.last_accessed or memory.created_at)
            
            effective_rate = SEMANTIC_DECAY_RATE / max(1.0, memory.access_count * 0.1)
            decay_factor = max(CONFIDENCE_FLOOR, 1.0 - (days_since_access * effective_rate))
            
            new_confidence = memory.confidence * decay_factor
            if new_confidence != memory.confidence:
                memory.confidence = max(CONFIDENCE_FLOOR, new_confidence)
                count += 1
        
        if count > 0:
            self.db.commit()
        return count
    
    async def _garbage_collect_episodic(self, user_id: int) -> int:
        """Remove episodic memories below confidence threshold."""
        memories = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id,
            EpisodicMemory.confidence < GC_THRESHOLD
        ).all()
        
        count = len(memories)
        for memory in memories:
            # Clean up graph references
            nodes = self.db.query(MemoryNode).filter(
                MemoryNode.user_id == user_id,
                MemoryNode.memory_type == "episodic",
                MemoryNode.memory_id == memory.id
            ).all()
            
            for node in nodes:
                self.db.query(MemoryEdge).filter(
                    (MemoryEdge.source_id == node.id) | (MemoryEdge.target_id == node.id)
                ).delete(synchronize_session="fetch")
                self.db.delete(node)
            
            self.db.delete(memory)
        
        if count > 0:
            self.db.commit()
        return count
    
    async def _garbage_collect_semantic(self, user_id: int) -> int:
        """Remove semantic memories below confidence threshold."""
        memories = self.db.query(SemanticMemory).filter(
            SemanticMemory.user_id == user_id,
            SemanticMemory.confidence < GC_THRESHOLD
        ).all()
        
        count = len(memories)
        for memory in memories:
            nodes = self.db.query(MemoryNode).filter(
                MemoryNode.user_id == user_id,
                MemoryNode.memory_type == "semantic",
                MemoryNode.memory_id == memory.id
            ).all()
            
            for node in nodes:
                self.db.query(MemoryEdge).filter(
                    (MemoryEdge.source_id == node.id) | (MemoryEdge.target_id == node.id)
                ).delete(synchronize_session="fetch")
                self.db.delete(node)
            
            self.db.delete(memory)
        
        if count > 0:
            self.db.commit()
        return count
    
    async def get_forgetting_stats(self, user_id: int) -> Dict[str, Any]:
        """Get statistics about memory decay state."""
        episodic_count = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id
        ).count()
        
        semantic_count = self.db.query(SemanticMemory).filter(
            SemanticMemory.user_id == user_id
        ).count()
        
        episodic_low = self.db.query(EpisodicMemory).filter(
            EpisodicMemory.user_id == user_id,
            EpisodicMemory.confidence < GC_THRESHOLD
        ).count()
        
        semantic_low = self.db.query(SemanticMemory).filter(
            SemanticMemory.user_id == user_id,
            SemanticMemory.confidence < GC_THRESHOLD
        ).count()
        
        avg_episodic_confidence = self.db.query(func.avg(EpisodicMemory.confidence)).filter(
            EpisodicMemory.user_id == user_id
        ).scalar() or 0.0
        
        avg_semantic_confidence = self.db.query(func.avg(SemanticMemory.confidence)).filter(
            SemanticMemory.user_id == user_id
        ).scalar() or 0.0
        
        return {
            "total_episodic": episodic_count,
            "total_semantic": semantic_count,
            "episodic_low_confidence": episodic_low,
            "semantic_low_confidence": semantic_low,
            "avg_episodic_confidence": round(float(avg_episodic_confidence), 3),
            "avg_semantic_confidence": round(float(avg_semantic_confidence), 3),
            "gc_candidates": episodic_low + semantic_low
        }
    
    def _days_since(self, dt: datetime) -> float:
        """Calculate days since a datetime."""
        if dt is None:
            return 0.0
        delta = datetime.utcnow() - dt
        return max(0.0, delta.total_seconds() / 86400.0)
```

**Key Design Decisions:**
- **Access count dampening:** Frequently accessed memories decay slower. Formula: `effective_rate = base_rate / (1 + access_count * 0.1)`. A memory accessed 10 times decays at 50% the base rate.
- **Importance damping:** Important memories decay slower. Formula: `importance_damping = 0.5 + importance * 0.5`. A memory with importance=1.0 decays at 50% the base rate.
- **Confidence floor:** Memories never decay below 0.05. They persist until garbage collected.
- **Garbage collection:** Memories below 0.1 confidence are permanently deleted (with graph cleanup). This is a hard delete, not soft delete.
- **`get_forgetting_stats`:** Provides monitoring data for the frontend dashboard.

**Risk Notes:**
- Risk: Aggressive decay could delete important memories. Mitigation: Importance damping ensures important memories survive. GC threshold is low (0.1).
- Risk: `apply_decay` iterates all memories (O(n)). Mitigation: Acceptable for personal use (<10K memories). Batch processing for larger datasets in v1.07.
- Risk: GC deletes graph nodes/edges. Mitigation: Cascade cleanup prevents orphaned references.

### Task 3: Implement Temporal Scoring

**Architecture:** Temporal scoring provides time-aware relevance ranking. It calculates scores based on:
1. **Recency:** How recently the memory was created or accessed
2. **Time-of-day patterns:** Memories created at similar times score higher
3. **Seasonal patterns:** Memories from similar seasons score higher (optional, v1.07)

The scoring uses exponential decay with configurable half-life.

**File:** `backend/app/services/memory/temporal.py`

```python
import math
from datetime import datetime
from typing import Optional


class TemporalScoring:
    """Time-aware relevance scoring for memories."""
    
    # Half-life in days (memory score halves after this many days)
    RECENCY_HALF_LIFE = 30.0  # 30 days
    ACCESS_HALF_LIFE = 7.0    # 7 days (access recency matters more)
    
    @staticmethod
    def recency_score(
        created_at: datetime,
        last_accessed: Optional[datetime] = None,
        half_life: float = 30.0
    ) -> float:
        """Calculate recency score using exponential decay.
        
        Args:
            created_at: When the memory was created
            last_accessed: When the memory was last accessed (optional)
            half_life: Days until score halves (default: 30)
            
        Returns:
            Score between 0.0 and 1.0
        """
        now = datetime.utcnow()
        
        if last_accessed:
            # Blend creation time and access time
            days_since_created = max(0.0, (now - created_at).total_seconds() / 86400.0)
            days_since_access = max(0.0, (now - last_accessed).total_seconds() / 86400.0)
            
            # Exponential decay
            creation_decay = math.exp(-0.693 * days_since_created / half_life)
            access_decay = math.exp(-0.693 * days_since_access / (half_life * 0.5))
            
            # Blend: 40% creation, 60% access (recent access matters more)
            return 0.4 * creation_decay + 0.6 * access_decay
        else:
            days_since_created = max(0.0, (now - created_at).total_seconds() / 86400.0)
            return math.exp(-0.693 * days_since_created / half_life)
    
    @staticmethod
    def importance_weight(importance: float, confidence: float) -> float:
        """Calculate importance weight.
        
        Formula: importance * confidence
        Range: 0.0-1.0
        """
        return max(0.0, min(1.0, importance * confidence))
    
    @staticmethod
    def access_frequency_weight(access_count: int) -> float:
        """Calculate access frequency weight using logarithmic scaling.
        
        Formula: min(1.0, log(access_count + 1) / 10)
        Range: 0.0-1.0
        
        0 accesses = 0.0
        1 access = 0.069
        10 accesses = 0.24
        100 accesses = 0.46
        1000 accesses = 0.69
        """
        if access_count <= 0:
            return 0.0
        return min(1.0, math.log(access_count + 1) / 10.0)
    
    @staticmethod
    def time_of_day_similarity(dt1: datetime, dt2: datetime) -> float:
        """Calculate similarity between two times of day.
        
        Returns 1.0 if same hour, 0.0 if 12 hours apart.
        """
        hour_diff = abs(dt1.hour - dt2.hour)
        if hour_diff > 12:
            hour_diff = 24 - hour_diff
        return 1.0 - (hour_diff / 12.0)
    
    @staticmethod
    def day_of_week_similarity(dt1: datetime, dt2: datetime) -> float:
        """Calculate similarity between two days of the week.
        
        Returns 1.0 if same day, 0.0 if 3+ days apart.
        """
        day_diff = abs(dt1.weekday() - dt2.weekday())
        if day_diff > 3:
            day_diff = 7 - day_diff
        return max(0.0, 1.0 - (day_diff / 3.0))
    
    @staticmethod
    def composite_temporal_score(
        created_at: datetime,
        last_accessed: Optional[datetime],
        importance: float,
        confidence: float,
        access_count: int,
        reference_time: Optional[datetime] = None
    ) -> float:
        """Calculate a composite temporal score.
        
        Blends recency, importance, and access frequency into a single score.
        """
        recency = TemporalScoring.recency_score(created_at, last_accessed)
        importance_w = TemporalScoring.importance_weight(importance, confidence)
        access_w = TemporalScoring.access_frequency_weight(access_count)
        
        # Weighted blend
        score = 0.4 * recency + 0.35 * importance_w + 0.25 * access_w
        
        # Optional: boost if reference_time matches time-of-day pattern
        if reference_time:
            tod_similarity = TemporalScoring.time_of_day_similarity(created_at, reference_time)
            score = 0.85 * score + 0.15 * tod_similarity
        
        return max(0.0, min(1.0, score))
```

**Key Design Decisions:**
- **Exponential decay:** More natural than linear decay. Score halves after `half_life` days, then quarters, etc.
- **Dual decay:** Creation time and access time both contribute. Recent access matters more (60% weight).
- **Logarithmic access frequency:** Log scaling prevents high-access-count memories from dominating. 1000 accesses only gives 0.69 weight.
- **Time-of-day similarity:** Optional boost for contextual retrieval. "What was I doing at this time last week?" queries.
- **Composite score:** Single score blending all temporal signals. Used by search service for ranking.
