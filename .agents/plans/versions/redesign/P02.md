### Task 2.1: Create Dock Component

**Create: `frontend/src/shared/layout/Dock.tsx`**

A glass-morphism bottom dock, centered, with 10 mode icons (emoji + label). Auto-hides after 3s idle in mode view. Shows on hover at bottom edge. Active mode gets red glow ring. Hover gets cyan glow + tooltip.

```tsx
"use client";

import { type ReactNode, useEffect, useRef, useState } from "react";
import { cn } from "@/shared/lib/utils";

interface DockItem {
  id: string;
  icon: string;
  label: string;
  shortcut: string; // "1" through "0"
}

const DOCK_ITEMS: DockItem[] = [
  { id: "chat", icon: "💬", label: "Chat", shortcut: "1" },
  { id: "search", icon: "🔍", label: "Search", shortcut: "2" },
  { id: "brain", icon: "🧠", label: "Brain", shortcut: "3" },
  { id: "vault", icon: "🔐", label: "Vault", shortcut: "4" },
  { id: "models", icon: "📚", label: "Models", shortcut: "5" },
  { id: "code", icon: "📐", label: "Code", shortcut: "6" },
  { id: "utility", icon: "🛠️", label: "Utility", shortcut: "7" },
  { id: "settings", icon: "⚙️", label: "Settings", shortcut: "8" },
  { id: "systems", icon: "🖥️", label: "Systems", shortcut: "9" },
  { id: "profile", icon: "👤", label: "Profile", shortcut: "0" },
];

interface DockProps {
  activeMode: string;
  onModeChange: (modeId: string) => void;
  visible: boolean;
}

export function Dock({ activeMode, onModeChange, visible }: DockProps) {
  const [hoveredItem, setHoveredItem] = useState<string | null>(null);
  const timeoutRef = useRef<ReturnType<typeof setTimeout>>();

  // Keyboard shortcuts: ⌘1-⌘0
  useEffect(() => {
    function handleKeyDown(e: KeyboardEvent) {
      if (!e.metaKey) return;
      const idx = parseInt(e.key, 10);
      if (idx >= 1 && idx <= 9) {
        onModeChange(DOCK_ITEMS[idx - 1].id);
      } else if (e.key === "0") {
        onModeChange("profile");
      }
    }
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [onModeChange]);

  return (
    <div
      className={cn(
        "fixed bottom-0 left-1/2 z-dock -translate-x-1/2",
        "transition-transform duration-300 ease-out",
        visible ? "translate-y-0" : "translate-y-[calc(100%-8px)]",
      )}
    >
      <div className="flex items-center gap-1 rounded-2xl border border-border-default bg-bg-glass px-3 py-2 backdrop-blur-2xl">
        {DOCK_ITEMS.map((item) => (
          <button
            key={item.id}
            onClick={() => onModeChange(item.id)}
            onMouseEnter={() => setHoveredItem(item.id)}
            onMouseLeave={() => setHoveredItem(null)}
            className={cn(
              "relative flex h-11 w-11 items-center justify-center rounded-xl text-lg",
              "transition-all duration-150 ease-out",
              "hover:bg-accent-red-muted hover:shadow-red",
              activeMode === item.id
                ? "bg-accent-red-muted text-accent-red shadow-red"
                : "text-text-secondary hover:text-text-primary",
            )}
            title={`${item.label} (⌘${item.shortcut})`}
          >
            <span className="text-xl">{item.icon}</span>
            {/* Tooltip */}
            <span
              className={cn(
                "absolute -top-8 left-1/2 -translate-x-1/2 whitespace-nowrap rounded-md bg-bg-elevated px-2 py-1 text-xs text-text-primary",
                "transition-opacity duration-150",
                hoveredItem === item.id ? "opacity-100" : "opacity-0 pointer-events-none",
              )}
            >
              {item.label}
            </span>
          </button>
        ))}
        {/* User avatar */}
        <div className="ml-2 border-l border-border-default pl-2">
          <button className="flex h-10 w-10 items-center justify-center rounded-full bg-accent-red/20 text-sm font-medium text-accent-red">
            {/* Will use real user data later */}
            A
          </button>
        </div>
      </div>
    </div>
  );
}
```

### Task 2.2: Create Neural Ribbon Component

**Create: `frontend/src/shared/layout/NeuralRibbon.tsx`**

Thin 24px status strip at top. Shows live system data: online status, active model, tps, vram, health dots.

```tsx
"use client";

import { useState, useEffect } from "react";
import { cn } from "@/shared/lib/utils";

interface StatusData {
  status: "online" | "degraded" | "offline";
  model: string;
  tps: number;
  vram: string;
}

export function NeuralRibbon() {
  const [status, setStatus] = useState<StatusData>({
    status: "online",
    model: "—",
    tps: 0,
    vram: "—",
  });

  // Fetch from /api/v1/models/health or /api/v1/system/health
  useEffect(() => {
    fetch("/api/v1/system/health")
      .then((r) => r.json())
      .then((data) => {
        setStatus((prev) => ({
          ...prev,
          status: data.status === "healthy" ? "online" : "degraded",
        }));
      })
      .catch(() => setStatus((prev) => ({ ...prev, status: "offline" })));

    fetch("/api/v1/models/ollama/catalog")
      .then((r) => r.json())
      .then((data) => {
        const active = data.models?.find((m: any) => m.active);
        if (active) {
          setStatus((prev) => ({
            ...prev,
            model: active.name,
            tps: active.tps ?? 0,
            vram: active.vram_used ?? "—",
          }));
        }
      })
      .catch(() => {});
  }, []);

  const statusDot = {
    online: "🟢",
    degraded: "🟡",
    offline: "🔴",
  }[status.status];

  return (
    <div
      className="fixed top-0 left-0 right-0 z-sticky flex h-6 items-center justify-center gap-3 bg-bg-base/80 backdrop-blur-sm text-xs text-text-muted font-mono border-b border-border-subtle"
    >
      <span className={cn(
        "flex items-center gap-1",
        status.status === "online" && "text-success",
        status.status === "degraded" && "text-warning",
        status.status === "offline" && "text-danger",
      )}>
        {statusDot} {status.status.toUpperCase()}
      </span>
      <span className="text-border-default">·</span>
      <span>🧠 {status.model}</span>
      {status.tps > 0 && (
        <>
          <span className="text-border-default">·</span>
          <span>📡 {status.tps.toFixed(1)} tps</span>
        </>
      )}
      {status.vram !== "—" && (
        <>
          <span className="text-border-default">·</span>
          <span>💾 {status.vram} VRAM</span>
        </>
      )}
    </div>
  );
}
```

### Task 2.3: Create Mode Shell Component

"use client";
