### Task 4.1: Create ModeStack Context

**Create: `frontend/src/shared/layout/ModeStack.tsx`**

Navigation stack with max depth 5. Each entry stores: modeId, scrollPosition, inputState (draft text). Pushes on mode switch, pops on Back. Oldest entry drops when depth exceeds 5.

```tsx
"use client";

import { createContext, useContext, useCallback, useState, type ReactNode } from "react";

interface ModeState {
  modeId: string;
  scrollPosition: number;
  inputDraft?: string;
}

interface ModeStackContextValue {
  stack: ModeState[];
  currentMode: string;
  pushMode: (modeId: string) => void;
  popMode: () => string | null;
  updateState: (modeId: string, updates: Partial<ModeState>) => void;
  goToHub: () => void;
}

const MAX_DEPTH = 5;

const ModeStackContext = createContext<ModeStackContextValue | null>(null);

export function useModeStack() {
  const ctx = useContext(ModeStackContext);
  if (!ctx) throw new Error("useModeStack must be inside ModeStackProvider");
  return ctx;
}

export function ModeStackProvider({ children }: { children: ReactNode }) {
  const [stack, setStack] = useState<ModeState[]>([{ modeId: "hub", scrollPosition: 0 }]);

  const currentMode = stack[stack.length - 1]?.modeId ?? "hub";

  const pushMode = useCallback((modeId: string) => {
    setStack((prev) => {
      const next = [...prev, { modeId, scrollPosition: 0 }];
      return next.length > MAX_DEPTH ? next.slice(next.length - MAX_DEPTH) : next;
    });
  }, []);

  const popMode = useCallback(() => {
    let popped: string | null = null;
    setStack((prev) => {
      if (prev.length <= 1) return prev;
      popped = prev[prev.length - 1].modeId;
      return prev.slice(0, -1);
    });
    return popped;
  }, []);

  const updateState = useCallback((modeId: string, updates: Partial<ModeState>) => {
    setStack((prev) =>
      prev.map((s) => (s.modeId === modeId ? { ...s, ...updates } : s)),
    );
  }, []);

  const goToHub = useCallback(() => {
    setStack([{ modeId: "hub", scrollPosition: 0 }]);
  }, []);

  return (
    <ModeStackContext.Provider value={{ stack, currentMode, pushMode, popMode, updateState, goToHub }}>
      {children}
    </ModeStackContext.Provider>
  );
}
```

### Task 4.2: Create ModeView Router

**Create: `frontend/src/shared/layout/ModeView.tsx`**

Switches between Hub view and Mode views. Handles crossfade animation (150ms). Reads `currentMode` from ModeStackContext. Renders hub or the appropriate mode component via lookup.

```tsx
"use client";

import { useModeStack } from "./ModeStack";
import { Dock } from "./Dock";
import { NeuralRibbon } from "./NeuralRibbon";
import { HubPage } from "@/features/hub/page";
import { ChatPage } from "@/features/chat/page";
import { SearchPage } from "@/features/search/page";
import { BrainPage } from "@/features/brain/page";
import { VaultPage } from "@/features/vault/page";
import { ModelsPage } from "@/features/models/page";
import { CodePage } from "@/features/code/page";
import { UtilityPage } from "@/features/utility/page";
import { SettingsPage } from "@/features/settings/page";
import { SystemsPage } from "@/features/systems/page";
import { ProfilePage } from "@/features/profile/page";
import { CommandBar } from "./CommandBar";
import { NeuralParticles } from "./NeuralParticles";
import { useState } from "react";

const MODE_COMPONENTS: Record<string, React.ComponentType<any>> = {
  chat: ChatPage,
  search: SearchPage,
  brain: BrainPage,
  vault: VaultPage,
  models: ModelsPage,
  code: CodePage,
  utility: UtilityPage,
  settings: SettingsPage,
  systems: SystemsPage,
  profile: ProfilePage,
};

export function ModeView() {
  const { currentMode } = useModeStack();
  const [dockVisible, setDockVisible] = useState(true);
  const [commandBarOpen, setCommandBarOpen] = useState(false);

  // Auto-hide dock after 3s in mode
  useEffect(() => {
    if (currentMode === "hub") { setDockVisible(true); return; }
    const t = setTimeout(() => setDockVisible(false), 3000);
    return () => clearTimeout(t);
  }, [currentMode]);

  const ModeComponent = currentMode !== "hub" ? MODE_COMPONENTS[currentMode] : null;

  return (
    <div className="relative h-dvh overflow-hidden bg-bg-base">
      <NeuralRibbon />
      <NeuralParticles />

      {currentMode === "hub" ? (
        <div className="pt-6">
          <HubPage />
        </div>
      ) : ModeComponent ? (
        <ModeComponent />
      ) : (
        <div className="pt-6 text-center text-text-muted">
          Mode {currentMode} not implemented
        </div>
      )}

      <Dock activeMode={currentMode} visible={dockVisible} />
      {commandBarOpen && <CommandBar onClose={() => setCommandBarOpen(false)} />}
    </div>
  );
}
```
