### Task 3.1: Create Hub Widget Component

**Create: `frontend/src/shared/layout/HubWidget.tsx`**

Glass-morphism card, single responsibility: render a mode preview widget. Props define icon, label, preview content, click handler, glow color.

```tsx
"use client";

import { type ReactNode } from "react";
import { cn } from "@/shared/lib/utils";

interface HubWidgetProps {
  icon: string;
  label: string;
  children: ReactNode;
  onClick: () => void;
  isActive?: boolean;
  glowColor?: "red" | "cyan";
  className?: string;
}

export function HubWidget({
  icon,
  label,
  children,
  onClick,
  isActive = false,
  glowColor = "cyan",
  className,
}: HubWidgetProps) {
  return (
    <button
      onClick={onClick}
      className={cn(
        "group relative flex flex-col rounded-2xl border border-border-subtle p-4 text-left",
        "bg-widget backdrop-blur-2xl",
        "transition-all duration-200 ease-out",
        "hover:border-border-default hover:-translate-y-0.5",
        glowColor === "red"
          ? "hover:shadow-red"
          : "hover:shadow-cyan",
        isActive && (glowColor === "red" ? "shadow-red" : "shadow-cyan"),
        className,
      )}
    >
      {/* Active glow indicator */}
      {isActive && (
        <span
          className={cn(
            "absolute -top-px left-4 right-4 h-px rounded-full",
            glowColor === "red" ? "bg-accent-red" : "bg-accent-cyan",
          )}
        />
      )}
      {/* Header */}
      <div className="flex items-center gap-2 mb-2">
        <span className="text-xl">{icon}</span>
        <span className="text-xs font-semibold text-text-secondary uppercase tracking-wider">
          {label}
        </span>
      </div>
      {/* Preview content */}
      <div className="flex-1 text-xs text-text-muted space-y-1">
        {children}
      </div>
    </button>
  );
}
```

### Task 3.3: Create Hub Greeting Component

**Create: `frontend/src/shared/layout/HubGreeting.tsx`**

Display heading at top of hub view. Reads user from AuthContext. Shows time-aware greeting.

```tsx
"use client";

import { useAuth } from "@/shared/auth/AuthProvider";

export function HubGreeting() {
  const { user } = useAuth();
  const hour = new Date().getHours();
  const greeting = hour < 12 ? "Good morning" : hour < 18 ? "Good afternoon" : "Good evening";

  return (
    <div className="text-center mb-6">
      <h1 className="text-display font-semibold text-text-primary">
        {greeting}, {user?.username ?? "Cortex"}.
      </h1>
      <p className="text-sm text-text-secondary mt-1">
        Your machine intelligence layer — always aware, always ready.
      </p>
    </div>
  );
}
```

### Task 3.5: Create Neural Particle Background

**Create: `frontend/src/shared/layout/NeuralParticles.tsx`**

Canvas-based particle animation. Extremely dim (max opacity 0.3). Cyan nodes + connecting lines. Nodes pulse on data activity. Fully removed on `prefers-reduced-motion: reduce`.

- Canvas fills viewport, `pointer-events: none`, `z-base`
- ~30 particles drifting slowly
- Connections drawn when particles within 150px
- Muted: 0.1-0.3 opacity on lines, 0.01 on far connections
- Animation loop: requestAnimationFrame, 60fps target, pause when tab hidden
- Performance: round-robin updates to avoid frame drops
