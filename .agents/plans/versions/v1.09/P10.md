```python
"""PersonalizationService — builds per-user profiles and personalizes content."""

from __future__ import annotations

import logging
from datetime import datetime
from typing import Any

from sqlalchemy.orm import Session

from backend.app.services.learning.preference_service import PreferenceLearningService
from backend.app.services.learning.habit_service import HabitDetectionService

logger = logging.getLogger(__name__)


class PersonalizationService:
    """Service for building user profiles and personalizing interactions."""

    def __init__(self, db: Session) -> None:
        self.db = db
        self._preference_service = PreferenceLearningService(db)
        self._habit_service = HabitDetectionService(db)

    def get_user_profile(self, user_id: int) -> dict[str, Any]:
        """Build a complete user profile from all learning data."""
        preferences = self._preference_service.get_preferences(user_id, min_confidence=0.3)
        habits = self._habit_service.get_habits(user_id)

        profile: dict[str, Any] = {
            "response_style": "normal",
            "suggestion_tolerance": "normal",
            "active_hours": None,
            "topics": [],
            "expertise_level": "general",
            "preferred_tools": [],
        }

        for pref in preferences:
            category = pref.category
            key = pref.key
            value = pref.value or {}

            if category == "response_style" and key == "verbosity":
                profile["response_style"] = value.get("level", "normal")
            elif category == "interaction" and key == "suggestion_density":
                profile["suggestion_tolerance"] = value.get("level", "normal")
            elif category == "expertise":
                profile["expertise_level"] = value.get("level", "general")
            elif category == "tools":
                profile["preferred_tools"] = value.get("tools", [])

        # Infer active hours from habits
        if habits:
            hours = set()
            for habit in habits:
                if habit.last_observed:
                    hours.add(habit.last_observed.hour)
            if hours:
                profile["active_hours"] = sorted(hours)

        return profile

    def personalize_content(self, content: str, user_id: int) -> str:
        """Adjust content length based on user's response_style preference."""
        pref = self._preference_service.get_preference(user_id, "response_style", "verbosity")
        if pref and pref.confidence >= 0.6:
            level = (pref.value or {}).get("level", "normal")
            if level == "concise" and len(content) > 500:
                return content[:500] + "\n\n[content trimmed]"
        return content

    def update_profile(self, user_id: int, profile_data: dict[str, Any]) -> dict[str, Any]:
        """Store profile overrides as explicit preferences."""
        for key, value in profile_data.items():
            self._preference_service.record_preference(
                user_id=user_id,
                category="profile",
                key=key,
                value={"value": value},
                source="explicit",
            )
        return self.get_user_profile(user_id)
```

```python
"""KnowledgeRefinementService — adjusts memory confidence based on outcomes."""

from __future__ import annotations

import logging
from typing import Any

from sqlalchemy.orm import Session

from backend.app.services.learning.feedback_service import FeedbackLearningService

logger = logging.getLogger(__name__)


class KnowledgeRefinementService:
    """Service for refining memory correctness based on user feedback."""

    def __init__(self, db: Session) -> None:
        self.db = db
        self._feedback_service = FeedbackLearningService(db)

    def refine_memory(self, user_id: int, memory_id: int, was_correct: bool) -> dict[str, Any]:
        """Adjust memory confidence based on outcome.
        
        Boosts confidence +0.1 for correct, reduces -0.2 for incorrect.
        """
        # This queries the MemoryNode model from v1.03
        from backend.app.models.memory.memory_node import MemoryNode

        memory = (
            self.db.query(MemoryNode)
            .filter(MemoryNode.id == memory_id, MemoryNode.user_id == user_id)
            .first()
        )

        if not memory:
            return {"status": "not_found"}

        old_confidence = memory.confidence or 0.5
        if was_correct:
            memory.confidence = min(1.0, old_confidence + 0.1)
        else:
            memory.confidence = max(0.0, old_confidence - 0.2)

        self.db.commit()

        return {
            "memory_id": memory_id,
            "old_confidence": old_confidence,
            "new_confidence": memory.confidence,
            "adjusted": was_correct,
        }

    def get_refinement_stats(self, user_id: int) -> dict[str, Any]:
        """Get refinement statistics for a user."""
        events = self._feedback_service.get_feedback_history(user_id, limit=100)
        affirmations = sum(1 for e in events if e.event_type == "affirmation")
        corrections = sum(1 for e in events if e.event_type == "correction")
        total = len(events)

        return {
            "total_events": total,
            "affirmations": affirmations,
            "corrections": corrections,
            "net_improvement": (affirmations - corrections) if total > 0 else 0,
        }

    def suggest_refinements(self, user_id: int, limit: int = 10) -> list[dict[str, Any]]:
        """Find low-confidence memories that might need refinement."""
        from backend.app.models.memory.memory_node import MemoryNode

        low_confidence = (
            self.db.query(MemoryNode)
            .filter(
                MemoryNode.user_id == user_id,
                MemoryNode.confidence < 0.3,
            )
            .limit(limit)
            .all()
        )

        return [
            {
                "id": m.id,
                "content": str(getattr(m, "content", ""))[:200],
                "confidence": m.confidence,
            }
            for m in low_confidence
        ]
```

```python
"""PatternRecognitionService — detects temporal, behavioral, and preference patterns."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.pattern import Pattern, MIN_EVIDENCE

logger = logging.getLogger(__name__)


class PatternRecognitionService:
    """Service for detecting and managing learned patterns."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def detect_pattern(
        self,
        user_id: int,
        pattern_type: str,
        description: str,
        evidence_item: dict[str, Any] | None = None,
    ) -> Pattern:
        """Detect a pattern. Creates new or strengthens existing."""
        existing = (
            self.db.query(Pattern)
            .filter(
                Pattern.user_id == user_id,
                Pattern.pattern_type == pattern_type,
                Pattern.description == description,
            )
            .first()
        )

        if existing:
            if evidence_item:
                existing.add_evidence(evidence_item)
            else:
                existing.occurrences = (existing.occurrences or 0) + 1
                existing.confidence = min(1.0, (existing.confidence or 0.3) + 0.1)
                existing.last_seen = datetime.now(timezone.utc)
            self.db.commit()
            self.db.refresh(existing)
            return existing

        pattern = Pattern(
            user_id=user_id,
            pattern_type=pattern_type,
            description=description,
            evidence=[evidence_item] if evidence_item else [],
            first_seen=datetime.now(timezone.utc),
            last_seen=datetime.now(timezone.utc),
        )
        self.db.add(pattern)
        self.db.commit()
        self.db.refresh(pattern)
        return pattern

    def get_patterns(
        self, user_id: int, min_confidence: float = 0.0
    ) -> list[Pattern]:
        """Get patterns for a user, optionally filtered by confidence."""
        query = self.db.query(Pattern).filter(Pattern.user_id == user_id)
        if min_confidence > 0:
            query = query.filter(Pattern.confidence >= min_confidence)
        return query.order_by(Pattern.confidence.desc()).all()

    def get_patterns_by_type(self, user_id: int, pattern_type: str) -> list[Pattern]:
        """Get patterns filtered by type."""
        return (
            self.db.query(Pattern)
            .filter(Pattern.user_id == user_id, Pattern.pattern_type == pattern_type)
            .order_by(Pattern.confidence.desc())
            .all()
        )

    def get_actionable_patterns(self, user_id: int) -> list[Pattern]:
        """Get patterns that are confident enough to act on."""
        return (
            self.db.query(Pattern)
            .filter(
                Pattern.user_id == user_id,
                Pattern.confidence >= 0.5,
                Pattern.occurrences >= MIN_EVIDENCE,
            )
            .all()
        )
```

```python
"""AnomalyDetectionService — flags unusual behavior against baselines.

Design constraint: NEVER auto-acts on anomalies. Only flags for review.
"""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

logger = logging.getLogger(__name__)

DEFAULT_BASELINES = {
    "active_hours_start": 9,
    "active_hours_end": 17,
    "expected_frequency": 10,
    "expected_volume": 100,
}


class AnomalyDetectionService:
    """Service for detecting anomalous user behavior."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def detect_anomaly(
        self,
        user_id: int,
        action: str,
        current_frequency: int | None = None,
        current_volume: int | None = None,
        current_hour: int | None = None,
    ) -> dict[str, Any]:
        """Check if an action is anomalous. Returns anomaly report.
        
        Never auto-acts — only flags.
        """
        baselines = self.get_baselines(user_id)
        reasons: list[str] = []
        is_anomaly = False

        if current_frequency is not None:
            expected = (baselines.get("expected_frequency") or DEFAULT_BASELINES["expected_frequency"])
            if current_frequency > expected * 3:
                reasons.append(
                    f"Frequency {current_frequency} exceeds 3x baseline {expected}"
                )
                is_anomaly = True

        if current_hour is not None:
            start = baselines.get("active_hours_start") or DEFAULT_BASELINES["active_hours_start"]
            end = baselines.get("active_hours_end") or DEFAULT_BASELINES["active_hours_end"]
            if current_hour < start or current_hour > end:
                reasons.append(
                    f"Activity at hour {current_hour} outside active hours {start}-{end}"
                )
                is_anomaly = True

        if current_volume is not None:
            expected = (baselines.get("expected_volume") or DEFAULT_BASELINES["expected_volume"])
            if current_volume > expected * 5:
                reasons.append(
                    f"Volume {current_volume} exceeds 5x baseline {expected}"
                )
                is_anomaly = True

        return {
            "is_anomaly": is_anomaly,
            "reasons": reasons,
            "action": action,
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "severity": len(reasons),
        }

    def get_baselines(self, user_id: int) -> dict[str, Any]:
        """Get current baselines for a user.
        
        v1.09 uses static defaults. Future versions adapt baselines.
        """
        return dict(DEFAULT_BASELINES)

    def update_baseline(self, user_id: int, baseline_updates: dict[str, Any]) -> dict[str, Any]:
        """Update baselines (placeholder for future adaptive baselines)."""
        logger.info("Baseline update requested for user %d (static in v1.09): %s", user_id, baseline_updates)
        return self.get_baselines(user_id)
```

```python
"""ContinuousImprovementService — tracks improvement metrics for learning quality."""

from __future__ import annotations

import logging
from typing import Any

from sqlalchemy.orm import Session

logger = logging.getLogger(__name__)


class ContinuousImprovementService:
    """Service for tracking learning improvement metrics.
    
    In-memory tracking for v1.09. Future versions persist to DB.
    """

    def __init__(self, db: Session) -> None:
        self.db = db
        self._metrics: dict[str, list[dict[str, Any]]] = {}

    def record_improvement(
        self,
        user_id: int,
        metric_name: str,
        before_value: float,
        after_value: float,
    ) -> dict[str, Any]:
        """Record an improvement measurement."""
        improvement = after_value - before_value
        improvement_pct = (
            (improvement / before_value * 100) if before_value != 0 else 0.0
        )

        record = {
            "user_id": user_id,
            "metric_name": metric_name,
            "before": before_value,
            "after": after_value,
            "improvement": improvement,
            "improvement_pct": improvement_pct,
        }

        self._metrics.setdefault(metric_name, []).append(record)
        return record

    def get_improvement_summary(self, user_id: int) -> dict[str, Any]:
        """Get improvement summary across all metrics."""
        improved = 0
        degraded = 0

        for metric_name, records in self._metrics.items():
            user_records = [r for r in records if r["user_id"] == user_id]
            if user_records:
                total = sum(r["improvement"] for r in user_records)
                if total > 0:
                    improved += 1
                elif total < 0:
                    degraded += 1

        total = improved + degraded
        return {
            "metrics_tracked": len(self._metrics),
            "improved": improved,
            "degraded": degraded,
            "overall_trend": "improving" if improved > degraded else (
                "degrading" if degraded > improved else "stable"
            ),
            "improvement_score": improved / total if total > 0 else 0.5,
        }

    def get_improvement_history(
        self, metric_name: str | None = None
    ) -> list[dict[str, Any]]:
        """Get improvement records, optionally filtered by metric."""
        if metric_name:
            return self._metrics.get(metric_name, [])
        return [r for records in self._metrics.values() for r in records]

    def calculate_improvement_score(self, user_id: int) -> float:
        """Calculate overall improvement score (0.0 to 1.0)."""
        summary = self.get_improvement_summary(user_id)
        return summary["improvement_score"]
```

```python
"""Integration tests for full learning cycle."""
import pytest


def test_full_learning_cycle(db_session, auth_user):
    """Verify complete learning flow: observe → learn → adapt → improve."""
    from backend.app.services.learning.preference_service import PreferenceLearningService
    from backend.app.services.learning.feedback_service import FeedbackLearningService
    from backend.app.services.learning.adaptation_service import BehaviorAdaptationService
    from backend.app.services.learning.improvement_service import ContinuousImprovementService

    pref_service = PreferenceLearningService(db_session)
    feedback_service = FeedbackLearningService(db_session)
    adaptation_service = BehaviorAdaptationService(db_session)
    improvement_service = ContinuousImprovementService(db_session)

    # 1. Record preference
    pref_service.record_preference(auth_user.id, "response_style", "verbosity", {"level": "concise"})

    # 2. Strengthen it (repeat observation)
    pref_service.record_preference(auth_user.id, "response_style", "verbosity", {"level": "concise"})
    pref_service.record_preference(auth_user.id, "response_style", "verbosity", {"level": "concise"})
    pref_service.record_preference(auth_user.id, "response_style", "verbosity", {"level": "concise"})

    pref = pref_service.get_preference(auth_user.id, "response_style", "verbosity")
    assert pref.confidence >= 0.6  # adaptation ready

    # 3. Adaptation should now trim responses
    adapted = adaptation_service.adapt_response("A" * 1000, auth_user.id)
    assert len(adapted) < 1000

    # 4. Feedback loop
    feedback_service.record_feedback(auth_user.id, "affirmation", context={"category": "response_style", "key": "verbosity"})
    learning_rate = feedback_service.get_learning_rate(auth_user.id)
    assert learning_rate > 0.5

    # 5. Improvement tracking
    improvement_service.record_improvement(auth_user.id, "response_accuracy", 0.5, 0.8)
    summary = improvement_service.get_improvement_summary(auth_user.id)
    assert summary["improved"] >= 1
    assert summary["overall_trend"] == "improving"


def test_anomaly_detection_never_auto_acts(db_session, auth_user):
    """Verify anomaly detection only flags, never auto-acts."""
    from backend.app.services.learning.anomaly_service import AnomalyDetectionService

    service = AnomalyDetectionService(db_session)
    result = service.detect_anomaly(
        auth_user.id, "bulk_delete", current_volume=1000, current_hour=3
    )
    assert result["is_anomaly"] is True
    assert len(result["reasons"]) > 0
    # Verify no auto-action field exists
    assert "auto_action" not in result
```
