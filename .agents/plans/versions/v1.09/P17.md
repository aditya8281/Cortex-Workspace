# P17: Semantic Hybrid Scoring in MemorySearchService

**Document:** Phase 17 of Version 1.09
**Authority:** Stage 6 — Master Versioned Implementation Roadmap
**Date:** 2026-06-30

---

## Objective

Replace the current keyword ILIKE/BM25 fallback in MemorySearchService with embedding-based semantic relevance scoring. Add caching, weighted composite scoring, and gradual rollout via feature flag. Improves search accuracy for semantically similar but lexically different matches (e.g. "code that handles errors" matches "except ValueError").

---

## Architecture Reference

Source: `backend/app/services/memory/memory_search.py` — `MemorySearchService.search()` currently uses keyword ILIKE for fallback/ensuring score.
Target: Add `SemanticMemoryScorer` that wraps embedding similarity + keyword score + temporal score into one composite.

```python
"""Semantic memory scorer — composite relevance with embedding + keyword + temporal signals."""
from __future__ import annotations

import math
import logging
from datetime import datetime, timezone
from functools import lru_cache

from backend.app.services.intelligence.embedding_service import EmbeddingService, get_embedding_service

logger = logging.getLogger(__name__)

# Composite weight configuration
SEMANTIC_WEIGHT: float = 0.55
KEYWORD_WEIGHT: float = 0.25
TEMPORAL_WEIGHT: float = 0.10
TYPE_BIAS_WEIGHT: float = 0.10

# Per-type bias (boost certain types in search results)
TYPE_BIAS: dict[str, float] = {
    "preference": 0.15,
    "fact": 0.10,
    "pattern": 0.10,
    "rule": 0.08,
    "concept": 0.05,
    "event": 0.0,
    "observation": 0.0,
    "noise": -0.10,
}

# Decay: entries older than this get 0 temporal bonus
MAX_TEMPORAL_DAYS: int = 365


class SemanticMemoryScorer:
    """Composite scorer for memory search results.

    Combines three signals:
    - Semantic: embedding cosine similarity (primary)
    - Keyword: TF-IDF/Dice keyword overlap (fallback / supplementary)
    - Temporal: recency bonus (logarithmic decay)
    - Type bias: per-memory-type boost
    """

    def __init__(self, embedding_service: EmbeddingService | None = None) -> None:
        self._embedder = embedding_service or get_embedding_service()
        self._embed_cache: dict[str, list[float]] = {}
        self._cache_max = 500

    def score(
        self,
        content: str,
        query_embedding: list[float],
        metadata: dict | None = None,
    ) -> float:
        """Compute composite relevance score for a single memory entry.

        Args:
            content: The text content of the memory entry
            query_embedding: Embedding vector of the search query
            metadata: Dict with optional keys:
                - created_at: ISO datetime string or None
                - type: memory type string for bias
                - keywords: pre-extracted keyword list

        Returns:
            Float 0.0-1.0 composite relevance score
        """
        metadata = metadata or {}

        # 1. Semantic score
        content_emb = self._get_content_embedding(content)
        semantic: float = (
            self._cosine_similarity(query_embedding, content_emb)
            if content_emb
            else 0.0
        )

        # 2. Keyword score (Dice coefficient on token overlap)
        keyword_score = self._keyword_score(
            content, metadata.get("keywords") or []
        )

        # 3. Temporal score (recency decay)
        created_at = metadata.get("created_at")
        temporal = self._temporal_score(created_at)

        # 4. Type bias
        mem_type = metadata.get("type", "")
        type_bias = TYPE_BIAS.get(mem_type, 0.0)

        # 5. Composite
        composite = (
            SEMANTIC_WEIGHT * semantic
            + KEYWORD_WEIGHT * keyword_score
            + TEMPORAL_WEIGHT * temporal
            + TYPE_BIAS_WEIGHT * max(0.0, semantic + type_bias)
        )

        # Normalize to 0.0-1.0 (type bias can push above 1.0)
        return min(1.0, max(0.0, composite))

    def score_batch(
        self,
        entries: list,
        query_embedding: list[float],
        content_attr: str = "content",
        metadata_attr: str | None = None,
    ) -> list[tuple]:
        """Score multiple memory entries. Returns sorted (entry, score) tuples.

        Args:
            entries: List of memory entry objects
            query_embedding: Query embedding vector
            content_attr: Attribute name for content text
            metadata_attr: Optional attribute name for metadata dict

        Returns:
            List of (entry, score) sorted descending by score
        """
        scored = []
        for entry in entries:
            content = getattr(entry, content_attr, str(entry))
            metadata = getattr(entry, metadata_attr, {}) if metadata_attr else {}
            if isinstance(entry, dict):
                content = entry.get(content_attr, str(entry))
                metadata = entry.get(metadata_attr or "", {})
            score = self.score(content, query_embedding, metadata)
            scored.append((entry, score))

        scored.sort(key=lambda x: x[1], reverse=True)
        return scored

    def _get_content_embedding(self, content: str) -> list[float] | None:
        """Get cached embedding for content. Returns None on failure."""
        cache_key = str(hash(content[:256]) % (2**32))
        if cache_key in self._embed_cache:
            return self._embed_cache[cache_key]

        try:
            emb = self._embedder.embed(content[:512])
            if self._is_mock_embedding(emb):
                return None
            self._embed_cache[cache_key] = emb
            if len(self._embed_cache) > self._cache_max:
                self._embed_cache.pop(next(iter(self._embed_cache)))
            return emb
        except Exception:
            logger.debug("Embedding failed for memory search content", exc_info=True)
            return None

    @staticmethod
    def _is_mock_embedding(vec: list[float]) -> bool:
        if not vec:
            return True
        mean = sum(vec) / len(vec)
        var = sum((v - mean) ** 2 for v in vec) / len(vec)
        return var < 0.0001

    @staticmethod
    def _cosine_similarity(a: list[float], b: list[float]) -> float:
        if not a or not b or len(a) != len(b):
            return 0.0
        dot = sum(ai * bi for ai, bi in zip(a, b))
        na = math.sqrt(sum(ai * ai for ai in a))
        nb = math.sqrt(sum(bi * bi for bi in b))
        if na == 0.0 or nb == 0.0:
            return 0.0
        return dot / (na * nb)

    @staticmethod
    def _keyword_score(content: str, keywords: list[str]) -> float:
        """Dice coefficient for keyword overlap with content."""
        content_lower = content.lower()
        content_words = set(content_lower.split())
        if not content_words or not keywords:
            return 0.0
        query_keywords = {k.lower() for k in keywords}
        intersection = content_words & query_keywords
        if not intersection:
            return 0.0
        dice = 2.0 * len(intersection) / (len(content_words) + len(query_keywords))
        return min(1.0, dice)

    @staticmethod
    def _temporal_score(created_at: str | datetime | None) -> float:
        """Compute recency bonus: newer = higher score.

        Uses logarithmic decay: score = 1.0 when fresh,
        approaches 0.0 after MAX_TEMPORAL_DAYS.
        """
        if created_at is None:
            return 0.5  # neutral for unknown age

        if isinstance(created_at, str):
            try:
                dt = datetime.fromisoformat(created_at)
            except ValueError:
                return 0.5
        else:
            dt = created_at

        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)

        age_days = (datetime.now(timezone.utc) - dt).days
        if age_days <= 0:
            return 1.0
        if age_days >= MAX_TEMPORAL_DAYS:
            return 0.0
        # Logarithmic decay
        return 1.0 - (math.log(age_days + 1) / math.log(MAX_TEMPORAL_DAYS + 1))
```

```python
from backend.app.services.memory.semantic_scorer import SemanticMemoryScorer
from backend.app.core.config import settings

class MemorySearchService:
    def __init__(self, db: Session):
        self._db = db
        self._semantic_enabled = getattr(settings, "SEMANTIC_MEMORY_SEARCH", True)
        self._semantic_scorer: SemanticMemoryScorer | None = (
            SemanticMemoryScorer() if self._semantic_enabled else None
        )

    def search(
        self,
        user_id: int,
        query: str,
        memory_type: str | None = None,
        limit: int = 20,
        min_score: float = 0.0,
    ) -> list[dict]:
        """Search memory with semantic scoring."""
        # 1. Base query (existing keyword/ILIKE search)
        base_results = self._base_search(user_id, query, memory_type, limit=limit * 2)

        if self._semantic_scorer and base_results and len(base_results) > 0:
            # 2. Get query embedding
            query_emb = self._get_query_embedding(query)

            # 3. Semantic re-scoring
            entries = []
            for r in base_results:
                entries.append(r)

            scored = self._semantic_scorer.score_batch(
                base_results,
                query_emb,
                content_attr="content",
                metadata_attr=None,
            )

            # 4. Apply min_score threshold
            filtered = [(e, s) for e, s in scored if s >= min_score]
            filtered.sort(key=lambda x: x[1], reverse=True)

            # 5. Return as dicts
            results = []
            for entry, score in filtered[:limit]:
                d = self._entry_to_dict(entry)
                d["relevance_score"] = round(score, 4)
                d["score_type"] = "semantic"
                results.append(d)

            return results

        # Fallback: keyword scoring
        results = []
        for entry in base_results[:limit]:
            d = self._entry_to_dict(entry)
            d["relevance_score"] = self._keyword_score(entry.content, query)
            d["score_type"] = "keyword"
            results.append(d)
        return results

            return embedder.embed(query)
        except Exception:
            return [0.0] * 10  # zero vector = keyword fallback

        # ... existing implementation unchanged ...

    def _entry_to_dict(self, entry) -> dict:
        """Convert memory entry to dict."""
        # ... existing implementation unchanged ...

    @staticmethod
    def _keyword_score(content: str, query: str) -> float:
        """Simple keyword match ratio."""
        if not content or not query:
            return 0.0
        content_lower = content.lower()
        query_words = query.lower().split()
        if not query_words:
            return 0.0
        matches = sum(1 for w in query_words if w in content_lower)
        return matches / len(query_words)
```

```python
"""Memory embedding cache pre-warming background task."""
import logging
from datetime import datetime, timedelta, timezone

from sqlalchemy import select

from backend.app.models.memory.episodic import EpisodicMemory
from backend.app.models.memory.semantic import SemanticMemory
from backend.app.services.intelligence.embedding_service import get_embedding_service

logger = logging.getLogger(__name__)

CACHE_WARM_DAYS = 7
BATCH_SIZE = 50


def warm_memory_embedding_cache(db_session_factory):
    """Pre-warm embedding cache for recent memories.

    Runs as a periodic task. Processes memories created in the last
    CACHE_WARM_DAYS and warms the SemanticMemoryScorer's cache.
    """
    from backend.app.services.memory.semantic_scorer import SemanticMemoryScorer

    embedder = get_embedding_service()
    scorer = SemanticMemoryScorer(embedder)
    cutoff = datetime.now(timezone.utc) - timedelta(days=CACHE_WARM_DAYS)

    with db_session_factory() as db:
        # Warm episodic memories
        stmt = select(EpisodicMemory).where(
            EpisodicMemory.created_at >= cutoff
        ).limit(BATCH_SIZE)
        episodic = db.execute(stmt).scalars().all()
        for mem in episodic:
            scorer._get_content_embedding(mem.content)
        logger.info(
            "Warmed %d episodic memory embeddings (past %d days)",
            len(episodic), CACHE_WARM_DAYS,
        )

        # Warm semantic memories
        stmt = select(SemanticMemory).where(
            SemanticMemory.created_at >= cutoff
        ).limit(BATCH_SIZE)
        semantic = db.execute(stmt).scalars().all()
        for mem in semantic:
            scorer._get_content_embedding(mem.content)
        logger.info(
            "Warmed %d semantic memory embeddings (past %d days)",
            len(semantic), CACHE_WARM_DAYS,
        )

    scorer._get_content_embedding.cache_clear()  # type: ignore
```
