## Objective

Build an automatic background consolidation pipeline that extracts facts, preferences, patterns, and rules from raw conversation episodes. Runs after N messages per conversation, building an abstraction hierarchy: Event → Episode → Pattern → Rule. Stores results as LongTermMemory entries with graph connections to source episodes.

```python
"""Automatic episodic→semantic consolidation pipeline."""
from __future__ import annotations

import json
import logging
import re
import time
from datetime import datetime, timezone

from sqlalchemy.orm import Session
from sqlalchemy import desc

from backend.app.models.interaction.conversation import Conversation, ConversationMessage
from backend.app.models.memory.consolidation import ConsolidationConfig, ConsolidationLog
from backend.app.models.memory.long_term_memory import LongTermMemory
from backend.app.models.memory.memory_graph import MemoryNode, MemoryEdge
from backend.app.services.memory.long_term import LongTermMemoryService
from backend.app.services.memory.memory_graph_service import MemoryGraphService
from backend.app.services.intelligence.llm.manager import llm_manager
from backend.app.services.intelligence.llm.provider import LLMMessage

logger = logging.getLogger(__name__)

VALID_TYPES = ("fact", "pattern", "preference", "rule")

EXTRACTION_SYSTEM_PROMPT = """You are a memory consolidation assistant. Analyze the conversation and extract key abstractions.

For each abstraction, provide:
- type: one of "fact" (specific, objective), "pattern" (repeated behavior), "preference" (user likes/dislikes), "rule" (generalizable principle)
- title: short title (max 50 chars)
- content: the insight (max 200 chars)
- confidence: float 0.0-1.0 how certain you are about this insight

Return a JSON array. If no insights, return [].
"""


class ConsolidationPipeline:
    """Background consolidation: raw episodes → facts/patterns/preferences/rules."""

    def __init__(self, db: Session) -> None:
        self.db = db
        self._ltm_service = LongTermMemoryService(db)
        self._graph_service = MemoryGraphService(db)

    def get_or_create_config(self, user_id: int) -> ConsolidationConfig:
        config = self.db.query(ConsolidationConfig).filter(ConsolidationConfig.user_id == user_id).first()
        if not config:
            config = ConsolidationConfig(user_id=user_id)
            self.db.add(config)
            self.db.commit()
            self.db.refresh(config)
        return config

    def run_for_user(self, user_id: int) -> dict:
        """Run consolidation for a user. Finds conversations needing consolidation."""
        config = self.get_or_create_config(user_id)
        if not config.auto_consolidate:
            return {"skipped": True, "reason": "auto_consolidate disabled"}

        candidates = self._find_consolidatable(user_id, config)
        if not candidates:
            return {"processed": 0, "conversations": []}

        start_time = time.time()
        results = []
        for c in candidates:
            try:
                result = self._consolidate_conversation(user_id, c["conversation_id"])
                results.append(result)
            except Exception as e:
                logger.error("Consolidation failed for conv %d: %s", c["conversation_id"], e)
                results.append({"conversation_id": c["conversation_id"], "error": str(e)})

        duration_ms = int((time.time() - start_time) * 1000)

        # Update last_consolidation_at
        config.last_consolidation_at = datetime.now(timezone.utc).replace(tzinfo=None)
        self.db.commit()

        # Log run
        log = ConsolidationLog(
            user_id=user_id,
            conversations_processed=len(candidates),
            facts_created=sum(r.get("facts_created", 0) for r in results),
            patterns_found=sum(r.get("patterns_found", 0) for r in results),
            preferences_updated=sum(r.get("preferences_updated", 0) for r in results),
            duration_ms=duration_ms,
        )
        self.db.add(log)
        self.db.commit()

        totals = {
            "processed": len(candidates),
            "facts_created": log.facts_created,
            "patterns_found": log.patterns_found,
            "preferences_updated": log.preferences_updated,
            "duration_ms": duration_ms,
        }
        logger.info("Consolidation for user %d: %s", user_id, totals)
        return totals

    def _find_consolidatable(self, user_id: int, config: ConsolidationConfig) -> list[dict]:
        """Find conversations with ≥min_messages_per_batch unconsolidated messages."""
        # Get already-consolidated conversation IDs from LongTermMemory source_ids
        consolidated_ids = [
            m.source_id
            for m in self.db.query(LongTermMemory).filter(
                LongTermMemory.user_id == user_id,
                LongTermMemory.source == "conversation",
            ).all()
            if m.source_id is not None
        ]
        consolidated_set = set(consolidated_ids)

        conversations = (
            self.db.query(Conversation)
            .filter(
                Conversation.user_id == user_id,
                Conversation.message_count >= config.min_messages_per_batch,
            )
            .order_by(Conversation.updated_at.desc())
            .limit(20)
            .all()
        )

        return [
            {"conversation_id": c.id, "message_count": c.message_count}
            for c in conversations
            if c.id not in consolidated_set
        ]

    def _consolidate_conversation(self, user_id: int, conversation_id: int) -> dict:
        """Consolidate a single conversation. Returns stats."""
        messages = (
            self.db.query(ConversationMessage)
            .filter(ConversationMessage.conversation_id == conversation_id)
            .order_by(ConversationMessage.created_at)
            .limit(50)
            .all()
        )

        if len(messages) < 4:
            return {"conversation_id": conversation_id, "skipped": True, "reason": "too few messages"}

        conversation_text = "\n".join(f"{m.role}: {m.content}" for m in messages)
        abstractions = self._extract_abstractions(conversation_text)

        facts_created = 0
        patterns_found = 0
        preferences_updated = 0

        for item in abstractions:
            atype = item.get("type", "fact")
            if atype not in VALID_TYPES:
                atype = "fact"

            category_map = {"fact": "fact", "pattern": "pattern", "preference": "preference", "rule": "context"}
            category = category_map[atype]

            memory = self._ltm_service.create(
                user_id=user_id,
                category=category,
                title=str(item.get("title", ""))[:50],
                content=str(item.get("content", ""))[:200],
                source="conversation",
                source_id=conversation_id,
                tags=[atype],
            )

            if atype == "fact":
                facts_created += 1
            elif atype == "pattern":
                patterns_found += 1
            elif atype == "preference":
                preferences_updated += 1

            self._link_to_graph(user_id, conversation_id, memory.id, atype)

        return {
            "conversation_id": conversation_id,
            "facts_created": facts_created,
            "patterns_found": patterns_found,
            "preferences_updated": preferences_updated,
        }

    def _extract_abstractions(self, conversation_text: str) -> list[dict]:
        """Send conversation to LLM and parse response into abstraction list."""
        try:
            llm_messages = [
                LLMMessage(role="system", content=EXTRACTION_SYSTEM_PROMPT),
                LLMMessage(role="user", content=f"Conversation:\n{conversation_text[:4000]}"),
            ]
            result = llm_manager.chat(llm_messages, max_tokens=2048, temperature=0.3)

            raw = result.content.strip()
            json_match = re.search(r"\[.*\]", raw, re.DOTALL)
            if not json_match:
                return []

            abstractions = json.loads(json_match.group())
            if not isinstance(abstractions, list):
                return []

            validated = []
            for item in abstractions:
                if isinstance(item, dict) and item.get("type") in VALID_TYPES:
                    validated.append(item)
            return validated

        except Exception as e:
            logger.warning("LLM extraction failed: %s", e)
            return []

    def _link_to_graph(self, user_id: int, conversation_id: int, memory_id: int, edge_type: str) -> None:
        """Create graph node for the new memory and link it to conversation episodes."""
        memory_node = self._graph_service.add_node(
            user_id=user_id,
            memory_type="long_term",
            memory_id=memory_id,
            label=f"consolidated:{edge_type}",
        )

        # Link to episodic nodes from this conversation
        episodic_nodes = (
            self.db.query(MemoryNode)
            .filter(
                MemoryNode.user_id == user_id,
                MemoryNode.memory_type == "episodic",
            )
            .all()
        )

        for enode in episodic_nodes:
            # Check if this episodic memory belongs to this conversation
            # (episodic memories have source context stored)
            self._graph_service.add_edge(
                source_id=memory_node.id,
                target_id=enode.id,
                edge_type=f"consolidated_{edge_type}",
                weight=0.6,
                bidirectional=True,
            )

    def _call_llm_extraction(self, text: str) -> list[dict]:
        """Wrapper for testing — mocked in tests."""
        return self._extract_abstractions(text)
```

```python
"""Consolidation pipeline API endpoints."""
from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session

from backend.app.api.deps import get_current_user, get_db
from backend.app.models.user import User
from backend.app.services.memory.consolidation import ConsolidationPipeline
from backend.app.models.memory.consolidation import ConsolidationConfig, ConsolidationLog

router = APIRouter(prefix="/memory/consolidation", tags=["memory"])


class ConsolidationConfigUpdate(BaseModel):
    min_messages_per_batch: int | None = None
    consolidation_interval_hours: int | None = None
    auto_consolidate: bool | None = None


@router.get("/status")
async def get_consolidation_status(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get last consolidation run and pending conversations."""
    pipeline = ConsolidationPipeline(db)
    config = pipeline.get_or_create_config(current_user.id)
    candidates = pipeline._find_consolidatable(current_user.id, config)

    last_log = (
        db.query(ConsolidationLog)
        .filter(ConsolidationLog.user_id == current_user.id)
        .order_by(ConsolidationLog.created_at.desc())
        .first()
    )

    return {
        "config": {
            "min_messages_per_batch": config.min_messages_per_batch,
            "consolidation_interval_hours": config.consolidation_interval_hours,
            "auto_consolidate": config.auto_consolidate,
            "last_consolidation_at": config.last_consolidation_at.isoformat() if config.last_consolidation_at else None,
        },
        "pending_conversations": len(candidates),
        "last_run": {
            "conversations_processed": last_log.conversations_processed if last_log else 0,
            "facts_created": last_log.facts_created if last_log else 0,
            "patterns_found": last_log.patterns_found if last_log else 0,
            "duration_ms": last_log.duration_ms if last_log else 0,
            "timestamp": last_log.created_at.isoformat() if last_log else None,
        } if last_log else None,
    }


@router.post("/run")
async def run_consolidation(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Trigger manual consolidation run."""
    pipeline = ConsolidationPipeline(db)
    result = pipeline.run_for_user(current_user.id)
    return result


@router.patch("/config")
async def update_consolidation_config(
    update: ConsolidationConfigUpdate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Update consolidation configuration."""
    pipeline = ConsolidationPipeline(db)
    config = pipeline.get_or_create_config(current_user.id)

    if update.min_messages_per_batch is not None:
        config.min_messages_per_batch = update.min_messages_per_batch
    if update.consolidation_interval_hours is not None:
        config.consolidation_interval_hours = update.consolidation_interval_hours
    if update.auto_consolidate is not None:
        config.auto_consolidate = update.auto_consolidate

    db.commit()
    return {"status": "updated"}
```

```python
if "pytest" not in sys.modules:

    async def _periodic_consolidation():
        while True:
            await asyncio.sleep(21600)  # 6 hours
            try:
                from backend.app.db.session import SessionLocal as _ConsolidationSession
                from backend.app.services.memory.consolidation import ConsolidationPipeline

                    pipeline = ConsolidationPipeline(_cdb)

                            result = pipeline.run_for_user(user.id)

                        except Exception as e:

                finally:
                    _cdb.close()
            except Exception as e:
                logger.warning("Periodic consolidation failed: %s", e)

    asyncio.create_task(_periodic_consolidation())
    logger.info("Memory consolidation scheduler started (21600s interval)")
```
