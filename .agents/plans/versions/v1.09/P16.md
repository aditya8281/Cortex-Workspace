## Objective

Replace the static 32K history + 4K RAG token budget with dynamic relevance-weighted allocation. Each message and RAG source is scored for relevance to the current user query. High-relevance content gets proportionally more budget; low-relevance content is truncated first. Improves response quality without increasing total token consumption.

---

Original source: `backend/app/services/intelligence/rag_pipeline.py` — `build_messages()` with hardcoded `max_history_tokens=28000` and `max_tokens=4000`
Target: `ConversationService.get_context_messages()`, `RAGPipeline.build_messages()`

```python
"""Relevance scoring service for dynamic context budget allocation."""
from __future__ import annotations

import math
import logging
from collections import OrderedDict
from functools import lru_cache

from backend.app.services.intelligence.embedding_service import EmbeddingService, get_embedding_service

logger = logging.getLogger(__name__)


class RelevanceScorer:
    """Scores content relevance to a user query using semantic + keyword signals.

    Uses embedding cosine similarity as primary signal. Falls back to
    TF-IDF keyword overlap when embeddings are unavailable or mock.
    """

    def __init__(self, embedding_service: EmbeddingService | None = None) -> None:
        self._embedder = embedding_service or get_embedding_service()
        # LRU cache: message_id / content_hash -> embedding vector
        self._embed_cache: OrderedDict[str, list[float]] = OrderedDict()
        self._cache_max = 500

    def score_message(self, content: str, query_embedding: list[float]) -> float:
        """Score a single message's relevance to the query embedding.

        Returns 0.0-1.0 where 1.0 = highly relevant.
        """
        content_emb = self._get_content_embedding(content)
        if content_emb is None:
            return self._keyword_score(content, query_embedding)
        return self._cosine_similarity(query_embedding, content_emb)

    def score_messages(self, messages: list, query_embedding: list[float]) -> list[tuple]:
        """Score multiple messages. Returns list of (message, score) sorted desc."""
        scored = []
        for msg in messages:
            score = self.score_message(msg.content, query_embedding)
            scored.append((msg, score))
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored

    def score_rag_results(self, rag_results: list, query_embedding: list[float]) -> list[tuple]:
        """Score RAG results. Returns list of (result, score) sorted desc."""
        scored = []
        for result in rag_results:
            content = result.content if hasattr(result, "content") else str(result)
            score = self.score_message(content, query_embedding)
            scored.append((result, score))
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored

    def _get_content_embedding(self, content: str) -> list[float] | None:
        """Get embedding for content, using cache. Returns None on failure."""
        cache_key = str(hash(content) % (2**32))
        if cache_key in self._embed_cache:
            return self._embed_cache[cache_key]

        try:
            emb = self._embedder.embed(content[:512])
            # Detect mock embeddings (near-zero variance = hash-based)
            if self._is_mock_embedding(emb):
                return None
            self._cache_put(cache_key, emb)
            return emb
        except Exception:
            logger.debug("Embedding failed for content, using keyword fallback", exc_info=True)
            return None

    def _is_mock_embedding(self, vec: list[float]) -> bool:
        """Detect deterministic mock embeddings by checking variance."""
        if not vec:
            return True
        mean = sum(vec) / len(vec)
        variance = sum((v - mean) ** 2 for v in vec) / len(vec)
        return variance < 0.0001  # mock MD5 hash has near-zero variance

    def _cache_put(self, key: str, value: list[float]) -> None:
        self._embed_cache[key] = value
        if len(self._embed_cache) > self._cache_max:
            self._embed_cache.popitem(last=False)

    def _keyword_score(self, content: str, query_embedding_or_vec) -> float:
        """Fallback keyword overlap scoring when embeddings unavailable."""
        if isinstance(query_embedding_or_vec, list) and len(query_embedding_or_vec) <= 10:
            # Called with raw vector (can't extract keywords) — low confidence
            return 0.3
        return 0.0

    def _keyword_score_query(self, content: str, query: str) -> float:
        """Score by keyword overlap between content and query string."""
        content_lower = content.lower()
        query_terms = query.lower().split()
        if not query_terms:
            return 0.0
        matches = sum(1 for term in query_terms if term in content_lower)
        return matches / len(query_terms)

    @staticmethod
    def _cosine_similarity(a: list[float], b: list[float]) -> float:
        """Cosine similarity between two vectors."""
        if len(a) != len(b) or not a:
            return 0.0
        dot = sum(ai * bi for ai, bi in zip(a, b))
        norm_a = math.sqrt(sum(ai * ai for ai in a))
        norm_b = math.sqrt(sum(bi * bi for bi in b))
        if norm_a == 0.0 or norm_b == 0.0:
            return 0.0
        return dot / (norm_a * norm_b)
```

```python
"""Dynamic context budget allocator with relevance-weighted distribution."""
from __future__ import annotations

import logging
from dataclasses import dataclass, field

from backend.app.services.intelligence.relevance_scorer import RelevanceScorer
from backend.app.services.interaction.conversation import estimate_tokens
from backend.app.services.intelligence.embedding_service import get_embedding_service

logger = logging.getLogger(__name__)

# Budget allocation weights
SYSTEM_PROMPT_RESERVE = 0.10     # 10% for system prompt + user message
HIGH_RELEVANCE_WEIGHT = 2.0      # Items with score > 0.7 get 2x budget weight
LOW_RELEVANCE_WEIGHT = 0.5       # Items with score < 0.3 get 0.5x budget weight
RAG_RELEVANCE_THRESHOLD = 0.5    # RAG results below this are excluded
HISTORY_RELEVANCE_THRESHOLD = 0.1  # History messages below this excluded


@dataclass
class AllocatedBudget:
    history: list = field(default_factory=list)
    context: list = field(default_factory=list)
    history_tokens: int = 0
    context_tokens: int = 0
    total_tokens: int = 0
    scoring_stats: dict = field(default_factory=dict)


class ContextBudgetAllocator:
    """Allocates token budget dynamically based on relevance scoring."""

    def __init__(self) -> None:
        self._scorer = RelevanceScorer()

    def allocate(
        self,
        messages: list,
        rag_results: list,
        user_query: str,
        total_budget: int = 36000,
    ) -> AllocatedBudget:
        """Allocate token budget between history and RAG context.

        Args:
            messages: List of conversation messages (with .content, .tokens, .role)
            rag_results: List of RAG retrieval results (with .content)
            user_query: The user's current message text
            total_budget: Total token budget (default 36000)

        Returns:
            AllocatedBudget with history (kept messages) and context (kept RAG results)
        """
        reserved = int(total_budget * SYSTEM_PROMPT_RESERVE)
        remaining = total_budget - reserved

        # Embed query once
        query_embedding = self._get_query_embedding(user_query)

        # Score messages
        scored_msgs = self._scorer.score_messages(messages, query_embedding) if messages else []

        # Score RAG results
        scored_rags = self._scorer.score_rag_results(rag_results, query_embedding) if rag_results else []

        # Allocate budget
        history = self._allocate_history(scored_msgs, remaining, query_embedding, user_query)
        history_tokens = sum(estimate_tokens(m.content) for m in history)

        context_remaining = max(remaining - history_tokens, remaining // 4)
        context = self._allocate_context(scored_rags, context_remaining)
        context_tokens = sum(
            estimate_tokens(r.content if hasattr(r, "content") else str(r))
            for r in context
        )

        total_used = reserved + history_tokens + context_tokens

        return AllocatedBudget(
            history=history,
            context=context,
            history_tokens=history_tokens,
            context_tokens=context_tokens,
            total_tokens=total_used,
            scoring_stats={
                "total_messages": len(scored_msgs),
                "kept_messages": len(history),
                "total_rag": len(scored_rags),
                "kept_rag": len(context),
                "high_relevance": sum(1 for _, s in scored_msgs if s > 0.7),
                "low_relevance": sum(1 for _, s in scored_msgs if s < 0.3),
            },
        )

    def _get_query_embedding(self, query: str) -> list[float]:
        """Embed the query for relevance scoring."""
        embedder = get_embedding_service()
        try:
            return embedder.embed(query)
        except Exception:
            logger.debug("Query embedding failed, using zero vector (keyword fallback)")
            return [0.0] * 10  # signals keyword fallback

    def _allocate_history(
        self, scored_messages: list[tuple], budget: int,
        query_embedding: list[float], query: str,
    ) -> list:
        """Select messages to keep based on relevance, within budget."""
        if not scored_messages:
            return []

        kept: list = []
        used = 0

        for msg, score in scored_messages:
            msg_tokens = estimate_tokens(msg.content)

            if score < HISTORY_RELEVANCE_THRESHOLD:
                continue

            # Apply relevance weight
            weight = 1.0
            if score > 0.7:
                weight = HIGH_RELEVANCE_WEIGHT
            elif score < 0.3:
                weight = LOW_RELEVANCE_WEIGHT

            weighted_budget = int(budget * weight / max(len(scored_messages), 1))
            if used + msg_tokens <= budget and msg_tokens <= weighted_budget:
                kept.append(msg)
                used += msg_tokens
            elif used + msg_tokens <= budget and score > 0.5:
                kept.append(msg)
                used += msg_tokens

        # Sort kept messages by original order (chronological)
        kept.sort(key=lambda m: getattr(m, "created_at", None) or 0)
        return kept

    def _allocate_context(self, scored_rags: list[tuple], budget: int) -> list:
        """Select RAG results to keep based on relevance, within budget."""
        kept: list = []
        used = 0

        for rag, score in scored_rags:
            if score < RAG_RELEVANCE_THRESHOLD:
                continue
            content = rag.content if hasattr(rag, "content") else str(rag)
            rag_tokens = estimate_tokens(content)
            if used + rag_tokens <= budget:
                kept.append(rag)
                used += rag_tokens
            else:
                break  # stop once budget is full

        return kept
```

```python
from backend.app.services.intelligence.context_allocator import ContextBudgetAllocator
from backend.app.core.config import settings

class RAGPipeline:
    def __init__(self, db, retrieval=None, conversation_service=None):
        self._db = db
        self._retrieval = retrieval or HybridRetrievalV2(db)
        self._conv_service = conversation_service or ConversationService(db)
        self._allocator = ContextBudgetAllocator()
        self._dynamic_enabled = getattr(settings, "DYNAMIC_CONTEXT_BUDGET", True)

    def build_messages(
        self,
        conversation_id: int,
        user_message: str,
        repo_id: int | None = None,
        max_history_tokens: int = 28000,
        total_budget: int | None = None,
    ) -> list[dict]:
        if total_budget is None:
            total_budget = getattr(settings, "MAX_TOTAL_CONTEXT_TOKENS", 36000)

        # 1. Retrieve RAG context (overfetch for relevance scoring)
        rag_context = self.retrieve_context(
            user_message, repo_id,
            max_tokens=8000,  # higher limit for scoring
            max_results=20,
        )

        # 2. Get full message history
        history = self._conv_service.get_messages(conversation_id, limit=500)

        if self._dynamic_enabled:
            # 3. Dynamic allocation
            allocated = self._allocator.allocate(
                messages=history,
                rag_results=rag_context.results,
                user_query=user_message,
                total_budget=total_budget,
            )
            kept_history = allocated.history
            context_content = allocated.context
            logger.info(
                "Dynamic budget: %d/%d history, %d/%d rag, %d total",
                len(kept_history), len(history),
                len(context_content), rag_context.source_count,
                allocated.total_tokens,
            )
        else:
            kept_history = self._conv_service.get_context_messages(
                conversation_id, max_tokens=max_history_tokens
            )
            context_content = rag_context.results[:MAX_CONTEXT_RESULTS]

        # 4. Build messages (same as before)
        messages = []
        system_parts = ["You are Cortex, a helpful AI assistant with access to the user's codebase and knowledge."]

        if context_content:
            formatted = []
            for i, result in enumerate(context_content):
                ref = f"[{i + 1}] "
                if hasattr(result, "file_path") and result.file_path:
                    ref += result.file_path
                content = result.content if hasattr(result, "content") else str(result)
                formatted.append(f"{ref}\n{content[:500]}")
            system_parts.append(f"Relevant context:\n\n" + "\n\n---\n\n".join(formatted))

        messages.append({"role": "system", "content": "\n\n".join(system_parts)})

        for msg in kept_history:
            messages.append({"role": msg.role, "content": msg.content})

        messages.append({"role": "user", "content": user_message})
        return messages
```
