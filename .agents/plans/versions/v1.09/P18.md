# P18: Prospective Memory — CORTEX Memory Architecture

**Document:** Phase 18 of Version 1.09
**Authority:** Stage 6 — Master Versioned Implementation Roadmap
**Date:** 2026-06-30

---

## Objective

Add a **prospective memory** system that encodes future intentions and automatically detects when trigger conditions are met. Enables CORTEX to proactively surface relevant information when the context matches a stored intention (e.g. "remind me about deployment when we discuss release" or "when this test fails again, check the mock setup").

---

## Architecture

```
User says "remember to X when Y happens"
    │
    ▼
ProspectiveMemoryService.create_intention()
    ├── Encodes intention text + trigger type + trigger config
    ├── Embeds trigger description for semantic matching
    └── Stores in DB + Qdrant (as cross-conversation memory type)
    
New conversation/event arrives
    │
    ▼
ProspectiveMemoryService.check_triggers()
    ├── Embed incoming content
    ├── Compare against active prospective memories
    ├── Score trigger match (semantic + keyword + structural)
    └── If match >= threshold → execute action (notify / log / inject)
```

---

## Prerequisites

P12 (Cross-Conversation Memory) for Qdrant storage. P17 (Semantic Scorer) for trigger matching. EmbeddingService operational.

"""Prospective memory model — stores future intentions with trigger conditions."""
from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

from sqlalchemy import Column, Integer, String, Float, DateTime, Boolean, Text, JSON
from sqlalchemy.orm import relationship, Mapped, mapped_column

from backend.app.models.base import Base


class ProspectiveMemory(Base):
    """Stores a future intention with trigger configuration.

    Each record represents "when X happens, do Y".
    
    Trigger types:
    - semantic: matched via embedding similarity to incoming content
    - keyword: matched via keyword presence in incoming content
    - structural: matched via specific data patterns (e.g. test failure)
    - temporal: time-based (matched by time elapsed since creation)
    
    Action types:
    - notify: push notification / chat message
    - log: log an event
    - context_inject: inject reminder into conversation context
    """

    __tablename__ = "prospective_memories"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    user_id: Mapped[int] = mapped_column(Integer, nullable=False, index=True)
    intention: Mapped[str] = mapped_column(Text, nullable=False)
    trigger_type: Mapped[str] = mapped_column(
        String(50), nullable=False, default="semantic"
    )
    trigger_config: Mapped[dict[str, Any]] = mapped_column(
        JSON, nullable=False, default=dict
    )
    action_type: Mapped[str] = mapped_column(
        String(50), nullable=False, default="notify"
    )
    action_config: Mapped[dict[str, Any]] = mapped_column(
        JSON, nullable=False, default=dict
    )
    status: Mapped[str] = mapped_column(
        String(20), nullable=False, default="active",
        index=True,
    )
    match_count: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
    last_matched_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True), nullable=True
    )
    match_threshold: Mapped[float] = mapped_column(
        Float, nullable=False, default=0.65,
    )
    cooldown_minutes: Mapped[int] = mapped_column(
        Integer, nullable=False, default=60,
    )
    is_auto_discovered: Mapped[bool] = mapped_column(
        Boolean, nullable=False, default=False,
    )
    qdrant_point_id: Mapped[str | None] = mapped_column(
        String(255), nullable=True,
    )
    source_conversation_id: Mapped[int | None] = mapped_column(
        Integer, nullable=True,
    )
    expires_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True), nullable=True,
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), default=lambda: datetime.now(timezone.utc),
        nullable=False,
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), default=lambda: datetime.now(timezone.utc),
        onupdate=lambda: datetime.now(timezone.utc),
        nullable=False,
    )

    def on_trigger_matched(self) -> None:
        """Update match tracking when this intention is triggered."""
        self.match_count += 1
        self.last_matched_at = datetime.now(timezone.utc)

    def is_on_cooldown(self) -> bool:
        """Check if this intention is on cooldown (avoid repeated triggers)."""
        if not self.last_matched_at:
            return False
        from datetime import timedelta
        elapsed = datetime.now(timezone.utc) - self.last_matched_at
        return elapsed.total_seconds() < self.cooldown_minutes * 60

    def is_expired(self) -> bool:
        """Check if this intention has expired."""
        if not self.expires_at:
            return False
        return datetime.now(timezone.utc) > self.expires_at

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "user_id": self.user_id,
            "intention": self.intention,
            "trigger_type": self.trigger_type,
            "trigger_config": self.trigger_config,
            "action_type": self.action_type,
            "action_config": self.action_config,
            "status": self.status,
            "match_count": self.match_count,
            "last_matched_at": self.last_matched_at.isoformat() if self.last_matched_at else None,
            "match_threshold": self.match_threshold,
            "cooldown_minutes": self.cooldown_minutes,
            "is_auto_discovered": self.is_auto_discovered,
            "expires_at": self.expires_at.isoformat() if self.expires_at else None,
            "created_at": self.created_at.isoformat() if self.created_at else None,
        }

```python
"""Prospective memory service — intention storage, trigger checking, auto-discovery."""
from __future__ import annotations

import logging
import json
from datetime import datetime, timezone, timedelta
from typing import Any

from sqlalchemy import select, and_, or_, delete, update
from sqlalchemy.orm import Session

from backend.app.models.memory.prospective import ProspectiveMemory
from backend.app.services.memory.semantic_scorer import SemanticMemoryScorer
from backend.app.services.intelligence.embedding_service import EmbeddingService

logger = logging.getLogger(__name__)


class ProspectiveMemoryService:
    """Manages prospective memory: creating intentions and checking triggers."""

    def __init__(
        self,
        db: Session,
        embedder: EmbeddingService | None = None,
    ) -> None:
        self._db = db
        self._embedder = embedder
        self._scorer = SemanticMemoryScorer(embedder) if embedder else SemanticMemoryScorer()

    # ── Create ────────────────────────────────────────────────

    def create_intention(
        self,
        user_id: int,
        intention: str,
        trigger_type: str = "semantic",
        trigger_config: dict[str, Any] | None = None,
        action_type: str = "notify",
        action_config: dict[str, Any] | None = None,
        match_threshold: float = 0.65,
        cooldown_minutes: int = 60,
        expires_in_days: int | None = None,
        source_conversation_id: int | None = None,
        is_auto_discovered: bool = False,
    ) -> ProspectiveMemory:
        """Create a new prospective memory intention."""
        expires_at = None
        if expires_in_days:
            expires_at = datetime.now(timezone.utc) + timedelta(days=expires_in_days)

        mem = ProspectiveMemory(
            user_id=user_id,
            intention=intention,
            trigger_type=trigger_type,
            trigger_config=trigger_config or {},
            action_type=action_type,
            action_config=action_config or {},
            status="active",
            match_threshold=match_threshold,
            cooldown_minutes=cooldown_minutes,
            source_conversation_id=source_conversation_id,
            is_auto_discovered=is_auto_discovered,
            expires_at=expires_at,
        )
        self._db.add(mem)
        self._db.commit()
        self._db.refresh(mem)
        logger.info("Created prospective intention %d: %s", mem.id, intention[:60])
        return mem

    # ── Check ──────────────────────────────────────────────────

    def check_triggers(
        self,
        user_id: int,
        content: str,
    ) -> list[dict[str, Any]]:
        """Check incoming content against all active prospective memories.

        Returns list of matched intention dicts with match scores.
        Deduplicates by cooldown and expiry.
        """
        intentions = self._query_active(user_id)
        if not intentions:
            return []

        # Embed content once for all semantic checks
        content_emb = self._get_content_embedding(content)
        matches: list[dict[str, Any]] = []

        for intention in intentions:
            # Skip on cooldown
            if intention.is_on_cooldown():
                continue

            # Auto-deactivate expired
            if intention.is_expired():
                intention.status = "expired"
                self._db.add(intention)
                self._db.flush()
                continue

            score = self._evaluate_trigger(intention, content, content_emb)
            if score >= intention.match_threshold:
                intention.on_trigger_matched()
                self._db.add(intention)
                matches.append({
                    "id": intention.id,
                    "intention": intention.intention,
                    "trigger_type": intention.trigger_type,
                    "score": round(score, 4),
                    "action_type": intention.action_type,
                    "action_config": intention.action_config,
                    "match_count": intention.match_count,
                })
                logger.info(
                    "Prospective trigger %d matched (score=%.2f): %s",
                    intention.id, score, intention.intention[:40],
                )

        if matches:
            self._db.commit()

        return matches

    def check_triggers_for_conversation(
        self,
        user_id: int,
        conversation_id: int,
        messages: list,
    ) -> list[dict[str, Any]]:
        """Check all messages in a conversation against intentions.

        Aggregates matches per intention (one match per cooldown period).
        """
        all_matches: dict[int, dict] = {}
        for msg in messages:
            content = msg.content if hasattr(msg, "content") else str(msg)
            matches = self.check_triggers(user_id, content)
            for m in matches:
                mid = m["id"]
                if mid not in all_matches:
                    all_matches[mid] = m
                    all_matches[mid]["triggered_by"] = []
                all_matches[mid]["triggered_by"].append(
                    getattr(msg, "created_at", None) or datetime.now(timezone.utc)
                )
        return list(all_matches.values())

    # ── Evaluation ────────────────────────────────────────────

    def _evaluate_trigger(
        self,
        intention: ProspectiveMemory,
        content: str,
        content_embedding: list[float] | None,
    ) -> float:
        """Evaluate whether content matches an intention's trigger condition."""
        trigger_type = intention.trigger_type

        if trigger_type == "semantic":
            return self._evaluate_semantic(intention, content, content_embedding)
        elif trigger_type == "keyword":
            return self._evaluate_keyword(intention, content)
        elif trigger_type == "structural":
            return self._evaluate_structural(intention, content)
        elif trigger_type == "temporal":
            return self._evaluate_temporal(intention)
        return 0.0

    def _evaluate_semantic(
        self,
        intention: ProspectiveMemory,
        content: str,
        content_embedding: list[float] | None,
    ) -> float:
        """Semantic trigger: score by embedding similarity to intention."""
        if content_embedding:
            score = self._scorer.score(
                content,
                content_embedding,
                {"type": "prospective", "created_at": None},
            )
            return score
        return 0.0

    def _evaluate_keyword(
        self,
        intention: ProspectiveMemory,
        content: str,
    ) -> float:
        """Keyword trigger: score by proportion of trigger keywords present."""
        keywords = intention.trigger_config.get("keywords", [])
        if not keywords:
            return 0.0
        content_lower = content.lower()
        matches = sum(1 for kw in keywords if kw.lower() in content_lower)
        return matches / len(keywords)

    def _evaluate_structural(
        self,
        intention: ProspectiveMemory,
        content: str,
    ) -> float:
        """Structural trigger: match JSON/pattern-based conditions."""
        pattern = intention.trigger_config.get("pattern", "")
        if not pattern:
            return 0.0
        if pattern in content:
            return 1.0
        return 0.0

    def _evaluate_temporal(
        self,
        intention: ProspectiveMemory,
    ) -> float:
        """Temporal trigger: check if enough time has passed."""
        interval_hours = intention.trigger_config.get("interval_hours", 24)
        if not intention.last_matched_at:
            return 1.0  # never triggered — fire now
        elapsed = datetime.now(timezone.utc) - intention.last_matched_at
        ratio = elapsed.total_seconds() / (interval_hours * 3600)
        return min(1.0, ratio)

    # ── Query ──────────────────────────────────────────────────

    def get_intention(self, user_id: int, intention_id: int) -> ProspectiveMemory | None:
        """Get a specific intention by ID."""
        stmt = select(ProspectiveMemory).where(
            and_(
                ProspectiveMemory.id == intention_id,
                ProspectiveMemory.user_id == user_id,
            )
        )
        return self._db.execute(stmt).scalar_one_or_none()

    def list_intentions(
        self,
        user_id: int,
        status: str | None = None,
        trigger_type: str | None = None,
    ) -> list[ProspectiveMemory]:
        """List intentions with optional filters."""
        stmt = select(ProspectiveMemory).where(
            ProspectiveMemory.user_id == user_id
        )
        if status:
            stmt = stmt.where(ProspectiveMemory.status == status)
        if trigger_type:
            stmt = stmt.where(ProspectiveMemory.trigger_type == trigger_type)
        stmt = stmt.order_by(ProspectiveMemory.created_at.desc())
        return list(self._db.execute(stmt).scalars().all())

    def deactivate_intention(
        self, user_id: int, intention_id: int,
    ) -> ProspectiveMemory | None:
        """Deactivate an intention (soft-delete by status)."""
        intention = self.get_intention(user_id, intention_id)
        if intention:
            intention.status = "deactivated"
            self._db.add(intention)
            self._db.commit()
            self._db.refresh(intention)
        return intention

    def delete_intention(self, user_id: int, intention_id: int) -> bool:
        """Permanently delete an intention."""
        intention = self.get_intention(user_id, intention_id)
        if intention:
            self._db.delete(intention)
            self._db.commit()
            return True
        return False

    # ── Internal ──────────────────────────────────────────────

    def _query_active(self, user_id: int) -> list[ProspectiveMemory]:
        """Get all active non-expired intentions for user."""
        stmt = select(ProspectiveMemory).where(
            and_(

```python
"""Intention detector — auto-discovers prospective memories from conversations."""
from __future__ import annotations

import json
import logging
from typing import Any

from backend.app.services.memory.prospective import ProspectiveMemoryService

logger = logging.getLogger(__name__)

# LLM prompt template for intention extraction
INTENTION_DETECTION_PROMPT = """You are analyzing a conversation to find **prospective intentions** — things the user wants to remember or be reminded about in the future.

Extract intentions as JSON array:
[
  {{
    "intention": "clear description of what to remember/do",
    "trigger_type": "semantic" | "keyword",
    "trigger_config": {{
      "keywords": ["trigger", "words", "here"]
    }},
    "action_type": "notify",
    "confidence": 0.0-1.0
  }}
]

Rules:
- Only extract CLEAR prospective intentions (future things to remember/do)
- "remind me when...", "I need to remember to...", "next time X happens, do Y"
- Set confidence low (< 0.5) if the intention is ambiguous
- Skip greetings, small talk, completed tasks
- Skip intentions already stated as completed

Conversation:
{conversation_text}

Return ONLY the JSON array, no other text.
"""


class IntentionDetector:
    """Detects prospective intentions in conversation text using LLM.

    Runs periodically or on-demand after significant conversations.
    Automatically creates ProspectiveMemory entries for detected intentions.
    """

    def __init__(
        self,
        llm_service=None,
        prospective_service: ProspectiveMemoryService | None = None,
    ) -> None:
        self._llm = llm_service
        self._prospective_service = prospective_service

    async def detect_intentions(
        self,
        user_id: int,
        conversation: str,
        conversation_id: int | None = None,
    ) -> list[dict[str, Any]]:
        """Analyze conversation and extract prospective intentions.

        Returns list of detected intention dicts (before creation).
        High-confidence intentions are auto-created.
        """
        if not self._llm:
            logger.warning("No LLM service available for intention detection")
            return []

        # Truncate conversation for LLM prompt
        truncated = conversation[:4000]

        prompt = INTENTION_DETECTION_PROMPT.format(conversation_text=truncated)
        try:
            response = await self._llm.generate(prompt)
            intentions = self._parse_response(response)
        except Exception:
            logger.exception("Intention detection LLM call failed")
            return []

        created = []
        for intention in intentions:
            confidence = intention.pop("confidence", 0.5)
            if confidence >= 0.6:
                if self._prospective_service:
                    mem = self._prospective_service.create_intention(
                        user_id=user_id,
                        intention=intention["intention"],
                        trigger_type=intention.get("trigger_type", "semantic"),
                        trigger_config=intention.get("trigger_config", {}),
                        action_type=intention.get("action_type", "notify"),
                        source_conversation_id=conversation_id,
                        is_auto_discovered=True,
                    )
                    intention["id"] = mem.id
                    intention["auto_created"] = True
                else:
                    intention["auto_created"] = False
            else:
                intention["auto_created"] = False
                intention["confidence"] = confidence

            created.append(intention)

        if created:
            logger.info("Detected %d prospective intentions (auto-created: %d)",
                       len(created), sum(1 for c in created if c.get("auto_created")))

        return created

    def _parse_response(self, response: str) -> list[dict[str, Any]]:
        """Parse LLM JSON response, with basic recovery."""
        try:
            # Strip markdown fences if present
            text = response.strip()
            if text.startswith("```"):
                text = text.split("\n", 1)[-1]
                text = text.rsplit("```", 1)[0]
            return json.loads(text.strip())
        except json.JSONDecodeError:
            logger.warning("Failed to parse intention detection response as JSON")
            return []
```

"""Periodic prospective memory trigger checker.

Runs on a schedule, checks recent conversations against active
prospective intentions, and executes actions for matched triggers.
"""
from __future__ import annotations

import logging
from datetime import datetime, timezone

from sqlalchemy import select

from backend.app.models.interaction.conversation import ConversationMessage, Conversation
from backend.app.services.memory.prospective import ProspectiveMemoryService
from backend.app.services.intelligence.embedding_service import get_embedding_service

logger = logging.getLogger(__name__)

CHECK_INTERVAL_SECONDS = 300  # 5 minutes
BATCH_SIZE = 20
RECENT_WINDOW_HOURS = 1  # Check messages from the last hour


def run_prospective_check(db_session_factory):
    """Check recent messages across all users against active intentions.

    Runs as a periodic background task. For each user with active
    intentions, checks their recent messages for trigger matches.
    """
    with db_session_factory() as db:
        embedder = get_embedding_service()
        service = ProspectiveMemoryService(db, embedder)

        # Find all users with active intentions
        from backend.app.models.memory.prospective import ProspectiveMemory
        user_ids = db.execute(
            select(ProspectiveMemory.user_id)
            .where(ProspectiveMemory.status == "active")
            .distinct()
        ).scalars().all()

        if not user_ids:
            return

        cutoff = datetime.now(timezone.utc)

        for user_id in user_ids:
            try:
                recent_messages = db.execute(
                    select(ConversationMessage)
                    .join(Conversation)
                    .where(
                        Conversation.user_id == user_id,
                        ConversationMessage.created_at >= cutoff,
                    )
                    .limit(BATCH_SIZE)
                ).scalars().all()

                for msg in recent_messages:
                    matches = service.check_triggers(
                        user_id=user_id,
                        content=msg.content,
                    )
                    for match in matches:
                        logger.info(
                            "Prospective trigger fired for user %d: %s (score=%.2f)",
                            user_id, match["intention"][:40], match["score"],
                        )
                        # Future: dispatch action (notify user, inject context)
            except Exception:
                logger.exception(
                    "Prospective check failed for user %d", user_id,
