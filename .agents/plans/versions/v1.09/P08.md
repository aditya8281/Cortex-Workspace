```python
"""PreferenceLearningService — records and retrieves user preferences."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.user_preference import UserPreference

logger = logging.getLogger(__name__)

SOURCE_PRIORITY = {"explicit": 3, "observed": 2, "inferred": 1}


class PreferenceLearningService:
    """Service for recording, strengthening, and retrieving preferences."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def record_preference(
        self,
        user_id: int,
        category: str,
        key: str,
        value: dict[str, Any],
        source: str = "observed",
    ) -> UserPreference:
        """Record a preference. Strengthens existing or creates new."""
        existing = (
            self.db.query(UserPreference)
            .filter(
                UserPreference.user_id == user_id,
                UserPreference.category == category,
                UserPreference.key == key,
            )
            .first()
        )

        if existing:
            new_priority = SOURCE_PRIORITY.get(source, 0)
            existing_priority = SOURCE_PRIORITY.get(existing.source, 0)

            if new_priority >= existing_priority:
                existing.value = value
                existing.source = source

            existing.strengthen()
            self.db.commit()
            self.db.refresh(existing)
            return existing

        pref = UserPreference(
            user_id=user_id,
            category=category,
            key=key,
            value=value,
            source=source,
            observation_count=1,
            last_observed=datetime.now(timezone.utc),
        )
        self.db.add(pref)
        self.db.commit()
        self.db.refresh(pref)
        return pref

    def get_preferences(
        self, user_id: int, category: str | None = None, min_confidence: float = 0.0
    ) -> list[UserPreference]:
        """Get preferences filtered by category and minimum confidence."""
        query = self.db.query(UserPreference).filter(UserPreference.user_id == user_id)

        if category:
            query = query.filter(UserPreference.category == category)
        if min_confidence > 0:
            query = query.filter(UserPreference.confidence >= min_confidence)

        return query.order_by(UserPreference.confidence.desc()).all()

    def get_preference(self, user_id: int, category: str, key: str) -> UserPreference | None:
        """Get a single preference."""
        return (
            self.db.query(UserPreference)
            .filter(
                UserPreference.user_id == user_id,
                UserPreference.category == category,
                UserPreference.key == key,
            )
            .first()
        )

    def get_adaptation_ready(self, user_id: int) -> list[UserPreference]:
        """Return preferences with confidence >= 0.6 (ready for adaptation)."""
        return self.get_preferences(user_id, min_confidence=0.6)

    def contradict_preference(self, user_id: int, category: str, key: str) -> UserPreference | None:
        """Weaken a preference (e.g., when user contradicts it)."""
        pref = self.get_preference(user_id, category, key)
        if pref:
            pref.weaken()
            self.db.commit()
            self.db.refresh(pref)
        return pref

    def delete_preference(self, user_id: int, category: str, key: str) -> bool:
        """Delete a preference. Returns True if deleted."""
        pref = self.get_preference(user_id, category, key)
        if pref:
            self.db.delete(pref)
            self.db.commit()
            return True
        return False

    def reset_all(self, user_id: int) -> int:
        """Delete all preferences for a user. Returns count deleted."""
        deleted = (
            self.db.query(UserPreference)
            .filter(UserPreference.user_id == user_id)
            .delete()
        )
        self.db.commit()
        return deleted
```

```python
"""WorkflowLearningService — detects and stores multi-step workflow patterns."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.workflow_pattern import WorkflowPattern

logger = logging.getLogger(__name__)

MIN_STEPS = 2
SIMILARITY_THRESHOLD = 0.6
MAX_PATTERNS_PER_USER = 200


class WorkflowLearningService:
    """Service for observing and retrieving workflow patterns."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def observe_workflow(
        self, user_id: int, steps: list[str], context: str | None = None
    ) -> WorkflowPattern:
        """Observe a workflow sequence. Creates new or strengthens existing."""
        if len(steps) < MIN_STEPS:
            raise ValueError(f"Workflow requires at least {MIN_STEPS} steps")

        existing = self._find_similar_pattern(user_id, steps)
        if existing:
            existing.observe()
            if context and context not in (existing.context or ""):
                existing.context = context
            self.db.commit()
            self.db.refresh(existing)
            return existing

        # Enforce max patterns per user
        count = (
            self.db.query(WorkflowPattern)
            .filter(WorkflowPattern.user_id == user_id)
            .count()
        )
        if count >= MAX_PATTERNS_PER_USER:
            # Evict least frequent
            oldest = (
                self.db.query(WorkflowPattern)
                .filter(WorkflowPattern.user_id == user_id)
                .order_by(WorkflowPattern.frequency.asc())
                .first()
            )
            if oldest:
                self.db.delete(oldest)
                self.db.commit()

        pattern = WorkflowPattern(
            user_id=user_id,
            pattern_name=f"workflow_{steps[0][:20]}_{steps[-1][:20]}",
            steps=steps,
            context=context,
            last_observed=datetime.now(timezone.utc),
        )
        self.db.add(pattern)
        self.db.commit()
        self.db.refresh(pattern)
        return pattern

    def get_common_workflows(self, user_id: int, limit: int = 10) -> list[WorkflowPattern]:
        """Get most frequent workflows."""
        return (
            self.db.query(WorkflowPattern)
            .filter(WorkflowPattern.user_id == user_id)
            .order_by(WorkflowPattern.frequency.desc())
            .limit(limit)
            .all()
        )

    def get_workflow(self, user_id: int, pattern_name: str) -> WorkflowPattern | None:
        """Get a specific workflow pattern by name."""
        return (
            self.db.query(WorkflowPattern)
            .filter(
                WorkflowPattern.user_id == user_id,
                WorkflowPattern.pattern_name == pattern_name,
            )
            .first()
        )

    def suggest_next_step(self, user_id: int, current_steps: list[str]) -> list[str]:
        """Predict next step given current workflow sequence."""
        patterns = (
            self.db.query(WorkflowPattern)
            .filter(WorkflowPattern.user_id == user_id)
            .order_by(WorkflowPattern.frequency.desc())
            .limit(5)
            .all()
        )

        suggestions = []
        for pattern in patterns:
            steps = list(pattern.steps or [])
            for i, step in enumerate(steps):
                if step == current_steps[-1] and i + 1 < len(steps):
                    suggestions.append(steps[i + 1])
                    break
        return suggestions[:3]  # top 3 suggestions

    def _find_similar_pattern(self, user_id: int, steps: list[str]) -> WorkflowPattern | None:
        """Find existing pattern with similar step sequence using LCS."""
        patterns = (
            self.db.query(WorkflowPattern)
            .filter(WorkflowPattern.user_id == user_id)
            .all()
        )

        for pattern in patterns:
            similarity = self._calculate_sequence_similarity(steps, list(pattern.steps or []))
            if similarity >= SIMILARITY_THRESHOLD:
                return pattern
        return None

    def _calculate_sequence_similarity(self, seq1: list[str], seq2: list[str]) -> float:
        """Calculate LCS-based similarity between two step sequences."""
        if not seq1 or not seq2:
            return 0.0

        m, n = len(seq1), len(seq2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]

        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if self._steps_similar(seq1[i - 1], seq2[j - 1]):
                    dp[i][j] = dp[i - 1][j - 1] + 1
                else:
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])

        lcs = dp[m][n]
        return (2 * lcs) / (m + n) if (m + n) > 0 else 0.0

    @staticmethod
    def _steps_similar(step1: str, step2: str) -> bool:
        """Check if two steps are similar (exact or substring match, case-insensitive)."""
        s1, s2 = step1.lower(), step2.lower()
        return s1 == s2 or s1 in s2 or s2 in s1
```

```python
# preferences.py
from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session

from backend.app.core.db import get_current_user, get_db
from backend.app.models.interaction.user import User
from backend.app.schemas.learning.preference import PreferenceCreate, PreferenceResponse
from backend.app.services.learning.preference_service import PreferenceLearningService

router = APIRouter(prefix="/preferences", tags=["learning-preferences"])


@router.get("", response_model=list[PreferenceResponse])
def list_preferences(
    category: str | None = Query(None),
    min_confidence: float = Query(0.0, ge=0.0, le=1.0),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    service = PreferenceLearningService(db)
    return service.get_preferences(current_user.id, category, min_confidence)


@router.post("", response_model=PreferenceResponse, status_code=201)
def create_preference(
    data: PreferenceCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    service = PreferenceLearningService(db)
    return service.record_preference(
        user_id=current_user.id,
        category=data.category,
        key=data.key,
        value=data.value,
        source=data.source,
    )


@router.delete("")
def reset_preferences(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    service = PreferenceLearningService(db)
    count = service.reset_all(current_user.id)
    return {"deleted": count}
```
