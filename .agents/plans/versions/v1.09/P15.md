```python
"""Memory compression service for old conversations."""
from __future__ import annotations

import logging
from datetime import datetime, timedelta, timezone

from sqlalchemy.orm import Session

from backend.app.models.interaction.conversation import Conversation, ConversationMessage
from backend.app.models.memory.compression import CompressedConversation, CompressionConfig
from backend.app.services.interaction.conversation import estimate_tokens
from backend.app.services.intelligence.llm.manager import llm_manager
from backend.app.services.intelligence.llm.provider import LLMMessage

logger = logging.getLogger(__name__)

COMPRESSION_PROMPT = """Summarize this conversation in a dense paragraph (max 512 tokens).
Preserve:
- Key topics discussed
- Decisions made or conclusions reached
- User preferences revealed
- Action items or follow-ups mentioned
- Any code snippets or technical details mentioned

Conversation:
{conversation_text}

Summary:"""


class CompressionService:
    """Compresses old conversations into dense summaries."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def get_or_create_config(self, user_id: int) -> CompressionConfig:
        config = self.db.query(CompressionConfig).filter(CompressionConfig.user_id == user_id).first()
        if not config:
            config = CompressionConfig(user_id=user_id)
            self.db.add(config)
            self.db.commit()
            self.db.refresh(config)
        return config

    def run_compression(self, user_id: int) -> dict:
        """Find and compress all eligible conversations."""
        config = self.get_or_create_config(user_id)
        if not config.auto_compress:
            return {"skipped": True, "reason": "auto_compress disabled"}

        candidates = self._get_compressible(user_id, config)
        if not candidates:
            return {"compressed": 0, "total_tokens_saved": 0}

        total_tokens_saved = 0
        compressed_count = 0

        for conv in candidates:
            try:
                result = self.compress_conversation(user_id, conv.id)
                if result.get("compressed_tokens", 0) > 0:
                    tokens_saved = result["original_tokens"] - result["compressed_tokens"]
                    total_tokens_saved += max(0, tokens_saved)
                    compressed_count += 1
            except Exception as e:
                logger.error("Failed to compress conv %d: %s", conv.id, e)

        config.last_compression_at = datetime.now(timezone.utc).replace(tzinfo=None)
        self.db.commit()

        logger.info("Compression: %d conversations compressed, ~%d tokens saved", compressed_count, total_tokens_saved)
        return {"compressed": compressed_count, "total_tokens_saved": total_tokens_saved}

    def _get_compressible(self, user_id: int, config: CompressionConfig) -> list[Conversation]:
        """Find conversations older than compression_age_days, not yet compressed, with ≥10 messages."""
        cutoff = datetime.now(timezone.utc).replace(tzinfo=None) - timedelta(days=config.compression_age_days)

        already_compressed = {
            row[0]
            for row in self.db.query(CompressedConversation.conversation_id)
            .filter(
                CompressedConversation.user_id == user_id,
                CompressedConversation.is_active == True,  # noqa: E712
            )
            .all()
        }

        conversations = (
            self.db.query(Conversation)
            .filter(
                Conversation.user_id == user_id,
                Conversation.updated_at <= cutoff,
                Conversation.message_count >= 10,
            )
            .order_by(Conversation.updated_at)
            .limit(50)
            .all()
        )

        return [c for c in conversations if c.id not in already_compressed]

    def compress_conversation(self, user_id: int, conversation_id: int) -> dict:
        """Compress a single conversation. Original messages preserved in DB."""
        messages = (
            self.db.query(ConversationMessage)
            .filter(ConversationMessage.conversation_id == conversation_id)
            .order_by(ConversationMessage.created_at)
            .limit(200)
            .all()
        )

        if len(messages) < 4:
            return {"conversation_id": conversation_id, "skipped": True, "reason": "too few messages"}

        conversation_text = "\n".join(f"{m.role}: {m.content[:500]}" for m in messages)
        original_tokens = sum(m.tokens or estimate_tokens(m.content) for m in messages)

        # Call LLM for summary
        prompt = COMPRESSION_PROMPT.format(conversation_text=conversation_text[:8000])
        llm_messages = [
            LLMMessage(role="system", content="You are a conversation summarizer. Be concise but thorough."),
            LLMMessage(role="user", content=prompt),
        ]
        result = llm_manager.chat(llm_messages, max_tokens=512, temperature=0.3)
        summary = result.content.strip()
        compressed_tokens = estimate_tokens(summary)

        ratio = compressed_tokens / max(original_tokens, 1)

        compressed = CompressedConversation(
            conversation_id=conversation_id,
            user_id=user_id,
            summary_text=summary,
            original_token_count=original_tokens,
            compressed_token_count=compressed_tokens,
            compression_ratio=round(ratio, 4),
            is_active=True,
        )
        self.db.add(compressed)
        self.db.commit()

        logger.info(
            "Compressed conv %d: %d → %d tokens (ratio=%.3f)",
            conversation_id, original_tokens, compressed_tokens, ratio,
        )

        return {
            "conversation_id": conversation_id,
            "original_tokens": original_tokens,
            "compressed_tokens": compressed_tokens,
            "compression_ratio": round(ratio, 4),
        }

    def restore_conversation(self, user_id: int, conversation_id: int) -> bool:
        """Restore a conversation from compressed to full (for archive search)."""
        compressed = (
            self.db.query(CompressedConversation)
            .filter(
                CompressedConversation.conversation_id == conversation_id,
                CompressedConversation.user_id == user_id,
            )
            .first()
        )
        if not compressed:
            return False
        compressed.is_active = False
        self.db.commit()
        return True

    def get_compression_stats(self, user_id: int) -> dict:
        """Get compression statistics."""
        compressed = (
            self.db.query(CompressedConversation)
            .filter(
                CompressedConversation.user_id == user_id,
                CompressedConversation.is_active == True,  # noqa: E712
            )
            .all()
        )

        if not compressed:
            return {"total_compressed": 0, "total_tokens_saved": 0, "avg_compression_ratio": 0}

        total_saved = sum(c.original_token_count - c.compressed_token_count for c in compressed)
        avg_ratio = sum(c.compression_ratio for c in compressed) / len(compressed)

        return {
            "total_compressed": len(compressed),
            "total_tokens_saved": max(0, total_saved),
            "avg_compression_ratio": round(avg_ratio, 4),
        }
```

def get_context_messages(
    self, conversation_id: int, max_tokens: int = MAX_CONTEXT_TOKENS
) -> list:
    """Get messages within token budget. Uses compressed summary if available."""
    # Check if conversation is compressed
    try:
        from backend.app.models.memory.compression import CompressedConversation

        compressed = (
            self._db.query(CompressedConversation)
            .filter(
                CompressedConversation.conversation_id == conversation_id,
                CompressedConversation.is_active == True,  # noqa: E712
            )
            .first()
        )
    except Exception:
        compressed = None

    if compressed:
        # Return summary as a synthetic message
        from backend.app.models.interaction.conversation import ConversationMessage

        summary_msg = ConversationMessage(
            conversation_id=conversation_id,
            role="system",
            content=f"[Compressed summary of older messages — {compressed.summary_text}]",
            tokens=compressed.compressed_token_count,
        )
        # Also load recent messages (post-compression)
        recent_msgs = (
            self._db.query(ConversationMessage)
            .filter(ConversationMessage.conversation_id == conversation_id)
            .order_by(ConversationMessage.created_at.desc())
            .limit(50)
            .all()
        )
        # Combine: summary + recent messages, within budget
        total = compressed.compressed_token_count
        kept = [summary_msg]
        for msg in reversed(recent_msgs):
            msg_tokens = msg.tokens or estimate_tokens(msg.content)
            if total + msg_tokens > max_tokens:
                break
            kept.append(msg)
            total += msg_tokens
        return kept

    # Original logic: load all, newest-first, within budget
    all_msgs = self.get_messages(conversation_id, limit=500)
    if not all_msgs:
        return []

    total = 0
    kept = []
    for msg in reversed(all_msgs):
        msg_tokens = msg.tokens or estimate_tokens(msg.content)
        if total + msg_tokens > max_tokens:
            break
        kept.append(msg)
        total += msg_tokens
    kept.reverse()
    return kept

"""Compression API endpoints."""
from fastapi import APIRouter, Depends
from pydantic import BaseModel
from sqlalchemy.orm import Session

from backend.app.api.deps import get_current_user, get_db
from backend.app.models.user import User
from backend.app.services.memory.compression import CompressionService

router = APIRouter(prefix="/memory/compression", tags=["memory"])


class CompressionConfigUpdate(BaseModel):
    compression_age_days: int | None = None
    max_summary_tokens: int | None = None
    auto_compress: bool | None = None


@router.get("/status")
async def get_status(db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    service = CompressionService(db)
    stats = service.get_compression_stats(current_user.id)
    config = service.get_or_create_config(current_user.id)
    return {"stats": stats, "config": {"compression_age_days": config.compression_age_days, "auto_compress": config.auto_compress}}


@router.post("/run")
async def run_compression(db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    service = CompressionService(db)
    result = service.run_compression(current_user.id)
    return result


@router.patch("/config")
async def update_config(update: CompressionConfigUpdate, db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    service = CompressionService(db)
    config = service.get_or_create_config(current_user.id)
    if update.compression_age_days is not None: config.compression_age_days = update.compression_age_days
    if update.max_summary_tokens is not None: config.max_summary_tokens = update.max_summary_tokens
    if update.auto_compress is not None: config.auto_compress = update.auto_compress
    db.commit()
    return {"status": "updated"}

    async def _periodic_compression():
        while True:
            await asyncio.sleep(86400)  # 24 hours
            try:
                from backend.app.db.session import SessionLocal as _CompressSession
                from backend.app.services.memory.compression import CompressionService

                _cdb = _CompressSession()
                try:
                    active_users = _cdb.query(_User).filter(_User.is_active == True).all()  # noqa: E712
                    for user in active_users:
                        try:
                            service = CompressionService(_cdb)

                                logger.info("Compression for user %d: %s", user.id, result)

                finally:
                    _cdb.close()
            except Exception as e:

    asyncio.create_task(_periodic_compression())
