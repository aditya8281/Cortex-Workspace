"""HabitDetectionService — detects recurring user actions."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.habit import Habit, MIN_OCCURRENCES

logger = logging.getLogger(__name__)

MAX_HABITS_PER_USER = 100


class HabitDetectionService:
    """Service for observing and detecting user habits."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def observe_action(
        self,
        user_id: int,
        action: str,
        trigger: str | None = None,
        habit_name: str | None = None,
    ) -> Habit:
        """Record an action occurrence. Creates or increments habit."""
        existing = (
            self.db.query(Habit)
            .filter(Habit.user_id == user_id, Habit.action == action)
            .first()
        )

        if existing:
            existing.observe()
            self.db.commit()
            self.db.refresh(existing)
            return existing

        # Enforce limit
        count = (
            self.db.query(Habit).filter(Habit.user_id == user_id).count()
        )
        if count >= MAX_HABITS_PER_USER:
            oldest = (
                self.db.query(Habit)
                .filter(Habit.user_id == user_id)
                .order_by(Habit.occurrences.asc())
                .first()
            )
            if oldest:
                self.db.delete(oldest)
                self.db.commit()

        habit = Habit(
            user_id=user_id,
            habit_name=habit_name or action,
            action=action,
            trigger=trigger,
            first_observed=datetime.now(timezone.utc),
            last_observed=datetime.now(timezone.utc),
        )
        self.db.add(habit)
        self.db.commit()
        self.db.refresh(habit)
        return habit

    def get_habits(self, user_id: int, confirmed_only: bool = True) -> list[Habit]:
        """Get habits for a user."""
        query = self.db.query(Habit).filter(Habit.user_id == user_id)
        if confirmed_only:
            query = query.filter(Habit.occurrences >= MIN_OCCURRENCES)
        return query.order_by(Habit.occurrences.desc()).all()

    def get_habit(self, user_id: int, action: str) -> Habit | None:
        """Get a specific habit by action name."""
        return (
            self.db.query(Habit)
            .filter(Habit.user_id == user_id, Habit.action == action)
            .first()
        )

    def get_habits_by_frequency(self, user_id: int, frequency: str) -> list[Habit]:
        """Get habits filtered by frequency classification."""
        return (
            self.db.query(Habit)
            .filter(Habit.user_id == user_id, Habit.frequency == frequency)
            .all()
        )

    def is_habitual(self, user_id: int, action: str) -> bool:
        """Check if an action is a confirmed habit."""
        habit = self.get_habit(user_id, action)
        return habit.is_confirmed_habit if habit else False

"""BehaviorAdaptationService — adapts Cortex behavior based on learned preferences."""

from __future__ import annotations

import logging
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.user_preference import UserPreference
from backend.app.services.learning.preference_service import PreferenceLearningService

logger = logging.getLogger(__name__)

ADAPTATION_THRESHOLD = 0.6


class BehaviorAdaptationService:
    """Service for adapting behavior based on high-confidence preferences.
    
    Design constraints:
    - Never removes options, only reorders/limits
    - User override always available
    - Confidence threshold of 0.6 for adaptation
    """

    def __init__(self, db: Session) -> None:
        self.db = db
        self._preference_service = PreferenceLearningService(db)

    def adapt_response(self, response: str, user_id: int) -> str:
        """Adapt response based on user's verbosity preference."""
        verbosity_pref = self._preference_service.get_preference(
            user_id, "response_style", "verbosity"
        )

        if verbosity_pref and verbosity_pref.confidence >= ADAPTATION_THRESHOLD:
            level = (verbosity_pref.value or {}).get("level", "normal")
            if level == "concise" and len(response) > 500:
                return response[:500] + "\n\n[response trimmed for concise preference]"
            if level == "detailed":
                return response  # return as-is

        return response

    def adapt_suggestions(
        self, suggestions: list[str], user_id: int
    ) -> list[str]:
        """Limit suggestions based on user's suggestion preference."""
        pref = self._preference_service.get_preference(
            user_id, "interaction", "suggestion_density"
        )

        if pref and pref.confidence >= ADAPTATION_THRESHOLD:
            density = (pref.value or {}).get("level", "normal")
            if density == "minimal":
                return suggestions[:2]
            if density == "normal":
                return suggestions[:5]

        return suggestions[:5]

    def should_suggest(self, user_id: int) -> bool:
        """Check if suggestions should be offered based on recent corrections."""
        from backend.app.models.learning.learning_event import LearningEvent

        recent = (
            self.db.query(LearningEvent)
            .filter(
                LearningEvent.user_id == user_id,
                LearningEvent.event_type.in_(["correction", "affirmation"]),
                LearningEvent.applied == True,
            )
            .order_by(LearningEvent.created_at.desc())
            .limit(10)
            .all()
        )

        if not recent:
            return True

        corrections = sum(1 for e in recent if e.is_negative)
        correction_rate = corrections / len(recent)
        return correction_rate < 0.5

    def get_adaptation_profile(self, user_id: int) -> dict[str, Any]:
        """Build adaptation profile from high-confidence preferences."""
        prefs = self._preference_service.get_adaptation_ready(user_id)
        profile = {
            "response_style": "normal",
            "suggestion_tolerance": "normal",
            "active_hours": None,
        }

        for pref in prefs:
            category = pref.category
            key = pref.key
            if category == "response_style" and key == "verbosity":
                profile["response_style"] = (pref.value or {}).get("level", "normal")
            elif category == "interaction" and key == "suggestion_density":
                profile["suggestion_tolerance"] = (pref.value or {}).get("level", "normal")

        return profile

"""FeedbackLearningService — records and processes user feedback."""

from __future__ import annotations

import logging
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.learning.learning_event import LearningEvent, MAX_DELTA
from backend.app.services.learning.preference_service import PreferenceLearningService

logger = logging.getLogger(__name__)


class FeedbackLearningService:
    """Service for recording feedback and calculating learning metrics."""

    def __init__(self, db: Session) -> None:
        self.db = db
        self._preference_service = PreferenceLearningService(db)

    def record_feedback(
        self,
        user_id: int,
        event_type: str,
        context: dict[str, Any] | None = None,
        input_data: dict[str, Any] | None = None,
        output_data: dict[str, Any] | None = None,
        apply_to_preference: bool = False,
    ) -> LearningEvent:
        """Record a feedback event. Optionally apply delta to related preference."""
        delta = MAX_DELTA if event_type == "affirmation" else -MAX_DELTA

        event = LearningEvent(
            user_id=user_id,
            event_type=event_type,
            context=context,
            input_data=input_data,
            output_data=output_data,
            delta=delta,
            applied=apply_to_preference,
        )
        self.db.add(event)

        if apply_to_preference and context and event_type in ("affirmation", "correction"):
            category = context.get("category")
            key = context.get("key")
            if category and key:
                if event_type == "affirmation":
                    self._preference_service.record_preference(
                        user_id, category, key, context.get("value", {}),
                    )
                else:
                    self._preference_service.contradict_preference(user_id, category, key)

        self.db.commit()
        self.db.refresh(event)
        return event

    def get_learning_rate(self, user_id: int) -> float:
        """Calculate learning rate: ratio of positive to total feedback."""
        events = (
            self.db.query(LearningEvent)
            .filter(LearningEvent.user_id == user_id)
            .all()
        )

        if not events:
            return 0.5  # neutral default

        positive = sum(1 for e in events if e.is_positive)
        return positive / len(events)

    def get_feedback_history(
        self, user_id: int, limit: int = 20
    ) -> list[LearningEvent]:
        """Get recent feedback events."""
        return (
            self.db.query(LearningEvent)
            .filter(LearningEvent.user_id == user_id)
            .order_by(LearningEvent.created_at.desc())
            .limit(limit)
            .all()
        )
