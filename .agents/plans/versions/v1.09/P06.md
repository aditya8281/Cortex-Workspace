```typescript
// api.ts
const API_PREFIX = "/api/v1/knowledge";

export interface KnowledgeSource {
  id: number;
  name: string;
  path: string;
  description: string | null;
  watch_enabled: boolean;
  auto_sync: boolean;
  embedding_model: string;
  chunk_strategy: string;
  chunk_size: number;
  chunk_overlap: number;
  status: string;
  progress: number;
  error_message: string | null;
  total_files: number;
  indexed_files: number;
  total_chunks: number;
  last_synced_at: string | null;
  created_at: string;
}

export interface KnowledgeSourceCreate {
  name: string;
  path: string;
  description?: string;
  watch_enabled?: boolean;
  auto_sync?: boolean;
  embedding_model?: string;
  chunk_strategy?: string;
  chunk_size?: number;
  chunk_overlap?: number;
}

export async function fetchSources(): Promise<KnowledgeSource[]> {
  const res = await fetch(`${API_PREFIX}/sources`);
  const data = await res.json();
  return data.sources ?? [];
}

export async function createSource(data: KnowledgeSourceCreate): Promise<KnowledgeSource> {
  const res = await fetch(`${API_PREFIX}/sources`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data),
  });
  if (!res.ok) throw new Error("Failed to create source");
  return res.json();
}

export async function triggerSync(sourceId: number): Promise<void> {
  await fetch(`${API_PREFIX}/sync/${sourceId}/sync`, { method: "POST" });
}

export async function forceSync(sourceId: number): Promise<void> {
  await fetch(`${API_PREFIX}/sync/${sourceId}/force-sync`, { method: "POST" });
}

export async function deleteSource(sourceId: number): Promise<void> {
  await fetch(`${API_PREFIX}/sources/${sourceId}`, { method: "DELETE" });
}
```

```tsx
  currentChunkSize: number;
  currentChunkOverlap: number;
  onUpdated: () => void;
}

export function EmbeddingConfig({
  sourceId,
  currentModel,
  currentStrategy,
  currentChunkSize,
  currentChunkOverlap,
  onUpdated,
}: EmbeddingConfigProps) {
  const [model, setModel] = useState(currentModel);
```

```tsx
import { useState, useCallback } from "react";
import { SearchResults } from "./SearchResults";

interface SearchResult {
  content: string;
  file_path: string;
  source_id: number | null;
  relevance_score: number;
}

export default function SearchPage() {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<SearchResult[]>([]);
  const [loading, setLoading] = useState(false);
  const [searched, setSearched] = useState(false);
  const [sourceFilter, setSourceFilter] = useState<number | undefined>();
  const [fileTypeFilter, setFileTypeFilter] = useState<string | undefined>();

  const handleSearch = useCallback(async () => {
    if (!query.trim()) return;
    setLoading(true);
    setSearched(true);

    try {
      const params = new URLSearchParams({ q: query });
      if (sourceFilter) params.set("source_id", String(sourceFilter));
      if (fileTypeFilter) params.set("file_type", fileTypeFilter);

      const res = await fetch(`/api/v1/knowledge/search?${params}`);
      const data = await res.json();
      setResults(data.results ?? []);
    } catch (err) {
      console.error("Search failed:", err);
      setResults([]);
    } finally {
      setLoading(false);
    }
  }, [query, sourceFilter, fileTypeFilter]);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "Enter") handleSearch();
  };

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-6">
      <h1 className="text-2xl font-bold text-text-primary">Search Knowledge</h1>

      <div className="flex gap-3">
```

```tsx
// GraphVisualization.tsx
"use client";

import { useEffect, useRef } from "react";
import * as d3 from "d3";

interface GraphEdge {
  source: string;
  target: string;
  type: string;
  strength: number;
}

interface GraphNode {
  id: string;
  group?: string;
}

interface Props {
  edges: GraphEdge[];
  width?: number;
  height?: number;
}

export function GraphVisualization({ edges, width = 600, height = 400 }: Props) {
  const svgRef = useRef<SVGSVGElement>(null);

  useEffect(() => {
    if (!svgRef.current || edges.length === 0) return;

    const svg = d3.select(svgRef.current);
    svg.selectAll("*").remove();

    // Build nodes from edges
    const nodeMap = new Map<string, GraphNode>();
    edges.forEach((e) => {
      if (!nodeMap.has(e.source)) nodeMap.set(e.source, { id: e.source });
      if (!nodeMap.has(e.target)) nodeMap.set(e.target, { id: e.target });
    });

    const nodes: GraphNode[] = Array.from(nodeMap.values());
    const linkData = edges.map((e) => ({
      source: e.source,
      target: e.target,
      type: e.type,
      strength: e.strength,
    }));

    // Color map for relationship types
    const colorMap: Record<string, string> = {
      cross_reference: "#f59e0b",
      shared_extension: "#3b82f6",
      embedding_proximity: "#10b981",
      co_occurrence: "#8b5cf6",
    };

    const simulation = d3.forceSimulation(nodes)
      .force("link", d3.forceLink(linkData).id((d: any) => d.id).distance(100))
      .force("charge", d3.forceManyBody().strength(-200))
      .force("center", d3.forceCenter(width / 2, height / 2));

    const zoom = d3.zoom<SVGSVGElement, unknown>()
      .scaleExtent([0.3, 3])
      .on("zoom", (event) => g.attr("transform", event.transform));
    svg.call(zoom);

    const g = svg.append("g");

    // Links
    const link = g.append("g")
      .selectAll("line")
      .data(linkData)
      .enter()
      .append("line")
      .attr("stroke", (d) => colorMap[d.type] || "#6b7280")
      .attr("stroke-width", (d) => d.strength * 3)
      .attr("stroke-opacity", 0.4);

    // Nodes
    const node = g.append("g")
      .selectAll("circle")
      .data(nodes)
      .enter()
      .append("circle")
      .attr("r", 6)
      .attr("fill", "#3b82f6")
      .attr("stroke", "#1e293b")
      .attr("stroke-width", 1.5)
      .call(
        d3.drag<SVGCircleElement, GraphNode>()
          .on("start", (event, d) => {
            if (!event.active) simulation.alphaTarget(0.3).restart();
            d.fx = d.x;
            d.fy = d.y;
          })
          .on("drag", (event, d) => {
            d.fx = event.x;
            d.fy = event.y;
          })
          .on("end", (event, d) => {
            if (!event.active) simulation.alphaTarget(0);
            d.fx = null;
            d.fy = null;
          }) as any
      );

    // Labels
    const label = g.append("g")
      .selectAll("text")
      .data(nodes)
      .enter()
      .append("text")
      .text((d) => d.id.split("/").pop() || d.id)
      .attr("font-size", "10px")
      .attr("fill", "#94a3b8")
      .attr("dx", 8)
      .attr("dy", 3);

    // Legend
    const legend = svg.append("g").attr("transform", `translate(10, ${height - 80})`);
    let legendY = 0;
    Object.entries(colorMap).forEach(([type, color]) => {
      legend.append("line")
        .attr("x1", 0).attr("y1", legendY)
        .attr("x2", 20).attr("y2", legendY)
        .attr("stroke", color).attr("stroke-width", 2);
      legend.append("text")
        .attr("x", 25).attr("y", legendY + 4)
        .attr("font-size", "10px")
        .attr("fill", "#94a3b8")
        .text(type.replace("_", " "));
      legendY += 16;
    });

    simulation.on("tick", () => {
      link
        .attr("x1", (d: any) => d.source.x)
        .attr("y1", (d: any) => d.source.y)
        .attr("x2", (d: any) => d.target.x)
        .attr("y2", (d: any) => d.target.y);
      node.attr("cx", (d: any) => d.x).attr("cy", (d: any) => d.y);
      label.attr("x", (d: any) => d.x).attr("y", (d: any) => d.y);
    });

    return () => {
      simulation.stop();
    };
  }, [edges, width, height]);

  if (edges.length === 0) {
    return (
      <div className="flex items-center justify-center h-64 bg-surface-800 rounded-lg border border-surface-700 text-text-muted">
        <p>No graph data available — sync a knowledge source first</p>
      </div>
    );
  }

  return (
    <svg
      ref={svgRef}
      width={width}
      height={height}
      className="bg-surface-900 rounded-lg border border-surface-700"
    />
  );
}
```

```tsx
// GraphPage.tsx
"use client";

import { useState, useEffect } from "react";
import { GraphVisualization } from "./GraphVisualization";
import { fetchSources, KnowledgeSource } from "../api";

interface GraphEdge {
  source: string;
  target: string;
  type: string;
  strength: number;
}

export default function GraphPage() {
  const [sources, setSources] = useState<KnowledgeSource[]>([]);
  const [selectedSourceId, setSelectedSourceId] = useState<number | null>(null);
  const [edges, setEdges] = useState<GraphEdge[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    fetchSources().then(setSources);
  }, []);

  const loadGraph = async (sourceId: number) => {
    setSelectedSourceId(sourceId);
    setLoading(true);
    try {
      const res = await fetch(`/api/v1/knowledge/graph/${sourceId}`);
      const data = await res.json();
      setEdges(data.edges ?? []);
    } catch (err) {
      console.error("Failed to load graph:", err);
      setEdges([]);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-6xl mx-auto p-6 space-y-6">
      <h1 className="text-2xl font-bold text-text-primary">Knowledge Graph</h1>

      <div className="flex gap-2">
        {sources.map((s) => (
          <button
            key={s.id}
            onClick={() => loadGraph(s.id)}
            className={`px-3 py-1.5 text-sm rounded ${
              selectedSourceId === s.id
                ? "bg-accent-500 text-white"
                : "bg-surface-800 text-text-muted hover:bg-surface-700"
            }`}
          >
            {s.name}
          </button>
        ))}
      </div>

      {loading && <div className="text-center text-text-muted py-8">Loading graph...</div>}

      {selectedSourceId && !loading && (
        <div className="flex items-center justify-center">
          <GraphVisualization edges={edges} width={800} height={500} />
        </div>
      )}

      {!selectedSourceId && (
        <div className="text-center py-16 text-text-muted">
          <p>Select a knowledge source to view its graph</p>
        </div>
      )}
    </div>
  );
}
```
