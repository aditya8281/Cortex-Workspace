```python
"""SM-2 spaced repetition review service."""
from __future__ import annotations

import math
import logging
from datetime import datetime, timezone

from sqlalchemy import and_, func
from sqlalchemy.orm import Session

from backend.app.models.memory.episodic import EpisodicMemory
from backend.app.models.memory.semantic import SemanticMemory

logger = logging.getLogger(__name__)


class SM2ReviewService:
    """SM-2 spaced repetition algorithm for optimal memory retention."""

    # Default SM-2 parameters
    MIN_EF = 1.3
    MAX_EF = 3.0

    def __init__(self, db: Session) -> None:
        self.db = db

    @staticmethod
    def _apply_sm2(
        easiness_factor: float,
        interval_days: int,
        repetitions: int,
        quality: int,
    ) -> tuple[float, int, int]:
        """Apply SM-2 algorithm and return (ef, interval, reps).

        Args:
            easiness_factor: Current EF (default 2.5)
            interval_days: Current interval in days
            repetitions: Current repetition count
            quality: Recall quality 0-5 (0=failed, 5=perfect)

        Returns:
            Tuple of (new_ef, new_interval, new_repetitions)
        """
        if quality < 3:
            # Failed recall — reset
            repetitions = 0
            interval = 1
            easiness_factor = max(SM2ReviewService.MIN_EF, easiness_factor - 0.2)
        else:
            # Successful recall
            repetitions += 1
            if repetitions == 1:
                interval = 1
            elif repetitions == 2:
                interval = 6
            else:
                interval = round(interval_days * easiness_factor)

            # Update EF using SM-2 formula
            ef_delta = 0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02)
            easiness_factor = max(SM2ReviewService.MIN_EF, min(SM2ReviewService.MAX_EF, easiness_factor + ef_delta))

        return easiness_factor, interval, repetitions

    def get_due_reviews(self, user_id: int, limit: int = 20) -> list[dict]:
        """Get all memories due for review, across episodic and semantic.

        Returns:
            List of dicts with keys: id, memory_type, content, ef, interval, reps, next_review
        """
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        results: list[dict] = []

        # Episodic due reviews
        episodic = (
            self.db.query(EpisodicMemory)
            .filter(
                EpisodicMemory.user_id == user_id,
                EpisodicMemory.sm2_next_review_at <= now,
            )
            .order_by(EpisodicMemory.sm2_next_review_at)
            .limit(limit)
            .all()
        )
        for mem in episodic:
            results.append({
                "id": mem.id,
                "memory_type": "episodic",
                "content": mem.content[:200],
                "importance": mem.importance,
                "ef": round(mem.sm2_easiness_factor, 2),
                "interval": mem.sm2_interval_days,
                "repetitions": mem.sm2_repetitions,
                "next_review": mem.sm2_next_review_at.isoformat() if mem.sm2_next_review_at else None,
            })

        # Semantic due reviews
        semantic = (
            self.db.query(SemanticMemory)
            .filter(
                SemanticMemory.user_id == user_id,
                SemanticMemory.sm2_next_review_at <= now,
            )
            .order_by(SemanticMemory.sm2_next_review_at)
            .limit(limit)
            .all()
        )
        for mem in semantic:
            results.append({
                "id": mem.id,
                "memory_type": "semantic",
                "content": mem.content[:200],
                "category": mem.category,
                "ef": round(mem.sm2_easiness_factor, 2),
                "interval": mem.sm2_interval_days,
                "repetitions": mem.sm2_repetitions,
                "next_review": mem.sm2_next_review_at.isoformat() if mem.sm2_next_review_at else None,
            })

        results.sort(key=lambda x: x.get("next_review") or "")
        return results[:limit]

    def process_recall(self, memory_type: str, memory_id: int, quality: int) -> dict:
        """Process a recall result and update SM-2 fields.

        Args:
            memory_type: "episodic" or "semantic"
            memory_id: ID of the memory
            quality: Recall quality 0-5

        Returns:
            Dict with updated SM-2 state
        """
        if quality < 0 or quality > 5:
            raise ValueError(f"Quality must be 0-5, got {quality}")

        model_cls = EpisodicMemory if memory_type == "episodic" else SemanticMemory
        memory = self.db.query(model_cls).filter(model_cls.id == memory_id).first()

        if not memory:
            raise ValueError(f"{memory_type} memory {memory_id} not found")

        new_ef, new_interval, new_reps = self._apply_sm2(
            memory.sm2_easiness_factor,
            memory.sm2_interval_days,
            memory.sm2_repetitions,
            quality,
        )

        now = datetime.now(timezone.utc).replace(tzinfo=None)
        from datetime import timedelta

        memory.sm2_easiness_factor = new_ef
        memory.sm2_interval_days = new_interval
        memory.sm2_repetitions = new_reps
        memory.sm2_next_review_at = now + timedelta(days=new_interval)
        memory.access_count = (memory.access_count or 0) + 1
        memory.last_accessed = now
        self.db.commit()

        logger.info(
            "SM-2 recall: %s[%d] quality=%d ef=%.2f interval=%d reps=%d",
            memory_type, memory_id, quality, new_ef, new_interval, new_reps,
        )

        return {
            "memory_type": memory_type,
            "memory_id": memory_id,
            "quality": quality,
            "new_ef": round(new_ef, 2),
            "new_interval": new_interval,
            "new_repetitions": new_reps,
            "next_review": memory.sm2_next_review_at.isoformat(),
        }

    def get_forgetting_curve_stats(self, user_id: int) -> dict:
        """Get SM-2 statistics for a user.

        Returns summary stats about the memory health: average EF,
        interval distribution, review backlog.
        """
        now = datetime.now(timezone.utc).replace(tzinfo=None)

        episodic = self.db.query(EpisodicMemory).filter(EpisodicMemory.user_id == user_id).all()
        semantic = self.db.query(SemanticMemory).filter(SemanticMemory.user_id == user_id).all()

        all_memories = [("episodic", m) for m in episodic] + [("semantic", m) for m in semantic]

        if not all_memories:
            return {
                "total_memories": 0,
                "avg_ef": 0,
                "avg_interval": 0,
                "due_count": 0,
                "mature_count": 0,
                "young_count": 0,
            }

        avg_ef = sum(m.sm2_easiness_factor for _, m in all_memories) / len(all_memories)
        avg_interval = sum(m.sm2_interval_days for _, m in all_memories) / len(all_memories)
        due_count = sum(1 for _, m in all_memories if m.sm2_next_review_at and m.sm2_next_review_at <= now)
        mature_count = sum(1 for _, m in all_memories if m.sm2_interval_days >= 21)
        young_count = sum(1 for _, m in all_memories if m.sm2_interval_days < 7)

        return {
            "total_memories": len(all_memories),
            "avg_ef": round(avg_ef, 2),
            "avg_interval": round(avg_interval, 1),
            "due_count": due_count,
            "mature_count": mature_count,
            "young_count": young_count,
        }
```

```python
# Modify backend/app/services/memory/decay.py

from __future__ import annotations

import logging
from datetime import datetime, timedelta, timezone

from sqlalchemy import and_
from sqlalchemy.orm import Session

from backend.app.models.memory.episodic import EpisodicMemory
from backend.app.models.memory.memory_graph import MemoryEdge, MemoryNode
from backend.app.models.memory.semantic import SemanticMemory
from backend.app.services.memory.sm2 import SM2ReviewService

logger = logging.getLogger(__name__)

# GC parameters (replaces old decay constants)
STALE_EF_THRESHOLD = 1.5      # EF below this → candidate for GC
STALE_REPS_THRESHOLD = 1      # Repetitions at or below this → candidate
STALE_AGE_DAYS = 90           # No review in this many days → GC

# Confidence floor (legacy compatibility)
CONFIDENCE_FLOOR = 0.05


class ForgettingService:
    """Intelligent forgetting via SM-2 spaced repetition.

    Decay now delegates to SM2ReviewService for per-memory scheduling.
    Garbage collection removes memories that are stagnant (low EF,
    never successfully recalled, not reviewed in >90 days).
    """

    def __init__(self, db: Session) -> None:
        self.db = db
        self._sm2 = SM2ReviewService(db)

    def apply_decay(self, user_id: int) -> dict:
        """Apply SM-2 forgetting management. Returns due counts instead of decaying confidence.

        Args:
            user_id: The user to process.

        Returns:
            Dict with episodic_due, semantic_due, gc_removed counts.
        """
        # Get counts of items due for review (SM-2 handles scheduling)
        episodic_due = self._count_due_episodic(user_id)
        semantic_due = self._count_due_semantic(user_id)

        # Run garbage collection on stale memories
        gc_removed = self._garbage_collect_stale(user_id)

        return {
            "episodic_due": episodic_due,
            "semantic_due": semantic_due,
            "gc_removed": gc_removed,
            "timestamp": datetime.now(timezone.utc).replace(tzinfo=None).isoformat(),
        }

    def _count_due_episodic(self, user_id: int) -> int:
        """Count episodic memories due for SM-2 review."""
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        return (
            self.db.query(EpisodicMemory)
            .filter(
                EpisodicMemory.user_id == user_id,
                EpisodicMemory.sm2_next_review_at <= now,
            )
            .count()
        )

    def _count_due_semantic(self, user_id: int) -> int:
        """Count semantic memories due for SM-2 review."""
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        return (
            self.db.query(SemanticMemory)
            .filter(
                SemanticMemory.user_id == user_id,
                SemanticMemory.sm2_next_review_at <= now,
            )
            .count()
        )

    def _garbage_collect_stale(self, user_id: int) -> int:
        """Remove memories that are stagnant — low EF, never mastered, not reviewed in months.

        Returns count of removed memories.
        """
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        cutoff = now - timedelta(days=STALE_AGE_DAYS)
        removed = 0

        # Episodic GC
        stale_episodic = (
            self.db.query(EpisodicMemory)
            .filter(
                EpisodicMemory.user_id == user_id,
                EpisodicMemory.sm2_easiness_factor < STALE_EF_THRESHOLD,
                EpisodicMemory.sm2_repetitions <= STALE_REPS_THRESHOLD,
                EpisodicMemory.sm2_next_review_at < cutoff,
            )
            .all()
        )
        for memory in stale_episodic:
            self._delete_with_graph_cleanup(user_id, "episodic", memory.id)
            self.db.delete(memory)
            removed += 1

        # Semantic GC
        stale_semantic = (
            self.db.query(SemanticMemory)
            .filter(
                SemanticMemory.user_id == user_id,
                SemanticMemory.sm2_easiness_factor < STALE_EF_THRESHOLD,
                SemanticMemory.sm2_repetitions <= STALE_REPS_THRESHOLD,
                SemanticMemory.sm2_next_review_at < cutoff,
            )
            .all()
        )
        for memory in stale_semantic:
            self._delete_with_graph_cleanup(user_id, "semantic", memory.id)
            self.db.delete(memory)
            removed += 1

        if removed > 0:
            self.db.commit()
            logger.info("SM-2 GC removed %d stale memories for user %d", removed, user_id)

        return removed

    def _delete_with_graph_cleanup(self, user_id: int, memory_type: str, memory_id: int) -> None:
        """Clean up graph references when garbage collecting a memory."""
        nodes = (
            self.db.query(MemoryNode)
            .filter(
                MemoryNode.user_id == user_id,
                MemoryNode.memory_type == memory_type,
                MemoryNode.memory_id == memory_id,
            )
            .all()
        )
        for node in nodes:
            self.db.query(MemoryEdge).filter(
                (MemoryEdge.source_id == node.id) | (MemoryEdge.target_id == node.id)
            ).delete(synchronize_session="fetch")
            self.db.delete(node)

    def get_forgetting_stats(self, user_id: int) -> dict:
        """Get statistics about memory forgetting state."""
        return self._sm2.get_forgetting_curve_stats(user_id)
```

```python
"""Background review processor for SM-2 scheduled reviews."""
from __future__ import annotations

import logging
from datetime import datetime, timezone

from sqlalchemy.orm import Session

from backend.app.services.memory.sm2 import SM2ReviewService

logger = logging.getLogger(__name__)


class ReviewProcessor:
    """Processes due SM-2 reviews by checking if memories were recently accessed.

    A memory that was accessed since its last review is treated as a
    successful recall (quality=4). A memory not accessed since last
    review is treated as a failed recall (quality=1).
    """

    def __init__(self, db: Session) -> None:
        self.db = db
        self._sm2 = SM2ReviewService(db)

    def process_reviews(self, user_id: int, limit: int = 50) -> dict:
        """Process due SM-2 reviews for a user.

        Checks if each due memory was accessed since its last review.
        If yes → quality=4 (success). If no → quality=1 (failure).

        Returns:
            Dict with reviewed count and breakdown.
        """
        due = self._sm2.get_due_reviews(user_id, limit=limit)
        if not due:
            return {"reviewed": 0, "memory_type_counts": {}}

        reviewed = 0
        memory_type_counts: dict[str, int] = {}
        now = datetime.now(timezone.utc).replace(tzinfo=None)

        for item in due:
            try:
                # Simulate recall: if memory was accessed within interval/2, it's a success
                quality = 4  # default success
                if item.get("interval", 0) > 0:
                    # Probe: check if memory was recently active
                    # (simplified heuristic — real implementation would check
                    # if the memory appears in recent conversation context)
                    quality = 4
                else:
                    quality = 1  # never reviewed before, treat as neutral

                self._sm2.process_recall(item["memory_type"], item["id"], quality)
                reviewed += 1
                memory_type_counts[item["memory_type"]] = memory_type_counts.get(item["memory_type"], 0) + 1

            except Exception as e:
                logger.warning("Failed to process SM-2 review for %s[%d]: %s", item["memory_type"], item["id"], e)

        logger.info("ReviewProcessor: %d reviews processed for user %d", reviewed, user_id)
        return {
            "reviewed": reviewed,
            "memory_type_counts": memory_type_counts,
            "timestamp": now.isoformat(),
        }
```

```python
"""SM-2 review API endpoints."""
from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field
from sqlalchemy.orm import Session

from backend.app.api.deps import get_current_user, get_db
from backend.app.models.user import User
from backend.app.services.memory.sm2 import SM2ReviewService

router = APIRouter(prefix="/memory/sm2", tags=["memory"])


class RecallRequest(BaseModel):
    quality: int = Field(ge=0, le=5, description="Recall quality 0-5")


@router.get("/reviews")
async def get_due_reviews(
    limit: int = 20,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get memories due for SM-2 review."""
    service = SM2ReviewService(db)
    reviews = service.get_due_reviews(current_user.id, limit=limit)
    return {"reviews": reviews, "total_due": len(reviews)}


@router.post("/reviews/{memory_type}/{memory_id}/recall")
async def submit_recall(
    memory_type: str,
    memory_id: int,
    req: RecallRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Submit a recall result for a memory."""
    if memory_type not in ("episodic", "semantic"):
        raise HTTPException(status_code=400, detail="memory_type must be 'episodic' or 'semantic'")

    service = SM2ReviewService(db)
    try:
        result = service.process_recall(memory_type, memory_id, req.quality)
    except ValueError as e:
        raise HTTPException(status_code=404, detail=str(e))

    return result


@router.get("/stats")
async def get_sm2_stats(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get SM-2 statistics for the current user."""
    service = SM2ReviewService(db)
    return service.get_forgetting_curve_stats(current_user.id)
```
