```python
from __future__ import annotations
from typing import Callable, Awaitable

FileChangeCallback = Callable[[str, str, str], Awaitable[None]]
# Args: (event_type: create|modify|delete|move, file_path: str, source_path: str)


class FileWatcherV2:
    """File system watcher using watchdog. Supports multi-callback routing."""

    def __init__(self, ...):
        ...existing init...
        self._knowledge_callbacks: dict[str, list[FileChangeCallback]] = {}
        """Mapping of source_path -> list of async callbacks."""

    def register_knowledge_source(self, source_path: str, callback: FileChangeCallback) -> None:
        """Register a callback for a specific knowledge source path."""
        abs_path = os.path.abspath(source_path)
        self._knowledge_callbacks.setdefault(abs_path, []).append(callback)
        logger.info("Registered knowledge watcher for %s", abs_path)

    def unregister_knowledge_source(self, source_path: str, callback: FileChangeCallback | None = None) -> None:
        """Remove a callback for a source path, or all callbacks for that path."""
        abs_path = os.path.abspath(source_path)
        if callback:
            self._knowledge_callbacks.get(abs_path, []).remove(callback)
            if not self._knowledge_callbacks[abs_path]:
                del self._knowledge_callbacks[abs_path]
        else:
            self._knowledge_callbacks.pop(abs_path, None)
        logger.info("Unregistered knowledge watcher for %s", abs_path)

    def _dispatch_knowledge_event(self, event_type: str, file_path: str) -> None:
        """Dispatch a file event to registered knowledge callbacks.
        
        Matches the file path against registered source paths (longest prefix wins).
        """
        abs_file = os.path.abspath(file_path)
        # Find the matching source path (longest prefix)
        matched_source = None
        matched_len = 0
        for source_path in self._knowledge_callbacks:
            if abs_file.startswith(source_path + os.sep) or abs_file == source_path:
                if len(source_path) > matched_len:
                    matched_source = source_path
                    matched_len = len(source_path)
        
        if matched_source:
            for callback in self._knowledge_callbacks[matched_source]:
                try:
                    import asyncio
                    asyncio.create_task(callback(event_type, abs_file, matched_source))
                except Exception as e:
                    logger.error("Knowledge callback error for %s: %s", abs_file, e)
```

```python
"""WatcherService — manages file watcher lifecycle for knowledge sources."""

from __future__ import annotations

import asyncio
import logging
from typing import Any

from sqlalchemy.orm import Session

from backend.app.services.awareness.file_watcher import FileWatcherV2, FileChangeCallback

logger = logging.getLogger(__name__)

_DEBOUNCE_SECONDS = 2.0
"""Debounce window in seconds — coalesce rapid file events into one sync."""


class WatcherService:
    """Manages file watcher registration and debounced sync for knowledge sources."""

    def __init__(self, db_factory: Any, sync_service_factory: Any) -> None:
        self._db_factory = db_factory
        self._sync_service_factory = sync_service_factory
        self._debounce_timers: dict[str, asyncio.Task | None] = {}
        self._watcher: FileWatcherV2 | None = None

    def set_watcher(self, watcher: FileWatcherV2) -> None:
        """Set the file watcher instance (injected, not created here)."""
        self._watcher = watcher

    def register_source(self, source_id: int, source_path: str) -> None:
        """Register a knowledge source for auto-watching."""
        if not self._watcher:
            logger.warning("FileWatcherV2 not available, cannot watch %s", source_path)
            return
        
        callback = self._make_callback(source_id)
        self._watcher.register_knowledge_source(source_path, callback)
        logger.info("Watcher registered for source %d: %s", source_id, source_path)

    def unregister_source(self, source_id: int, source_path: str) -> None:
        """Unregister a knowledge source from watching."""
        if not self._watcher:
            return
        self._watcher.unregister_knowledge_source(source_path)
        # Cancel any pending debounce
        key = f"{source_id}:{source_path}"
        if key in self._debounce_timers and self._debounce_timers[key]:
            self._debounce_timers[key].cancel()
            del self._debounce_timers[key]
        logger.info("Watcher unregistered for source %d: %s", source_id, source_path)

    def _make_callback(self, source_id: int) -> FileChangeCallback:
        """Create a debounced callback for the given source."""
        
        async def debounced_callback(event_type: str, file_path: str, source_path: str) -> None:
            key = f"{source_id}:{source_path}"
            
            # Cancel existing debounce timer
            if key in self._debounce_timers and self._debounce_timers[key]:
                self._debounce_timers[key].cancel()
            
            # Create new debounce timer
            self._debounce_timers[key] = asyncio.create_task(
                self._debounce_then_sync(source_id, file_path, source_path, event_type)
            )
        
        return debounced_callback

    async def _debounce_then_sync(self, source_id: int, file_path: str, source_path: str, event_type: str) -> None:
        """Wait for debounce window, then sync the file."""
        try:
            await asyncio.sleep(_DEBOUNCE_SECONDS)
            
            db = self._db_factory()
            try:
                service = self._sync_service_factory(db)
                
                if event_type == "deleted":
                    # Remove chunks for this file
                    service.remove_file(source_id, file_path, source_path)
                else:
                    # Sync the single file (created or modified)
                    service.sync_file(source_id, file_path, source_path)
            finally:
                db.close()
        except asyncio.CancelledError:
            pass
        except Exception as e:
            logger.error("Debounced sync failed for %s: %s", file_path, e)
```

```python
def remove_file(self, source_id: int, file_path: str, source_path: str) -> dict[str, Any]:
    """Remove a file's chunks from Qdrant and set its status to deleted."""
    from backend.app.models.knowledge.knowledge_file import KnowledgeFile
    
    rel_path = os.path.relpath(file_path, source_path)
    
    file_record = (
        self.db.query(KnowledgeFile)
        .filter(KnowledgeFile.source_id == source_id, KnowledgeFile.file_path == rel_path)
        .first()
    )
    if not file_record:
        return {"status": "not_found"}
    
    # Delete from Qdrant
    self._delete_file_chunks(source_id, file_record.id)
    
    # Delete DB record
    self.db.delete(file_record)
    self.db.commit()
    
    logger.info("Removed file %s (source %d)", rel_path, source_id)
    return {"status": "deleted", "file_path": rel_path}
```

```python
# In backend/app/main.py or backend/app/events.py startup handler

@app.on_event("startup")
async def startup_watcher_registration():
    """Register watchers for all active knowledge sources."""
    from backend.app.services.knowledge.watcher_service import WatcherService
    from backend.app.services.knowledge.sync_service import KnowledgeSyncService
    from backend.app.core.database import SessionLocal
    from backend.app.models.knowledge.knowledge_source import KnowledgeSource
    from backend.app.services.awareness.file_watcher import file_watcher
    
    db = SessionLocal()
    try:
        watcher_service = WatcherService(SessionLocal, lambda db: KnowledgeSyncService(db))
        watcher_service.set_watcher(file_watcher)
        
        sources = db.query(KnowledgeSource).filter(
            KnowledgeSource.watch_enabled == True,
            KnowledgeSource.status == "idle",
            KnowledgeSource.path.isnot(None),
        ).all()
        
        for source in sources:
            if os.path.isdir(source.path):
                watcher_service.register_source(source.id, source.path)
                logger.info("Registered watcher for source %d: %s", source.id, source.path)
    finally:
        db.close()
```

```python
import os

@pytest.mark.asyncio
async def test_watcher_detects_create(db_session, auth_user):
    """Verify watcher detects file creation and triggers sync."""
    from backend.app.services.knowledge.watcher_service import WatcherService
    from backend.app.models.knowledge.knowledge_source import KnowledgeSource
    
    with tempfile.TemporaryDirectory() as d:
        # Create source
        source = KnowledgeSource(user_id=auth_user.id, name="Watcher Test", path=d)
        db_session.add(source)
        db_session.commit()
        
        # Register watcher
        callbacks = []
        async def fake_callback(event_type, file_path, source_path):
            callbacks.append((event_type, file_path, source_path))
        
        # Simulate a file creation
        new_file = os.path.join(d, "new_file.txt")
        with open(new_file, "w") as f:

        # Manual trigger
        await fake_callback("created", new_file, d)
        
        assert len(callbacks) == 1
        assert callbacks[0][0] == "created"
        assert callbacks[0][1] == new_file
```
