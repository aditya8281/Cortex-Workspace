@router.patch("/{conversation_id}/memory", response_model=ConversationResponse)
def toggle_conversation_memory(
    conversation_id: int,
    data: MemoryToggleRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Enable or disable knowledge memory for a conversation."""
    conversation = (
        db.query(Conversation)
        .filter(Conversation.id == conversation_id, Conversation.user_id == current_user.id)
        .first()
    )
    if not conversation:
        raise HTTPException(status_code=404, detail="Conversation not found")
    
    conversation.memory_enabled = data.memory_enabled
    db.commit()
    db.refresh(conversation)
    return conversation


class MemoryToggleRequest(BaseModel):
    memory_enabled: bool

```python
"""Knowledge search service — RAG retrieval and semantic search."""

from __future__ import annotations

import logging
from typing import Any

logger = logging.getLogger(__name__)

DEFAULT_TOP_K = 5
"""Number of chunks to retrieve for RAG context."""


class KnowledgeSearchService:
    """Service for retrieving knowledge chunks via hybrid search."""

    def __init__(self, db_session: Any) -> None:
        self.db = db_session

    def search(self, query: str, user_id: int, top_k: int = DEFAULT_TOP_K, **filters: Any) -> list[dict[str, Any]]:
        """Hybrid search across a user's knowledge sources.
        
        Returns list of {content, file_path, source_name, score, ...}.
        """
        # 1. Generate query embedding
        query_embedding = self._embed_query(query)
        if not query_embedding:
            logger.warning("Failed to embed query, returning empty results")
            return []
        
        # 2. Search Qdrant
        from backend.app.core.vector_db import qdrant_client
        from qdrant_client.http import models as qmodels
        
        search_filter = self._build_filter(user_id, **filters)
        
        try:
            results = qdrant_client.search(
                collection_name="knowledge_chunks",
                query_vector=query_embedding,
                limit=top_k,
                query_filter=search_filter,
                with_payload=["content", "file_path", "source_id", "chunk_index"],
            )
        except Exception as e:
            logger.error("Qdrant search failed: %s", e)
            return []
        
        # 3. Enrich results with file paths and source names
        enriched = []
        for hit in results:
            payload = hit.payload or {}
            enriched.append({
                "content": payload.get("content", ""),
                "file_path": payload.get("file_path", ""),
                "source_id": payload.get("source_id"),
                "chunk_index": payload.get("chunk_index"),
                "relevance_score": hit.score,
            })
        
        return enriched

    def search_for_rag(self, query: str, user_id: int, top_k: int = DEFAULT_TOP_K) -> str:
        """Search and format results as a RAG context string for LLM injection.
        
        Returns formatted string suitable for prepending to chat prompt.
        """
        results = self.search(query, user_id, top_k=top_k)
        if not results:
            return ""
        
        lines = ["Based on your knowledge base:", ""]
        for r in results:
            lines.append(f"[file: {r['file_path']}]")
            lines.append(r["content"][:1000])
            lines.append("")
        
        return "\n".join(lines)

    def _embed_query(self, text: str) -> list[float] | None:
        """Generate embedding for query text."""
        try:
            from backend.app.services.intelligence.embedding_service import embedding_service
            result = embedding_service.embed([text])
            return result[0] if result else None
        except Exception as e:
            logger.error("Query embedding failed: %s", e)
            return None

    def _build_filter(self, user_id: int, **filters: Any) -> Any:
        """Build Qdrant filter from kwargs."""
        from qdrant_client.http import models as qmodels
        
        conditions = [
            qmodels.FieldCondition(key="user_id", match=qmodels.MatchValue(value=user_id)),
        ]
        
        if filters.get("source_id"):
            conditions.append(
                qmodels.FieldCondition(key="source_id", match=qmodels.MatchValue(value=filters["source_id"]))
            )
        if filters.get("file_type"):
            conditions.append(
                qmodels.FieldCondition(key="file_type", match=qmodels.MatchValue(value=filters["file_type"]))
            )
        
        return qmodels.Filter(must=conditions) if conditions else None
```

MAX_RAG_CHARS = 4000
"""Max characters for RAG context to avoid overflowing LLM context window."""


def _truncate_rag_context(context: str, max_chars: int = MAX_RAG_CHARS) -> str:
    """Truncate RAG context to fit within allowed size.
    
    Preserves file headers and early chunks; truncates middle.
    """
    if len(context) <= max_chars:
        return context
    
    # Find the last file boundary within limit
    truncated = context[:max_chars]
    last_boundary = truncated.rfind("[file:")
    if last_boundary > max_chars // 2:
        truncated = truncated[:last_boundary]
    
    return truncated + "\n[context truncated due to length]\n"
```

// ChatMemoryToggle.tsx
"use client";

import { useState } from "react";

interface ChatMemoryToggleProps {
  conversationId: number;
  initialEnabled: boolean;
}

export function ChatMemoryToggle({ conversationId, initialEnabled }: ChatMemoryToggleProps) {
  const [enabled, setEnabled] = useState(initialEnabled);
  const [loading, setLoading] = useState(false);

  const toggle = async () => {
    setLoading(true);
    try {
      const res = await fetch(`/api/v1/conversations/${conversationId}/memory`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ memory_enabled: !enabled }),
      });
      if (res.ok) {
        setEnabled(!enabled);
      }
    } finally {
      setLoading(false);
    }
  };

  return (
    <button
      onClick={toggle}
      disabled={loading}
      className={`px-3 py-1 rounded text-sm flex items-center gap-2 ${
        enabled
          ? "bg-accent-500/20 text-accent-400 border border-accent-500/30"
          : "bg-surface-800 text-text-muted border border-surface-600"
      }`}
    >
      <span className={`w-2 h-2 rounded-full ${enabled ? "bg-accent-400 animate-pulse" : "bg-surface-500"}`} />
      {enabled ? "Memory On" : "Memory Off"}
    </button>
  );
}
```
