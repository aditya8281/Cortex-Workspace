"""UserPreference model — stores observed, explicit, and inferred preferences."""

from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

from sqlalchemy import Column, DateTime, Float, ForeignKey, Integer, String, UniqueConstraint
from sqlalchemy.dialects.postgresql import JSONB

from backend.app.models.base import Base

JSON = JSONB


class UserPreference(Base):
    __tablename__ = "user_preferences"

    id: int = Column(Integer, primary_key=True, autoincrement=True)
    user_id: int = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    category: str = Column(String(100), nullable=False)
    key: str = Column(String(100), nullable=False)
    value: dict[str, Any] = Column(JSON, nullable=False, default=dict)
    confidence: float = Column(Float, default=0.5, nullable=False)
    source: str = Column(String(20), default="observed", nullable=False)  # explicit | observed | inferred
    observation_count: int = Column(Integer, default=0, nullable=False)
    last_observed: datetime | None = Column(DateTime(timezone=True), default=None)
    created_at: datetime = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    updated_at: datetime = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc),
                                   onupdate=lambda: datetime.now(timezone.utc))

    __table_args__ = (
        UniqueConstraint("user_id", "category", "key", name="uq_user_preference"),
    )

    def strengthen(self) -> None:
        self.observation_count = (self.observation_count or 0) + 1
        self.confidence = min(1.0, (self.confidence or 0.5) + 0.1)
        self.last_observed = datetime.now(timezone.utc)

    def weaken(self) -> None:
        self.confidence = max(0.0, (self.confidence or 0.5) - 0.1)
        self.last_observed = datetime.now(timezone.utc)

    @property
    def is_high_confidence(self) -> bool:
        return (self.confidence or 0.0) >= 0.7

    @property
    def source_priority(self) -> int:
        return {"explicit": 3, "observed": 2, "inferred": 1}.get(self.source, 0)
```

"""Habit model — stores recurring user behaviors."""

from __future__ import annotations

from datetime import datetime, timezone

from sqlalchemy import Column, DateTime, ForeignKey, Integer, String

from backend.app.models.base import Base

MIN_OCCURRENCES = 5


class Habit(Base):
    __tablename__ = "habits"

    id: int = Column(Integer, primary_key=True, autoincrement=True)
    user_id: int = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    habit_name: str = Column(String(200), nullable=False)
    description: str | None = Column(String(500), nullable=True)
    trigger: str | None = Column(String(200), nullable=True)
    action: str = Column(String(200), nullable=False)
    frequency: str = Column(String(20), default="irregular", nullable=False)
    # daily | weekly | hourly | irregular
    occurrences: int = Column(Integer, default=1, nullable=False)
    first_observed: datetime = Column(DateTime(timezone=True), nullable=True)
    last_observed: datetime = Column(DateTime(timezone=True), nullable=True)
    created_at: datetime = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))

    def observe(self) -> None:
        self.occurrences = (self.occurrences or 0) + 1
        self.last_observed = datetime.now(timezone.utc)
        self._classify_frequency()

    def _classify_frequency(self) -> None:
        occ = self.occurrences or 0
        if occ >= 30:
            self.frequency = "daily"
        elif occ >= 7:
            self.frequency = "weekly"
        elif occ >= 3:
            self.frequency = "hourly"
        else:
            self.frequency = "irregular"

    @property
    def is_confirmed_habit(self) -> bool:
        return (self.occurrences or 0) >= MIN_OCCURRENCES

    @property
    def confidence(self) -> float:
        return min(1.0, (self.occurrences or 0) / (MIN_OCCURRENCES * 2))
```

"""Pattern model — stores temporal, behavioral, and preference patterns."""

from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

from sqlalchemy import Column, DateTime, Float, ForeignKey, Integer, String, Text
from sqlalchemy.dialects.postgresql import JSONB

from backend.app.models.base import Base

MIN_EVIDENCE = 3
JSON = JSONB


class Pattern(Base):
    __tablename__ = "patterns"

    id: int = Column(Integer, primary_key=True, autoincrement=True)
    user_id: int = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    pattern_type: str = Column(String(30), nullable=False)
    # temporal | behavioral | preference
    description: str = Column(Text, nullable=False)
    evidence: list[dict[str, Any]] = Column(JSON, nullable=False, default=list)
    confidence: float = Column(Float, default=0.3, nullable=False)
    first_seen: datetime = Column(DateTime(timezone=True), nullable=True)
    last_seen: datetime = Column(DateTime(timezone=True), nullable=True)
    occurrences: int = Column(Integer, default=1, nullable=False)

    def add_evidence(self, evidence_item: dict[str, Any]) -> None:
        self.evidence = list(self.evidence or []) + [evidence_item]
        self.occurrences = (self.occurrences or 0) + 1
        self.confidence = min(1.0, (self.confidence or 0.3) + 0.1)
        self.last_seen = datetime.now(timezone.utc)

    @property
    def is_confident(self) -> bool:
        return (self.confidence or 0) >= 0.5 and (self.occurrences or 0) >= MIN_EVIDENCE
