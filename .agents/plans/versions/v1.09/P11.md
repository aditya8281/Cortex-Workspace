"""Habits API."""
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from backend.app.core.db import get_current_user, get_db
from backend.app.models.interaction.user import User
from backend.app.schemas.learning.habit import HabitResponse
from backend.app.services.learning.habit_service import HabitDetectionService

router = APIRouter(prefix="/habits", tags=["learning-habits"])


@router.get("", response_model=list[HabitResponse])
def list_habits(
    confirmed_only: bool = Query(True),
    frequency: str | None = Query(None),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    service = HabitDetectionService(db)
    if frequency:
        return service.get_habits_by_frequency(current_user.id, frequency)
    return service.get_habits(current_user.id, confirmed_only=confirmed_only)


@router.post("/observe", response_model=HabitResponse)
def observe_action(
    action: str,
    trigger: str | None = None,
    habit_name: str | None = None,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    service = HabitDetectionService(db)
    return service.observe_action(current_user.id, action, trigger, habit_name)

// api.ts
const API_PREFIX = "/api/v1/learning";

export interface Preference {
  id: number;
  user_id: number;
  category: string;
  key: string;
  value: Record<string, unknown>;
  confidence: number;
  source: "explicit" | "observed" | "inferred";
  observation_count: number;
  last_observed: string | null;
  created_at: string;
}

export interface WorkflowPattern {
  id: number;
  user_id: number;
  pattern_name: string;
  description: string | null;
  steps: string[];
  context: string | null;
  frequency: number;
  confidence: number;
  last_observed: string | null;
}

export interface Habit {
  id: number;
  user_id: number;
  habit_name: string;
  action: string;
  trigger: string | null;
  frequency: string;
  occurrences: number;
  is_confirmed_habit: boolean;
}

export interface LearningEvent {
  id: number;
  event_type: string;
  delta: number;
  applied: boolean;
  created_at: string;
}

export interface UserProfile {
  response_style: string;
  suggestion_tolerance: string;
  active_hours: number[] | null;
  expertise_level: string;
  preferred_tools: string[];
}

export const learningApi = {
  // Preferences
  getPreferences: (category?: string, minConfidence?: number): Promise<Preference[]> => {
    const params = new URLSearchParams();
    if (category) params.set("category", category);
    if (minConfidence) params.set("min_confidence", String(minConfidence));
    return fetch(`${API_PREFIX}/preferences?${params}`).then((r) => r.json());
  },

  recordPreference: (category: string, key: string, value: Record<string, unknown>, source = "observed"): Promise<Preference> =>
    fetch(`${API_PREFIX}/preferences`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ category, key, value, source }),
    }).then((r) => r.json()),

  // Workflows
  observeWorkflow: (steps: string[], context?: string): Promise<WorkflowPattern> =>
    fetch(`${API_PREFIX}/workflows/observe`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ steps, context }),
    }).then((r) => r.json()),

  getCommonWorkflows: (limit = 10): Promise<WorkflowPattern[]> =>
    fetch(`${API_PREFIX}/workflows/common?limit=${limit}`).then((r) => r.json()),

  // Habits
  getHabits: (confirmedOnly = true): Promise<Habit[]> =>
    fetch(`${API_PREFIX}/habits?confirmed_only=${confirmedOnly}`).then((r) => r.json()),

  observeAction: (action: string, trigger?: string): Promise<Habit> =>
    fetch(`${API_PREFIX}/habits/observe?action=${encodeURIComponent(action)}${trigger ? `&trigger=${encodeURIComponent(trigger)}` : ""}`, {
      method: "POST",
    }).then((r) => r.json()),

  // Feedback
  recordFeedback: (eventType: string, context?: Record<string, unknown>, applyToPreference = false): Promise<LearningEvent> =>
    fetch(`${API_PREFIX}/feedback`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ event_type: eventType, context, apply_to_preference: applyToPreference }),
    }).then((r) => r.json()),

  getLearningRate: (): Promise<{ learning_rate: number }> =>
    fetch(`${API_PREFIX}/feedback/learning-rate`).then((r) => r.json()),

  // Profile
  getUserProfile: (): Promise<UserProfile> =>
    fetch(`${API_PREFIX}/personalization/profile`).then((r) => r.json()),

  // Anomaly detection
  detectAnomaly: (action: string, currentHour?: number, currentVolume?: number): Promise<Record<string, unknown>> =>
    fetch(`${API_PREFIX}/anomalies/detect`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ action, current_hour: currentHour, current_volume: currentVolume }),
    }).then((r) => r.json()),

  // Improvement
  getImprovementSummary: (): Promise<Record<string, unknown>> =>
    fetch(`${API_PREFIX}/improvement/summary`).then((r) => r.json()),
};
```

// LearningDashboard.tsx
"use client";

import { useEffect, useState } from "react";
import { Preference, UserProfile, learningApi } from "../api";
import { PreferencePanel } from "./PreferencePanel";
import { AdaptationProfile } from "./AdaptationProfile";
import { WorkflowTimeline } from "./WorkflowTimeline";
import { PatternDisplay } from "./PatternDisplay";

export function LearningDashboard() {
  const [preferences, setPreferences] = useState<Preference[]>([]);
  const [profile, setProfile] = useState<UserProfile | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    setLoading(true);
    try {
      const [prefs, prof] = await Promise.all([
        learningApi.getPreferences(),
        learningApi.getUserProfile(),
      ]);
      setPreferences(prefs);
      setProfile(prof);
    } catch (err) {
      console.error("Failed to load learning data:", err);
    } finally {
      setLoading(false);
    }
  };

  if (loading) return <div className="p-4 text-text-muted">Loading learning data...</div>;

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-6">
      <h1 className="text-2xl font-bold text-text-primary">Learning & Adaptation</h1>

      {profile && <AdaptationProfile profile={profile} />}
      {preferences.length > 0 && <PreferencePanel preferences={preferences} />}
      <WorkflowTimeline />
      <PatternDisplay />

      <div className="text-center text-sm text-text-muted py-4">
        Learning data stays local. Preferences adapt with repeated observation.
      </div>
    </div>
  );
}
```

"use client";

import { Preference } from "../api";

interface Props {
  preferences: Preference[];
}

export function PreferencePanel({ preferences }: Props) {
  if (preferences.length === 0) {
    return (
      <div className="p-4 bg-surface-800 rounded-lg border border-surface-700 text-text-muted text-center">
        No preferences learned yet. Preferences grow with repeated observations.
      </div>
    );
  }

  return (
    <div className="space-y-3">
      <h2 className="text-lg font-semibold text-text-primary">Learned Preferences</h2>
      {preferences.map((pref) => (
        <div key={pref.id} className="p-3 bg-surface-800 rounded-lg border border-surface-700">
          <div className="flex items-center justify-between">
            <span className="font-medium text-text-primary">{pref.key}</span>
            <span className={`text-xs px-2 py-0.5 rounded-full ${
              pref.source === "explicit" ? "bg-purple-900/30 text-purple-400" :
              pref.source === "observed" ? "bg-blue-900/30 text-blue-400" :
              "bg-surface-600 text-text-muted"
            }`}>
              {pref.source}
            </span>
          </div>
          <div className="text-sm text-text-muted mt-1">{pref.category}</div>

            <span className="text-xs text-text-muted">{Math.round(pref.confidence * 100)}%</span>

def test_unauthorized_access(client):
    """Verify all endpoints require auth."""
    endpoints = [
        ("GET", "/api/v1/learning/preferences"),
        ("POST", "/api/v1/learning/preferences"),
        ("GET", "/api/v1/learning/habits"),
        ("POST", "/api/v1/learning/feedback"),
        ("GET", "/api/v1/learning/personalization/profile"),
