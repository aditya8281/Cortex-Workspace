### Task 1: Create Learning API Router

**File:** `backend/app/api/v1/learning/router.py`
**Purpose:** Expose all learning capabilities via REST endpoints. Route order: specific routes before parameterized (e.g., `/patterns/type/{type}` after `/patterns`). All endpoints require authentication and enforce user ownership.

```python
# backend/app/api/v1/learning/router.py

from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session
from typing import Optional
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.learning.preference import PreferenceLearningService
from backend.app.services.learning.workflow import WorkflowLearningService
from backend.app.services.learning.habit import HabitDetectionService, BehaviorAdaptationService, FeedbackLearningService
from backend.app.services.learning.patterns import PatternRecognitionService
from backend.app.services.learning.personalization import PersonalizationService
from backend.app.services.learning.refinement import KnowledgeRefinementService
from backend.app.services.learning.anomaly import AnomalyDetectionService
from backend.app.schemas.learning.preference import PreferenceResponse, CreatePreferenceRequest
from backend.app.schemas.learning.workflow import WorkflowPatternResponse, ObserveWorkflowRequest
from backend.app.schemas.learning.habit import HabitResponse, ObserveHabitRequest
from backend.app.schemas.learning.event import LearningEventResponse, RecordFeedbackRequest
from backend.app.schemas.learning.pattern import PatternResponse

router = APIRouter(prefix="/learning", tags=["learning"])


# --- Preferences ---

@router.get("/preferences", response_model=list[PreferenceResponse])
async def get_preferences(
    category: Optional[str] = None,
    min_confidence: float = Query(0.0, ge=0.0, le=1.0),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get user preferences with optional filtering."""
    service = PreferenceLearningService(db)
    return await service.get_preferences(current_user.id, category, min_confidence)


@router.get("/preferences/ready", response_model=list[PreferenceResponse])
async def get_adaptation_ready_preferences(
    category: Optional[str] = None,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get preferences ready for adaptation (confidence ≥ 0.6)."""
    service = PreferenceLearningService(db)
    return await service.get_adaptation_ready(current_user.id, category)


@router.get("/preferences/{category}/{key}")
async def get_preference(
    category: str,
    key: str,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get a specific preference."""
    service = PreferenceLearningService(db)
    result = await service.get_preference(current_user.id, category, key)
    if not result:
        raise HTTPException(status_code=404, detail="Preference not found")
    return result


@router.post("/preferences", response_model=PreferenceResponse)
async def record_preference(
    request: CreatePreferenceRequest,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Record a user preference (explicit, observed, or inferred)."""
    service = PreferenceLearningService(db)
    return await service.record_preference(
        current_user.id, request.category, request.key,
        request.value, request.source
    )


@router.delete("/preferences/{category}/{key}")
async def delete_preference(
    category: str,
    key: str,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Delete a specific preference."""
    service = PreferenceLearningService(db)
    deleted = await service.delete_preference(current_user.id, category, key)
    if not deleted:
        raise HTTPException(status_code=404, detail="Preference not found")
    return {"deleted": True}


@router.post("/preferences/reset")
async def reset_all_preferences(
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Reset all preferences for the current user."""
    service = PreferenceLearningService(db)
    count = await service.reset_all(current_user.id)
    return {"deleted": count}

@router.get("/workflows/common", response_model=list[WorkflowPatternResponse])

@router.get("/workflows/suggest")
async def suggest_next_step(
    steps: str = Query(..., description="JSON array of current steps"),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Suggest the next workflow step based on learned patterns."""
    import json
    try:
        current_steps = json.loads(steps)
    except json.JSONDecodeError:
        raise HTTPException(status_code=400, detail="Steps must be valid JSON array")
    
    service = WorkflowLearningService(db)
    suggestion = await service.suggest_next_step(current_user.id, current_steps)
    return {"suggestion": suggestion}
```

```typescript
// frontend/features/learning/api.ts

import { apiClient } from '@/shared/api/client';

export interface Preference {
  id: number;
  user_id: number;
  category: string;
  key: string;
  value: string | null;
  confidence: number;
  source: 'explicit' | 'observed' | 'inferred';
  observation_count: number;
}

export interface WorkflowPattern {
  id: number;
  pattern_name: string;
  steps: string[];
  frequency: number;
  confidence: number;
  context: string | null;
}

export interface Habit {
  id: number;
  habit_name: string;
  action: string;
  trigger: string | null;
  frequency: string | null;
  occurrences: number;
}

export interface Pattern {
  id: number;
  pattern_type: string;
  description: string;
  confidence: number;
  occurrences: number;
  evidence: string;
}

export interface LearningEvent {
  id: number;
  event_type: string;
  delta: number | null;
  context: string | null;
  created_at: string;
}

export interface UserProfile {
  user_id: number;
  response_style: string;
  preferred_topics: string[];
  active_hours: string;
  expertise_level: string;
  suggestion_tolerance: string;
}

export const learningApi = {
  // Preferences
  getPreferences: (category?: string, minConfidence: number = 0) =>
    apiClient.get<Preference[]>(`/learning/preferences${category ? '?category=' + category : ''}${minConfidence > 0 ? '&min_confidence=' + minConfidence : ''}`),
  getAdaptationReady: (category?: string) =>
    apiClient.get<Preference[]>(`/learning/preferences/ready${category ? '?category=' + category : ''}`),
  getPreference: (category: string, key: string) =>
    apiClient.get(`/learning/preferences/${category}/${key}`),
  recordPreference: (category: string, key: string, value: string, source: string = 'observed') =>
    apiClient.post<Preference>('/learning/preferences', { category, key, value, source }),
  deletePreference: (category: string, key: string) =>
    apiClient.delete(`/learning/preferences/${category}/${key}`),
  resetPreferences: () => apiClient.post('/learning/preferences/reset'),

  // Workflows
  getWorkflows: (limit: number = 10) =>
    apiClient.get<WorkflowPattern[]>(`/learning/workflows/common?limit=${limit}`),
  observeWorkflow: (steps: string[], context?: string) =>
    apiClient.post<WorkflowPattern>('/learning/workflows/observe', { steps, context }),
  suggestNextStep: (currentSteps: string[]) =>
    apiClient.get(`/learning/workflows/suggest?steps=${JSON.stringify(currentSteps)}`),

  // Habits
  getHabits: (minOccurrences?: number) =>
    apiClient.get<Habit[]>(`/learning/habits${minOccurrences ? '?min_occurrences=' + minOccurrences : ''}`),
  getHabitsByFrequency: (frequency: string) =>
    apiClient.get<Habit[]>(`/learning/habits/frequency/${frequency}`),
  checkHabit: (action: string) =>
    apiClient.get<{ action: string; is_habit: boolean }>(`/learning/habits/check/${action}`),
  observeHabit: (action: string, trigger?: string) =>
    apiClient.post<Habit>('/learning/habits/observe', { action, trigger }),

  // Patterns
  getPatterns: (minConfidence: number = 0.5) =>
    apiClient.get<Pattern[]>(`/learning/patterns?min_confidence=${minConfidence}`),
  getActionablePatterns: () => apiClient.get<Pattern[]>('/learning/patterns/actionable'),
  getPatternsByType: (type: string) =>
    apiClient.get<Pattern[]>(`/learning/patterns/type/${type}`),

  // Feedback
  getFeedbackHistory: (limit: number = 20) =>
    apiClient.get<LearningEvent[]>(`/learning/feedback/history?limit=${limit}`),
  getLearningRate: () =>
    apiClient.get<{ learning_rate: number }>('/learning/feedback/learning-rate'),
  recordFeedback: (eventType: string, context?: Record<string, unknown>) =>
    apiClient.post<LearningEvent>('/learning/feedback', { event_type: eventType, context }),

  // Personalization
  getProfile: () => apiClient.get<UserProfile>('/learning/personalization/profile'),
  personalizeContent: (content: string) =>
    apiClient.post(`/learning/personalization/personalize?content=${encodeURIComponent(content)}`),

  // Refinement
  getRefinementStats: () => apiClient.get('/learning/refinement/stats'),
  suggestRefinements: () => apiClient.get('/learning/refinement/suggestions'),

  // Anomaly
  getBaselines: () => apiClient.get('/learning/anomalies/baselines'),
  detectAnomaly: (eventType: string, eventData: Record<string, unknown>) =>
    apiClient.post(`/learning/anomalies/detect?event_type=${eventType}`, eventData),

  // Adaptation
  getAdaptationProfile: () => apiClient.get<UserProfile>('/learning/adaptation/profile'),
  adaptSuggestions: (suggestions: string[]) =>
    apiClient.post('/learning/adaptation/suggestions', suggestions),
};
```

## Cross-Domain Integration Notes

- **v1.02 (Backend Architecture)** — Router follows established pattern. Auth dependencies reused.
- **v1.03 (Memory Foundation)** — Refinement stats reference memory confidence. Feedback history stored in learning_events.
- **v1.07 (Memory Evolution)** — Learning patterns become graph nodes. Preferences link to memories via cross-domain edges.
- **v1.08 (Awareness Expansion)** — Awareness events feed into preference/workflow/habit learning. API endpoints for awareness coexist with learning endpoints.

## Security Considerations

1. **Authentication required** — All endpoints use `Depends(get_current_user)`.
2. **Ownership enforcement** — All queries filter by `current_user.id`.
3. **Input validation** — Pydantic schemas validate all inputs. Min/max constraints on query params.
4. **Rate limiting** — Feedback recording should be rate-limited (10/minute per user).
5. **Data exposure** — Learning data is user-specific. Never expose cross-user data.

---

## Performance Considerations

1. **Preference queries** — O(1) via unique constraint index.
2. **Habit queries** — O(n) where n=habits per user (capped at 100).
3. **Pattern queries** — O(n) with confidence index. Default limit 100.
4. **Feedback history** — Limited to 100 most recent. No full table scan.
5. **Profile building** — Aggregate query across 3 tables. Cache for 5 minutes.
