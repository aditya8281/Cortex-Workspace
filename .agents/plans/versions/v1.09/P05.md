"""KnowledgeGraphBuilder — extracts relationships between files."""

from __future__ import annotations

import itertools
import logging
import os
import re
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.knowledge.knowledge_file import KnowledgeFile
from backend.app.models.knowledge.knowledge_graph_edge import KnowledgeGraphEdge

logger = logging.getLogger(__name__)

# Cross-reference patterns for common file types
_CROSS_REF_PATTERNS: dict[str, list[str]] = {
    ".py": [r'import\s+(\S+)', r'from\s+(\S+)\s+import'],
    ".js": [r'require\([\'"](\S+?)[\'"]\)', r'from\s+[\'"](\S+?)[\'"]'],
    ".ts": [r'from\s+[\'"](\S+?)[\'"]', r'import\s+.*\s+from\s+[\'"](\S+?)[\'"]'],
    ".md": [r'\[.*?\]\((\S+?\.\w+)\)', r'!\[.*?\]\((\S+?\.\w+)\)'],
    ".html": [r'(?:src|href)=[\'"](\S+?)[\'"]'],
}


class KnowledgeGraphBuilder:
    """Extracts relationships between KnowledgeFiles."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def build_graph_for_source(self, source_id: int) -> int:
        """Build all edges for a source. Returns number of edges created."""
        files = (
            self.db.query(KnowledgeFile)
            .filter(KnowledgeFile.source_id == source_id, KnowledgeFile.status == "indexed")
            .all()
        )
        if len(files) < 2:
            return 0
        
        edges_added = 0
        edges_added += self._extract_shared_extension(files, source_id)
        edges_added += self._extract_cross_references(files, source_id)
        edges_added += self._extract_embedding_proximity(files, source_id)
        edges_added += self._extract_co_occurrence(files, source_id)
        
        return edges_added

    def _extract_shared_extension(self, files: list[KnowledgeFile], source_id: int) -> int:
        """Connect files that share the same extension (with >2 count)."""
        counts: dict[str, list[KnowledgeFile]] = {}
        for f in files:
            ext = os.path.splitext(f.file_path)[1]
            if ext:
                counts.setdefault(ext, []).append(f)
        
        added = 0
        threshold = 3  # Only create edges for extensions with 3+ files
        for ext, group in counts.items():
            if len(group) >= threshold:
                for a, b in self._pairwise(group, limit=10):
                    edge = KnowledgeGraphEdge(
                        source_id=source_id,
                        file_id_a=a.id,
                        file_id_b=b.id,
                        relationship_type="shared_extension",
                        strength=0.3,
                        metadata={"extension": ext},
                    )
                    self.db.add(edge)
                    added += 1
        
        self.db.commit()
        return added

    def _extract_cross_references(self, files: list[KnowledgeFile], source_id: int) -> int:
        """Find files that reference each other via imports/links."""
        ref_map: dict[int, set[int]] = {}
        name_to_id: dict[str, int] = {}
        
        for f in files:
            name_to_id[os.path.splitext(os.path.basename(f.file_path))[0]] = f.id
        
        added = 0
        for f in files:
            ext = os.path.splitext(f.file_path)[1]
            patterns = _CROSS_REF_PATTERNS.get(ext, [])
            if not patterns:
                continue
            
            refs = set()
            for pattern in patterns:
                try:
                    # Read file content from DB or filesystem
                    content = self._read_file_content(f, source_id)
                    if content:
                        matches = re.findall(pattern, content)
                        for m in matches:
                            ref_name = m.split(".")[0] if "." in m else m
                            if ref_name in name_to_id and name_to_id[ref_name] != f.id:
                                refs.add(name_to_id[ref_name])
                except Exception:
                    continue
            
            ref_map[f.id] = refs
        
        for file_id, refs in ref_map.items():
            for ref_id in refs:
                edge = KnowledgeGraphEdge(
                    source_id=source_id,
                    file_id_a=min(file_id, ref_id),
                    file_id_b=max(file_id, ref_id),
                    relationship_type="cross_reference",
                    strength=0.6,
                    metadata=None,
                )
                self.db.add(edge)
                added += 1
        
        self.db.commit()
        return added

    def _extract_embedding_proximity(self, files: list[KnowledgeFile], source_id: int) -> int:
        """Connect files whose chunks have high embedding similarity.
        
        Uses centroid of each file's chunks and finds nearest neighbors.
        """
        import numpy as np
        
        try:
            from backend.app.core.vector_db import qdrant_client
            from qdrant_client.http import models as qmodels
        except ImportError:
            return 0
        
        file_centroids: dict[int, list[float]] = {}
        for f in files:
            try:
                points = qdrant_client.scroll(
                    collection_name="knowledge_chunks",
                    scroll_filter=qmodels.Filter(
                        must=[qmodels.FieldCondition(key="file_id", match=qmodels.MatchValue(value=f.id))]
                    ),
                    limit=100,
                    with_vectors=True,
                )[0]
                if points:
                    vecs = [p.vector for p in points if p.vector]
                    if vecs:
                        file_centroids[f.id] = list(np.mean(vecs, axis=0))
            except Exception:
                continue
        
        added = 0
        threshold = 0.85
        
        file_items = list(file_centroids.items())
        for i in range(len(file_items)):
            for j in range(i + 1, len(file_items)):
                fid_a, vec_a = file_items[i]
                fid_b, vec_b = file_items[j]
                similarity = float(np.dot(vec_a, vec_b) / (
                    np.linalg.norm(vec_a) * np.linalg.norm(vec_b) + 1e-10
                ))
                if similarity >= threshold:
                    edge = KnowledgeGraphEdge(
                        source_id=source_id,
                        file_id_a=fid_a,
                        file_id_b=fid_b,
                        relationship_type="embedding_proximity",
                        strength=similarity,
                        metadata=None,
                    )
                    self.db.add(edge)
                    added += 1
        
        self.db.commit()
        return added

    def _extract_co_occurrence(self, files: list[KnowledgeFile], source_id: int) -> int:
        """Connect files that share frequently co-occurring terms.
        
        Extracts noun phrases (via simple capitalized-word heuristic) 
        and connects files sharing >2 significant terms.
        """
        import collections
        
        term_to_files: dict[str, set[int]] = {}
        stop_terms = {"This", "That", "The", "A", "An", "It", "We", "They", "I", "You", "He", "She", "Also"}
        
        for f in files:
            content = self._read_file_content(f, source_id)
            if not content:
                continue
            
            # Extract capitalized terms as simple topic signal
            terms = set(re.findall(r'\b[A-Z][a-z]{2,}\b', content))
            terms -= stop_terms
            
            for term in terms:
                term_to_files.setdefault(term, set()).add(f.id)
        
        # Only consider terms that appear in 2+ files
        significant_terms = {t for t, fs in term_to_files.items() if len(fs) >= 2}
        
        added = 0
        for term in significant_terms:
            files_with_term = list(term_to_files[term])
            for fid_a, fid_b in itertools.combinations(files_with_term, 2):
                existing = (
                    self.db.query(KnowledgeGraphEdge)
                    .filter(
                        KnowledgeGraphEdge.file_id_a == min(fid_a, fid_b),
                        KnowledgeGraphEdge.file_id_b == max(fid_a, fid_b),
                        KnowledgeGraphEdge.relationship_type == "co_occurrence",
                    )
                    .first()
                )
                if existing:
                    existing.strength = min(1.0, existing.strength + 0.1)
                else:
                    edge = KnowledgeGraphEdge(
                        source_id=source_id,
                        file_id_a=min(fid_a, fid_b),
                        file_id_b=max(fid_a, fid_b),
                        relationship_type="co_occurrence",
                        strength=0.4,
                        metadata={"term": term},
                    )
                    self.db.add(edge)
                added += 1
        
        self.db.commit()
        return added

    def _read_file_content(self, file: KnowledgeFile, source_id: int) -> str | None:
        """Try to read file content from DB chunks or filesystem."""
        from backend.app.models.knowledge.knowledge_chunk import KnowledgeChunk
        
        chunks = (
            self.db.query(KnowledgeChunk)
            .filter(KnowledgeChunk.file_id == file.id)
            .order_by(KnowledgeChunk.chunk_index)
            .limit(5)
            .all()
        )
        if chunks:
            return " ".join(c.content for c in chunks)
        return None

    @staticmethod
    def _pairwise(items: list[Any], limit: int | None = None) -> list[tuple[Any, Any]]:
        """Generate unique pairs with optional limit."""
        pairs = []
        for i in range(len(items)):
            for j in range(i + 1, len(items)):
                if limit and len(pairs) >= limit:
                    return pairs
                pairs.append((items[i], items[j]))
        return pairs

"""Knowledge Graph API."""
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from backend.app.core.db import get_current_user, get_db
from backend.app.models.interaction.user import User
from backend.app.models.knowledge.knowledge_graph_edge import KnowledgeGraphEdge
from backend.app.models.knowledge.knowledge_file import KnowledgeFile
from backend.app.schemas.knowledge.knowledge_graph import KnowledgeGraphResponse, GraphEdgeResponse

router = APIRouter(prefix="/graph", tags=["knowledge-graph"])


@router.get("/{source_id}", response_model=KnowledgeGraphResponse)
def get_source_graph(
    source_id: int,
    relationship_type: str | None = Query(None, description="Filter by relationship type"),
    min_strength: float = Query(0.0, ge=0.0, le=1.0),
    limit: int = Query(200, le=1000),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get the knowledge graph for a source (list of edges with file info)."""
    query = (
        db.query(KnowledgeGraphEdge)
        .filter(KnowledgeGraphEdge.source_id == source_id)
    )
    
    if relationship_type:
        query = query.filter(KnowledgeGraphEdge.relationship_type == relationship_type)
    if min_strength > 0:
        query = query.filter(KnowledgeGraphEdge.strength >= min_strength)
    
    edges = query.order_by(KnowledgeGraphEdge.strength.desc()).limit(limit).all()
    
    # Collect unique files referenced
    file_ids: set[int] = set()
    for e in edges:
        file_ids.add(e.file_id_a)
        file_ids.add(e.file_id_b)
    
    files = (
        db.query(KnowledgeFile)
        .filter(KnowledgeFile.id.in_(file_ids))
        .all()
    )
    file_map = {f.id: f.file_path for f in files}
    
    edge_responses = [
        GraphEdgeResponse(
            source=file_map.get(e.file_id_a, f"file:{e.file_id_a}"),
            target=file_map.get(e.file_id_b, f"file:{e.file_id_b}"),
            type=e.relationship_type,
            strength=e.strength,
            metadata=e.metadata,
        )
        for e in edges
    ]
    
    return KnowledgeGraphResponse(
        source_id=source_id,
        edges=edge_responses,
        total=len(edge_responses),
    )


@router.post("/{source_id}/rebuild")
def rebuild_graph(
    source_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Delete all edges for a source and rebuild from scratch."""
    from backend.app.services.knowledge.graph_builder import KnowledgeGraphBuilder
    
    # Clear existing edges
    db.query(KnowledgeGraphEdge).filter(KnowledgeGraphEdge.source_id == source_id).delete()
    db.commit()
    
    # Rebuild
    builder = KnowledgeGraphBuilder(db)
    edges = builder.build_graph_for_source(source_id)
    
    return {"status": "success", "edges_created": edges}

    self, query: str, user_id: int, top_k: int = DEFAULT_TOP_K,
    source_id: int | None = None, file_type: str | None = None,
    date_from: str | None = None, date_to: str | None = None,

            embedding_service=self._embedding_service,

            query=query,
            collection="knowledge_chunks",
            user_id=user_id,
            top_k=top_k,
            source_id=source_id,
            file_type=file_type,
            date_from=date_from,
            date_to=date_to,
        )
        
        return [
            {
                "content": r.payload.get("content", ""),
                "file_path": r.payload.get("file_path", ""),
                "source_id": r.payload.get("source_id"),
                "relevance_score": r.score,
            }
            for r in results
        ]
    except ImportError:
        logger.warning("HybridRetrievalV2 not available, falling back to vector search")
        return self.search(query, user_id, top_k=top_k, source_id=source_id)

def extract_topics(self, source_id: int, max_topics: int = 10) -> dict[str, list[int]]:
    """Extract topics from source files and group files by topic.
    
    Returns {topic_name: [file_id, ...]}
    """
    import collections
    import math
    
    files = (
        self.db.query(KnowledgeFile)
        .filter(KnowledgeFile.source_id == source_id, KnowledgeFile.status == "indexed")
        .all()
    )
    
    # Collect term frequencies per file and across files
    doc_term_freqs: dict[int, collections.Counter] = {}
    global_term_freq: collections.Counter = collections.Counter()
    
    for f in files:
        content = self._read_file_content(f, source_id)
        if not content:
            continue
        # Simple tokenization on alphanumeric terms
        terms = [t.lower() for t in re.findall(r'\b[a-zA-Z]{3,}\b', content)]
        stop_words = {"the", "and", "for", "are", "but", "not", "you", "all", "can", "had", "her", "was", "one", "our", "out", "has", "have", "been", "this", "that", "with", "from", "file", "each", "also", "its", "than", "they", "will", "what", "when", "where", "which", "were", "their"}
        terms = [t for t in terms if t not in stop_words]
        doc_term_freqs[f.id] = collections.Counter(terms)
        global_term_freq.update(set(terms))
    
    # Calculate TF-IDF-like score for each term in each document
    n_docs = len(files)
    topic_groups: dict[str, list[tuple[int, float]]] = {}
    
    for fid, term_counter in doc_term_freqs.items():
        for term, tf in term_counter.most_common(5):  # top 5 terms per file
            df = global_term_freq[term]
            idf = math.log1p(n_docs / max(1, df))
            score = tf * idf
            if score > 0.5:
                topic_groups.setdefault(term, []).append((fid, score))
    
    # Sort by total score and limit
    sorted_topics = sorted(
        topic_groups.items(),
        key=lambda x: sum(s for _, s in x[1]),
        reverse=True,
    )[:max_topics]
    
    return {topic: [fid for fid, _ in files] for topic, files in sorted_topics}
