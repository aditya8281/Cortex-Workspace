```python
"""KnowledgeSyncService — orchestrates the file-to-embedding pipeline."""

from __future__ import annotations

import hashlib
import logging
import os
from pathlib import Path
from typing import Any

from sqlalchemy.orm import Session

from backend.app.models.knowledge.knowledge_file import KnowledgeFile
from backend.app.models.knowledge.knowledge_source import KnowledgeSource

logger = logging.getLogger(__name__)

# Directories to skip during walking — mirrors file_indexer.py
SKIP_DIRS: set[str] = {
    ".git", "node_modules", "__pycache__", "venv", ".venv",
    ".tox", ".mypy_cache", ".pytest_cache", "dist", "build",
    ".next", ".nuxt", "target", ".idea", ".vscode",
    ".hg", ".svn", ".bzr",
}

_MAX_FILE_SIZE = 50 * 1024 * 1024  # 50MB


class KnowledgeSyncService:
    """Orchestrates file walking, parsing, chunking, embedding, and storage."""

    def __init__(self, db: Session) -> None:
        self.db = db

    def walk_files(self, directory: str) -> list[dict[str, Any]]:
        """Walk *directory* recursively, return list of file info dicts.
        
        Returns [{path, size, mtime, extension}, ...].
        Skips ignored dirs, hidden files (dotfiles), and oversized files.
        """
        results: list[dict[str, Any]] = []
        abs_dir = os.path.abspath(directory)

        for root, dirs, filenames in os.walk(abs_dir):
            # Prune ignored dirs in-place
            dirs[:] = [d for d in dirs if d not in SKIP_DIRS and not d.startswith(".")]

            for filename in filenames:
                filepath = os.path.join(root, filename)
                try:
                    stat = os.stat(filepath)
                except (OSError, PermissionError):
                    continue

                if stat.st_size > _MAX_FILE_SIZE:
                    logger.debug("Skipping oversized file: %s (%d bytes)", filepath, stat.st_size)
                    continue

                results.append({
                    "path": filepath,
                    "size": stat.st_size,
                    "mtime": stat.st_mtime,
                    "extension": os.path.splitext(filename)[1].lower(),
                })

        return results

    def _compute_hash(self, filepath: str) -> str:
        """Compute SHA-256 hex digest via chunked reading."""
        hasher = hashlib.sha256()
        try:
            with open(filepath, "rb") as f:
                for chunk in iter(lambda: f.read(8192), b""):
                    hasher.update(chunk)
            return hasher.hexdigest()
        except (OSError, PermissionError):
            return ""

    def get_progress(self, source_id: int) -> dict[str, Any]:
        """Return current sync progress for a source."""
        source = self.db.query(KnowledgeSource).filter(KnowledgeSource.id == source_id).first()
        if not source:
            return {"status": "not_found"}
        return {
            "status": source.status,
            "progress": source.progress,
            "total_files": source.total_files,
            "indexed_files": source.indexed_files,
            "total_chunks": source.total_chunks,
            "error_message": source.error_message,
            "last_synced_at": source.last_synced_at,
        }
```

import mimetypes

# Re-export parsers
_AVAILABLE_PARSERS: dict[str, Any] = {}

try:
    from backend.app.services.intelligence.parsers import PARSER_REGISTRY as _pr
    _AVAILABLE_PARSERS = _pr
except ImportError:
    logger.warning("Parser registry not available, using fallback text parsing")


def _detect_mime(self, filepath: str) -> str:
    """Detect MIME type from file extension."""
    mime, _ = mimetypes.guess_type(filepath)
    return mime or "application/octet-stream"


def _parse_file(self, filepath: str) -> str | None:
    """Parse a file into plain text using available parsers or fallback."""
    mime = self._detect_mime(filepath)
    ext = os.path.splitext(filepath)[1].lower()
    
    # Try registered parser first
    if ext in _AVAILABLE_PARSERS:
        try:
            return _AVAILABLE_PARSERS[ext](filepath)
        except Exception as e:
            logger.warning("Parser failed for %s: %s", filepath, e)
            return None
    
    # Fallback: try reading as text
    try:
        with open(filepath, "r", encoding="utf-8", errors="replace") as f:
            return f.read()
    except (UnicodeDecodeError, OSError):
        return None

def _chunk_text(self, text: str, strategy: str, chunk_size: int, overlap: int) -> list[dict[str, Any]]:
    """Split text into chunks using the specified strategy.
    
    Returns list of [{index, content, token_count}, ...].
    """
    if strategy == "fixed":
        return self._chunk_fixed(text, chunk_size, overlap)
    elif strategy == "paragraph":
        return self._chunk_paragraph(text, chunk_size)
    else:  # semantic (default) — use existing chunker
        return self._chunk_semantic(text, chunk_size)


def _chunk_fixed(self, text: str, size: int, overlap: int) -> list[dict[str, Any]]:
    """Fixed-size character chunking with overlap."""
    chunks = []
    start = 0
    idx = 0
    while start < len(text):
        end = min(start + size, len(text))
        content = text[start:end]
        chunks.append({
            "index": idx,
            "content": content,
            "token_count": len(content) // 4,  # rough estimate
        })
        idx += 1
        start += size - overlap
    return chunks


def _chunk_paragraph(self, text: str, max_size: int) -> list[dict[str, Any]]:
    """Paragraph-based chunking (split on double newlines)."""
    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]
    chunks = []
    current = ""
    idx = 0
    for para in paragraphs:
        if len(current) + len(para) < max_size:
            current += para + "\n\n"
        else:
            if current:
                chunks.append({"index": idx, "content": current.strip(), "token_count": len(current) // 4})
                idx += 1
            current = para + "\n\n"
    if current:
        chunks.append({"index": idx, "content": current.strip(), "token_count": len(current) // 4})
    return chunks


def _chunk_semantic(self, text: str, max_size: int) -> list[dict[str, Any]]:
    """Semantic chunking using existing chunker."""
    try:
        from backend.app.services.intelligence.chunker import SemanticChunker
        chunker = SemanticChunker(max_chunk_size=max_size)
        return chunker.chunk(text)
    except ImportError:
        logger.warning("SemanticChunker not available, falling back to fixed")
        return self._chunk_fixed(text, max_size, max_size // 8)

def _store_chunks(
    self,
    source_id: int,
    file_id: int,
    chunks: list[dict[str, Any]],
    embedding_model: str,
) -> int:
    """Generate embeddings for chunks and store in Qdrant.
    
    Returns number of chunks stored.
    """
    if not chunks:
        return 0
    
    texts = [c["content"] for c in chunks]
    
    # Get embeddings (batch)
    embeddings = self._generate_embeddings(texts, embedding_model)
    if not embeddings:
        logger.error("Failed to generate embeddings for file %d", file_id)
        return 0
    
    # Prepare Qdrant points
    from qdrant_client.http import models as qmodels
    
    points = []
    for i, (chunk, emb) in enumerate(zip(chunks, embeddings)):
        point_id = f"{source_id}:{file_id}:{i}"
        points.append(qmodels.PointStruct(
            id=hash(point_id) % (2**63),
            vector=emb,
            payload={
                "source_id": source_id,
                "file_id": file_id,
                "chunk_index": chunk["index"],
                "content": chunk["content"][:500],  # store preview in payload
                "file_path": "",  # set during sync
                "user_id": 0,     # set during sync
            },
        ))
    
    # Batch upsert
    try:
        from backend.app.core.vector_db import qdrant_client
        batch_size = 100
        for i in range(0, len(points), batch_size):
            batch = points[i : i + batch_size]
            qdrant_client.upsert(
                collection_name="knowledge_chunks",
                points=batch,
                wait=False,
            )
        return len(points)
    except Exception as e:
        logger.error("Qdrant batch upsert failed: %s", e)
        return 0


def _generate_embeddings(self, texts: list[str], model: str) -> list[list[float]] | None:
    """Generate embeddings for a list of texts using the specified model."""
    try:
        from backend.app.services.intelligence.embedding_service import embedding_service
        return embedding_service.embed(texts, model=model)
    except Exception as e:
        logger.error("Embedding generation failed: %s", e)
        return None


def _delete_file_chunks(self, source_id: int, file_id: int) -> None:
    """Delete all Qdrant chunks for a file."""
    try:
        from qdrant_client.http import models as qmodels
        from backend.app.core.vector_db import qdrant_client
        qdrant_client.delete(
            collection_name="knowledge_chunks",
            points_selector=qmodels.Filter(
                must=[
                    qmodels.FieldCondition(key="file_id", match=qmodels.MatchValue(value=file_id)),
                    qmodels.FieldCondition(key="source_id", match=qmodels.MatchValue(value=source_id)),
                ]
            ),
        )
    except Exception as e:
        logger.error("Qdrant delete failed for file %d: %s", file_id, e)

"""Knowledge sources API."""
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session

from backend.app.core.db import get_current_user, get_db
from backend.app.models.interaction.user import User
from backend.app.schemas.knowledge.knowledge_source import (
    KnowledgeSourceCreate, KnowledgeSourceResponse, KnowledgeSourceListResponse,
)
from backend.app.services.knowledge.sync_service import KnowledgeSyncService

router = APIRouter(prefix="/sources", tags=["knowledge-sources"])


@router.post("", response_model=KnowledgeSourceResponse)
def create_source(
    data: KnowledgeSourceCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Create a new knowledge source."""
    from backend.app.models.knowledge.knowledge_source import KnowledgeSource
    
    source = KnowledgeSource(
        user_id=current_user.id,
        name=data.name,
        path=data.path,
        description=data.description,
        watch_enabled=data.watch_enabled,
        auto_sync=data.auto_sync,
        embedding_model=data.embedding_model,
        chunk_strategy=data.chunk_strategy,
        chunk_size=data.chunk_size,
        chunk_overlap=data.chunk_overlap,
    )
    db.add(source)
    db.commit()
    db.refresh(source)
    return source


@router.get("", response_model=KnowledgeSourceListResponse)
def list_sources(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """List all knowledge sources for the current user."""
    from backend.app.models.knowledge.knowledge_source import KnowledgeSource
    
    sources = (
        db.query(KnowledgeSource)
        .filter(KnowledgeSource.user_id == current_user.id)
        .order_by(KnowledgeSource.created_at.desc())
        .all()
    )
    return {"sources": sources, "total": len(sources)}

def _update_progress(self, source_id: int, indexed: int, total: int, chunks: int, status: str = "indexing"):
    """Update KnowledgeSource progress and emit event."""
    from backend.app.models.knowledge.knowledge_source import KnowledgeSource
    from datetime import datetime, timezone
    
    source = self.db.query(KnowledgeSource).filter(KnowledgeSource.id == source_id).first()
    if not source:
        return
    
    source.indexed_files = indexed
    source.total_files = total
    source.total_chunks = chunks
    source.progress = (indexed / total) if total > 0 else 0.0
    source.status = status
    
    if status in ("idle", "error"):
        source.last_synced_at = datetime.now(timezone.utc)
    
    self.db.commit()
    
    # Emit progress event
    try:
        from backend.app.core.event_bus import event_bus
        import asyncio
        asyncio.create_task(event_bus.emit("knowledge_sync_progress", {
            "source_id": source_id,
            "status": status,
            "progress": source.progress,
            "indexed_files": indexed,
            "total_files": total,
            "total_chunks": chunks,
        }))
    except Exception:
        pass
