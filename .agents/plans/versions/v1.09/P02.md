### Task 1: Implement Preference Learning Service

**File:** `backend/app/services/learning/preference.py`
**Purpose:** Record, retrieve, strengthen, and weaken user preferences. Handles the preference lifecycle: explicit recording → repeated observation → confidence growth → adaptation readiness. Source priority: explicit > observed > inferred.

```python
# backend/app/services/learning/preference.py

import json
from typing import Dict, List, Optional
from sqlalchemy.orm import Session
from backend.app.models.learning.preference import UserPreference


class PreferenceLearningService:
    """Learns and manages user preferences with confidence tracking.
    
    Preference lifecycle:
    1. Record (explicit, observed, or inferred)
    2. Strengthen on repeated observation (+0.1 per observation)
    3. Weaken on contradiction (-0.1 per contradiction)
    4. Adapt behavior when confidence ≥ 0.6
    
    Source priority: explicit (3) > observed (2) > inferred (1)
    Explicit preferences override observed/inferred on conflict.
    
    Architecture ref: 1.6 (Evidence Over Opinion) — all preferences grounded in evidence.
    """
    
    STRENGTHEN_AMOUNT = 0.1
    WEAKEN_AMOUNT = 0.1
    ADAPTATION_THRESHOLD = 0.6
    
    def __init__(self, db: Session):
        self.db = db
    
    async def record_preference(self, user_id: int, category: str, key: str,
                                 value: str, source: str = "observed") -> UserPreference:
        """Record a user preference. Strengthens existing or creates new."""
        existing = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == category,
            UserPreference.key == key
        ).first()
        
        if existing:
            # Check source priority — explicit overrides observed/inferred
            new_priority = {"explicit": 3, "observed": 2, "inferred": 1}.get(source, 0)
            existing_priority = {"explicit": 3, "observed": 2, "inferred": 1}.get(existing.source, 0)
            
            if new_priority >= existing_priority:
                existing.value = value
                existing.source = source
            existing.strengthen(self.STRENGTHEN_AMOUNT)
            self.db.commit()
            return existing
        
        pref = UserPreference(
            user_id=user_id, category=category, key=key,
            value=value, confidence=0.5, source=source,
            observation_count=1
        )
        self.db.add(pref)
        self.db.commit()
        return pref
    
    async def get_preferences(self, user_id: int, category: str = None,
                               min_confidence: float = 0.0) -> List[Dict]:
        """Get user preferences with optional filtering."""
        query = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.confidence >= min_confidence
        )
        if category:
            query = query.filter(UserPreference.category == category)
        
        prefs = query.order_by(UserPreference.confidence.desc()).all()
        return [
            {"id": p.id, "category": p.category, "key": p.key, "value": p.value,
             "confidence": p.confidence, "source": p.source,
             "observation_count": p.observation_count}
            for p in prefs
        ]
    
    async def get_preference(self, user_id: int, category: str, key: str) -> Optional[Dict]:
        """Get a specific preference."""
        pref = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == category,
            UserPreference.key == key
        ).first()
        
        if pref:
            return {"value": pref.value, "confidence": pref.confidence, "source": pref.source}
        return None
    
    async def get_adaptation_ready(self, user_id: int, category: str = None) -> List[Dict]:
        """Get preferences ready for adaptation (confidence ≥ 0.6)."""
        return await self.get_preferences(user_id, category, min_confidence=self.ADAPTATION_THRESHOLD)
    
    async def contradict_preference(self, user_id: int, category: str, key: str) -> Optional[Dict]:
        """Weaken a preference when contradicted."""
        pref = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == category,
            UserPreference.key == key
        ).first()
        
        if not pref:
            return None
        
        pref.weaken(self.WEAKEN_AMOUNT)
        self.db.commit()
        return {"value": pref.value, "confidence": pref.confidence}
    
    async def delete_preference(self, user_id: int, category: str, key: str) -> bool:
        """Delete a preference (user reset)."""
        pref = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id,
            UserPreference.category == category,
            UserPreference.key == key
        ).first()
        
        if pref:
            self.db.delete(pref)
            self.db.commit()
            return True
        return False
    
    async def reset_all(self, user_id: int) -> int:
        """Reset all preferences for a user. Returns count deleted."""
        count = self.db.query(UserPreference).filter(
            UserPreference.user_id == user_id
        ).delete()
        self.db.commit()
        return count
```

### Task 2: Implement Workflow Learning Service

**File:** `backend/app/services/learning/workflow.py`
**Purpose:** Observe and detect multi-step workflow patterns. When a user performs a sequence of steps, the service checks if a matching pattern exists. If yes, it strengthens the pattern. If no, it creates a new one. Patterns rank by frequency.

```python
# backend/app/services/learning/workflow.py

import json
from datetime import datetime
from typing import Dict, List, Optional
from sqlalchemy.orm import Session
from backend.app.models.learning.workflow_pattern import WorkflowPattern


class WorkflowLearningService:
    """Learns multi-step workflow patterns from user behavior.
    
    When the user performs a sequence of steps (e.g., "edit → test → commit"),
    this service detects if a matching pattern exists. If yes, strengthens it.
    If no, creates a new pattern. Patterns rank by frequency for automation.
    
    Architecture ref: 4.7 (Workflow Architecture) — workflow detection as
    event processing via event bus.
    """
    
    SIMILARITY_THRESHOLD = 0.6  # How similar step sequences must be to match
    MAX_PATTERNS_PER_USER = 200
    
    def __init__(self, db: Session):
        self.db = db
    
    async def observe_workflow(self, user_id: int, steps: List[str],
                                context: str = None) -> WorkflowPattern:
        """Observe a workflow step sequence. Matches existing or creates new."""
        if len(steps) < 2:
            raise ValueError("Workflow must have at least 2 steps")
        
        # Generate pattern name from first 3 steps
        pattern_name = " → ".join(steps[:3])
        steps_json = json.dumps(steps)
        
        # Check for existing similar pattern
        existing = self._find_similar_pattern(user_id, steps)
        
        if existing:
            existing.observe()
            if context:
                existing.context = context
            self.db.commit()
            return existing
        
        # Check pattern limit
        current_count = self.db.query(WorkflowPattern).filter(
            WorkflowPattern.user_id == user_id
        ).count()
        
        if current_count >= self.MAX_PATTERNS_PER_USER:
            # Remove least frequent pattern
            least_frequent = self.db.query(WorkflowPattern).filter(
                WorkflowPattern.user_id == user_id
            ).order_by(WorkflowPattern.frequency.asc()).first()
            if least_frequent and least_frequent.frequency <= 1:
                self.db.delete(least_frequent)
                self.db.commit()
        
        pattern = WorkflowPattern(
            user_id=user_id, pattern_name=pattern_name,
            steps=steps_json, context=context,
            frequency=1, confidence=0.3
        )
        self.db.add(pattern)
        self.db.commit()
        return pattern
    
    async def get_common_workflows(self, user_id: int, limit: int = 10) -> List[Dict]:
        """Get most common workflows ranked by frequency."""
        patterns = self.db.query(WorkflowPattern).filter(
            WorkflowPattern.user_id == user_id
        ).order_by(WorkflowPattern.frequency.desc()).limit(limit).all()
        
        return [
            {"id": p.id, "name": p.pattern_name, "steps": json.loads(p.steps),
             "frequency": p.frequency, "confidence": p.confidence,
             "context": p.context}
            for p in patterns
        ]
    
    async def get_workflow(self, user_id: int, pattern_name: str) -> Optional[Dict]:
        """Get a specific workflow pattern."""
        pattern = self.db.query(WorkflowPattern).filter(
            WorkflowPattern.user_id == user_id,
            WorkflowPattern.pattern_name == pattern_name
        ).first()
        
        if pattern:
            return {"name": pattern.pattern_name, "steps": json.loads(pattern.steps),
                    "frequency": pattern.frequency, "confidence": pattern.confidence}
        return None
    
    async def suggest_next_step(self, user_id: int, current_steps: List[str]) -> Optional[str]:
        """Suggest the next step based on learned workflows."""
        patterns = self.db.query(WorkflowPattern).filter(
            WorkflowPattern.user_id == user_id,
            WorkflowPattern.frequency >= 3
        ).order_by(WorkflowPattern.frequency.desc()).all()
        
        for pattern in patterns:
            steps = json.loads(pattern.steps)
            # Check if current steps match the beginning of a known workflow
            if len(current_steps) < len(steps):
                match = all(
                    self._steps_similar(current_steps[i], steps[i])
                    for i in range(len(current_steps))
                )
                if match:
                    return steps[len(current_steps)]
        
        return None
    
    def _find_similar_pattern(self, user_id: int, steps: List[str]) -> Optional[WorkflowPattern]:
        """Find an existing pattern similar to the given steps."""
        patterns = self.db.query(WorkflowPattern).filter(
            WorkflowPattern.user_id == user_id
        ).all()
        
        best_match = None
        best_similarity = 0.0
        
        for pattern in patterns:
            existing_steps = json.loads(pattern.steps)
            similarity = self._calculate_sequence_similarity(steps, existing_steps)
            if similarity > best_similarity and similarity >= self.SIMILARITY_THRESHOLD:
                best_similarity = similarity
                best_match = pattern
        
        return best_match
    
    def _calculate_sequence_similarity(self, seq1: List[str], seq2: List[str]) -> float:
        """Calculate similarity between two step sequences using LCS."""
        if not seq1 or not seq2:
            return 0.0
        
        # Longest Common Subsequence
        m, n = len(seq1), len(seq2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if self._steps_similar(seq1[i-1], seq2[j-1]):
                    dp[i][j] = dp[i-1][j-1] + 1
                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])
        
        lcs_length = dp[m][n]
        max_len = max(m, n)
        return lcs_length / max_len if max_len > 0 else 0.0
    
    def _steps_similar(self, step1: str, step2: str) -> bool:
        """Check if two steps are similar (exact or substring match)."""
        s1, s2 = step1.lower().strip(), step2.lower().strip()
        return s1 == s2 or s1 in s2 or s2 in s1
```

        pref = await service.record_preference(1, "test", "k", "explicit_val", "explicit")

        assert pref.source == "explicit"

        assert pref.source == "explicit"
    
    @pytest.mark.asyncio
    async def test_get_preferences_by_category(self, db_session):

        assert prefs[0]["category"] == "response_style"
    
    @pytest.mark.asyncio
    async def test_contradict_preference(self, db_session):

            await service.record_preference(1, "test", "k", "val")
        ready = await service.get_adaptation_ready(1)

## Cross-Domain Integration Notes

- **v1.03 (Memory Foundation)** — LearningEvent can reference memory IDs. Preferences stored independently.
- **v1.04 (Awareness Foundation)** — Awareness events (terminal history, app focus) feed into workflow observation.
- **v1.07 (Memory Evolution)** — Preferences become graph nodes. Workflow patterns link to memory nodes.

## Security Considerations

1. **User isolation** — All queries filter by `user_id`. No cross-user preference access.
2. **Data sensitivity** — Preferences reveal personal habits. Store locally only.
3. **Reset capability** — Users can delete all preferences via `reset_all()`.
4. **Confidence bounds** — Confidence capped at 1.0, floored at 0.0.

---

## Performance Considerations

1. **Preference lookup** — O(1) via unique constraint index on (user_id, category, key).
2. **Workflow similarity** — O(n*m) LCS comparison. Acceptable for <200 patterns.
3. **Next step suggestion** — O(p*s) where p=patterns, s=steps. Limit to frequent patterns only.
4. **Pattern limit** — Max 200 patterns per user. Evict least frequent when exceeded.
