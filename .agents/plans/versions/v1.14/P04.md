## Architecture Decisions

### ADR-14-10: Cognition API as Separate Router

**Decision:** Advanced cognition endpoints are in a separate `api/v1/cognition/advanced.py` router, not mixed with basic cognition endpoints.

**Rationale:** Clean separation between basic cognition (v1.06) and advanced cognition (v1.14). Different authentication requirements may emerge (advanced operations may require higher-tier access).

**Consequence:** Two routers for cognition: basic (`/cognition/`) and advanced (`/cognition/advanced/`). Frontend imports both.

### ADR-14-11: System Validation as Automated Suite

**Decision:** Final system validation is a comprehensive test suite that can be run via `make validate-system`. It tests cross-domain scenarios, performance benchmarks, and security checks.

**Rationale:** Manual validation is error-prone and not repeatable. An automated suite ensures the system remains valid through future maintenance.

**Consequence:** The validation suite must be maintained. Tests may be slow (cross-domain). Consider tagging them as `slow` in pytest.

const BASE = "/api/v1/cognition/advanced";

async function fetchJSON<T>(url: string, options?: RequestInit): Promise<T> {
  const res = await fetch(url, {
    credentials: "include",
    headers: { "Content-Type": "application/json" },
    ...options,
  });
  if (!res.ok) {
    const error = await res.json().catch(() => ({ detail: res.statusText }));
    throw new Error(error.detail || `HTTP ${res.status}`);
  }
  return res.json();
}

// Reasoning
export const reasoningAPI = {
  createChain: (data: ReasoningChainCreate) =>
    fetchJSON<ReasoningChain>(`${BASE}/reasoning/chain`, {
      method: "POST", body: JSON.stringify(data),
    }),
  listChains: (chainType?: string, limit = 20) => {
    const params = new URLSearchParams();
    if (chainType) params.set("chain_type", chainType);
    params.set("limit", String(limit));
    return fetchJSON<ReasoningChain[]>(`${BASE}/reasoning/chains?${params}`);
  },
  getChain: (id: number) =>
    fetchJSON<ReasoningChain>(`${BASE}/reasoning/chains/${id}`),
  infer: (premises: string[]) =>
    fetchJSON(`${BASE}/reasoning/infer`, {
      method: "POST", body: JSON.stringify(premises),
    }),
  stats: () =>
    fetchJSON(`${BASE}/reasoning/stats`),
};

class TestSecurityAudit:
    """Validate security invariants across all endpoints."""

    def test_authentication_required(self, client: TestClient):
        endpoints = [
            ("GET", "/api/v1/cognition/advanced/reasoning/chains"),
            ("POST", "/api/v1/cognition/advanced/reasoning/chain"),
            ("GET", "/api/v1/cognition/advanced/consciousness/state"),
            ("POST", "/api/v1/cognition/advanced/meta/reflect"),
            ("GET", "/api/v1/cognition/advanced/creative/saved"),
        ]
        for method, url in endpoints:
            resp = client.request(method, url)
            assert resp.status_code in (401, 403), f"{method} {url} should require auth"

    def test_ownership_enforced(self, client: TestClient, auth_headers, auth_headers_user2):
        """User 1 creates data, User 2 cannot see it."""
        # User 1 creates a chain
        resp1 = client.post(
            "/api/v1/cognition/advanced/reasoning/chain",
            json={"problem": "User 1 private problem"},
            headers=auth_headers,
        )
        chain_id = resp1.json()["id"]

        # User 2 tries to access User 1's chain
        resp2 = client.get(
            f"/api/v1/cognition/advanced/reasoning/chains/{chain_id}",
            headers=auth_headers_user2,
        )
        assert resp2.status_code == 404, "User 2 should not access User 1's data"

    def test_reasoning_depth_limit(self, client: TestClient, auth_headers):
        """Verify max_depth cannot exceed 10."""
        response = client.post(
            "/api/v1/cognition/advanced/reasoning/chain",
            json={"problem": "Depth test", "max_depth": 999},
            headers=auth_headers,
        )
        # Should either reject (422) or clamp to 10
        assert response.status_code in (200, 422)
