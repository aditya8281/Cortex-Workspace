## Architecture Decisions

### ADR-13-16: API Response Envelope

**Decision:** All utility/integration API responses use a consistent envelope: `{"data": ..., "meta": {"count": N, "has_more": bool}}` for list endpoints and `{"data": ...}` for single-item endpoints.

**Rationale:** Consistent response format makes frontend API client simpler. Meta field enables pagination without custom headers. Matches the existing API patterns from earlier versions.

**Consequence:** Frontend always unwraps `.data`. List endpoints must compute `has_more` by querying count+1.

### ADR-13-17: Query Parameters for Filtering

**Decision:** Use query parameters for filtering, sorting, and pagination on GET endpoints. POST/PUT bodies for data mutations.

**Rationale:** REST convention. Query params are bookmarkable and cacheable. POST body for mutations avoids accidental data exposure in URLs.

**Consequence:** Complex filters (tags array) use repeated query params: `?tags=python&tags=fastapi`.

```python
async def create_event(
    data: CalendarEventCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),

    """Create a calendar event."""
    service = CalendarService(db)
    event = await service.create_event(current_user.id, data)
    return event

async def get_events(
    start: Optional[datetime] = None,
    end: Optional[datetime] = None,
    status: Optional[str] = None,
    limit: int = Query(100, ge=1, le=500),
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get calendar events in a time range."""
    service = CalendarService(db)
    events = await service.get_events(current_user.id, start, end, status)
    return {"data": events, "meta": {"count": len(events)}}


@router.get("/events/upcoming")
async def get_upcoming(
    hours: int = Query(24, ge=1, le=168),
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),

    """Get upcoming events in the next N hours."""
    service = CalendarService(db)
    events = await service.get_upcoming(current_user.id, hours)
    return {"data": events, "meta": {"count": len(events)}}

async def get_event(
    event_id: int,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get a single calendar event."""
    service = CalendarService(db)
    events = await service.get_events(current_user.id)
    for e in events:
        if e["id"] == event_id:
            return e
    raise HTTPException(status_code=404, detail="Event not found")

    current_user=Depends(get_current_user),

    """Update a calendar event."""
    service = CalendarService(db)
    event = await service.update_event(event_id, current_user.id, data)
    if not event:
        raise HTTPException(status_code=404, detail="Event not found")
    return event
```

```tsx
'use client';

import { useEffect, useState } from 'react';
import { dashboardApi } from '../api';

interface DashboardData {
  greeting: string;
  calendar: { upcoming_count: number; upcoming: any[] };
  tasks: { overdue_count: number; overdue: any[]; today_count: number; today: any[] };
  email: { unread: number; action_required: number };
  habits: { completed_today: number; total_habits: number; best_streak: number };
  focus: { today_minutes: number; today_sessions: number };
  notes: { recent: any[] };
}

export function Dashboard() {
  const [data, setData] = useState<DashboardData | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    dashboardApi.getDashboard().then(res => {
      setData(res.data);
      setLoading(false);
    });
  }, []);

  if (loading) return <div className="animate-pulse p-6">Loading dashboard...</div>;
  if (!data) return <div>Failed to load dashboard</div>;

  return (
    <div className="p-6 space-y-6">
      <h1 className="text-2xl font-bold">{data.greeting}</h1>

      {/* Calendar Widget */}
      <section className="rounded-lg border p-4">
        <h2 className="text-lg font-semibold mb-2">
          Upcoming ({data.calendar.upcoming_count})
        </h2>
        {data.calendar.upcoming.slice(0, 3).map((event, i) => (
          <div key={i} className="py-1 text-sm">
            {event.title} — {new Date(event.start_time).toLocaleTimeString()}
          </div>
        ))}
      </section>

      {/* Tasks Widget */}
      <section className="rounded-lg border p-4">
        <h2 className="text-lg font-semibold mb-2">
          Tasks
          {data.tasks.overdue_count > 0 && (
            <span className="ml-2 text-red-500 text-sm">
              {data.tasks.overdue_count} overdue
            </span>
          )}
        </h2>
        {data.tasks.today.slice(0, 5).map((task, i) => (
          <div key={i} className="py-1 text-sm">
            <span className={`priority-${task.priority}`}>●</span> {task.title}
          </div>
        ))}
      </section>

      {/* Email Widget */}
      <section className="rounded-lg border p-4">
        <h2 className="text-lg font-semibold mb-2">Email</h2>
        <p className="text-sm">{data.email.unread} unread, {data.email.action_required} need action</p>
      </section>

      {/* Habits Widget */}
      <section className="rounded-lg border p-4">
        <h2 className="text-lg font-semibold mb-2">Habits</h2>
        <p className="text-sm">
          {data.habits.completed_today}/{data.habits.total_habits} today
          {data.habits.best_streak > 0 && ` · Best streak: ${data.habits.best_streak}`}
        </p>
      </section>

      {/* Focus Widget */}
      <section className="rounded-lg border p-4">
        <h2 className="text-lg font-semibold mb-2">Focus</h2>
        <p className="text-sm">
          {data.focus.today_minutes} min today in {data.focus.today_sessions} sessions
        </p>
      </section>
    </div>
  );
}
```
