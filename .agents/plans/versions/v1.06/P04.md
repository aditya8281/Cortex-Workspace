### Task 1: Implement Tool Registry

**Architecture:** The ToolRegistry is a central catalog of all available tools. Each tool has:
- A name, description, and typed parameter schema
- A callable function
- A `requires_confirmation` flag for dangerous operations
- A `max_timeout_ms` for execution limits

Tools are registered at application startup. The registry provides lookup, validation, and execution.

**Implementation:**
```python
# backend/app/services/execution/tool_registry.py

from typing import Dict, Any, Callable, Optional, List
import inspect
import asyncio
import json


class ToolValidationError(Exception):
    """Raised when tool parameters fail validation."""
    pass


class ToolNotFoundError(Exception):
    """Raised when a requested tool is not registered."""
    pass


class ToolRegistry:
    """Central registry of available tools for the execution engine.
    
    Features:
    - Tool registration with typed parameter schemas
    - Parameter validation against JSON Schema-like definitions
    - Confirmation gate for dangerous operations
    - Execution timeout enforcement
    - Tool listing with metadata
    
    Usage:
        registry = get_tool_registry()
        registry.register(
            name="web_search",
            func=my_search_function,
            description="Search the web for information",
            parameters={"query": {"type": "string", "required": True}},
            requires_confirmation=False,
            max_timeout_ms=30000,
        )
        result = await registry.execute("web_search", {"query": "test"})
    """
    
    def __init__(self):
        self._tools: Dict[str, Dict[str, Any]] = {}
    
    def register(self, name: str, func: Callable,
                description: str, parameters: Dict[str, Any],
                requires_confirmation: bool = False,
                max_timeout_ms: int = 60000,
                category: str = "general") -> None:
        """Register a tool.
        
        Args:
            name: Unique tool name
            func: Async callable to execute
            description: Human-readable description
            parameters: Parameter schema {name: {type, required, default, description}}
            requires_confirmation: If True, execution requires explicit confirmation
            max_timeout_ms: Maximum execution time in milliseconds
            category: Tool category for grouping (general, memory, file, web, etc.)
        """
        if name in self._tools:
            raise ValueError(f"Tool '{name}' is already registered")
        
        # Validate func is callable
        if not callable(func):
            raise ValueError(f"Tool function must be callable, got {type(func)}")
        
        self._tools[name] = {
            "name": name,
            "func": func,
            "description": description,
            "parameters": parameters,
            "requires_confirmation": requires_confirmation,
            "max_timeout_ms": max_timeout_ms,
            "category": category,
        }
    
    def unregister(self, name: str) -> bool:
        """Unregister a tool."""
        if name in self._tools:
            del self._tools[name]
            return True
        return False
    
    def get_tool(self, name: str) -> Optional[Dict[str, Any]]:
        """Get tool metadata (without the function reference)."""
        tool = self._tools.get(name)
        if not tool:
            return None
        return {
            "name": tool["name"],
            "description": tool["description"],
            "parameters": tool["parameters"],
            "requires_confirmation": tool["requires_confirmation"],
            "max_timeout_ms": tool["max_timeout_ms"],
            "category": tool["category"],
        }
    
    def list_tools(self, category: Optional[str] = None) -> List[Dict[str, Any]]:
        """List all registered tools, optionally filtered by category."""
        tools = []
        for name, tool in self._tools.items():
            if category and tool["category"] != category:
                continue
            tools.append({
                "name": name,
                "description": tool["description"],
                "parameters": tool["parameters"],
                "requires_confirmation": tool["requires_confirmation"],
                "category": tool["category"],
            })
        return tools
    
    def has_tool(self, name: str) -> bool:
        """Check if a tool is registered."""
        return name in self._tools
    
    def validate_params(self, name: str, params: Dict[str, Any]) -> List[str]:
        """Validate parameters against the tool's schema.
        
        Returns list of validation errors (empty if valid).
        """
        tool = self._tools.get(name)
        if not tool:
            return [f"Tool '{name}' not found"]
        
        errors = []
        schema = tool["parameters"]
        
        for param_name, param_def in schema.items():
            if param_def.get("required", False) and param_name not in params:
                errors.append(f"Missing required parameter: {param_name}")
            
            if param_name in params:
                expected_type = param_def.get("type")
                value = params[param_name]
                
                type_map = {
                    "string": str, "str": str,
                    "integer": int, "int": int,
                    "number": (int, float), "float": (int, float),
                    "boolean": bool, "bool": bool,
                    "array": list, "list": list,
                    "object": dict, "dict": dict,
                }
                
                if expected_type and expected_type in type_map:
                    expected = type_map[expected_type]
                    if not isinstance(value, expected):
                        errors.append(f"Parameter '{param_name}' expected {expected_type}, got {type(value).__name__}")
        
        # Check for unexpected parameters
        for param_name in params:
            if param_name not in schema:
                errors.append(f"Unexpected parameter: {param_name}")
        
        return errors
    
    async def execute(self, name: str, params: Dict[str, Any],
                     confirmed: bool = False) -> Any:
        """Execute a tool with validation and timeout.
        
        Args:
            name: Tool name
            params: Tool parameters
            confirmed: Whether dangerous operation is confirmed
        
        Returns:
            Tool execution result
        
        Raises:
            ToolNotFoundError: If tool not registered
            ToolValidationError: If parameters invalid
            PermissionError: If confirmation required but not given
            TimeoutError: If execution exceeds max_timeout_ms
        """
        tool = self._tools.get(name)
        if not tool:
            raise ToolNotFoundError(f"Tool '{name}' is not registered")
        
        # Validate parameters
        errors = self.validate_params(name, params)
        if errors:
            raise ToolValidationError(f"Parameter validation failed: {'; '.join(errors)}")
        
        # Check confirmation requirement
        if tool["requires_confirmation"] and not confirmed:
            raise PermissionError(
                f"Tool '{name}' requires confirmation. "
                f"Set confirmed=True to proceed."
            )
        
        # Execute with timeout
        timeout = tool["max_timeout_ms"] / 1000.0
        func = tool["func"]
        
        try:
            if asyncio.iscoroutinefunction(func):
                result = await asyncio.wait_for(func(**params), timeout=timeout)
            else:
                result = func(**params)
            return result
        except asyncio.TimeoutError:
            raise TimeoutError(
                f"Tool '{name}' exceeded timeout of {tool['max_timeout_ms']}ms"
            )


# Global tool registry instance
_tool_registry: Optional[ToolRegistry] = None


def get_tool_registry() -> ToolRegistry:
    """Get the global tool registry singleton."""
    global _tool_registry
    if _tool_registry is None:
        _tool_registry = ToolRegistry()
    return _tool_registry


def register_default_tools(registry: ToolRegistry) -> None:
    """Register built-in tools."""
    
    async def echo_tool(message: str = "ok") -> dict:
        """Echo tool for testing."""
        return {"echo": message}
    
    async def add_numbers(a: float = 0, b: float = 0) -> dict:
        """Add two numbers."""
        return {"result": a + b}
    
    registry.register(
        name="echo",
        func=echo_tool,
        description="Echoes a message back",
        parameters={"message": {"type": "string", "required": False, "default": "ok"}},
        category="general",
    )
    
    registry.register(
        name="add_numbers",
        func=add_numbers,
        description="Adds two numbers",
        parameters={
            "a": {"type": "number", "required": True},
            "b": {"type": "number", "required": True},
        },
        category="math",
    )
```

### Task 2: Implement Action Verifier

**Architecture:** The ActionVerifier performs pre-flight safety checks on every tool execution. It detects dangerous patterns, validates resource limits, checks permissions, and analyzes side effects. Actions that fail verification are blocked before execution.

**Implementation:**
```python
# backend/app/services/execution/action_verifier.py

import re
from typing import Dict, Any, List, Optional, Callable


class ActionVerifier:
    """Pre-flight safety verification for tool executions.
    
    Verification layers:
    1. Dangerous pattern detection (shell commands, SQL injection, etc.)
    2. Resource limit checks (file size, API calls, etc.)
    3. Permission checks (via RBAC from v1.05)
    4. Custom rules (user-defined checks)
    
    Usage:
        verifier = ActionVerifier()
        result = verifier.verify("shell_command", {"command": "ls -la"})
        if not result["approved"]:
            print(f"Blocked: {result['errors']}")
    """
    
    # Dangerous patterns that should trigger warnings or blocks
    DANGEROUS_PATTERNS = [
        (r"rm\s+-rf\s+/", "CRITICAL", "Recursive deletion of root filesystem"),
        (r"DROP\s+TABLE", "CRITICAL", "Database table deletion"),
        (r"DELETE\s+FROM.*WHERE\s+1", "CRITICAL", "Unconditional database deletion"),
        (r"sudo\s+", "HIGH", "Privilege escalation attempt"),
        (r"chmod\s+777", "HIGH", "Overly permissive file permissions"),
        (r">\s*/dev/sd", "CRITICAL", "Direct disk write"),
        (r"curl\s+.*\|\s*sh", "HIGH", "Remote code execution via pipe"),
        (r"eval\s*\(", "MEDIUM", "Dynamic code evaluation"),
        (r"exec\s*\(", "MEDIUM", "Dynamic code execution"),
        (r"__import__", "MEDIUM", "Dynamic module import"),
        (r"subprocess\.(call|run|Popen)", "MEDIUM", "Subprocess execution"),
        (r"os\.system\s*\(", "MEDIUM", "System command execution"),
        (r"\bpassword\b.*=", "LOW", "Potential password in parameters"),
        (r"\bsecret\b.*=", "LOW", "Potential secret in parameters"),
        (r"\bapi_key\b.*=", "LOW", "Potential API key in parameters"),
    ]
    
    # Resource limits
    RESOURCE_LIMITS = {
        "max_file_size_bytes": 100 * 1024 * 1024,  # 100MB
        "max_api_calls_per_minute": 60,
        "max_text_length": 1_000_000,
        "max_list_length": 10_000,
    }
    
    def __init__(self):
        self._rules: List[Dict[str, Any]] = []
        self._custom_checks: List[Callable] = []
    
    def add_rule(self, name: str, check_fn: Callable, description: str,
                severity: str = "MEDIUM") -> None:
        """Add a custom verification rule."""
        self._rules.append({
            "name": name,
            "check": check_fn,
            "description": description,
            "severity": severity,
        })
    
    def add_dangerous_pattern(self, pattern: str, severity: str, description: str) -> None:
        """Add a custom dangerous pattern."""
        self.DANGEROUS_PATTERNS.append((pattern, severity, description))
    
    def verify(self, tool_name: str, params: Dict[str, Any],
              context: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """Verify an action before execution.
        
        Args:
            tool_name: Name of the tool to execute
            params: Parameters that will be passed to the tool
            context: Additional context (user_id, role, etc.)
        
        Returns:
            {
                "approved": bool,
                "warnings": [str],
                "errors": [str],
                "severity": str (highest severity found),
                "tool_name": str,
                "checks_performed": [str],
            }
        """
        context = context or {}
        warnings: List[str] = []
        errors: List[str] = []
        checks_performed: List[str] = []
        max_severity = "INFO"
        
        severity_order = {"INFO": 0, "LOW": 1, "MEDIUM": 2, "HIGH": 3, "CRITICAL": 4}
        
        def update_severity(sev: str):
            nonlocal max_severity
            if severity_order.get(sev, 0) > severity_order.get(max_severity, 0):
                max_severity = sev
        
        # 1. Dangerous pattern detection
        param_str = str(params)
        for pattern, severity, description in self.DANGEROUS_PATTERNS:
            if re.search(pattern, param_str, re.IGNORECASE):
                update_severity(severity)
                if severity in ("CRITICAL", "HIGH"):
                    errors.append(f"[{severity}] {description} (pattern: {pattern})")
                else:
                    warnings.append(f"[{severity}] {description} (pattern: {pattern})")
        checks_performed.append("dangerous_patterns")
        
        # 2. Resource limit checks
        if "text" in params and isinstance(params["text"], str):
            if len(params["text"]) > self.RESOURCE_LIMITS["max_text_length"]:
                errors.append(f"Text length {len(params['text'])} exceeds limit {self.RESOURCE_LIMITS['max_text_length']}")
                update_severity("HIGH")
        checks_performed.append("resource_limits")
        
        if "items" in params and isinstance(params["items"], list):
            if len(params["items"]) > self.RESOURCE_LIMITS["max_list_length"]:
                errors.append(f"List length {len(params['items'])} exceeds limit {self.RESOURCE_LIMITS['max_list_length']}")
                update_severity("MEDIUM")
        checks_performed.append("list_limits")
        
        # 3. Custom rules
        for rule in self._rules:
            try:
                result = rule["check"](tool_name, params, context)
                if result is False:
                    errors.append(f"Rule '{rule['name']}' failed: {rule['description']}")
                    update_severity(rule.get("severity", "MEDIUM"))
                elif result is True:
                    warnings.append(f"Rule '{rule['name']}' warning: {rule['description']}")
                checks_performed.append(f"rule:{rule['name']}")
            except Exception as e:
                warnings.append(f"Rule '{rule['name']}' error: {str(e)}")
                checks_performed.append(f"rule:{rule['name']}:error")
        
        return {
            "approved": len(errors) == 0,
            "warnings": warnings,
            "errors": errors,
            "severity": max_severity,
            "tool_name": tool_name,
            "checks_performed": checks_performed,
        }
```

### Task 3: Implement Execution Engine

**Architecture:** The ExecutionEngine orchestrates tool execution with full lifecycle tracking. It creates a ToolExecution record, runs the tool, captures results/errors, measures timing, and stores everything for learning and debugging.

**Implementation:**
```python
# backend/app/services/execution/engine.py

import time
import traceback
from datetime import datetime
from typing import Dict, Any, Optional, List
from sqlalchemy.orm import Session
from backend.app.models.execution.tool_execution import ToolExecution
from backend.app.services.execution.tool_registry import ToolRegistry, ToolValidationError, ToolNotFoundError
from backend.app.services.execution.action_verifier import ActionVerifier


class ExecutionEngine:
    """Core execution engine for running tools with full lifecycle tracking.
    
    Flow:
    1. Verify action (ActionVerifier)
    2. Create execution record (status: running)
    3. Execute tool via registry
    4. Record result, timing, errors
    5. Return execution record
    
    The engine provides:
    - Execution history for learning and debugging
    - Performance statistics (duration, success rate)
    - Retry support for transient failures
    - Workflow integration (execution can be part of a workflow)
    
    Usage:
        engine = ExecutionEngine(db, registry)
        execution = await engine.execute_tool(
            user_id=1,
            tool_name="web_search",
            params={"query": "python best practices"}
        )
    """
    
    def __init__(self, db: Session, tool_registry: ToolRegistry):
        self.db = db
        self.registry = tool_registry
        self.verifier = ActionVerifier()
    
    async def execute_tool(self, user_id: int, tool_name: str,
                          params: Dict[str, Any],
                          auto_verify: bool = True,
                          confirmed: bool = False,
                          workflow_id: Optional[int] = None,
                          metadata: Optional[Dict[str, Any]] = None) -> ToolExecution:
        """Execute a tool and record the execution.
        
        Args:
            user_id: ID of the requesting user
            tool_name: Name of the tool to execute
            params: Tool parameters
            auto_verify: Whether to run action verification before execution
            confirmed: Whether dangerous operations are confirmed
            workflow_id: ID of parent workflow if this is a workflow step
            metadata: Additional metadata to attach to the execution
        
        Returns:
            ToolExecution record with result or error
        """
        start_time = time.time()
        
        # 1. Verify action
        verification_result = None
        if auto_verify:
            verification_result = self.verifier.verify(
                tool_name, params, {"user_id": user_id}
            )
            if not verification_result["approved"]:
                execution = ToolExecution(
                    user_id=user_id,
                    tool_name=tool_name,
                    parameters=params,
                    status="blocked",
                    error_message=f"Action verification failed: {'; '.join(verification_result['errors'])}",
                    verification_result=verification_result,
                    started_at=datetime.utcnow(),
                    completed_at=datetime.utcnow(),
                    duration_ms=0,
                    workflow_id=workflow_id,
                    metadata=metadata,
                )
                self.db.add(execution)
                self.db.commit()
                self.db.refresh(execution)
                return execution
        
        # 2. Create execution record
        execution = ToolExecution(
            user_id=user_id,
            tool_name=tool_name,
            parameters=params,
            status="running",
            started_at=datetime.utcnow(),
            verification_result=verification_result,
            workflow_id=workflow_id,
            metadata=metadata,
        )
        self.db.add(execution)
        self.db.commit()
        self.db.refresh(execution)
        
        # 3. Execute tool
        try:
            result = await self.registry.execute(tool_name, params, confirmed=confirmed)
            execution.result = result
            execution.status = "success"
        except ToolValidationError as e:
            execution.status = "failed"
            execution.error_type = "ToolValidationError"
            execution.error_message = str(e)
        except ToolNotFoundError as e:
            execution.status = "failed"
            execution.error_type = "ToolNotFoundError"
            execution.error_message = str(e)
        except PermissionError as e:
            execution.status = "failed"
            execution.error_type = "PermissionError"
            execution.error_message = str(e)
        except TimeoutError as e:
            execution.status = "timeout"
            execution.error_type = "TimeoutError"
            execution.error_message = str(e)
        except Exception as e:
            execution.status = "failed"
            execution.error_type = type(e).__name__
            execution.error_message = str(e)
        
        # 4. Record timing
        end_time = time.time()
        execution.completed_at = datetime.utcnow()
        execution.duration_ms = int((end_time - start_time) * 1000)
        
        self.db.commit()
        self.db.refresh(execution)
        return execution
    
    async def execute_with_retry(self, user_id: int, tool_name: str,
                                params: Dict[str, Any],
                                max_retries: int = 3,
                                retry_delay_ms: int = 1000) -> ToolExecution:
        """Execute a tool with automatic retry on failure.
        
        Uses exponential backoff: delay * 2^attempt
        """
        import asyncio
        
        last_execution = None
        for attempt in range(max_retries + 1):
            execution = await self.execute_tool(
                user_id=user_id,
                tool_name=tool_name,
                params=params,
                auto_verify=True,
                metadata={"retry_attempt": attempt, "max_retries": max_retries},
            )
            
            if execution.status == "success":
                return execution
            
            last_execution = execution
            
            if attempt < max_retries:
                delay = retry_delay_ms * (2 ** attempt) / 1000.0
                await asyncio.sleep(delay)
                execution.retry_count = attempt + 1
                self.db.commit()
        
        return last_execution
    
    async def get_execution(self, execution_id: int) -> Optional[ToolExecution]:
        """Get an execution by ID."""
        return self.db.query(ToolExecution).filter(
            ToolExecution.id == execution_id
        ).first()
    
    async def get_user_executions(self, user_id: int,
                                 tool_name: Optional[str] = None,
                                 status: Optional[str] = None,
                                 limit: int = 50) -> List[ToolExecution]:
        """Get executions for a user."""
        query = self.db.query(ToolExecution).filter(
            ToolExecution.user_id == user_id
        )
        if tool_name:
            query = query.filter(ToolExecution.tool_name == tool_name)
        if status:
            query = query.filter(ToolExecution.status == status)
        return query.order_by(ToolExecution.started_at.desc()).limit(limit).all()
    
    async def get_execution_stats(self, user_id: int) -> Dict[str, Any]:
        """Get execution statistics for a user."""
        executions = await self.get_user_executions(user_id, limit=1000)
        
        total = len(executions)
        successful = sum(1 for e in executions if e.status == "success")
        failed = sum(1 for e in executions if e.status == "failed")
        blocked = sum(1 for e in executions if e.status == "blocked")
        timeout = sum(1 for e in executions if e.status == "timeout")
        
        durations = [e.duration_ms for e in executions if e.duration_ms is not None]
        avg_duration = sum(durations) // len(durations) if durations else 0
        
        # Tool breakdown
        tool_counts: Dict[str, int] = {}
        for e in executions:
            tool_counts[e.tool_name] = tool_counts.get(e.tool_name, 0) + 1
        
        return {
            "total": total,
            "successful": successful,
            "failed": failed,
            "blocked": blocked,
            "timeout": timeout,
            "success_rate": (successful / total * 100) if total > 0 else 0,
            "average_duration_ms": avg_duration,
            "tool_breakdown": tool_counts,
        }
```

### Task 4: Implement Workflow Orchestrator

**Architecture:** The WorkflowOrchestrator chains tool executions into multi-step workflows with DAG-based dependency resolution. Key features:
- Steps execute in topological order based on dependencies
- Independent steps can run in parallel (future: async executor)
- Conditional branching based on previous step results
- Error recovery with retry and fallback
- Checkpointing after each step for resumability
- Cancellation support

**Implementation:**
```python
# backend/app/services/execution/workflow.py

import time
from datetime import datetime
from typing import List, Dict, Any, Optional
from sqlalchemy.orm import Session
from backend.app.models.execution.workflow import Workflow
from backend.app.services.execution.engine import ExecutionEngine


class WorkflowOrchestrator:
    """Multi-step workflow orchestrator with DAG dependency resolution.
    
    Features:
    - DAG-based step ordering (topological sort)
    - Dependency validation (cycle detection)
    - Conditional branching
    - Error recovery (retry, skip, fail-fast)
    - Checkpointing (resumable workflows)
    - Cancellation support
    
    Step definition:
    {
        "tool": "tool_name",
        "params": {"key": "value"},
        "depends_on": [0],          // step indices this depends on
        "condition": {              // optional: skip if condition not met
            "type": "previous_step_success",
            "step_index": 0
        },
        "max_retries": 2,           // optional: retry count
        "on_failure": "skip"        // optional: skip | fail | retry
    }
    
    Usage:
        engine = ExecutionEngine(db, registry)
        orchestrator = WorkflowOrchestrator(db, engine)
        workflow = await orchestrator.create_workflow(
            user_id=1,
            name="Data pipeline",
            steps=[
                {"tool": "fetch_data", "params": {"url": "..."}},
                {"tool": "process_data", "params": {}, "depends_on": [0]},
                {"tool": "store_data", "params": {}, "depends_on": [1]},
            ]
        )
        completed = await orchestrator.run_workflow(workflow.id)
    """
    
    def __init__(self, db: Session, execution_engine: ExecutionEngine):
        self.db = db
        self.engine = execution_engine
    
    async def create_workflow(self, user_id: int, name: str,
                             steps: List[Dict[str, Any]],
                             description: Optional[str] = None) -> Workflow:
        """Create a workflow.
        
        Validates:
        - No circular dependencies
        - All depends_on indices are valid
        - Steps have required fields
        """
        # Validate step structure
        for i, step in enumerate(steps):
            if "tool" not in step:
                raise ValueError(f"Step {i} missing 'tool' field")
            step["status"] = step.get("status", "pending")
            step["depends_on"] = step.get("depends_on", [])
            step["max_retries"] = step.get("max_retries", 0)
            step["on_failure"] = step.get("on_failure", "fail")
        
        # Validate DAG (no cycles)
        if self._has_cycle(steps):
            raise ValueError("Workflow step dependencies contain a cycle")
        
        # Validate dependency indices
        for i, step in enumerate(steps):
            for dep in step.get("depends_on", []):
                if dep < 0 or dep >= len(steps):
                    raise ValueError(f"Step {i} depends on invalid index {dep}")
        
        workflow = Workflow(
            user_id=user_id,
            name=name,
            description=description,
            steps=steps,
            status="idle",
            current_step=0,
            created_at=datetime.utcnow(),
        )
        self.db.add(workflow)
        self.db.commit()
        self.db.refresh(workflow)
        return workflow
    
    async def run_workflow(self, workflow_id: int) -> Workflow:
        """Execute a workflow from the beginning (or from checkpoint).
        
        Steps are executed in dependency order. Each step is:
        1. Checked for dependency satisfaction
        2. Evaluated for conditions
        3. Executed via the execution engine
        4. Result stored in step
        5. Status updated
        """
        workflow = self.db.query(Workflow).filter(Workflow.id == workflow_id).first()
        if not workflow:
            raise ValueError(f"Workflow {workflow_id} not found")
        
        workflow.status = "running"
        workflow.last_run = datetime.utcnow()
        start_time = time.time()
        self.db.commit()
        
        try:
            for i, step in enumerate(workflow.steps):
                workflow.current_step = i
                self.db.commit()
                
                # Check dependencies
                deps = step.get("depends_on", [])
                deps_ok = True
                for dep_idx in deps:
                    if dep_idx < len(workflow.steps):
                        dep_step = workflow.steps[dep_idx]
                        if dep_step.get("status") not in ("completed", "skipped"):
                            deps_ok = False
                            break
                
                if not deps_ok:
                    step["status"] = "failed"
                    step["error"] = "Dependencies not satisfied"
                    workflow.status = "failed"
                    workflow.error_message = f"Step {i}: dependencies not satisfied"
                    self.db.commit()
                    return workflow
                
                # Check condition
                condition = step.get("condition")
                if condition and not self._evaluate_condition(condition, workflow, i):
                    step["status"] = "skipped"
                    continue
                
                # Execute step
                tool_name = step.get("tool")
                params = step.get("params", {})
                
                # Parameter substitution: reference previous step results
                params = self._resolve_params(params, workflow.steps)
                
                max_retries = step.get("max_retries", 0)
                last_error = None
                
                for attempt in range(max_retries + 1):
                    execution = await self.engine.execute_tool(
                        user_id=workflow.user_id,
                        tool_name=tool_name,
                        params=params,
                        auto_verify=True,
                        workflow_id=workflow.id,
                        metadata={"workflow_step": i, "attempt": attempt},
                    )
                    
                    step["result"] = execution.result
                    step["status"] = execution.status
                    
                    if execution.status == "success":
                        last_error = None
                        break
                    else:
                        last_error = execution.error_message
                        step["error"] = last_error
                
                if step.get("status") != "success":
                    on_failure = step.get("on_failure", "fail")
                    if on_failure == "skip":
                        step["status"] = "skipped"
                        continue
                    else:  # "fail"
                        workflow.status = "failed"
                        workflow.error_message = f"Step {i} failed: {last_error}"
                        self.db.commit()
                        return workflow
            
            # All steps completed
            workflow.status = "completed"
            workflow.last_run_status = "completed"
            workflow.run_count = (workflow.run_count or 0) + 1
            end_time = time.time()
            workflow.total_duration_ms = int((end_time - start_time) * 1000)
            self.db.commit()
            self.db.refresh(workflow)
            return workflow
            
        except Exception as e:
            workflow.status = "failed"
            workflow.error_message = str(e)
            self.db.commit()
            self.db.refresh(workflow)
            return workflow
    
    async def cancel_workflow(self, workflow_id: int) -> Workflow:
        """Cancel a running workflow."""
        workflow = self.db.query(Workflow).filter(Workflow.id == workflow_id).first()
        if not workflow:
            raise ValueError(f"Workflow {workflow_id} not found")
        
        workflow.status = "cancelled"
        workflow.last_run_status = "cancelled"
        self.db.commit()
        self.db.refresh(workflow)
        return workflow
    
    async def get_workflow(self, workflow_id: int) -> Optional[Workflow]:
        """Get a workflow by ID."""
        return self.db.query(Workflow).filter(Workflow.id == workflow_id).first()
    
    async def get_user_workflows(self, user_id: int, status: Optional[str] = None,
                                limit: int = 50) -> List[Workflow]:
        """Get workflows for a user."""
        query = self.db.query(Workflow).filter(Workflow.user_id == user_id)
        if status:
            query = query.filter(Workflow.status == status)
        return query.order_by(Workflow.created_at.desc()).limit(limit).all()
    
    async def duplicate_workflow(self, workflow_id: int, new_name: str) -> Workflow:
        """Create a copy of an existing workflow."""
        original = self.db.query(Workflow).filter(Workflow.id == workflow_id).first()
        if not original:
            raise ValueError(f"Workflow {workflow_id} not found")
        
        return await self.create_workflow(
            user_id=original.user_id,
            name=new_name,
            steps=[
                {k: v for k, v in step.items() if k not in ("status", "result", "error")}
                for step in original.steps
            ],
            description=f"Copy of {original.name}",
        )
    
    def _has_cycle(self, steps: List[Dict[str, Any]]) -> bool:
        """Detect cycles in the dependency graph using DFS."""
        n = len(steps)
        WHITE, GRAY, BLACK = 0, 1, 2
        color = {i: WHITE for i in range(n)}
        
        adj = {}
        for step in steps:
            idx = step.get("step", 0)
            adj[idx] = step.get("depends_on", [])
        
        def dfs(node: int) -> bool:
            color[node] = GRAY
            for neighbor in adj.get(node, []):
                if neighbor >= n:
                    continue
                if color[neighbor] == GRAY:
                    return True
                if color[neighbor] == WHITE and dfs(neighbor):
                    return True
            color[node] = BLACK
            return False
        
        for i in range(n):
            if color[i] == WHITE and dfs(i):
                return True
        return False
    
    def _evaluate_condition(self, condition: Dict[str, Any],
                           workflow: Workflow, current_step: int) -> bool:
        """Evaluate a workflow step condition."""
        condition_type = condition.get("type", "")
        
        if condition_type == "always":
            return True
        elif condition_type == "previous_step_success":
            step_idx = condition.get("step_index", current_step - 1)
            if 0 <= step_idx < len(workflow.steps):
                return workflow.steps[step_idx].get("status") == "completed"
            return False
        elif condition_type == "variable_equals":
            var_name = condition.get("variable", "")
            expected = condition.get("value")
            # Check previous step results for variable
            for i in range(current_step - 1, -1, -1):
                result = workflow.steps[i].get("result", {})
                if isinstance(result, dict) and var_name in result:
                    return result[var_name] == expected
            return False
        
        return True
    
    def _resolve_params(self, params: Dict[str, Any],
                       steps: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Resolve parameter references to previous step results.
        
        Supports:
        - {"$ref": "step.0.result.key"} → steps[0].result["key"]
        - {"$ref": "step.1.status"} → steps[1].status
        """
        import re
        
        resolved = {}
        ref_pattern = re.compile(r'^step\.(\d+)\.(result|status)\.(.+)$')
        
        for key, value in params.items():
            if isinstance(value, dict) and "$ref" in value:
                ref = value["$ref"]
                match = ref_pattern.match(ref)
                if match:
                    step_idx = int(match.group(1))
                    field = match.group(2)
                    nested_key = match.group(3)
                    
                    if step_idx < len(steps):
                        step = steps[step_idx]
                        if field == "result" and step.get("result"):
                            resolved[key] = step["result"].get(nested_key)
                        elif field == "status":
                            resolved[key] = step.get("status")
                    else:
                        resolved[key] = None
                else:
                    resolved[key] = value
            else:
                resolved[key] = value
        
        return resolved
```

## Security Considerations

1. **Tool validation:** All parameters validated against schema before execution — prevents type confusion attacks
2. **Dangerous pattern detection:** Blocks destructive operations (rm -rf, DROP TABLE, etc.) before execution
3. **Confirmation gate:** Dangerous tools require explicit confirmation — prevents accidental destructive actions
4. **Timeout enforcement:** All tool executions have maximum timeout — prevents resource exhaustion
5. **Execution history:** All executions recorded with full context — enables security auditing
6. **Workflow parameter resolution:** `$ref` resolution only reads from existing step data — no code injection
