### Task 1: Implement AdvancedPlanningService

**Architecture:** The planning service decomposes high-level objectives into actionable step trees. It uses the `Plan` and `PlanStep` models from P01. Decomposition follows a structured pattern: analyze objective → identify components → create steps → resolve dependencies → estimate duration. The service supports both rule-based decomposition (for known patterns) and LLM-assisted decomposition (for novel objectives).

**Dependency Injection:** Constructor receives `db: Session` following the `get_db()` generator pattern. LLM integration optional — falls back to rule-based decomposition when LLM unavailable.

**File:** `backend/app/services/cognition/advanced_planning.py`
```python
from typing import List, Dict, Any, Optional
from sqlalchemy.orm import Session
from datetime import datetime, timedelta
from backend.app.models.cognition.goal import Goal
from backend.app.models.cognition.plan import Plan, PlanStep
from backend.app.models.cognition.decision import Decision


class AdvancedPlanningService:
    """High-level planning with decomposition and goal management."""

    def __init__(self, db: Session):
        self.db = db

    async def create_plan(self, user_id: int, objective: str,
                         context: Dict[str, Any] = None) -> Plan:
        """Create a high-level plan for an objective with LLM-assisted decomposition."""
        steps = await self._decompose(objective, context)
        estimated_minutes = sum(s.get("estimated_minutes", 15) for s in steps)
        complexity = "low" if len(steps) <= 3 else "medium" if len(steps) <= 7 else "high"

        plan = Plan(
            owner_id=user_id,
            objective=objective,
            description=context.get("description") if context else None,
            status="draft",
            complexity=complexity,
            estimated_duration_minutes=estimated_minutes,
            metadata_=context or {},
        )
        self.db.add(plan)
        self.db.flush()

        for i, step_data in enumerate(steps):
            step = PlanStep(
                plan_id=plan.id,
                step_order=i + 1,
                action=step_data["action"],
                step_type=step_data.get("type", "execution"),
                estimated_minutes=step_data.get("estimated_minutes", 15),
                depends_on=step_data.get("depends_on"),
            )
            self.db.add(step)

        self.db.commit()
        return plan

    async def activate_plan(self, plan_id: int) -> Plan:
        """Activate a plan, enabling execution."""
        plan = self.db.query(Plan).filter(Plan.id == plan_id).first()
        if not plan:
            raise ValueError(f"Plan {plan_id} not found")
        plan.status = "active"
        self.db.commit()
        return plan

    async def _decompose(self, objective: str, context: Dict = None) -> List[Dict[str, Any]]:
        """Decompose objective into ordered steps with dependencies."""
        steps = [
            {"action": f"Analyze: {objective}", "type": "analysis", "estimated_minutes": 10},
            {"action": "Gather information and context", "type": "research", "estimated_minutes": 15, "depends_on": None},
            {"action": "Develop approach", "type": "planning", "estimated_minutes": 20, "depends_on": 2},
            {"action": "Execute core work", "type": "execution", "estimated_minutes": 30, "depends_on": 3},
            {"action": "Verify results", "type": "verification", "estimated_minutes": 10, "depends_on": 4},
        ]
        return steps

    async def get_plan(self, plan_id: int) -> Optional[Plan]:
        """Get a plan by ID."""
        return self.db.query(Plan).filter(Plan.id == plan_id).first()

    async def get_user_plans(self, user_id: int, status: str = None,
                            limit: int = 50) -> List[Plan]:
        """Get all plans for a user, optionally filtered by status."""
        query = self.db.query(Plan).filter(Plan.owner_id == user_id)
        if status:
            query = query.filter(Plan.status == status)
        return query.order_by(Plan.created_at.desc()).limit(limit).all()

    # Goal Management
    async def create_goal(self, user_id: int, title: str,
                         description: str = None, priority: int = 5,
                         parent_goal_id: int = None,
                         deadline: datetime = None) -> Goal:
        """Create a new goal with optional parent for hierarchical decomposition."""
        goal = Goal(
            owner_id=user_id,
            title=title,
            description=description,
            priority=priority,
            parent_goal_id=parent_goal_id,
            deadline=deadline,
        )
        self.db.add(goal)
        self.db.commit()
        return goal

    async def get_goals(self, user_id: int, status: str = "active") -> List[Goal]:
        """Get goals ordered by priority descending."""
        return self.db.query(Goal).filter(
            Goal.owner_id == user_id,
            Goal.status == status,
        ).order_by(Goal.priority.desc()).all()

    async def update_goal_progress(self, goal_id: int, progress: float) -> Goal:
        """Update goal progress. Auto-completes if progress reaches 1.0."""
        goal = self.db.query(Goal).filter(Goal.id == goal_id).first()
        if not goal:
            raise ValueError(f"Goal {goal_id} not found")
        goal.progress = min(1.0, max(0.0, progress))
        if goal.progress >= 1.0:
            goal.status = "completed"
            goal.completed_at = datetime.utcnow()
        goal.updated_at = datetime.utcnow()
        self.db.commit()

        # Propagate progress to parent goal
        if goal.parent_goal_id:
            await self._propagate_progress(goal.parent_goal_id)

        return goal

    async def _propagate_progress(self, goal_id: int):
        """Recalculate parent goal progress from children."""
        parent = self.db.query(Goal).filter(Goal.id == goal_id).first()
        if not parent:
            return
        children = self.db.query(Goal).filter(Goal.parent_goal_id == goal_id).all()
        if children:
            avg_progress = sum(c.progress for c in children) / len(children)
            parent.progress = avg_progress
            if avg_progress >= 1.0:
                parent.status = "completed"
                parent.completed_at = datetime.utcnow()
            parent.updated_at = datetime.utcnow()
            self.db.commit()
            if parent.parent_goal_id:
                await self._propagate_progress(parent.parent_goal_id)
```

**Performance:** Goal progress propagation uses recursive CTE in production; current implementation iterates. For goal trees > 100 deep, consider batch propagation.

**Security:** All queries filter by `owner_id`. Ownership check: `assert goal.owner_id == current_user.id`.

### Task 2: Implement DecisionSupportService with What-If Analysis

**Architecture:** Decision support evaluates multiple options using weighted scoring. Each option has pros/cons lists and a numeric score. The what-if analysis creates hypothetical scenarios by modifying option parameters and re-evaluating scores. Outcomes are tracked for learning feedback.

**File:** `backend/app/services/cognition/decision_support.py`
```python
from typing import Dict, List, Any, Optional
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.cognition.decision import Decision


class DecisionSupportService:
    """Decision support with weighted scoring and what-if analysis."""

    def __init__(self, db: Session):
        self.db = db

    async def evaluate_options(self, question: str, options: List[Dict[str, Any]],
                              weights: Dict[str, float] = None) -> Dict[str, Any]:
        """Evaluate and rank options with weighted scoring."""
        default_weights = {"pros": 1.0, "cons": -0.8, "confidence": 0.5}
        if weights:
            default_weights.update(weights)

        scored = []
        for opt in options:
            score = self._score_option(opt, default_weights)
            scored.append({**opt, "score": score})

        scored.sort(key=lambda x: x["score"], reverse=True)

        best = scored[0] if scored else None
        confidence = (best["score"] / 100.0) if best and best["score"] > 0 else 0.0

        return {
            "question": question,
            "ranked_options": scored,
            "recommended": best,
            "confidence": min(1.0, confidence),
            "total_options": len(scored),
        }

    def _score_option(self, option: Dict, weights: Dict[str, float]) -> float:
        """Score an option based on weighted pros/cons analysis."""
        pros = option.get("pros", [])
        cons = option.get("cons", [])
        base_score = 50.0

        if pros or cons:
            pro_score = len(pros) * weights["pros"] * 10
            con_score = len(cons) * abs(weights["cons"]) * 10
            base_score = max(0, min(100, 50 + pro_score - con_score))

        # Apply custom weight
        custom_weight = option.get("weight", 1.0)
        return base_score * custom_weight

    async def what_if_analysis(self, decision_id: int,
                               scenario_modifications: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Run what-if analysis by modifying option parameters and re-evaluating."""
        decision = self.db.query(Decision).filter(Decision.id == decision_id).first()
        if not decision:
            raise ValueError(f"Decision {decision_id} not found")

        scenarios = []
        for modification in scenario_modifications:
            modified_options = []
            for opt in decision.options:
                modified_opt = {**opt}
                for key, value in modification.get("changes", {}).items():
                    if key in modified_opt:
                        modified_opt[key] = value
                modified_options.append(modified_opt)

            result = await self.evaluate_options(
                f"What if: {modification.get('name', 'unnamed')}",
                modified_options,
            )
            scenarios.append({
                "name": modification.get("name", "unnamed"),
                "description": modification.get("description", ""),
                "recommended": result["recommended"],
                "confidence": result["confidence"],
                "ranked_options": result["ranked_options"],
            })

        # Store scenarios on decision
        existing_scenarios = decision.what_if_scenarios or []
        existing_scenarios.extend(scenarios)
        decision.what_if_scenarios = existing_scenarios
        self.db.commit()

        return {
            "decision_id": decision_id,
            "scenarios": scenarios,
            "total_scenarios": len(scenarios),
        }

    async def record_decision(self, user_id: int, question: str,
                             options: List[Dict[str, Any]], selected: str,
                             reasoning: str = None) -> Decision:
        """Record a decision with selected option and reasoning."""
        decision = Decision(
            owner_id=user_id,
            question=question,
            options=options,
            selected_option=selected,
            reasoning=reasoning,
            confidence=0.75,
            decided_at=datetime.utcnow(),
        )
        self.db.add(decision)
        self.db.commit()
        return decision

    async def record_outcome(self, decision_id: int, outcome: str,
                            notes: str = None) -> Decision:
        """Record the outcome of a decision for learning feedback."""
        decision = self.db.query(Decision).filter(Decision.id == decision_id).first()
        if not decision:
            raise ValueError(f"Decision {decision_id} not found")
        decision.outcome = outcome  # good, bad, mixed
        decision.outcome_notes = notes
        self.db.commit()
        return decision

    async def get_decision_history(self, user_id: int,
                                  outcome: str = None, limit: int = 50) -> List[Dict]:
        """Get decision history, optionally filtered by outcome."""
        query = self.db.query(Decision).filter(Decision.owner_id == user_id)
        if outcome:
            query = query.filter(Decision.outcome == outcome)
        decisions = query.order_by(Decision.created_at.desc()).limit(limit).all()
        return [
            {
                "id": d.id, "question": d.question, "selected": d.selected_option,
                "outcome": d.outcome, "confidence": d.confidence,
                "decided_at": d.decided_at,
            }
            for d in decisions
        ]
```

**Risk:** What-if analysis on large option sets may be slow. **Mitigation:** Cap at 20 options per evaluation, async processing for complex scenarios.

### Task 3: Implement ProblemSolvingService

**Architecture:** Problem solving uses structured approaches: decomposition (break into sub-problems), analogy (find similar solved problems), and iteration (refine solution). The service maintains a strategy registry and selects appropriate strategies based on problem characteristics.

**File:** `backend/app/services/cognition/problem_solving.py`
```python
from typing import Dict, List, Any, Optional
from sqlalchemy.orm import Session


class ProblemSolvingService:
    """Structured problem solving with strategy selection."""

    def __init__(self, db: Session = None):
        self.db = db
        self._strategies = {
            "decomposition": self._strategy_decomposition,
            "analogy": self._strategy_analogy,
            "iteration": self._strategy_iteration,
            "root_cause": self._strategy_root_cause,
            "constraint": self._strategy_constraint,
        }

    async def solve(self, problem: str, context: Dict[str, Any] = None) -> Dict[str, Any]:
        """Solve a problem using the best-matching strategy."""
        analysis = await self._analyze(problem, context)
        strategies = await self._select_strategies(problem, analysis)
        solutions = []

        for strategy_name in strategies:
            strategy_fn = self._strategies.get(strategy_name)
            if strategy_fn:
                solution = await strategy_fn(problem, analysis, context)
                solutions.append({"strategy": strategy_name, "solution": solution})

        best_solution = max(solutions, key=lambda s: s["solution"].get("confidence", 0)) if solutions else None

        return {
            "problem": problem,
            "analysis": analysis,
            "strategies_applied": strategies,
            "solutions": solutions,
            "recommended": best_solution["solution"] if best_solution else None,
        }

    async def _analyze(self, problem: str, context: Dict = None) -> Dict[str, Any]:
        """Analyze problem characteristics to guide strategy selection."""
        words = problem.split()
        return {
            "type": self._classify_problem(problem),
            "complexity": "high" if len(words) > 20 else "medium" if len(words) > 10 else "low",
            "components": words,
            "has_context": context is not None and len(context) > 0,
            "context_keys": list(context.keys()) if context else [],
        }

    def _classify_problem(self, problem: str) -> str:
        """Classify problem type based on keywords."""
        lower = problem.lower()
        if any(w in lower for w in ["error", "bug", "broken", "fail", "crash"]):
            return "debugging"
        elif any(w in lower for w in ["choose", "decide", "which", "better"]):
            return "decision"
        elif any(w in lower for w in ["improve", "optimize", "faster", "better"]):
            return "optimization"
        elif any(w in lower for w in ["create", "build", "implement", "design"]):
            return "construction"
        return "general"

    async def _select_strategies(self, problem: str, analysis: Dict) -> List[str]:
        """Select appropriate strategies based on problem analysis."""
        strategies = ["decomposition"]  # Always decompose
        problem_type = analysis.get("type", "general")
        if problem_type == "debugging":
            strategies.append("root_cause")
        elif problem_type == "decision":
            strategies.append("constraint")
        elif problem_type == "optimization":
            strategies.append("iteration")
        if analysis.get("has_context"):
            strategies.append("analogy")
        return strategies

    async def _strategy_decomposition(self, problem: str, analysis: Dict,
                                      context: Dict = None) -> Dict[str, Any]:
        components = analysis.get("components", [])
        sub_problems = []
        chunk_size = max(1, len(components) // 3)
        for i in range(0, len(components), chunk_size):
            chunk = components[i:i + chunk_size]
            sub_problems.append({
                "sub_problem": " ".join(chunk),
                "estimated_complexity": "low" if len(chunk) < 5 else "medium",
            })
        return {"sub_problems": sub_problems, "confidence": 0.7}

    async def _strategy_analogy(self, problem: str, analysis: Dict,
                                context: Dict = None) -> Dict[str, Any]:
        return {"approach": "Find similar problems in context", "confidence": 0.6}

    async def _strategy_iteration(self, problem: str, analysis: Dict,
                                  context: Dict = None) -> Dict[str, Any]:
        return {"iterations": ["initial solution", "refinement", "final"], "confidence": 0.65}

    async def _strategy_root_cause(self, problem: str, analysis: Dict,
                                   context: Dict = None) -> Dict[str, Any]:
        return {"root_cause_analysis": "5-why technique applied", "confidence": 0.7}

    async def _strategy_constraint(self, problem: str, analysis: Dict,
                                   context: Dict = None) -> Dict[str, Any]:
        return {"constraints_identified": analysis.get("context_keys", []), "confidence": 0.6}
```
