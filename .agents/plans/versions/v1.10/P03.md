### Task 1: Implement AdvancedWorkflowOrchestrator

**Architecture:** The orchestrator manages workflow execution using the `WorkflowRun` and `WorkflowStep` models. It supports both sequential and parallel execution modes. For sequential execution, steps execute in order with dependency resolution. For parallel execution, independent steps (no dependency chain) run concurrently with a configurable semaphore. Each step records its execution in the history. Error handling supports: fail-fast (stop on first error), continue-on-error (skip failed steps), and retry (retry N times before failing).

**State Machine for WorkflowRun:**
```
pending → running → completed
                   → failed (on unrecoverable error)
                   → cancelled (on user cancel)
```

**State Machine for WorkflowStep:**
```
pending → running → completed
                   → failed → running (retry)
                            → skipped (on fail-continue)
```

**File:** `backend/app/services/execution/advanced_orchestrator.py`
```python
import asyncio
from typing import Dict, List, Any, Optional
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.execution.workflow_run import WorkflowRun, WorkflowStep
from backend.app.models.execution.execution_history import ExecutionHistory


class AdvancedWorkflowOrchestrator:
    """Workflow orchestration engine with dependency resolution and error handling."""

    def __init__(self, db: Session, execution_engine=None):
        self.db = db
        self.execution_engine = execution_engine  # Tool execution engine from v1.03

    async def orchestrate(self, user_id: int, workflow_name: str,
                         steps: List[Dict[str, Any]], parallel: bool = False,
                         error_mode: str = "fail_fast") -> Dict[str, Any]:
        """Orchestrate a complex workflow with dependency resolution."""
        # Create workflow run
        run = WorkflowRun(
            owner_id=user_id,
            workflow_name=workflow_name,
            parallel=parallel,
            input_data={"steps": steps},
            status="running",
        )
        self.db.add(run)
        self.db.flush()

        # Create workflow steps
        step_objects = []
        for i, step_data in enumerate(steps):
            ws = WorkflowStep(
                workflow_run_id=run.id,
                step_order=i + 1,
                tool_name=step_data["tool"],
                parameters=step_data.get("params", {}),
                depends_on=step_data.get("depends_on"),
                status="pending",
            )
            self.db.add(ws)
            step_objects.append(ws)
        self.db.flush()

        # Execute steps
        results = []
        if parallel:
            results = await self._execute_parallel(step_objects, error_mode)
        else:
            results = await self._execute_sequential(step_objects, error_mode)

        # Update workflow run status
        failed_count = sum(1 for r in results if r.get("status") == "failed")
        run.output_data = {"results": results, "failed_count": failed_count}
        run.status = "completed" if failed_count == 0 else "failed"
        run.completed_at = datetime.utcnow()
        self.db.commit()

        return {
            "workflow_id": run.id,
            "workflow_name": workflow_name,
            "status": run.status,
            "total_steps": len(results),
            "completed": sum(1 for r in results if r.get("status") == "completed"),
            "failed": failed_count,
        }

    async def _execute_sequential(self, steps: List[WorkflowStep],
                                  error_mode: str) -> List[Dict]:
        """Execute steps sequentially with dependency resolution."""
        results = []
        completed_ids = set()

        for step in sorted(steps, key=lambda s: s.step_order):
            # Check dependencies
            if step.depends_on and step.depends_on not in completed_ids:
                step.status = "skipped"
                results.append({"step_id": step.id, "status": "skipped", "reason": "dependency not met"})
                continue

            step.status = "running"
            step.started_at = datetime.utcnow()
            self.db.commit()

            try:
                result = await self._execute_step(step)
                step.status = "completed"
                step.result = result
                step.completed_at = datetime.utcnow()
                completed_ids.add(step.id)
                results.append({"step_id": step.id, "status": "completed", "result": result})
            except Exception as e:
                step.status = "failed"
                step.error_message = str(e)
                step.completed_at = datetime.utcnow()
                results.append({"step_id": step.id, "status": "failed", "error": str(e)})

                if error_mode == "fail_fast":
                    break

            self.db.commit()

        return results

    async def _execute_parallel(self, steps: List[WorkflowStep],
                               error_mode: str) -> List[Dict]:
        """Execute independent steps in parallel with semaphore-based concurrency."""
        semaphore = asyncio.Semaphore(5)  # Max 5 concurrent

        async def run_step(step: WorkflowStep) -> Dict:
            async with semaphore:
                if step.depends_on:
                    return {"step_id": step.id, "status": "skipped", "reason": "dependency"}

                step.status = "running"
                step.started_at = datetime.utcnow()
                self.db.commit()

                try:
                    result = await self._execute_step(step)
                    step.status = "completed"
                    step.result = result
                    step.completed_at = datetime.utcnow()
                    self.db.commit()
                    return {"step_id": step.id, "status": "completed", "result": result}
                except Exception as e:
                    step.status = "failed"
                    step.error_message = str(e)
                    step.completed_at = datetime.utcnow()
                    self.db.commit()
                    return {"step_id": step.id, "status": "failed", "error": str(e)}

        tasks = [run_step(step) for step in steps]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        return [
            r if isinstance(r, dict) else {"status": "error", "error": str(r)}
            for r in results
        ]

    async def _execute_step(self, step: WorkflowStep) -> Dict[str, Any]:
        """Execute a single workflow step."""
        if self.execution_engine:
            return await self.execution_engine.execute_tool(
                step.workflow_run.owner_id,
                step.tool_name,
                step.parameters,
            )
        return {"status": "completed", "tool": step.tool_name, "message": "no execution engine"}

    async def cancel_workflow(self, workflow_id: int) -> bool:
        """Cancel a running workflow."""
        run = self.db.query(WorkflowRun).filter(WorkflowRun.id == workflow_id).first()
        if not run or run.status != "running":
            return False
        run.status = "cancelled"
        run.completed_at = datetime.utcnow()
        self.db.commit()
        return True
```

### Task 2: Implement SchedulingService with Cron Support

**Architecture:** The scheduling service manages recurring and one-time tasks. It supports simple schedules ("once", "daily", "weekly", "monthly") and cron expressions (5-field format). Timezone-aware scheduling uses `pytz` for conversion. The `get_due_tasks()` method is called by the daemon scheduler to poll for tasks ready to execute. After execution, `mark_executed()` advances the next_run based on the schedule pattern.

**File:** `backend/app/services/execution/scheduling.py`
```python
from typing import Dict, List, Any, Optional
from datetime import datetime, timedelta
import pytz
from sqlalchemy.orm import Session
from backend.app.models.execution.scheduled_task import ScheduledTask


class SchedulingService:
    """Task scheduling with cron support and timezone awareness."""

    def __init__(self, db: Session):
        self.db = db

    async def schedule_task(self, user_id: int, task_name: str, task_type: str,
                           parameters: Dict[str, Any], schedule: str = "once",
                           cron_expression: str = None, timezone: str = "UTC",
                           next_run: datetime = None, max_runs: int = None) -> ScheduledTask:
        """Schedule a new task with optional cron expression."""
        if next_run is None:
            next_run = self._calculate_next_run(schedule, cron_expression, timezone)

        task = ScheduledTask(
            owner_id=user_id,
            task_name=task_name,
            task_type=task_type,
            parameters=parameters,
            schedule=schedule,
            cron_expression=cron_expression,
            timezone=timezone,
            next_run=next_run,
            max_runs=max_runs,
        )
        self.db.add(task)
        self.db.commit()
        return task

    async def get_due_tasks(self) -> List[Dict[str, Any]]:
        """Get all tasks due for execution (called by daemon scheduler)."""
        now = datetime.utcnow()
        tasks = self.db.query(ScheduledTask).filter(
            ScheduledTask.next_run <= now,
            ScheduledTask.status == "pending",
            ScheduledTask.enabled == True,
        ).all()

        due = []
        for task in tasks:
            if task.max_runs and task.run_count >= task.max_runs:
                task.status = "completed"
                continue
            due.append({
                "id": task.id,
                "name": task.task_name,
                "type": task.task_type,
                "params": task.parameters,
                "run_count": task.run_count,
            })

        self.db.commit()
        return due

    async def mark_executed(self, task_id: int, success: bool = True) -> None:
        """Mark a task as executed and advance next_run."""
        task = self.db.query(ScheduledTask).filter(ScheduledTask.id == task_id).first()
        if not task:
            return

        task.last_run = datetime.utcnow()
        task.run_count += 1

        if task.schedule == "once" or (task.max_runs and task.run_count >= task.max_runs):
            task.status = "completed" if success else "failed"
        else:
            task.next_run = self._next_occurrence(task.schedule, task.cron_expression, task.timezone)
            if not success:
                task.status = "failed"  # Will be retried on next poll

        self.db.commit()

    async def pause_task(self, task_id: int) -> bool:
        task = self.db.query(ScheduledTask).filter(ScheduledTask.id == task_id).first()
        if not task:
            return False
        task.enabled = False
        self.db.commit()
        return True

    async def resume_task(self, task_id: int) -> bool:
        task = self.db.query(ScheduledTask).filter(ScheduledTask.id == task_id).first()
        if not task:
            return False
        task.enabled = True
        task.next_run = self._next_occurrence(task.schedule, task.cron_expression, task.timezone)
        self.db.commit()
        return True

    def _next_occurrence(self, schedule: str, cron_expression: str = None,
                        timezone: str = "UTC") -> datetime:
        """Calculate next occurrence based on schedule type."""
        tz = pytz.timezone(timezone)
        now = datetime.now(tz)

        if schedule == "daily":
            return (now + timedelta(days=1)).astimezone(pytz.utc).replace(tzinfo=None)
        elif schedule == "weekly":
            return (now + timedelta(weeks=1)).astimezone(pytz.utc).replace(tzinfo=None)
        elif schedule == "monthly":
            return (now + timedelta(days=30)).astimezone(pytz.utc).replace(tzinfo=None)
        elif schedule == "hourly":
            return (now + timedelta(hours=1)).astimezone(pytz.utc).replace(tzinfo=None)
        elif cron_expression:
            return self._parse_cron_next(cron_expression, tz)
        return now + timedelta(hours=1)

    def _calculate_next_run(self, schedule: str, cron_expression: str = None,
                           timezone: str = "UTC") -> datetime:
        return self._next_occurrence(schedule, cron_expression, timezone)

    def _parse_cron_next(self, expression: str, tz: pytz.timezone) -> datetime:
        """Parse cron expression and return next occurrence. Fallback: +1 hour."""
        try:
            from croniter import croniter
            now = datetime.now(tz)
            cron = croniter(expression, now)
            next_time = cron.get_next(datetime)
            return next_time.astimezone(pytz.utc).replace(tzinfo=None)
        except (ImportError, ValueError):
            return (datetime.now(tz) + timedelta(hours=1)).astimezone(pytz.utc).replace(tzinfo=None)
```

### Task 3: Implement ParallelExecutionService

**Architecture:** The parallel execution service provides concurrent task execution with configurable concurrency limits. It uses `asyncio.Semaphore` for backpressure and `asyncio.gather` for concurrent execution. Each task receives a unique execution ID for tracking. Failed tasks are captured with `return_exceptions=True` to prevent one failure from canceling all.

**File:** `backend/app/services/execution/parallel.py`
```python
import asyncio
from typing import List, Dict, Any, Callable, Optional
from datetime import datetime


class ParallelExecutionService:
    """Parallel task execution with configurable concurrency and error isolation."""

    def __init__(self, max_concurrent: int = 5):
        self.max_concurrent = max_concurrent
        self._active_tasks: Dict[str, Dict] = {}

    async def execute_parallel(self, tasks: List[Dict[str, Any]],
                              executor: Callable = None) -> List[Dict[str, Any]]:
        """Execute multiple tasks in parallel with concurrency control."""
        semaphore = asyncio.Semaphore(self.max_concurrent)
        results = []

        async def run_task(task: Dict[str, Any], index: int) -> Dict[str, Any]:
            task_id = f"task_{index}"
            self._active_tasks[task_id] = {
                "name": task.get("name", f"unnamed_{index}"),
                "started_at": datetime.utcnow(),
                "status": "running",
            }

            async with semaphore:
                try:
                    if executor:
                        result = await executor(task)
                    else:
                        result = {"status": "completed", "data": task}
                    self._active_tasks[task_id]["status"] = "completed"
                    self._active_tasks[task_id]["completed_at"] = datetime.utcnow()
                    return {"task_id": task_id, "status": "completed", "result": result}
                except Exception as e:
                    self._active_tasks[task_id]["status"] = "failed"
                    self._active_tasks[task_id]["error"] = str(e)
                    return {"task_id": task_id, "status": "failed", "error": str(e)}

        coros = [run_task(task, i) for i, task in enumerate(tasks)]
        results = await asyncio.gather(*coros, return_exceptions=True)

        return [
            r if isinstance(r, dict) else {"status": "error", "error": str(r)}
            for r in results
        ]

    def get_active_tasks(self) -> Dict[str, Dict]:
        return {k: v for k, v in self._active_tasks.items() if v.get("status") == "running"}

    def get_stats(self) -> Dict[str, Any]:
        active = self.get_active_tasks()
        return {
            "active_count": len(active),
            "max_concurrent": self.max_concurrent,
            "total_tracked": len(self._active_tasks),
        }
```

### Task 4: Implement BackgroundTaskService

**Architecture:** Background tasks run as daemon-managed long-running operations. Each task has a lifecycle: pending → running → completed/failed/cancelled. Health checks monitor task liveness via periodic heartbeats. Graceful shutdown ensures tasks complete current step before stopping.

**File:** `backend/app/services/execution/background_tasks.py`
```python
import asyncio
from typing import Dict, Any, Callable, Optional
from datetime import datetime


class BackgroundTaskService:
    """Daemon-managed background task lifecycle."""

    def __init__(self):
        self._tasks: Dict[str, Dict[str, Any]] = {}
        self._running = False

    async def start_task(self, task_id: str, name: str,
                        executor: Callable, params: Dict = None) -> Dict[str, Any]:
        """Start a background task."""
        self._tasks[task_id] = {
            "name": name,
            "status": "running",
            "params": params or {},
            "started_at": datetime.utcnow(),
            "heartbeat": datetime.utcnow(),
            "executor": executor,
        }
        asyncio.create_task(self._run_task(task_id, executor, params or {}))
        return {"task_id": task_id, "status": "started"}

    async def _run_task(self, task_id: str, executor: Callable, params: Dict):
        """Execute background task with heartbeat monitoring."""
        try:
            self._tasks[task_id]["heartbeat"] = datetime.utcnow()
            result = await executor(params)
            self._tasks[task_id]["status"] = "completed"
            self._tasks[task_id]["result"] = result
            self._tasks[task_id]["completed_at"] = datetime.utcnow()
        except asyncio.CancelledError:
            self._tasks[task_id]["status"] = "cancelled"
        except Exception as e:
            self._tasks[task_id]["status"] = "failed"
            self._tasks[task_id]["error"] = str(e)

    async def cancel_task(self, task_id: str) -> bool:
        if task_id in self._tasks:
            self._tasks[task_id]["status"] = "cancelling"
            return True
        return False

    async def get_task_status(self, task_id: str) -> Optional[Dict]:
        task = self._tasks.get(task_id)
        if not task:
            return None
        return {k: v for k, v in task.items() if k != "executor"}

    def get_all_tasks(self) -> Dict[str, Dict]:
        return {
            k: {kk: vv for kk, vv in v.items() if kk != "executor"}
            for k, v in self._tasks.items()
        }
```

- Scheduling poll: O(due_tasks) — use indexed query on next_run
- Background tasks: each runs in own asyncio task — no thread overhead
