## Implementation Tasks

### Task 1: Implement RecoveryService

**Architecture:** The recovery service creates state snapshots before operations and restores state on failure. Recovery points are stored in the `recovery_points` table. The service supports three recovery actions: rollback (undo changes), retry (re-execute from checkpoint), and skip (mark as completed without execution). Recovery points are automatically cleaned up after successful completion with configurable retention.

**File:** `backend/app/services/execution/recovery.py`
```python
from typing import Dict, Optional, Any
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from backend.app.models.execution.recovery_point import RecoveryPoint
from backend.app.models.execution.execution_history import ExecutionHistory


class RecoveryService:
    """Recovery mechanism with state snapshots and configurable recovery actions."""

    def __init__(self, db: Session):
        self.db = db

    async def create_recovery_point(self, execution_id: int,
                                   state_snapshot: Dict[str, Any],
                                   recovery_action: str = "rollback") -> RecoveryPoint:
        """Create a recovery point before an operation."""
        rp = RecoveryPoint(
            execution_id=execution_id,
            state_snapshot=state_snapshot,
            recovery_action=recovery_action,
        )
        self.db.add(rp)
        self.db.commit()
        return rp

    async def recover(self, execution_id: int) -> Dict[str, Any]:
        """Attempt to recover from a failed execution."""
        rp = self.db.query(RecoveryPoint).filter(
            RecoveryPoint.execution_id == execution_id
        ).order_by(RecoveryPoint.id.desc()).first()

        if not rp:
            return {"recovered": False, "reason": "No recovery point found"}

        return {
            "recovered": True,
            "state": rp.state_snapshot,
            "action": rp.recovery_action,
            "recovery_point_id": rp.id,
        }

    async def cleanup_old_recovery_points(self, retention_days: int = 30) -> int:
        """Remove recovery points older than retention period."""
        cutoff = datetime.utcnow() - timedelta(days=retention_days)
        old_points = self.db.query(RecoveryPoint).filter(
            RecoveryPoint.created_at < cutoff
        ).all()
        count = len(old_points)
        for rp in old_points:
            self.db.delete(rp)
        self.db.commit()
        return count

    async def get_recovery_points(self, execution_id: int) -> list:
        """Get all recovery points for an execution."""
        points = self.db.query(RecoveryPoint).filter(
            RecoveryPoint.execution_id == execution_id
        ).order_by(RecoveryPoint.id.desc()).all()
        return [
            {
                "id": p.id, "state": p.state_snapshot,
                "action": p.recovery_action, "created_at": p.created_at,
            }
            for p in points
        ]
```

**Risk:** State snapshots may contain sensitive data. **Mitigation:** Encrypt snapshots at rest using the vault system, implement snapshot size limits (max 10MB), and log access to recovery points.

### Task 2: Implement RollbackService

**Architecture:** The rollback service reverses completed operations by restoring state from recovery points. It supports compensating transactions (inverse operations) for operations that cannot be simply undone. The service marks operations as "rolled_back" and records the rollback in execution history. Non-rollbackable operations (e.g., network calls, file deletions) are flagged with `rollback_available=0`.

**File:** `backend/app/services/execution/rollback.py`
```python
from typing import Dict, Any, Optional
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.execution.execution_history import ExecutionHistory
from backend.app.models.execution.recovery_point import RecoveryPoint


class RollbackService:
    """Rollback completed operations with state restoration."""

    def __init__(self, db: Session):
        self.db = db

    async def rollback(self, execution_id: int) -> Dict[str, Any]:
        """Rollback a completed execution."""
        history = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.id == execution_id
        ).first()

        if not history:
            return {"rolled_back": False, "reason": "Execution not found"}

        if not history.rollback_available:
            return {
                "rolled_back": False,
                "reason": "Rollback not available for this operation type",
            }

        # Get latest recovery point
        rp = self.db.query(RecoveryPoint).filter(
            RecoveryPoint.execution_id == execution_id
        ).order_by(RecoveryPoint.id.desc()).first()

        # Update execution status
        history.status = "rolled_back"
        history.completed_at = datetime.utcnow()

        # Create rollback history entry
        rollback_entry = ExecutionHistory(
            owner_id=history.owner_id,
            operation=f"rollback:{history.operation}",
            operation_type="rollback",
            input_data={"original_execution_id": execution_id, "recovery_state": rp.state_snapshot if rp else None},
            output_data={"rolled_back": True},
            status="success",
            rollback_available=0,
        )
        self.db.add(rollback_entry)
        self.db.commit()

        return {
            "rolled_back": True,
            "original_operation": history.operation,
            "original_status": history.status,
            "rollback_execution_id": rollback_entry.id,
        }

    async def can_rollback(self, execution_id: int) -> Dict[str, Any]:
        """Check if an execution can be rolled back."""
        history = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.id == execution_id
        ).first()

        if not history:
            return {"can_rollback": False, "reason": "Execution not found"}

        return {
            "can_rollback": history.rollback_available == 1,
            "status": history.status,
            "reason": "Rollback available" if history.rollback_available else "Rollback not supported",
        }

    async def get_rollback_history(self, user_id: int, limit: int = 20) -> list:
        """Get recent rollback operations."""
        rollbacks = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.owner_id == user_id,
            ExecutionHistory.operation_type == "rollback",
        ).order_by(ExecutionHistory.started_at.desc()).limit(limit).all()

        return [
            {
                "id": r.id, "operation": r.operation,
                "status": r.status, "started_at": r.started_at,
            }
            for r in rollbacks
        ]
```

**Risk:** Rollback of side-effecting operations may not be safe. **Mitigation:** Mark non-rollbackable operations with `rollback_available=0`, implement compensating transactions for common patterns, and require explicit rollback approval for destructive operations.

### Task 3: Implement ExecutionHistoryService

**Architecture:** The execution history service provides a complete audit trail of all operations. It records input/output data, duration, status, and rollback availability. The service supports filtering by operation type, time range, and status. Analytics include success rate, average duration, and failure categorization. A retention policy manages old records.

**File:** `backend/app/services/execution/history.py`
```python
from typing import List, Dict, Any, Optional
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from sqlalchemy import func
from backend.app.models.execution.execution_history import ExecutionHistory


class ExecutionHistoryService:
    """Execution history with audit trail and analytics."""

    def __init__(self, db: Session):
        self.db = db

    async def record(self, user_id: int, operation: str, operation_type: str,
                    input_data: Dict = None, output_data: Dict = None,
                    status: str = "success", duration_ms: int = 0,
                    parent_execution_id: int = None) -> ExecutionHistory:
        """Record an execution in history."""
        history = ExecutionHistory(
            owner_id=user_id,
            operation=operation,
            operation_type=operation_type,
            input_data=input_data,
            output_data=output_data,
            status=status,
            duration_ms=duration_ms,
            parent_execution_id=parent_execution_id,
            rollback_available=1 if status == "success" and operation_type != "rollback" else 0,
            started_at=datetime.utcnow() - timedelta(milliseconds=duration_ms),
            completed_at=datetime.utcnow(),
        )
        self.db.add(history)
        self.db.commit()
        return history

    async def get_history(self, user_id: int, limit: int = 50,
                         operation_type: str = None, status: str = None,
                         since: datetime = None) -> List[Dict[str, Any]]:
        """Get execution history with filters."""
        query = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.owner_id == user_id,
        )
        if operation_type:
            query = query.filter(ExecutionHistory.operation_type == operation_type)
        if status:
            query = query.filter(ExecutionHistory.status == status)
        if since:
            query = query.filter(ExecutionHistory.started_at >= since)

        histories = query.order_by(ExecutionHistory.started_at.desc()).limit(limit).all()
        return [
            {
                "id": h.id, "operation": h.operation,
                "operation_type": h.operation_type, "status": h.status,
                "duration_ms": h.duration_ms, "rollback_available": h.rollback_available,
                "started_at": h.started_at, "completed_at": h.completed_at,
            }
            for h in histories
        ]

    async def get_stats(self, user_id: int, days: int = 30) -> Dict[str, Any]:
        """Get execution statistics for the last N days."""
        since = datetime.utcnow() - timedelta(days=days)
        histories = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.owner_id == user_id,
            ExecutionHistory.started_at >= since,
        ).all()

        if not histories:
            return {"total": 0, "success_rate": 0.0, "avg_duration_ms": 0}

        total = len(histories)
        success_count = sum(1 for h in histories if h.status == "success")
        failed_count = sum(1 for h in histories if h.status == "failed")
        durations = [h.duration_ms for h in histories if h.duration_ms]

        return {
            "total": total,
            "success_count": success_count,
            "failed_count": failed_count,
            "success_rate": success_count / total if total > 0 else 0.0,
            "avg_duration_ms": sum(durations) / len(durations) if durations else 0,
            "by_type": self._group_by_type(histories),
        }

    def _group_by_type(self, histories: list) -> Dict[str, int]:
        types = {}
        for h in histories:
            types[h.operation_type] = types.get(h.operation_type, 0) + 1
        return types

    async def cleanup_old_history(self, retention_days: int = 90) -> int:
        """Remove execution history older than retention period."""
        cutoff = datetime.utcnow() - timedelta(days=retention_days)
        old = self.db.query(ExecutionHistory).filter(
            ExecutionHistory.started_at < cutoff
        ).all()
        count = len(old)
        for h in old:
            self.db.delete(h)
        self.db.commit()
        return count
```
