### Task 1: Create Planning API Router

**Architecture:** The planning router groups goal, decision, and plan endpoints under `/api/v1/cognition/planning`. All endpoints use dependency injection for `get_db()` and `get_current_user`. Response models defined on every endpoint. Ownership checks enforced: `resource.owner_id == current_user.id`.

**File:** `backend/app/api/v1/cognition/planning.py`
```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import Optional, List
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.schemas.cognition.goal import GoalCreate, GoalUpdate, GoalResponse
from backend.app.schemas.cognition.decision import DecisionCreate, DecisionResponse
from backend.app.schemas.cognition.plan import PlanCreate, PlanResponse
from backend.app.services.cognition.advanced_planning import AdvancedPlanningService
from backend.app.services.cognition.decision_support import DecisionSupportService

router = APIRouter()


# Goals
@router.post("/goals", response_model=GoalResponse)
async def create_goal(
    body: GoalCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Create a new goal."""
    service = AdvancedPlanningService(db)
    goal = await service.create_goal(
        user_id=current_user.id, title=body.title,
        description=body.description, priority=body.priority,
        parent_goal_id=body.parent_goal_id, deadline=body.deadline,
    )
    return goal


@router.get("/goals", response_model=List[GoalResponse])
async def get_goals(
    status: str = Query("active", regex="^(active|completed|abandoned|paused)$"),
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get all goals for the current user."""
    service = AdvancedPlanningService(db)
    return await service.get_goals(user_id=current_user.id, status=status)


@router.patch("/goals/{goal_id}", response_model=GoalResponse)
async def update_goal(
    goal_id: int,
    body: GoalUpdate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Update goal progress or status."""
    service = AdvancedPlanningService(db)
    return await service.update_goal_progress(goal_id, body.progress or 0.0)


# Plans
@router.post("/plans", response_model=PlanResponse)
async def create_plan(
    body: PlanCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Create a plan for an objective."""
    service = AdvancedPlanningService(db)
    return await service.create_plan(
        user_id=current_user.id, objective=body.objective,
    )


@router.get("/plans", response_model=List[PlanResponse])
async def get_plans(
    status: Optional[str] = None,
    limit: int = Query(50, ge=1, le=100),
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get all plans for the current user."""
    service = AdvancedPlanningService(db)
    return await service.get_user_plans(
        user_id=current_user.id, status=status, limit=limit,
    )


# Decisions
@router.post("/decisions", response_model=DecisionResponse)
async def create_decision(
    body: DecisionCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Record a decision with options and reasoning."""
    service = DecisionSupportService(db)
    options = [opt.model_dump() for opt in body.options]
    return await service.record_decision(
        user_id=current_user.id, question=body.question,
        options=options, selected=options[0]["option"],
    )
```

### Task 2: Create Orchestration API Router

**File:** `backend/app/api/v1/execution/orchestration.py`
```python
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from typing import Optional, List
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.schemas.execution.scheduled_task import ScheduledTaskCreate, ScheduledTaskResponse
from backend.app.schemas.execution.workflow_run import WorkflowRunCreate, WorkflowRunResponse
from backend.app.schemas.execution.execution_history import ExecutionHistoryResponse
from backend.app.services.execution.advanced_orchestrator import AdvancedWorkflowOrchestrator
from backend.app.services.execution.scheduling import SchedulingService
from backend.app.services.execution.history import ExecutionHistoryService
from backend.app.services.execution.rollback import RollbackService

router = APIRouter()


@router.post("/workflows", response_model=WorkflowRunResponse)
async def create_workflow(
    body: WorkflowRunCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Create and execute a workflow."""
    orchestrator = AdvancedWorkflowOrchestrator(db)
    steps = [step.model_dump() for step in body.steps]
    return await orchestrator.orchestrate(
        user_id=current_user.id, workflow_name=body.workflow_name,
        steps=steps, parallel=body.parallel,
    )


@router.post("/schedule", response_model=ScheduledTaskResponse)
async def schedule_task(
    body: ScheduledTaskCreate,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Schedule a task for later execution."""
    service = SchedulingService(db)
    return await service.schedule_task(
        user_id=current_user.id, task_name=body.task_name,
        task_type=body.task_type, parameters=body.parameters,
        schedule=body.schedule, cron_expression=body.cron_expression,
        timezone=body.timezone,
    )


@router.get("/history", response_model=List[ExecutionHistoryResponse])
async def get_history(
    limit: int = Query(50, ge=1, le=200),
    operation_type: Optional[str] = None,
    status: Optional[str] = None,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get execution history with filters."""
    service = ExecutionHistoryService(db)
    return await service.get_history(
        user_id=current_user.id, limit=limit,
        operation_type=operation_type, status=status,
    )


@router.post("/history/{execution_id}/rollback")
async def rollback(
    execution_id: int,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Rollback a completed execution."""
    service = RollbackService(db)
    return await service.rollback(execution_id)


@router.get("/stats")
async def get_stats(
    days: int = Query(30, ge=1, le=365),
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get execution statistics."""
    service = ExecutionHistoryService(db)
    return await service.get_stats(user_id=current_user.id, days=days)
```

### Task 3: Create Frontend API Client

**File:** `frontend/features/cognition/planning-api.ts`
```typescript
import { apiClient } from '@/shared/api/client';

export const planningApi = {
  // Goals
  createGoal: (title: string, description?: string, priority: number = 5, deadline?: string) =>
    apiClient.post('/cognition/planning/goals', { title, description, priority, deadline }),
  getGoals: (status: string = 'active') =>
    apiClient.get(`/cognition/planning/goals?status=${status}`),
  updateGoalProgress: (goalId: number, progress: number) =>
    apiClient.patch(`/cognition/planning/goals/${goalId}`, { progress }),

  // Plans
  createPlan: (objective: string) =>
    apiClient.post('/cognition/planning/plans', { objective }),
  getPlans: (status?: string, limit: number = 50) =>
    apiClient.get(`/cognition/planning/plans?limit=${limit}${status ? `&status=${status}` : ''}`),

  // Decisions
  createDecision: (question: string, options: any[]) =>
    apiClient.post('/cognition/planning/decisions', { question, options }),
};

export const orchestrationApi = {
  // Workflows
  createWorkflow: (name: string, steps: any[], parallel: boolean = false) =>
    apiClient.post('/execution/orchestration/workflows', { workflow_name: name, steps, parallel }),

  // Scheduling
  scheduleTask: (name: string, type: string, schedule: string, params: any = {}) =>
    apiClient.post('/execution/orchestration/schedule', {
      task_name: name, task_type: type, schedule, parameters: params,
    }),

  // History
  getHistory: (limit: number = 50, operationType?: string, status?: string) =>
    apiClient.get(`/execution/orchestration/history?limit=${limit}${
      operationType ? `&operation_type=${operationType}` : ''
    }${status ? `&status=${status}` : ''}`),

  // Rollback
  rollback: (executionId: number) =>
    apiClient.post(`/execution/orchestration/history/${executionId}/rollback`),

  // Stats
  getStats: (days: number = 30) =>
    apiClient.get(`/execution/orchestration/stats?days=${days}`),
};
```

## Security Considerations

- All endpoints require JWT authentication via `get_current_user`
- Ownership checks enforced: `resource.owner_id == current_user.id`
- Request body validated via Pydantic schemas
- Rate limiting applied to all endpoints
- CSRF protection enabled for state-changing operations

## Performance Considerations

- History query limited to 200 records max to prevent OOM
- Stats aggregation uses indexed queries
- Goal hierarchy queries use eager loading with `selectin`
- Response models exclude large JSON fields by default
