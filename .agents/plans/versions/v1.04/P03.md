### Task 1: Implement Project Scanner Service

**Architecture:** The project scanner detects project type, frameworks, configuration, and feature flags (tests, CI, Docker). It's similar to the repository scanner but focused on project-level metadata rather than code statistics.

**File:** `backend/app/services/awareness/project.py`

```python
import os
import json
from datetime import datetime
from typing import Optional, List, Dict
from sqlalchemy.orm import Session
from backend.app.models.awareness.project_index import ProjectIndex

# Project type detection indicators
PROJECT_TYPE_INDICATORS = {
    'package.json': 'node',
    'requirements.txt': 'python',
    'pyproject.toml': 'python',
    'setup.py': 'python',
    'Cargo.toml': 'rust',
    'go.mod': 'go',
    'Gemfile': 'ruby',
    'pom.xml': 'java',
    'build.gradle': 'java',
    'build.gradle.kts': 'java',
    'Makefile': 'c',
    'CMakeLists.txt': 'cpp',
    'composer.json': 'php',
    'pubspec.yaml': 'dart',
    'Package.swift': 'swift',
    'build.sbt': 'scala',
}

# Framework detection indicators
FRAMEWORK_INDICATORS = {
    # Python
    'django': ['django'],
    'manage.py': ['django'],
    'asgi.py': ['fastapi'],
    'flask': ['flask'],
    
    # Node.js
    'next.config.js': ['next.js'],
    'next.config.mjs': ['next.js'],
    'nuxt.config.js': ['nuxt'],
    'nuxt.config.ts': ['nuxt'],
    'gatsby-config.js': ['gatsby'],
    'remix.config.js': ['remix'],
    'vite.config.js': ['vite'],
    'vite.config.ts': ['vite'],
    'angular.json': ['angular'],
    'svelte.config.js': ['svelte'],
    'express': ['express'],
    
    # Rust
    'actix-web': ['actix-web'],
    'axum': ['axum'],
    'rocket': ['rocket'],
    
    # Go
    'gin': ['gin'],
    'echo': ['echo'],
    'fiber': ['fiber'],
}

# Feature detection indicators
FEATURE_INDICATORS = {
    'has_tests': ['tests/', 'test/', '__tests__/', 'spec/', 'pytest.ini', 'jest.config.js', 'vitest.config.ts'],
    'has_ci': ['.github/workflows/', '.gitlab-ci.yml', '.circleci/', 'Jenkinsfile', '.travis.yml'],
    'has_docker': ['Dockerfile', 'docker-compose.yml', 'docker-compose.yaml', '.dockerignore'],
}


class ProjectScannerService:
    """Project type and framework detection service."""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def scan_project(
        self, user_id: int, project_path: str
    ) -> ProjectIndex:
        """Scan a project and detect its type, frameworks, and features.
        
        Returns:
            ProjectIndex with detected metadata
        """
        project_type = self._detect_type(project_path)
        frameworks = self._detect_frameworks(project_path)
        configuration = self._load_configuration(project_path, project_type)
        features = self._detect_features(project_path)
        
        # Upsert
        existing = self.db.query(ProjectIndex).filter(
            ProjectIndex.user_id == user_id,
            ProjectIndex.project_path == project_path
        ).first()
        
        if existing:
            existing.project_type = project_type
            existing.frameworks = json.dumps(frameworks)
            existing.configuration = json.dumps(configuration)
            existing.has_tests = features.get("has_tests", 0)
            existing.has_ci = features.get("has_ci", 0)
            existing.has_docker = features.get("has_docker", 0)
            existing.last_scanned = datetime.utcnow()
            self.db.commit()
            self.db.refresh(existing)
            return existing
        
        project_index = ProjectIndex(
            user_id=user_id,
            project_path=project_path,
            project_name=os.path.basename(project_path),
            project_type=project_type,
            frameworks=json.dumps(frameworks),
            configuration=json.dumps(configuration),
            has_tests=features.get("has_tests", 0),
            has_ci=features.get("has_ci", 0),
            has_docker=features.get("has_docker", 0),
            last_scanned=datetime.utcnow()
        )
        self.db.add(project_index)
        self.db.commit()
        self.db.refresh(project_index)
        return project_index
    
    def _detect_type(self, project_path: str) -> str:
        """Detect project type from config files."""
        for filename, ptype in PROJECT_TYPE_INDICATORS.items():
            if os.path.exists(os.path.join(project_path, filename)):
                return ptype
        return "unknown"
    
    def _detect_frameworks(self, project_path: str) -> List[str]:
        """Detect frameworks from file indicators."""
        frameworks = []
        for indicator, fw_list in FRAMEWORK_INDICATORS.items():
            if os.path.exists(os.path.join(project_path, indicator)):
                frameworks.extend(fw_list)
        return list(set(frameworks))  # Deduplicate
    
    def _detect_features(self, project_path: str) -> Dict[str, int]:
        """Detect project features (tests, CI, Docker)."""
        features = {}
        for feature, indicators in FEATURE_INDICATORS.items():
            features[feature] = 1 if any(
                os.path.exists(os.path.join(project_path, ind))
                for ind in indicators
            ) else 0
        return features
    
    def _load_configuration(
        self, project_path: str, project_type: str
    ) -> Dict:
        """Load key configuration values based on project type."""
        config = {}
        
        if project_type == "python":
            config = self._load_python_config(project_path)
        elif project_type == "node":
            config = self._load_node_config(project_path)
        elif project_type == "rust":
            config = self._load_rust_config(project_path)
        
        return config
    
    def _load_python_config(self, project_path: str) -> Dict:
        """Load Python project configuration."""
        config = {}
        
        # pyproject.toml
        pyproject = os.path.join(project_path, "pyproject.toml")
        if os.path.exists(pyproject):
            try:
                with open(pyproject) as f:
                    content = f.read()
                    # Basic parsing for python version and test framework
                    if "pytest" in content:
                        config["test_framework"] = "pytest"
                    if "black" in content:
                        config["formatter"] = "black"
                    if "ruff" in content:
                        config["linter"] = "ruff"
            except Exception:
                pass
        
        # requirements.txt
        req_file = os.path.join(project_path, "requirements.txt")
        if os.path.exists(req_file):
            try:
                with open(req_file) as f:
                    deps = [line.strip().split("==")[0] for line in f if line.strip() and not line.startswith("#")]
                    config["dependency_count"] = len(deps)
            except Exception:
                pass
        
        return config
    
    def _load_node_config(self, project_path: str) -> Dict:
        """Load Node.js project configuration."""
        config = {}
        pkg_file = os.path.join(project_path, "package.json")
        if os.path.exists(pkg_file):
            try:
                with open(pkg_file) as f:
                    pkg = json.load(f)
                    config["name"] = pkg.get("name", "")
                    config["version"] = pkg.get("version", "")
                    config["node_version"] = pkg.get("engines", {}).get("node", "")
                    config["scripts"] = list(pkg.get("scripts", {}).keys())
            except (json.JSONDecodeError, Exception):
                pass
        return config
    
    def _load_rust_config(self, project_path: str) -> Dict:
        """Load Rust project configuration."""
        config = {}
        cargo_file = os.path.join(project_path, "Cargo.toml")
        if os.path.exists(cargo_file):
            try:
                with open(cargo_file) as f:
                    content = f.read()
                    # Basic parsing
                    for line in content.split("\n"):
                        line = line.strip()
                        if line.startswith("name"):
                            config["name"] = line.split("=")[1].strip().strip('"')
                        elif line.startswith("version"):
                            config["version"] = line.split("=")[1].strip().strip('"')
            except Exception:
                pass
        return config
```

**Key Design Decisions:**
- **File-existence-based detection:** Simple and fast. No content analysis required for v1.03.
- **Framework detection via indicators:** Multiple indicators can map to the same framework. Deduplication ensures clean results.
- **Feature detection:** Binary flags for tests, CI, Docker. Simple directory/file existence checks.
- **Configuration loading:** Basic parsing of config files. Not comprehensive but covers common cases.

**Risk Notes:**
- Risk: Framework detection may produce false positives. Mitigation: Multiple indicators per framework reduce false positive rate.
- Risk: Configuration parsing is fragile. Mitigation: Try-except blocks prevent crashes. Empty config on failure.

### Task 2: Implement Device Info Service

**Architecture:** The device info service collects hardware and OS details using Python's `platform` module and optional `psutil`. It performs an upsert: if device info already exists for the user, update it; otherwise, create new.

**File:** `backend/app/services/awareness/device.py`

```python
import platform
import sys
from datetime import datetime
from typing import Optional
from sqlalchemy.orm import Session
from backend.app.models.awareness.device_info import DeviceInfo

# Try to import psutil (optional)
try:
    import psutil
    HAS_PSUTIL = True
except ImportError:
    HAS_PSUTIL = False


class DeviceInfoService:
    """Device hardware and OS information service."""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def get_device_info(self, user_id: int) -> DeviceInfo:
        """Get or create device information.
        
        Performs upsert: updates existing record or creates new one.
        """
        hostname = platform.node()
        os_type = platform.system().lower()
        os_version = platform.version()
        python_version = platform.python_version()
        
        cpu_cores = self._get_cpu_cores()
        cpu_model = self._get_cpu_model()
        total_memory_gb = self._get_total_memory()
        available_memory_gb = self._get_available_memory()
        disk_total_gb = self._get_disk_total()
        disk_used_gb = self._get_disk_used()
        
        existing = self.db.query(DeviceInfo).filter(
            DeviceInfo.user_id == user_id
        ).first()
        
        if existing:
            existing.hostname = hostname
            existing.os_type = os_type
            existing.os_version = os_version
            existing.python_version = python_version
            existing.cpu_cores = cpu_cores
            existing.cpu_model = cpu_model
            existing.total_memory_gb = total_memory_gb
            existing.available_memory_gb = available_memory_gb
            existing.disk_total_gb = disk_total_gb
            existing.disk_used_gb = disk_used_gb
            existing.last_checked = datetime.utcnow()
            self.db.commit()
            self.db.refresh(existing)
            return existing
        
        device = DeviceInfo(
            user_id=user_id,
            hostname=hostname,
            os_type=os_type,
            os_version=os_version,
            python_version=python_version,
            cpu_cores=cpu_cores,
            cpu_model=cpu_model,
            total_memory_gb=total_memory_gb,
            available_memory_gb=available_memory_gb,
            disk_total_gb=disk_total_gb,
            disk_used_gb=disk_used_gb,
            last_checked=datetime.utcnow()
        )
        self.db.add(device)
        self.db.commit()
        self.db.refresh(device)
        return device
    
    def _get_cpu_cores(self) -> Optional[int]:
        """Get CPU core count."""
        if HAS_PSUTIL:
            return psutil.cpu_count(logical=True)
        return platform.cpu_count()
    
    def _get_cpu_model(self) -> Optional[str]:
        """Get CPU model name."""
        if HAS_PSUTIL:
            # psutil doesn't provide CPU model directly on all platforms
            pass
        # Fallback to platform
        return platform.processor() or None
    
    def _get_total_memory(self) -> int:
        """Get total memory in GB."""
        if HAS_PSUTIL:
            return psutil.virtual_memory().total // (1024 ** 3)
        return 0
    
    def _get_available_memory(self) -> int:
        """Get available memory in GB."""
        if HAS_PSUTIL:
            return psutil.virtual_memory().available // (1024 ** 3)
        return 0
    
    def _get_disk_total(self) -> int:
        """Get total disk in GB."""
        if HAS_PSUTIL:
            return psutil.disk_usage('/').total // (1024 ** 3)
        return 0
    
    def _get_disk_used(self) -> int:
        """Get used disk in GB."""
        if HAS_PSUTIL:
            return psutil.disk_usage('/').used // (1024 ** 3)
        return 0
```

**Key Design Decisions:**
- **Graceful psutil fallback:** If `psutil` is not installed, the service still works with basic `platform` info. Memory and disk will be 0.
- **Upsert pattern:** Each user has exactly one device info record. Updated on each check.
- **Non-real-time:** This is a snapshot, not streaming metrics. Updated on-demand via API.

**Risk Notes:**
- Risk: `psutil` not available on all platforms. Mitigation: Try-except import. Graceful degradation.
- Risk: Disk usage changes between checks. Mitigation: Acceptable for snapshot-based design.

### Task 3: Implement Environment Scanner Service

**Architecture:** The environment scanner returns safe, non-secret environment variables. It uses a strict allowlist of variable names that are safe to expose. Never returns `SECRET_KEY`, `DATABASE_URL`, `PASSWORD`, `TOKEN`, or `API_KEY`.

**File:** `backend/app/services/awareness/environment.py`

```python
import os
from typing import Dict, List


# Strict allowlist of safe environment variables
SAFE_ENV_VARS = [
    "HOME", "SHELL", "LANG", "PATH", "USER", "LOGNAME",
    "HOSTNAME", "TERM", "EDITOR", "VISUAL",
    "CORTEX_ROOT", "APP_NAME", "API_V1_PREFIX",
    "PYTHON_VERSION", "NODE_VERSION",
]

# Patterns that indicate secrets (never return these)
SECRET_PATTERNS = [
    "SECRET", "PASSWORD", "TOKEN", "API_KEY", "PRIVATE",
    "CREDENTIALS", "AUTH", "DATABASE_URL", "REDIS_URL",
]


class EnvironmentScannerService:
    """Safe environment variable scanner."""
    
    async def get_environment(self, user_id: int) -> Dict[str, str]:
        """Get relevant (safe) environment variables.
        
        Returns only variables in the allowlist.
        Never returns secrets.
        """
        safe_vars = {}
        for var_name in SAFE_ENV_VARS:
            value = os.environ.get(var_name)
            if value:
                safe_vars[var_name] = value
        return safe_vars
    
    async def get_system_paths(self) -> Dict[str, str]:
        """Get important system paths."""
        return {
            "home": os.path.expanduser("~"),
            "temp": os.path.expanduser("~/tmp"),
            "downloads": os.path.expanduser("~/Downloads"),
            "desktop": os.path.expanduser("~/Desktop"),
            "documents": os.path.expanduser("~/Documents"),
        }
    
    async def get_safe_env_check(self) -> List[Dict[str, str]]:
        """Check which safe variables are set.
        
        Returns list of {name, is_set} for all safe variables.
        """
        return [
            {"name": var, "is_set": os.environ.get(var) is not None}
            for var in SAFE_ENV_VARS
        ]
```

**Key Design Decision:**
- **Strict allowlist:** Only explicitly listed variables are returned. This is a whitelist approach, not a blacklist. Much safer than trying to filter out secrets.

**Risk Notes:**
- Risk: New safe variables need code changes to add. Mitigation: Allowlist is small and focused. Easy to maintain.
- Risk: `CORTEX_ROOT` could expose internal paths. Mitigation: Acceptable for local-first system.

### Task 4: Implement System Health Service

**Architecture:** The system health service monitors individual services by running health check functions. Each service has a check function that returns success/failure. The service tracks status, response time, and error messages.

**File:** `backend/app/services/awareness/health.py`

```python
import time
from datetime import datetime
from typing import Callable, Awaitable, List, Optional
from sqlalchemy.orm import Session
from backend.app.models.awareness.system_health import SystemHealth


# Default health check timeout (seconds)
HEALTH_CHECK_TIMEOUT = 5.0


class SystemHealthService:
    """System health monitoring service."""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def check_service(
        self,
        service_name: str,
        check_func: Callable[[], Awaitable[None]]
    ) -> SystemHealth:
        """Check a service's health.
        
        Args:
            service_name: Name of the service
            check_func: Async function that raises on failure
            
        Returns:
            Updated SystemHealth record
        """
        start_time = time.monotonic()
        try:
            await check_func()
            response_time = int((time.monotonic() - start_time) * 1000)
            
            status = "healthy"
            error_message = None
        except Exception as e:
            response_time = int((time.monotonic() - start_time) * 1000)
            status = "down"
            error_message = str(e)[:500]  # Truncate error messages
        
        # Determine degraded status (response time > 2s)
        if status == "healthy" and response_time > 2000:
            status = "degraded"
        
        # Upsert
        existing = self.db.query(SystemHealth).filter(
            SystemHealth.service_name == service_name
        ).first()
        
        if existing:
            existing.status = status
            existing.response_time_ms = response_time
            existing.error_message = error_message
            existing.last_check = datetime.utcnow()
            self.db.commit()
            self.db.refresh(existing)
            return existing
        
        health = SystemHealth(
            service_name=service_name,
            status=status,
            response_time_ms=response_time,
            error_message=error_message,
            last_check=datetime.utcnow()
        )
        self.db.add(health)
        self.db.commit()
        self.db.refresh(health)
        return health
    
    async def check_all_services(
        self, checks: dict[str, Callable[[], Awaitable[None]]]
    ) -> List[SystemHealth]:
        """Check all registered services.
        
        Args:
            checks: Dict of {service_name: check_func}
            
        Returns:
            List of SystemHealth records
        """
        results = []
        for service_name, check_func in checks.items():
            health = await self.check_service(service_name, check_func)
            results.append(health)
        return results
    
    async def get_all_health(self) -> List[SystemHealth]:
        """Get health status of all services."""
        return self.db.query(SystemHealth).all()
    
    async def get_service_health(self, service_name: str) -> Optional[SystemHealth]:
        """Get health status of a specific service."""
        return self.db.query(SystemHealth).filter(
            SystemHealth.service_name == service_name
        ).first()
    
    async def get_overall_status(self) -> dict:
        """Get overall system health status."""
        services = self.db.query(SystemHealth).all()
        
        if not services:
            return {
                "overall_status": "unknown",
                "services": 0,
                "healthy": 0,
                "degraded": 0,
                "down": 0
            }
        
        healthy = sum(1 for s in services if s.status == "healthy")
        degraded = sum(1 for s in services if s.status == "degraded")
        down = sum(1 for s in services if s.status == "down")
        
        if down > 0:
            overall = "down"
        elif degraded > 0:
            overall = "degraded"
        else:
            overall = "healthy"
        
        return {
            "overall_status": overall,
            "services": len(services),
            "healthy": healthy,
            "degraded": degraded,
            "down": down
        }
```

**Key Design Decisions:**
- **Degraded status:** Response time > 2000ms triggers "degraded" status. This catches slow-but-functional services.
- **Error truncation:** Error messages truncated to 500 chars to prevent DB bloat.
- **Upsert pattern:** Each service has one health record. Updated on each check.
- **`get_overall_status`:** Provides a dashboard-ready summary.

**Risk Notes:**
- Risk: Health check functions may hang. Mitigation: Implement timeout in check functions (asyncio.wait_for).
- Risk: Concurrent health checks could overwhelm services. Mitigation: Sequential checks in `check_all_services`. Parallel in v1.08.
