## Objective

Create REST API endpoints for all awareness operations, build the frontend API client and React hooks, write comprehensive integration and E2E tests, and validate the complete awareness system end-to-end. This phase wires together all services from P02-P03 into a consumable API surface with a frontend dashboard.

---

## Purpose

Awareness services need REST API endpoints for the frontend to interact with. This phase creates the HTTP layer that maps HTTP methods to service calls, validates input via Pydantic schemas, handles authentication via JWT, and returns properly typed responses. The frontend provides a dashboard view of device info, system health, and repository structure.

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.awareness.file_watcher import FilesystemWatcherService
from backend.app.schemas.awareness.file_index import FileIndexList

router = APIRouter(prefix="/files", tags=["awareness-files"])


@router.post("/scan", response_model=dict)
async def scan_directory(
    directory: str = Query(..., min_length=1, max_length=1000),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Scan a directory and index all files."""
    service = FilesystemWatcherService(db)
    files, stats = await service.scan_directory(current_user.id, directory)
    return {
        "files_indexed": len(files),
        "stats": stats,
        "directory": directory
    }


@router.get("/changes", response_model=dict)
async def detect_changes(
    directory: str = Query(..., min_length=1, max_length=1000),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Detect file changes since last scan."""
    service = FilesystemWatcherService(db)
    changes = await service.detect_changes(current_user.id, directory)
    return {
        "created": len(changes["created"]),
        "modified": len(changes["modified"]),
        "deleted": len(changes["deleted"]),
        "details": changes
    }


@router.get("/summary", response_model=dict)
async def get_directory_summary(
    directory: str = Query(..., min_length=1, max_length=1000),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """Get a summary of files in a directory."""
    service = FilesystemWatcherService(db)
    return await service.get_directory_summary(current_user.id, directory)

import { useState, useEffect, useCallback } from 'react';
import { awarenessApi, DeviceInfo } from '../api';

export function useDeviceAwareness() {
  const [deviceInfo, setDeviceInfo] = useState<DeviceInfo | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const loadDeviceInfo = useCallback(async () => {
    try {
      setLoading(true);
      const response = await awarenessApi.getDeviceInfo();
      setDeviceInfo(response.data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load device info');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    loadDeviceInfo();
  }, [loadDeviceInfo]);

  return { deviceInfo, loading, error, reload: loadDeviceInfo };
}
```

import { useState, useEffect, useCallback } from 'react';
import { awarenessApi, SystemHealth, HealthSummary } from '../api';

export function useSystemHealth() {
  const [health, setHealth] = useState<SystemHealth | null>(null);
  const [summary, setSummary] = useState<HealthSummary | null>(null);
  const [loading, setLoading] = useState(true);

  const loadHealth = useCallback(async () => {
    try {
      setLoading(true);
      const [healthResp, summaryResp] = await Promise.all([
        awarenessApi.getSystemHealth(),
        awarenessApi.getHealthSummary()
      ]);
      setHealth(healthResp.data);
      setSummary(summaryResp.data);
    } catch (err) {
      console.error('Failed to load health:', err);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    loadHealth();
  }, [loadHealth]);

  // Auto-refresh every 30 seconds
  useEffect(() => {
    const interval = setInterval(loadHealth, 30000);
    return () => clearInterval(interval);
  }, [loadHealth]);

  return { health, summary, loading, reload: loadHealth };
}
```

```typescript
import { useState, useCallback } from 'react';
import { awarenessApi, RepositoryIndex } from '../api';

export function useRepositoryScan() {
  const [repo, setRepo] = useState<RepositoryIndex | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const scan = useCallback(async (repoPath: string) => {
    try {
      setLoading(true);
      setError(null);
      const response = await awarenessApi.scanRepository(repoPath);
      setRepo(response.data);
      return response.data;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Scan failed');
      return null;
    } finally {
      setLoading(false);
    }
  }, []);

  return { repo, loading, error, scan };
}
```

## Security Considerations

- **Authentication:** All endpoints require valid JWT.
- **Environment safety:** Strict allowlist prevents secret leakage. Code-reviewed.
- **Path validation:** File operations validate paths. No `../` traversal.
- **Rate limiting:** Scan endpoints rate-limited to prevent resource exhaustion.
- **Health check auth:** Health data is not publicly accessible.
