# services/developer/git_intelligence.py

import subprocess
from typing import Dict, List, Any


class GitIntelligenceService:
    """Provides git intelligence and repository analysis."""
    
    async def get_log(self, repo_path: str, limit: int = 20) -> List[Dict[str, str]]:
        """Get git log with details."""
        try:
            result = subprocess.run(
                ["git", "log", f"-{limit}", "--format=%H|%s|%an|%ai"],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if result.returncode != 0:
                return []
            
            commits = []
            for line in result.stdout.strip().split("\n"):
                if "|" in line:
                    parts = line.split("|", 3)
                    commits.append({
                        "hash": parts[0][:8],
                        "full_hash": parts[0],
                        "message": parts[1],
                        "author": parts[2],
                        "date": parts[3] if len(parts) > 3 else "",
                    })
            
            return commits
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return []
    
    async def analyze_commit(self, repo_path: str, commit_hash: str) -> Dict[str, Any]:
        """Analyze a specific commit."""
        try:
            info_result = subprocess.run(
                ["git", "show", "--stat", "--format=%H|%s|%an|%ai", commit_hash],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if info_result.returncode != 0:
                return {"error": "Commit not found"}
            
            lines = info_result.stdout.strip().split("\n")
            first_line = lines[0] if lines else ""
            parts = first_line.split("|", 3)
            
            files_changed = 0
            changed_files = []
            
            for line in lines[1:]:
                if "|" in line:
                    file_parts = line.strip().split("|")
                    if len(file_parts) == 2:
                        changed_files.append({
                            "file": file_parts[0].strip(),
                            "changes": file_parts[1].strip(),
                        })
                        files_changed += 1
            
            # Get insertion/deletion counts
            insertions = 0
            deletions = 0
            stat_result = subprocess.run(
                ["git", "diff", "--numstat", f"{commit_hash}~1", commit_hash],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if stat_result.returncode == 0:
                for line in stat_result.stdout.strip().split("\n"):
                    stat_parts = line.split("\t")
                    if len(stat_parts) == 3:
                        if stat_parts[0] != "-":
                            insertions += int(stat_parts[0])
                        if stat_parts[1] != "-":
                            deletions += int(stat_parts[1])
            
            return {
                "hash": parts[0] if parts else commit_hash,
                "short_hash": parts[0][:8] if parts else commit_hash[:8],
                "message": parts[1] if len(parts) > 1 else "",
                "author": parts[2] if len(parts) > 2 else "",
                "date": parts[3] if len(parts) > 3 else "",
                "files_changed": files_changed,
                "insertions": insertions,
                "deletions": deletions,
                "changed_files": changed_files,
            }
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return {"error": "Git not available"}
    
    async def get_branch_info(self, repo_path: str) -> Dict[str, Any]:
        """Get branch information."""
        try:
            current_result = subprocess.run(
                ["git", "branch", "--show-current"],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            current = current_result.stdout.strip()
            
            all_result = subprocess.run(
                ["git", "branch", "-a"],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            branches = []
            for line in all_result.stdout.strip().split("\n"):
                branch = line.strip().replace("* ", "")
                if branch:
                    branches.append({
                        "name": branch,
                        "is_remote": branch.startswith("remotes/"),
                        "is_current": branch == current,
                    })
            
            local = [b for b in branches if not b["is_remote"]]
            remote = [b for b in branches if b["is_remote"]]
            
            return {
                "current": current,
                "local": local,
                "remote": remote,
                "total": len(branches),
                "local_count": len(local),
                "remote_count": len(remote),
            }
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return {"error": "Git not available"}
    
    async def get_diff_summary(
        self, repo_path: str, base: str = "HEAD~1", head: str = "HEAD"
    ) -> Dict[str, Any]:
        """Get diff summary between two commits."""
        try:
            numstat_result = subprocess.run(
                ["git", "diff", "--numstat", f"{base}...{head}"],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if numstat_result.returncode != 0:
                return {"error": "Diff failed"}
            
            files_changed = []
            total_insertions = 0
            total_deletions = 0
            
            for line in numstat_result.stdout.strip().split("\n"):
                parts = line.split("\t")
                if len(parts) == 3:
                    ins = int(parts[0]) if parts[0] != "-" else 0
                    dels = int(parts[1]) if parts[1] != "-" else 0
                    files_changed.append({
                        "file": parts[2],
                        "insertions": ins,
                        "deletions": dels,
                    })
                    total_insertions += ins
                    total_deletions += dels
            
            return {
                "base": base,
                "head": head,
                "files_changed": files_changed,
                "total_files": len(files_changed),
                "total_insertions": total_insertions,
                "total_deletions": total_deletions,
            }
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return {"error": "Git not available"}
    
    async def blame_file(self, repo_path: str, file_path: str) -> Dict[str, Any]:
        """Git blame a file for authorship information."""
        try:
            result = subprocess.run(
                ["git", "blame", "--line-porcelain", file_path],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if result.returncode != 0:
                return {"error": "Blame failed"}
            
            author_lines: Dict[str, int] = {}
            for line in result.stdout.split("\n"):
                if line.startswith("author "):
                    author = line[len("author "):]
                    author_lines[author] = author_lines.get(author, 0) + 1
            
            sorted_authors = sorted(author_lines.items(), key=lambda x: x[1], reverse=True)
            
            return {
                "file": file_path,
                "authors": [
                    {"name": name, "lines": count}
                    for name, count in sorted_authors
                ],
                "total_lines": sum(author_lines.values()),
            }
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return {"error": "Git not available"}
    
    async def is_git_repo(self, path: str) -> bool:
        """Check if a path is inside a git repository."""
        try:
            result = subprocess.run(
                ["git", "rev-parse", "--git-dir"],
                cwd=path,
                capture_output=True,
                text=True,
                timeout=5,
            )
            return result.returncode == 0
        except (subprocess.TimeoutExpired, FileNotFoundError):
            return False

```python
# services/developer/cicd.py

import subprocess
import os
import json
from typing import Dict, List, Any


class CICDIntegrationService:
    """Integrates with CI/CD pipelines."""
    
    CI_DETECTORS = [
        (".github/workflows", "github_actions", "GitHub Actions"),
        (".gitlab-ci.yml", "gitlab_ci", "GitLab CI"),
        ("Jenkinsfile", "jenkins", "Jenkins"),
        (".circleci", "circle_ci", "CircleCI"),
        ("azure-pipelines.yml", "azure_devops", "Azure DevOps"),
        (".travis.yml", "travis_ci", "Travis CI"),
        ("bitbucket-pipelines.yml", "bitbucket_pipelines", "Bitbucket Pipelines"),
        ("cloudbuild.yaml", "cloud_build", "Google Cloud Build"),
    ]
    
    async def detect_ci(self, repo_path: str) -> Dict[str, Any]:
        """Detect CI/CD configuration."""
        detected = []
        
        for path, ci_type, display_name in self.CI_DETECTORS:
            full_path = os.path.join(repo_path, path)
            if os.path.exists(full_path):
                detected.append({
                    "type": ci_type,
                    "display_name": display_name,
                    "path": path,
                })
        
        return {
            "detected": detected,
            "count": len(detected),
            "primary": detected[0] if detected else None,
        }
    
    async def get_github_actions_workflows(self, repo_path: str) -> Dict[str, Any]:
        """Get GitHub Actions workflow details."""
        workflows_dir = os.path.join(repo_path, ".github", "workflows")
        if not os.path.exists(workflows_dir):
            return {"error": "No GitHub Actions workflows found"}
        
        workflows = []
        for fname in os.listdir(workflows_dir):
            if fname.endswith((".yml", ".yaml")):
                fpath = os.path.join(workflows_dir, fname)
                try:
                    with open(fpath) as f:
                        content = f.read()
                    
                    name = fname
                    triggers = []
                    jobs = []
                    
                    for line in content.split("\n"):
                        stripped = line.strip()
                        if stripped.startswith("name:"):
                            name = stripped.split(":", 1)[1].strip().strip('"').strip("'")
                        elif stripped.startswith("- ") and "push" in stripped:
                            triggers.append("push")
                        elif stripped.startswith("- ") and "pull_request" in stripped:
                            triggers.append("pull_request")
                        elif stripped and not stripped.startswith("#") and stripped.endswith(":"):
                            indent = len(line) - len(line.lstrip())
                            if indent == 2 and not stripped.startswith(("on:", "jobs:", "steps:", "env:")):
                                job_name = stripped.rstrip(":").strip('"').strip("'")
                                if job_name not in ("name", "runs-on", "steps"):
                                    jobs.append(job_name)
                    
                    workflows.append({
                        "file": fname,
                        "name": name,
                        "triggers": triggers if triggers else ["push"],
                        "jobs": jobs,
                    })
                except Exception:
                    workflows.append({"file": fname, "name": fname, "error": "Failed to parse"})
        
        return {
            "ci_type": "github_actions",
            "workflows": workflows,
            "total": len(workflows),
        }
    
    async def get_gitlab_pipeline_config(self, repo_path: str) -> Dict[str, Any]:
        """Get GitLab CI pipeline configuration."""
        ci_file = os.path.join(repo_path, ".gitlab-ci.yml")
        if not os.path.exists(ci_file):
            return {"error": "No .gitlab-ci.yml found"}
        
        try:
            with open(ci_file) as f:
                content = f.read()
            
            stages = []
            jobs = []
            
            for line in content.split("\n"):
                stripped = line.strip()
                if stripped.startswith("- ") and not stripped.startswith("- " + '"'):
                    stage_name = stripped[2:].strip().strip('"').strip("'")
                    skip_words = {"before_script", "after_script", "script", "only", "when", "tags", "cache", "services", "variables"}
                    if stage_name and stage_name not in skip_words:
                        stages.append(stage_name)
                elif stripped and ":" in stripped and not stripped.startswith(("#", "-", "  ")):
                    job_name = stripped.split(":")[0].strip()
                    skip_keys = {"image", "services", "variables", "stages", "before_script", "after_script", "script", "only", "when", "tags", "cache"}
                    if job_name and job_name not in skip_keys:
                        jobs.append(job_name)
            
            return {
                "ci_type": "gitlab_ci",
                "stages": stages,
                "jobs": jobs,
                "total_jobs": len(jobs),
            }
        except Exception as e:
            return {"error": str(e)}
    
    async def analyze_ci_health(self, repo_path: str) -> Dict[str, Any]:
        """Analyze CI/CD configuration health."""
        detection = await self.detect_ci(repo_path)
        
        if not detection["detected"]:
            return {
                "status": "no_ci",
                "recommendations": ["Add CI/CD pipeline for automated testing and deployment"],
            }
        
        recommendations = []
        primary = detection["primary"]
        
        if primary["type"] == "github_actions":
            workflows = await self.get_github_actions_workflows(repo_path)
            if "error" not in workflows:
                if workflows["total"] == 0:
                    recommendations.append("No workflow files found in .github/workflows/")
                elif workflows["total"] == 1:
                    recommendations.append("Consider adding separate workflows for lint, test, and deploy")
        
        if not recommendations:
            recommendations.append("CI/CD configuration detected and appears healthy")
        
        return {
            "status": "configured",
            "detected": detection["detected"],
            "recommendations": recommendations,
        }
    
    async def get_recent_runs(self, repo_path: str) -> Dict[str, Any]:
        """Get recent CI/CD run information (GitHub Actions)."""
        try:
            result = subprocess.run(
                ["gh", "run", "list", "--limit", "10", "--json",
                 "status,conclusion,headBranch,createdAt,event,name"],
                cwd=repo_path,
                capture_output=True,
                text=True,
                timeout=10,
            )
            
            if result.returncode != 0:
                return {"error": "Could not fetch runs — is gh CLI available?"}
            
            runs = json.loads(result.stdout)
            
            return {
                "runs": [
                    {
                        "name": run.get("name", "unknown"),
                        "status": run.get("status", "unknown"),
                        "conclusion": run.get("conclusion", "pending"),
                        "branch": run.get("headBranch", "unknown"),
                        "event": run.get("event", "unknown"),
                        "created_at": run.get("createdAt", ""),
                    }
                    for run in runs
                ],
                "total": len(runs),
            }
        except (subprocess.TimeoutExpired, FileNotFoundError, json.JSONDecodeError):
            return {"error": "Could not fetch runs"}
```
