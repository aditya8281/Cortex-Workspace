# P05: Agent-in-the-Loop Coding Protocol — CORTEX

**Document:** Phase 05 of Version 1.12
**Date:** 2026-06-30

---

## Objective

Build the agent-in-the-loop coding protocol: a session manager that orchestrates autonomous write→review→test→rewrite cycles. The agent receives a coding task, proposes changes, runs tests, reviews results, and iterates — with user checkpoints before destructive operations and a hard iteration limit.

---

## Protocol Flow

```
User: "Add input validation to the API endpoint"
    │
    ▼
AgentCodingSession.start()
    │
    ├── [1] UNDERSTAND ── read file, get LSP diagnostics, parse AST
    │       │
    │       ▼
    ├── [2] PLAN ── propose changes with edit_file / rename / extract
    │       │
    │       ▼
    ├── [3] USER_CHECKPOINT ── show diff, ask approval (if destructive)
    │       │
    │       ▼
    ├── [4] EXECUTE ── apply changes via edit_file / write_file
    │       │
    │       ▼
    ├── [5] REVIEW ── run lint, get diagnostics, code review
    │       │
    │       ▼
    ├── [6] TEST ── run relevant tests, parse output
    │       │
    │       ▼
    ├── [7] EVALUATE ── are diagnostics clean? tests passing?
    │       ├── YES → complete, return diff
    │       └── NO  → increment iteration, if < max → goto [2]
    │                   else → return partial with failure details
    │
    ▼
Session ends. User sees diff + test results + review.
```

## Implementation Tasks

### Task 1: Coding Session State Machine

```python
"""Agent-in-the-loop coding session manager."""
from __future__ import annotations

import enum
import logging
from datetime import datetime
from typing import Any, Callable
from pathlib import Path

from backend.app.models.developer.agent_coding_session import AgentCodingSession
from backend.app.models.developer.editor_action_log import EditorActionLog

logger = logging.getLogger(__name__)

MAX_ITERATIONS = 5
MAX_ITERATIONS_BEFORE_USER_ESCALATION = 3  # Ask after 3 failed attempts


class SessionState(enum.Enum):
    INITIALIZED = "initialized"
    UNDERSTANDING = "understanding"
    PLANNING = "planning"
    AWAITING_USER = "awaiting_user"
    EXECUTING = "executing"
    REVIEWING = "reviewing"
    TESTING = "testing"
    EVALUATING = "evaluating"
    COMPLETED = "completed"
    FAILED = "failed"


class CodingSessionManager:
    """Manages the state machine for agent-in-the-loop coding sessions."""

    def __init__(self, db_session=None):
        self._db = db_session
        self._state: SessionState = SessionState.INITIALIZED
        self._session: AgentCodingSession | None = None
        self._context: dict[str, Any] = {}
        self._user_approved: bool = False

    async def start(self, user_id: int, task: str, file_path: str | None = None,
                    conversation_id: int | None = None) -> dict[str, Any]:
        """Start a new coding session."""
        self._session = AgentCodingSession(
            owner_id=user_id,
            task_description=task,
            target_file=file_path,
            conversation_id=conversation_id,
            max_iterations=MAX_ITERATIONS,
            status="active",
        )
        self._state = SessionState.UNDERSTANDING
        self._context = {
            "task": task,
            "file_path": file_path,
            "iteration": 0,
            "changes": [],
            "test_results": [],
            "review_findings": [],
        }

        if self._db:
            self._db.add(self._session)
            self._db.commit()
            self._db.refresh(self._session)

        return {
            "session_id": self._session.id,
            "state": self._state.value,
            "task": task,
        }

    async def step(self, action_result: dict[str, Any] | None = None) -> dict[str, Any]:
        """Advance the state machine one step."""
        if self._state == SessionState.UNDERSTANDING:
            return await self._step_understand()
        elif self._state == SessionState.PLANNING:
            return await self._step_plan(action_result)
        elif self._state == SessionState.AWAITING_USER:
            return {"action": "awaiting_user", "requires_approval": True}
        elif self._state == SessionState.EXECUTING:
            return await self._step_execute(action_result)
        elif self._state == SessionState.REVIEWING:
            return await self._step_review()
        elif self._state == SessionState.TESTING:
            return await self._step_test()
        elif self._state == SessionState.EVALUATING:
            return await self._step_evaluate()
        else:
            return {"state": self._state.value, "completed": True}

    async def _step_understand(self) -> dict[str, Any]:
        """Read target file, get diagnostics, parse AST."""
        file_path = self._context.get("file_path")
        context = {"file_content": "", "ast": None, "diagnostics": []}

        if file_path and Path(file_path).exists():
            with open(file_path) as f:
                context["file_content"] = f.read()

            # Get diagnostics
            try:
                from backend.app.services.developer.lsp_service import LspService
                from backend.app.services.developer.lsp_manager import LspSessionManager
                lsp = LspService(LspSessionManager())
                context["diagnostics"] = await lsp.get_diagnostics(file_path)
            except Exception:
                context["diagnostics"] = []

            # Parse AST
            try:
                from backend.app.services.developer.ast_service import AstService
                ast = AstService.parse(file_path, context["file_content"])
                context["ast"] = ast
            except Exception:
                context["ast"] = None

        self._context["understanding"] = context
        self._state = SessionState.PLANNING
        return {
            "state": "understood",
            "file": file_path,
            "diagnostics_count": len(context["diagnostics"]),
            "file_size": len(context["file_content"]),
        }

    async def _step_plan(self, understanding: dict[str, Any] | None = None) -> dict[str, Any]:
        """Generate plan for changes."""
        self._context["iteration"] += 1
        task = self._context["task"]
        file_path = self._context.get("file_path")
        diagnostics = self._context.get("understanding", {}).get("diagnostics", [])

        # Build the plan from task context + diagnostics
        plan = {
            "task": task,
            "file_path": file_path,
            "diagnostics_before": diagnostics,
            "proposed_actions": self._generate_proposed_actions(task, diagnostics),
        }

        self._context["plan"] = plan
        self._state = SessionState.AWAITING_USER
        return plan

    def _generate_proposed_actions(self, task: str, diagnostics: list) -> list[dict]:
        """Generate proposed edit actions based on task and current diagnostics.
        
        In production, this calls the LLM with the task + file content + diagnostics.
        The LLM returns structured edit actions (edit_file, write_file calls).
        """
        return [
            {
                "tool": "edit_file",
                "description": f"Implement: {task}",
                "estimated_impact": "moderate",
            }
        ]

    async def approve_plan(self, approved: bool = True) -> dict[str, Any]:
        """User approves or rejects the proposed plan."""
        self._user_approved = approved
        if approved:
            self._state = SessionState.EXECUTING
        else:
            self._state = SessionState.FAILED
            if self._session:
                self._session.status = "failed"
        return {"approved": approved, "state": self._state.value}

    async def _step_execute(self, actions: dict[str, Any] | None = None) -> dict[str, Any]:
        """Execute the planned edits."""
        file_path = self._context.get("file_path")
        if not file_path:
            self._state = SessionState.FAILED
            return {"error": "No file target"}

        # In production: call the LLM to generate edit_file calls
        # For the plan, the agent tools handle the actual edits
        executed = []
        for action in self._context.get("plan", {}).get("proposed_actions", []):
            executed.append({
                "action": action["tool"],
                "file": file_path,
                "status": "pending_agent_tool_call",
            })

        self._context["executed"] = executed
        self._state = SessionState.REVIEWING
        return {"executed": executed}

    async def _step_review(self) -> dict[str, Any]:
        """Run lint + diagnostics after edit."""
        file_path = self._context.get("file_path")
        findings = []

        if file_path and Path(file_path).exists():
            # Get post-edit diagnostics
            try:
                from backend.app.services.developer.lsp_service import LspService
                from backend.app.services.developer.lsp_manager import LspSessionManager
                lsp = LspService(LspSessionManager())
                diags = await lsp.get_diagnostics(file_path)
                findings = [
                    {"type": "diagnostic", "severity": d.get("severity", "unknown"),
                     "message": d.get("message", ""), "line": d.get("range", {}).get("start", {}).get("line")}
                    for d in diags
                ]
            except Exception:
                findings = []

            # Run lint
            try:
                import subprocess
                ext = Path(file_path).suffix.lower()
                if ext == ".py":
                    lint_result = subprocess.run(
                        ["ruff", "check", file_path],
                        capture_output=True, text=True, timeout=15,
                    )
                    if lint_result.stdout:
                        for line in lint_result.stdout.split("\n")[:10]:
                            findings.append({"type": "lint", "message": line[:200]})
            except Exception:
                pass

        self._context["review_findings"] = findings
        self._state = SessionState.TESTING
        return {"findings": findings, "count": len(findings)}

    async def _step_test(self) -> dict[str, Any]:
        """Run relevant tests."""
        file_path = self._context.get("file_path")
        test_result = {"pass": 0, "fail": 0, "output": ""}

        if file_path and Path(file_path).exists():
            try:
                import subprocess
                import re

                # Find relevant test file
                base_name = Path(file_path).stem
                test_paths = [
                    f"tests/test_{base_name}.py",
                    f"tests/test_{base_name}_test.py",
                    "tests/",
                ]

                for test_path in test_paths:
                    if Path(test_path).exists():
                        result = subprocess.run(
                            ["python", "-m", "pytest", test_path, "-x", "--tb=short", "-q"],
                            capture_output=True, text=True, timeout=60,
                        )
                        test_result["output"] = result.stdout + result.stderr
                        # Parse results
                        match = re.search(r"(\d+) passed", result.stdout)
                        if match:
                            test_result["pass"] = int(match.group(1))
                        match = re.search(r"(\d+) failed", result.stdout)
                        if match:
                            test_result["fail"] = int(match.group(1))
                        break
            except Exception as e:
                test_result["output"] = str(e)

        self._context["test_results"] = test_result
        self._state = SessionState.EVALUATING
        return test_result

    async def _step_evaluate(self) -> dict[str, Any]:
        """Evaluate if we're done or need another iteration."""
        test_results = self._context["test_results"]
        findings = self._context["review_findings"]
        iteration = self._context["iteration"]

        # Success: clean diagnostics + tests passing or no failures
        diagnostics_issues = [f for f in findings if f["type"] == "diagnostic"
                            and f.get("severity") in ("error", "warning")]
        test_failures = test_results.get("fail", 0)
        tests_run = test_results.get("pass", 0) + test_failures

        all_clean = len(diagnostics_issues) == 0 and (tests_run == 0 or test_failures == 0)

        if all_clean:
            self._state = SessionState.COMPLETED
            if self._session:
                self._session.status = "completed"
                self._session.current_iteration = iteration
            return {
                "state": "completed",
                "iterations": iteration,
                "tests_passed": test_results.get("pass", 0),
                "issues_remaining": len(diagnostics_issues),
            }

        if iteration >= MAX_ITERATIONS:
            self._state = SessionState.FAILED
            if self._session:
                self._session.status = "failed"
                self._session.current_iteration = iteration
            return {
                "state": "failed",
                "reason": "Max iterations reached",
                "iterations": iteration,
                "remaining_issues": len(diagnostics_issues),
                "test_failures": test_failures,
            }

        # Iterate: go back to planning
        self._state = SessionState.PLANNING
        return {
            "state": "iterating",
            "iteration": iteration + 1,
            "remaining_issues": len(diagnostics_issues),
            "test_failures": test_failures,
            "next_step": "planning",
        }

    def get_state(self) -> dict[str, Any]:
        """Get current session state."""
        return {
            "state": self._state.value,
            "iteration": self._context.get("iteration", 0),
            "changes": self._context.get("changes", []),
            "test_results": self._context.get("test_results", {}),
            "review_findings": self._context.get("review_findings", []),
        }

    def log_action(self, action_type: str, file_path: str,
                   before: str | None, after: str | None, approved: bool = True) -> None:
        """Log an editor action for audit trail."""
        if not self._db or not self._session:
            return
        log = EditorActionLog(
            owner_id=self._session.owner_id,
            session_id=self._session.id,
            action_type=action_type,
            file_path=file_path,
            before_content={"text": before} if before else None,
            after_content={"text": after} if after else None,
            approved=approved,
        )
        self._db.add(log)
        self._db.commit()
```

"""User checkpoint handling for coding sessions."""
from backend.app.agents.tool_defs import tool, ask_user


@tool(
    description="Send a checkpoint to the user for approval",
    category="coding",
    params={
        "message": "What to ask the user",
        "diff": "The proposed diff to show",
        "session_id": "Session ID",
    }
)
async def coding_checkpoint(message: str, diff: str, session_id: int) -> dict:
    """Pause and ask user for approval before destructive edits."""
    response = await ask_user(
        f"{message}\n\nProposed changes:\n```diff\n{diff}\n```\n\nApprove? (yes/no)"
    )
    approved = response.strip().lower() in ("yes", "y", "approve", "true")

    from backend.app.services.developer.coding_session import CodingSessionManager
    manager = CodingSessionManager()
    result = await manager.approve_plan(approved)
    return {
        "approved": approved,
        "user_response": response,
    }

    """State machine follows UNDERSTAND→PLAN→AWAIT→EXECUTE→REVIEW→TEST→EVAL."""

    assert manager._state == SessionState.UNDERSTANDING

    result = await manager.step()
    assert manager._state == SessionState.PLANNING

    result = await manager.step()
    assert manager._state == SessionState.AWAITING_USER

    """Session fails after max iterations."""
