# services/developer/dependency_analysis.py

import os
import json
from typing import Dict, List, Any


class DependencyAnalysisService:
    """Analyzes project dependencies."""
    
    async def analyze_requirements(self, repo_path: str) -> Dict[str, Any]:
        """Analyze requirements.txt dependencies."""
        req_file = os.path.join(repo_path, "requirements.txt")
        if not os.path.exists(req_file):
            return {"error": "No requirements.txt found"}
        
        with open(req_file) as f:
            lines = f.readlines()
        
        deps = []
        for line in lines:
            line = line.strip()
            if line and not line.startswith("#") and not line.startswith("-"):
                parts = line.split("==")
                version = parts[1].strip() if len(parts) > 1 else "unpinned"
                deps.append({
                    "name": parts[0].strip(),
                    "version": version,
                    "pinned": "==" in line,
                })
        
        unpinned = [d for d in deps if not d["pinned"]]
        
        return {
            "file": "requirements.txt",
            "dependencies": deps,
            "total": len(deps),
            "pinned_count": len(deps) - len(unpinned),
            "unpinned_count": len(unpinned),
            "unpinned": [d["name"] for d in unpinned],
        }
    
    async def analyze_package_json(self, repo_path: str) -> Dict[str, Any]:
        """Analyze package.json dependencies."""
        pkg_file = os.path.join(repo_path, "package.json")
        if not os.path.exists(pkg_file):
            return {"error": "No package.json found"}
        
        with open(pkg_file) as f:
            pkg = json.load(f)
        
        deps = pkg.get("dependencies", {})
        dev_deps = pkg.get("devDependencies", {})
        
        return {
            "file": "package.json",
            "dependencies": [{"name": k, "version": v} for k, v in deps.items()],
            "devDependencies": [{"name": k, "version": v} for k, v in dev_deps.items()],
            "total": len(deps) + len(dev_deps),
            "production_count": len(deps),
            "dev_count": len(dev_deps),
        }
    
    async def analyze_pyproject(self, repo_path: str) -> Dict[str, Any]:
        """Analyze pyproject.toml dependencies."""
        pyproject_file = os.path.join(repo_path, "pyproject.toml")
        if not os.path.exists(pyproject_file):
            return {"error": "No pyproject.toml found"}
        
        try:
            content = {}
            with open(pyproject_file) as f:
                current_section = None
                for line in f:
                    line = line.strip()
                    if line.startswith("[") and line.endswith("]"):
                        current_section = line[1:-1]
                        content[current_section] = []
                    elif current_section and "=" in line:
                        content[current_section].append(line)
            
            deps = []
            for section, items in content.items():
                if "dependencies" in section.lower():
                    for item in items:
                        if "=" in item:
                            name = item.split("=")[0].strip()
                            version = item.split("=", 1)[1].strip().strip('"').strip("'")
                            deps.append({"name": name, "version": version, "section": section})
            
            return {
                "file": "pyproject.toml",
                "dependencies": deps,
                "total": len(deps),
            }
        except Exception as e:
            return {"error": str(e)}
    
    async def detect_dependency_files(self, repo_path: str) -> List[Dict[str, str]]:
        """Detect which dependency files exist."""
        known_files = [
            ("requirements.txt", "pip"),
            ("requirements-dev.txt", "pip-dev"),
            ("Pipfile", "pipenv"),
            ("pyproject.toml", "poetry/pip"),
            ("setup.py", "setuptools"),
            ("package.json", "npm"),
            ("yarn.lock", "yarn"),
            ("Cargo.toml", "cargo"),
            ("go.mod", "go"),
            ("Gemfile", "bundler"),
        ]
        
        found = []
        for filename, ecosystem in known_files:
            fpath = os.path.join(repo_path, filename)
            if os.path.exists(fpath):
                found.append({"file": filename, "ecosystem": ecosystem})
        
        return found

# services/developer/security_analysis.py

import re
import os
from typing import Dict, List, Any


class SecurityAnalysisService:
    """Analyzes code for security vulnerabilities."""
    
    SECRET_PATTERNS = [
        (r'(?i)password\s*=\s*["\'][^"\']{4,}["\']', "Hardcoded password"),
        (r'(?i)api[_-]?key\s*=\s*["\'][^"\']{4,}["\']', "Hardcoded API key"),
        (r'(?i)secret\s*=\s*["\'][^"\']{4,}["\']', "Hardcoded secret"),
        (r'(?i)token\s*=\s*["\'][A-Za-z0-9_\-\.]{20,}["\']', "Hardcoded token"),
        (r'(?i)private[_-]?key\s*=\s*["\'][^"\']{4,}["\']', "Hardcoded private key"),
        (r'(?i)aws[_-]?(?:access|secret)[_-]?(?:key|id)\s*=\s*["\'][^"\']{4,}["\']', "AWS credentials"),
    ]
    
    INJECTION_PATTERNS = [
        (r'\.execute\s*\(\s*f["\']', "Potential SQL injection via f-string"),
        (r'\.execute\s*\(\s*["\'].*%\s', "Potential SQL injection via string formatting"),
        (r'\.execute\s*\(\s*["\'].*\+', "Potential SQL injection via concatenation"),
        (r'\.raw\s*\(\s*f["\']', "Raw query with f-string"),
    ]
    
    INSECURE_PATTERNS = [
        (r'verify\s*=\s*False', "SSL verification disabled"),
        (r'(?i)debug\s*=\s*True', "Debug mode enabled"),
        (r'(?i)allow_all_origins|origins\s*=\s*\["\*"\]', "Overly permissive CORS"),
        (r'(?i)pickle\.loads?\s*\(', "Unsafe deserialization with pickle"),
        (r'(?i)eval\s*\(', "Use of eval() — potential code injection"),
        (r'(?i)exec\s*\(', "Use of exec() — potential code injection"),
        (r'(?i)subprocess\.call\s*\(\s*["\']', "Shell command via subprocess"),
        (r'(?i)os\.system\s*\(', "Shell command via os.system"),
    ]
    
    async def scan_file(self, file_path: str) -> Dict[str, Any]:
        """Scan file for security issues."""
        try:
            with open(file_path, "r") as f:
                content = f.read()
            
            issues: List[Dict[str, Any]] = []
            lines = content.split("\n")
            
            for i, line in enumerate(lines, 1):
                for pattern, message in self.SECRET_PATTERNS:
                    if re.search(pattern, line):
                        issues.append({
                            "line": i,
                            "severity": "critical",
                            "category": "hardcoded_secret",
                            "message": message,
                            "code": line.strip()[:80],
                        })
                
                for pattern, message in self.INJECTION_PATTERNS:
                    if re.search(pattern, line):
                        issues.append({
                            "line": i,
                            "severity": "high",
                            "category": "injection",
                            "message": message,
                            "code": line.strip()[:80],
                        })
                
                for pattern, message in self.INSECURE_PATTERNS:
                    if re.search(pattern, line):
                        issues.append({
                            "line": i,
                            "severity": "medium",
                            "category": "insecure_practice",
                            "message": message,
                            "code": line.strip()[:80],
                        })
            
            critical = sum(1 for i in issues if i["severity"] == "critical")
            high = sum(1 for i in issues if i["severity"] == "high")
            medium = sum(1 for i in issues if i["severity"] == "medium")
            
            risk_level = "low"
            if critical > 0:
                risk_level = "critical"
            elif high > 0:
                risk_level = "high"
            elif medium > 0:
                risk_level = "medium"
            
            return {
                "file": file_path,
                "issues": issues,
                "total_issues": len(issues),
                "risk_level": risk_level,
                "summary": {
                    "critical": critical,
                    "high": high,
                    "medium": medium,
                },
            }
        except Exception as e:
            return {"error": str(e)}
    
    async def scan_directory(self, dir_path: str) -> Dict[str, Any]:
        """Scan all source files in a directory."""
        results = []
        
        for root, _dirs, files in os.walk(dir_path):
            if any(part.startswith(".") for part in root.split(os.sep)):
                continue
            
            for fname in files:
                ext = os.path.splitext(fname)[1]
                if ext in (".py", ".js", ".ts", ".jsx", ".tsx", ".java", ".go", ".rs"):
                    fpath = os.path.join(root, fname)
                    result = await self.scan_file(fpath)
                    if "error" not in result and result["total_issues"] > 0:
                        results.append(result)
        
        all_issues = []
        for r in results:
            all_issues.extend(r["issues"])
        
        return {
            "directory": dir_path,
            "files_scanned": len(results),
            "files_with_issues": len(results),
            "total_issues": len(all_issues),
            "by_severity": {
                "critical": sum(1 for i in all_issues if i["severity"] == "critical"),
                "high": sum(1 for i in all_issues if i["severity"] == "high"),
                "medium": sum(1 for i in all_issues if i["severity"] == "medium"),
            },
            "results": results,
        }

# services/developer/performance_analysis.py

import ast
import re
from typing import Dict, List, Any


class PerformanceAnalysisService:
    """Analyzes code for performance issues."""
    
    async def analyze_file(self, file_path: str) -> Dict[str, Any]:
        """Analyze file for performance issues."""
        try:
            with open(file_path, "r") as f:
                content = f.read()
            
            issues: List[Dict[str, Any]] = []
            lines = content.split("\n")
            
            # N+1 query patterns
            if ".all()" in content and "for " in content:
                for i, line in enumerate(lines, 1):
                    if ".all()" in line:
                        issues.append({
                            "type": "n_plus_one",
                            "severity": "medium",
                            "line": i,
                            "message": "Potential N+1 query — consider eager loading or join",
                        })
            
            # Large range loops
            for i, line in enumerate(lines, 1):
                range_match = re.search(r"range\((\d+)", line)
                if range_match and int(range_match.group(1)) > 1000:
                    issues.append({
                        "type": "large_range",
                        "severity": "low",
                        "line": i,
                        "message": f"Large range({range_match.group(1)}) — consider batching or streaming",
                    })
            
            # String concatenation in loops
            in_loop = False
            for i, line in enumerate(lines, 1):
                stripped = line.strip()
                if stripped.startswith("for ") or stripped.startswith("while "):
                    in_loop = True
                elif in_loop and "+=" in line and ("str" in line or '"' in line or "'" in line):
                    issues.append({
                        "type": "string_concat_loop",
                        "severity": "medium",
                        "line": i,
                        "message": "String concatenation in loop — use list append + join",
                    })
                    in_loop = False
            
            # Heavy imports at module level
            heavy_modules = {"pandas", "numpy", "tensorflow", "torch", "scipy", "cv2"}
            for i, line in enumerate(lines, 1):
                for module in heavy_modules:
                    if re.search(rf"import\s+{module}\b|from\s+{module}\b", line):
                        issues.append({
                            "type": "heavy_import",
                            "severity": "low",
                            "line": i,
                            "message": f"Module-level import of '{module}' — consider lazy loading",
                        })
            
            # Sync I/O in async functions
            try:
                tree = ast.parse(content)
                for node in ast.walk(tree):
                    if isinstance(node, ast.AsyncFunctionDef):
                        for child in ast.walk(node):
                            if isinstance(child, ast.Call):
                                func_name = ""
                                if isinstance(child.func, ast.Name):
                                    func_name = child.func.id
                                elif isinstance(child.func, ast.Attribute):
                                    func_name = child.func.attr
                                if func_name in ("open", "time.sleep", "requests.get", "requests.post"):
                                    issues.append({
                                        "type": "sync_io_in_async",
                                        "severity": "medium",
                                        "line": child.lineno,
                                        "message": f"Sync call '{func_name}' in async function — use async equivalent",
                                    })
            except SyntaxError:
                pass
            
            return {
                "file": file_path,
                "issues": issues,
                "total_issues": len(issues),
                "by_severity": {
                    "medium": sum(1 for i in issues if i["severity"] == "medium"),
                    "low": sum(1 for i in issues if i["severity"] == "low"),
                },
            }
        except Exception as e:
            return {"error": str(e)}
    
    async def analyze_file_complexity(self, file_path: str) -> Dict[str, Any]:
        """Analyze cyclomatic complexity of a Python file."""
        try:
            with open(file_path, "r") as f:
                content = f.read()
            
            tree = ast.parse(content)
            complexities = []
            
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    complexity = 1
                    
                    for child in ast.walk(node):
                        if isinstance(child, (ast.If, ast.While, ast.For, ast.ExceptHandler)):
                            complexity += 1
                        elif isinstance(child, ast.BoolOp):
                            complexity += len(child.values) - 1
                        elif isinstance(child, ast.comprehension):
                            complexity += 1
                    
                    rating = "low"
                    if complexity > 10:
                        rating = "high"
                    elif complexity > 5:
                        rating = "medium"
                    
                    complexities.append({
                        "name": node.name,
                        "line": node.lineno,
                        "complexity": complexity,
                        "rating": rating,
                    })
            
            avg = (
                sum(c["complexity"] for c in complexities) / max(1, len(complexities))
            )
            
            return {
                "file": file_path,
                "functions": complexities,
                "average_complexity": round(avg, 2),
                "highest_complexity": max(c["complexity"] for c in complexities) if complexities else 0,
                "total_functions": len(complexities),
            }
        except Exception as e:
            return {"error": str(e)}
