```python
import re

@tool(
    description="Search codebase for symbols, patterns, or text",
    category="code",
    params={
        "query": "Search query (symbol name, pattern, or natural language)",
        "file_pattern": "Optional glob pattern to filter files (e.g. '**/*.py')",
        "max_results": "Maximum results (default 20)",
    }
)
async def search_code(query: str, file_pattern: Optional[str] = None,
                      max_results: int = 20) -> list[dict]:
    """Search code using AST symbol index + fallback grep."""
    results = []

    # 1. Try symbol index first
    from backend.app.services.developer.symbol_index_service import SymbolIndexService
    svc = SymbolIndexService()
    symbols = await svc.search_symbols(query, file_pattern=file_pattern, limit=max_results)
    for s in symbols:
        results.append({
            "type": "symbol",
            "name": s.symbol_name,
            "kind": s.symbol_kind,
            "file": s.file_path,
            "line": s.line_start,
        })

    # 2. If too few results, fallback to grep
    if len(results) < max_results:
        import subprocess
        remaining = max_results - len(results)
        seen_files = {r["file"] for r in results}

        root = _find_project_root(".")
        if root:
            grep_result = subprocess.run(
                ["grep", "-rn", "--include", file_pattern or "*", query, "."],
                cwd=root, capture_output=True, text=True, timeout=10,
            )
            for line in grep_result.stdout.split("\n")[:remaining]:
                if ":" in line:
                    parts = line.split(":", 2)
                    file_path = parts[0].strip()
                    if file_path not in seen_files:
                        results.append({
                            "type": "match",
                            "file": file_path,
                            "line": int(parts[1]) if parts[1].isdigit() else 0,
                            "snippet": parts[2][:200] if len(parts) > 2 else "",
                        })
                        seen_files.add(file_path)

    return results[:max_results]
```

```python
@tool(
    description="Surgical edit: replace lines in a file",
    category="code",
    params={
        "file_path": "Path to the file",
        "start_line": "Start line (1-indexed, inclusive)",
        "end_line": "End line (1-indexed, inclusive)",
        "new_text": "Replacement text",
    },
    requires_approval=True,
)
async def edit_file(file_path: str, start_line: int, end_line: int,
                    new_text: str) -> dict:
    """Replace a range of lines with new text. Validates file exists."""
    path = Path(file_path)
    if not path.exists():
        return {"error": f"File not found: {file_path}"}

    with open(path) as f:
        lines = f.readlines()

    if start_line < 1 or end_line > len(lines):
        return {"error": f"Line range {start_line}-{end_line} out of bounds (1-{len(lines)})"}

    before = "".join(lines[start_line - 1:end_line])
    new_lines = lines[:start_line - 1] + [new_text + "\n"] + lines[end_line:]
    after = "".join(new_lines)

    # Write
    with open(path, "w") as f:
        f.writelines(new_lines)

    return {
        "status": "ok",
        "file": file_path,
        "start_line": start_line,
        "end_line": end_line,
        "preview": new_text[:500],
        "diff": _generate_diff(before, after),
    }


@tool(
    description="Rename a symbol across all files",
    category="code",
    params={
        "old_name": "Current symbol name",
        "new_name": "New symbol name",
        "file_path": "Optional: only rename in this file (dry-run shows all)",
    },
    requires_approval=True,  # Multi-file rename always requires approval
)
async def rename_symbol(old_name: str, new_name: str,
                        file_path: Optional[str] = None) -> dict:
    """Rename a symbol across files with dry-run preview."""
    from backend.app.services.developer.symbol_index_service import SymbolIndexService
    svc = SymbolIndexService()
    references = await svc.find_references_by_name(old_name)

    if file_path:
        references = [r for r in references if r.file_path == file_path]

    if not references:
        return {"error": f"No references found for '{old_name}'"}

    affected_files = set(r.file_path for r in references)
    changes = []

    for ref_file in affected_files:
        path = Path(ref_file)
        with open(path) as f:
            content = f.read()

        count = content.count(old_name)
        new_content = content.replace(old_name, new_name)

        changes.append({
            "file": ref_file,
            "occurrences": count,
            "preview": _generate_diff(content, new_content)[:500],
        })

    return {
        "status": "preview",
        "old_name": old_name,
        "new_name": new_name,
        "affected_files": len(affected_files),
        "total_occurrences": len(references),
        "changes": changes,
        "instruction": "To apply, call rename_symbol with apply=true",
    }


@tool(
    description="Extract selected lines into a new function/method",
    category="code",
    params={
        "file_path": "Path to the file",
        "start_line": "First line to extract",
        "end_line": "Last line to extract",
        "function_name": "Name for the new function",
        "insert_after_line": "Line number to insert the new function after",
    },
    requires_approval=True,
)
async def extract_function(file_path: str, start_line: int, end_line: int,
                           function_name: str, insert_after_line: int) -> dict:
    """Extract lines into a new function and replace with a call."""
    path = Path(file_path)
    if not path.exists():
        return {"error": f"File not found: {file_path}"}

    with open(path) as f:
        lines = f.readlines()

    # Get extracted lines
    extracted = "".join(lines[start_line - 1:end_line])
    indent = _detect_indent(lines[start_line - 1]) if lines else "    "

    # Build new function
    if file_path.endswith(".py"):
        new_function = f"\ndef {function_name}():\n{indent}{extracted.strip().replace(chr(10), chr(10) + indent)}\n"
        call_line = f"{function_name}()\n"
    else:
        new_function = f"\nfunction {function_name}() {{\n{indent}{extracted.strip()}\n}}\n"
        call_line = f"{function_name}();\n"

    # Insert function after insert_after_line
    new_lines = lines[:insert_after_line] + [new_function] + lines[insert_after_line:]

    # Replace extracted lines with call
    # (indices shift due to insertion)
    actual_start = insert_after_line if start_line > insert_after_line else start_line
    adj_end = actual_start + (end_line - start_line)
    new_lines = new_lines[:actual_start - 1] + [call_line] + new_lines[adj_end:]

    with open(path, "w") as f:
        f.writelines(new_lines)

    return {
        "status": "ok",
        "file": file_path,
        "function_name": function_name,
        "extracted_lines": end_line - start_line + 1,
    }


@tool(
    description="Format a file using its language formatter",
    category="code",
    params={"file_path": "Path to the file to format"},
)
async def format_file(file_path: str) -> dict:
    """Format a file using LSP or native formatter."""
    ext = Path(file_path).suffix.lower()
    formatters = {
        ".py": ["ruff", "format", file_path],
        ".js": ["npx", "prettier", "--write", file_path],
        ".ts": ["npx", "prettier", "--write", file_path],
        ".tsx": ["npx", "prettier", "--write", file_path],
        ".jsx": ["npx", "prettier", "--write", file_path],
        ".rs": ["rustfmt", file_path],
        ".go": ["gofmt", "-w", file_path],
        ".java": ["java", "-jar", "google-java-format.jar", "--replace", file_path],
    }

    cmd = formatters.get(ext)
    if not cmd:
        return {"error": f"No formatter configured for {ext}"}

    import subprocess
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
    return {
        "status": "ok" if result.returncode == 0 else "error",
        "file": file_path,
        "output": result.stdout or result.stderr,
    }
```

```python
@tool(
    description="Lint a file and return diagnostics",
    category="code",
    params={"file_path": "Path to the file"},
)
async def lint_file(file_path: str) -> dict:
    """Run linter on a file and return results."""
    ext = Path(file_path).suffix.lower()
    linters = {
        ".py": ["ruff", "check", file_path],
        ".js": ["npx", "eslint", file_path],
        ".ts": ["npx", "eslint", file_path],
        ".tsx": ["npx", "eslint", file_path],
    }

    cmd = linters.get(ext)
    if not cmd:
        return {"error": f"No linter configured for {ext}"}

    import subprocess
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
    issues = []
    for line in result.stdout.split("\n"):
        if ":" in line and any(s in line.lower() for s in ["error", "warning"]):
            issues.append(line)

    return {
        "file": file_path,
        "issues": issues[:50],
        "issue_count": len(issues),
        "exit_code": result.returncode,
    }


@tool(
    description="Auto-fix a specific diagnostic in a file",
    category="code",
    params={
        "file_path": "Path to the file",
        "diagnostic_line": "Line number of the diagnostic",
        "rule": "The diagnostic rule/code to fix (e.g. 'F401')",
    }
)
async def fix_diagnostic(file_path: str, diagnostic_line: int, rule: str) -> dict:
    """Apply auto-fix for a specific diagnostic."""
    import subprocess
    ext = Path(file_path).suffix.lower()

    if ext == ".py":
        result = subprocess.run(
            ["ruff", "check", "--fix", "--select", rule, file_path],
            capture_output=True, text=True, timeout=30,
        )
        return {
            "status": "ok" if result.returncode == 0 else "error",
            "stdout": result.stdout[:500],
            "stderr": result.stderr[:500],
        }

    return {"error": f"No auto-fix for {ext}"}
```

### Task 5: Helper Functions

```python
def _find_project_root(start_path: str) -> str | None:
    """Walk up to find project root."""
    from pathlib import Path
    path = Path(start_path).absolute()
    for parent in [path] + list(path.parents):
        markers = [".git", "pyproject.toml", "package.json", "Cargo.toml", "go.mod"]
        if any((parent / m).exists() for m in markers):
            return str(parent)
    return None


def _generate_diff(before: str, after: str) -> str:
    """Generate a simple unified diff."""
    import difflib
    return "".join(difflib.unified_diff(
        before.splitlines(keepends=True),
        after.splitlines(keepends=True),
        n=3,
    ))


def _detect_indent(line: str) -> str:
    """Detect indentation from a line."""
    stripped = line.lstrip()
    if stripped and line != stripped:
        return line[:len(line) - len(stripped)]
    return "    "
```
