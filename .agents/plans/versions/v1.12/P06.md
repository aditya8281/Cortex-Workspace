```python
    """Cross-file symbol resolution backed by AST-extracted symbols."""

    def __init__(self, db=None):
        self._db = db

    async def build_index(self, project_path: str) -> dict:
        """Scan all source files and build symbol index."""
        from pathlib import Path
        from backend.app.services.developer.ast_service import AstService

        source_exts = {".py", ".js", ".ts", ".tsx", ".jsx", ".rs", ".go",
                       ".java", ".cpp", ".cc", ".hpp", ".kt", ".swift",
                       ".rb", ".php", ".cs", ".scala", ".ex"}
        indexed = 0
        errors = 0

        for ext in source_exts:
            for path in Path(project_path).rglob(f"*{ext}"):
                if any(p.startswith(".") for p in path.parts):
                    continue
                if "node_modules" in path.parts or ".venv" in path.parts:
                    continue
                try:
                    source = path.read_text(encoding="utf-8", errors="replace")
                    symbols = AstService.extract_symbols(source, str(path))
                    await self._store_symbols(symbols)
                    indexed += 1
                except Exception:
                    errors += 1

        return {"indexed": indexed, "errors": errors, "total": indexed + errors}
```

### Task 3: CodeReviewService — Multi-Language Expansion

Enhance original CodeReviewService to cover all supported languages. The existing service handles Python; expand with language-agnostic checks:

**Universal checks (all languages):**
- Function/method too long (>50 lines)
- Too many parameters (>5)
- Deep nesting (>4 levels)
- Duplicate code blocks
- Unused variables/imports
- Missing docstrings/comments
- Magic numbers

**Language-specific checks (from AST queries):**
- Python: no `except:`, mutable defaults, bare `assert`
- JS/TS: `==` vs `===`, `var` usage, implicit any
- Rust: unwrap() in non-test code, missing error handling
- Go: shadowed variables, missing error checks

**File:** `backend/app/services/developer/code_review.py`

```python
class CodeReviewService:
    def __init__(self):
        self._rules = self._load_rules()

    async def review_file(self, file_path: str) -> dict:
        """Run all applicable review rules."""
        ext = Path(file_path).suffix.lower()
        rules = self._rules.get(ext, []) + self._rules.get("universal", [])
        findings = []
        for rule in rules:
            finding = await rule.check(file_path)
            if finding:
                findings.append(finding)
        return {
            "file": file_path,
            "findings": findings,
            "score": self._compute_score(findings),
        }

    async def review_diff(self, diff_text: str) -> list[dict]:
        """Review only the changed lines in a diff."""

    def _load_rules(self) -> dict:
        """Load rules from rules/ directory, one file per language."""
```
