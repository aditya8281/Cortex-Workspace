```python
class SecurityAnalysisService:
    """Security vulnerability scanning for multiple languages."""

    PATTERNS = {
        "python": [
            (r"eval\s*\(", "dangerous", "Use of eval() is dangerous"),
            (r"exec\s*\(", "dangerous", "Use of exec() is dangerous"),
            (r"pickle\.loads", "dangerous", "Unsafe deserialization via pickle"),
            (r"subprocess\.call.*shell=True", "dangerous", "Shell injection risk"),
            (r"os\.system\s*\(", "dangerous", "OS command injection risk"),
            (r"sqlite3\.execute.*f['\"]", "sql_injection", "Potential SQL injection (f-string)"),
            (r"request\.get\(", "ssrf", "SSRF risk — validate URLs"),
            (r"SECRET_KEY\s*=\s*['\"][^'\"]{10,}", "hardcoded_secret", "Hardcoded secret detected"),
        ],
        "javascript": [
            (r"eval\s*\(", "dangerous", "Use of eval() is dangerous"),
            (r"innerHTML\s*=.*user", "xss", "Potential XSS via innerHTML"),
            (r"child_process\.exec\(", "dangerous", "Shell execution risk"),
            (r"process\.env\.\w+", "env_leak", "Environment variable exposure"),
        ],
        "rust": [
            (r"\.unwrap\(\)", "safety", "unwrap() may panic in production"),
            (r"unsafe\s*\{", "unsafe", "Unsafe block — verify invariants"),
            (r"std::process::Command", "security", "External command execution"),
        ],
        "go": [
            (r"sql\.Query\(`", "sql_injection", "Raw SQL query injection risk"),
            (r"exec\.Command\(", "security", "Shell execution risk"),
        ],
    }

    async def scan_file(self, file_path: str) -> list[dict]:
        """Scan a file for security vulnerabilities."""
        ext = Path(file_path).suffix.lower()
        lang_map = {
            ".py": "python", ".js": "javascript", ".ts": "javascript",
            ".tsx": "javascript", ".jsx": "javascript",
            ".rs": "rust", ".go": "go",
        }
        lang = lang_map.get(ext)
        if not lang:
            return []

        patterns = self.PATTERNS.get(lang, [])
        content = Path(file_path).read_text(encoding="utf-8", errors="replace")
        findings = []

        for pattern, severity, message in patterns:
            for m in re.finditer(pattern, content, re.MULTILINE):
                line_num = content[:m.start()].count("\n") + 1
                findings.append({
                    "file": file_path,
                    "line": line_num,
                    "severity": severity,
                    "message": message,
                    "match": m.group(0)[:80],
                })

        return findings
```

### Task 3: PerformanceAnalysisService (Enhanced)

```python
class PerformanceAnalysisService:
    """Performance analysis for multi-language codebases."""

    async def analyze_file(self, file_path: str) -> dict:
        """Analyze a file for performance issues."""
        ext = Path(file_path).suffix.lower()
        content = Path(file_path).read_text()

        analysis = {
            "file": file_path,
            "language": ext,
            "metrics": {},
            "issues": [],
        }

        # Cyclomatic complexity
        analysis["metrics"]["cyclomatic_complexity"] = self._calc_complexity(content, ext)

        # N+1 query detection
        analysis["issues"].extend(self._detect_nplus1(content, ext))

        # Large functions
        analysis["issues"].extend(self._detect_large_functions(content, ext))

        # Memory-intensive patterns
        analysis["issues"].extend(self._detect_memory_issues(content, ext))

        return analysis
```
