### Task 1: Enhanced GitIntelligenceService

```python
class GitIntelligenceService:
    """Full git intelligence with branch, blame, diff, and commit analysis."""

    async def get_log(self, repo_path: str, limit: int = 20) -> list[dict]:
        """Get formatted git log."""

    async def get_blame(self, repo_path: str, file_path: str) -> list[dict]:
        """Get blame annotations per line."""
        result = subprocess.run(
            ["git", "blame", "--line-porcelain", file_path],
            cwd=repo_path, capture_output=True, text=True, timeout=30,
        )
        # Parse porcelain format → per-line dict

    async def get_diff(self, repo_path: str, ref: str = "HEAD",
                       file_path: str = None) -> str:
        """Get diff for a ref or unstaged changes."""

    async def list_branches(self, repo_path: str) -> list[dict]:
        """List branches with last commit info."""

    async def create_branch(self, repo_path: str, branch_name: str,
                            base: str = "main") -> dict:
        """Create and switch to a new branch."""

    async def commit(self, repo_path: str, message: str,
                     files: list[str] = None) -> dict:
        """Stage and commit files."""

    async def get_status(self, repo_path: str) -> dict:
        """Get working tree status (staged, unstaged, untracked)."""

    async def analyze_commit(self, repo_path: str, commit_hash: str) -> dict:
        """Analyze a commit: diff, impact, risk assessment."""
```

### Task 2: CI/CD Integration (Enhanced)

```python
class CICDIntegrationService:
    """Detect and analyze CI/CD configuration files."""

    async def detect_platform(self, repo_path: str) -> str | None:
        """Detect CI/CD platform from config files."""
        if os.path.exists(os.path.join(repo_path, ".github/workflows")):
            return "github_actions"
        if os.path.exists(os.path.join(repo_path, ".gitlab-ci.yml")):
            return "gitlab_ci"
        if os.path.exists(os.path.join(repo_path, "Jenkinsfile")):
            return "jenkins"
        # CircleCI, Travis, Drone, Buildkite, etc.

    async def analyze_pipeline(self, repo_path: str) -> dict:
        """Parse CI/CD pipeline configuration for jobs, steps, deps."""
```

### Task 3: Agent Tools for Git

```python
@tool(description="Stage and commit files", category="git",
      params={"message": "Commit message", "files": "Optional list of files to stage"})
async def git_commit(message: str, files: list[str] = None) -> dict:
    """Commit changes with message."""
    from backend.app.services.developer.git_intelligence import GitIntelligenceService
    svc = GitIntelligenceService()
    return await svc.commit(".", message, files)


@tool(description="Create and switch to a new branch", category="git",
```
