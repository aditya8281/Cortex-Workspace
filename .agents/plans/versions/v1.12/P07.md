### Task 2: Enhanced TestGenerationService

Expand to generate tests for any language, using AST to extract function signatures:

class TestGenerationService:
    """Multi-language test scaffold generation."""

    TEST_FRAMEWORKS = {
        ".py": "pytest",
        ".js": "jest",
        ".ts": "jest",
        ".rs": "cargo test",
        ".go": "go test",
        ".java": "junit",
    }

    async def generate_scaffold(self, file_path: str, function_name: str = None) -> dict:
        """Generate test scaffold for a function or file."""
        ext = Path(file_path).suffix.lower()
        framework = self.TEST_FRAMEWORKS.get(ext)
        if not framework:
            return {"error": f"No test framework for {ext}"}

        # Parse AST to get function signatures
        from backend.app.services.developer.ast_service import AstService
        symbols = AstService.extract_symbols(Path(file_path).read_text(), file_path)

        tests = []
        for sym in symbols:
            if sym["kind"] in ("function", "method", "async_function"):
                test_code = self._generate_test_case(sym["name"], ext, framework)
                tests.append({"function": sym["name"], "test": test_code})

        return {
            "framework": framework,
            "file": file_path,
            "tests": tests,
            "test_file": self._test_file_path(file_path),
        }

    def _generate_test_case(self, name: str, ext: str, framework: str) -> str:
        """Generate a test case for a function."""
        if ext == ".py":
            return (
                f"def test_{name}():\n"
                f'    """Test {name}."""\n'
                f"    # Arrange\n"
                f"    # Act\n"
                f"    result = {name}()\n"
                f"    # Assert\n"
                f"    assert result is not None\n"
            )
        elif ext in (".js", ".ts"):
            return (
                f"describe('{name}', () => {{\n"
                f"    it('should work', () => {{\n"
                f"        // Arrange\n"
                f"        // Act\n"
                f"        const result = {name}();\n"
                f"        // Assert\n"
                f"        expect(result).toBeDefined();\n"
                f"    }});\n"
                f"}});\n"
            )

    async def analyze_stack_trace(self, traceback: str) -> list[dict]:
        """Parse stack trace into structured frames."""
        frames = []
        for line in traceback.split("\n"):
            # Parse Python traceback lines
            import re
            m = re.match(r'\s*File "([^"]+)", line (\d+), in (\S+)', line)
            if m:
                frames.append({
                    "file": m.group(1),
                    "line": int(m.group(2)),
                    "function": m.group(3),
                })
        return frames
