# P02: Multi-Language AST Engine — CORTEX

**Document:** Phase 02 of Version 1.12
**Date:** 2026-06-30

---

## Objective

Expand the existing Python-only tree-sitter crate to support 13+ languages. Build a unified AST schema, cross-file symbol index, tree-sitter query patterns for symbol extraction, and the Python FFI bridge. All parsing is local via tree-sitter grammars — no cloud calls.

---

## Architecture

```
Python (agent tools / services)
    │
    ▼
PyO3 FFI ──> rust/crates/code-intel/
    │
    ├── ast_engine.rs         # Language-aware dispatcher
    ├── parser/               # Per-language tree-sitter wrappers
    │   ├── python.rs, javascript.rs, typescript.rs
    │   ├── rust.rs, go.rs, java.rs, cpp.rs
    │   ├── kotlin.rs, swift.rs, ruby.rs, php.rs
    │   ├── csharp.rs, scala.rs
    │   └── config.rs          # Grammar registration
    ├── symbol_index.rs        # Extract definitions + references
    ├── query.rs               # Pre-built tree-sitter queries
    └── ffi.rs                 # PyO3 exports
```

```rust
pub struct LanguageConfig {

    pub name: &'static str,
    pub extensions: &'static [&'static str],
    pub grammar: fn() -> Language,
    pub symbol_query: &'static str,

impl SourceLanguage {
    pub fn from_extension(ext: &str) -> Option<SourceLanguage> {
        match ext.to_lowercase().as_str() {
            "py" => Some(SourceLanguage::Python),
            "js" => Some(SourceLanguage::JavaScript),
            "mjs" => Some(SourceLanguage::JavaScript),
            "ts" => Some(SourceLanguage::TypeScript),
            "tsx" => Some(SourceLanguage::Tsx),
            "jsx" => Some(SourceLanguage::Tsx),
            "rs" => Some(SourceLanguage::Rust),
            "go" => Some(SourceLanguage::Go),
            "java" => Some(SourceLanguage::Java),
            "cpp" | "cc" | "cxx" | "hpp" => Some(SourceLanguage::Cpp),
            "kt" | "kts" => Some(SourceLanguage::Kotlin),
            "swift" => Some(SourceLanguage::Swift),
            "rb" => Some(SourceLanguage::Ruby),

            "cs" => Some(SourceLanguage::CSharp),

            "ex" | "exs" => Some(SourceLanguage::Elixir),

    }
}
```

### Task 3: Implement AST Engine

```rust
// src/ast_engine.rs — Multi-language AST parser

use pyo3::prelude::*;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;

use crate::config::{self, SourceLanguage};

#[derive(Debug, Serialize, Deserialize)]
pub struct AstNode {
    pub id: u32,
    pub kind: String,
    pub text: String,
    pub start_line: u32,
    pub end_line: u32,
    pub start_column: u32,
    pub end_column: u32,
    pub children: Vec<AstNode>,
    pub is_named: bool,
}

#[derive(Debug, Serialize, Deserialize)]
pub struct AstResult {
    pub language: String,
    pub root: AstNode,
    pub node_count: u32,
    pub error: Option<String>,
}

#[derive(Debug, Serialize, Deserialize)]
pub struct SymbolDef {
    pub name: String,
    pub kind: String,
    pub file_path: String,
    pub line_start: u32,
    pub line_end: u32,
    pub column_start: u32,
    pub column_end: u32,
    pub parent_name: Option<String>,
    pub doc_text: Option<String>,
    pub is_public: bool,
}

#[pymethods]
impl AstEngine {
    #[staticmethod]
    pub fn parse_file(file_path: &str, source: &str) -> PyResult<AstResult> {
        let ext = file_path.rsplit('.').next().unwrap_or("");
        let lang = match config::SourceLanguage::from_extension(ext) {
            Some(l) => l,
            None => return Ok(AstResult {
                language: ext.to_string(),
                root: AstNode { id: 0, kind: "unsupported".into(), text: "".into(),
                    start_line: 0, end_line: 0, start_column: 0, end_column: 0,
                    children: vec![], is_named: false },
                node_count: 0,
                error: Some(format!("Unsupported language: {}", ext)),
            }),
        };

        let grammar = config::get_language(lang);
        let mut parser = tree_sitter::Parser::new();
        parser.set_language(&grammar)
            .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(
                format!("Failed to set language {:?}: {}", lang, e)))?;

        let tree = parser.parse(source, None)
            .ok_or_else(|| pyo3::exceptions::PyRuntimeError::new_err("Parse returned None"))?;

        let root = tree.root_node();
        let mut counter = NodeCounter(0);
        let ast_root = walk_node(root, source, &mut counter);

        Ok(AstResult {
            language: format!("{:?}", lang).to_lowercase(),
            root: ast_root,
            node_count: counter.0,
            error: None,
        })
    }
}

fn walk_node(node: tree_sitter::Node, source: &str, counter: &mut NodeCounter) -> AstNode {
    counter.0 += 1;
    let mut children = Vec::new();
    let mut cursor = node.walk();
    for child in node.named_children(&mut cursor) {
        children.push(walk_node(child, source, counter));
    }
    AstNode {
        id: node.id(),
        kind: node.kind().to_string(),
        text: node.utf8_text(source.as_bytes()).unwrap_or("").to_string(),
        start_line: node.start_position().row as u32,
        end_line: node.end_position().row as u32,
        start_column: node.start_position().column as u32,
        end_column: node.end_position().column as u32,
        children,
        is_named: node.is_named(),
    }
}

struct NodeCounter(u32);
```

### Task 4: Symbol Index Extraction

```rust
// src/symbol_index.rs — Extract definitions and references via queries

use tree_sitter::{Query, QueryCursor};
use crate::config::{self, SourceLanguage};

pub fn extract_symbols(source: &str, lang: SourceLanguage, file_path: &str)
    -> Vec<SymbolDef>
{
    let grammar = config::get_language(lang);
    let mut parser = tree_sitter::Parser::new();
    if parser.set_language(&grammar).is_err() {
        return vec![];
    }
    let tree = match parser.parse(source, None) {
        Some(t) => t,
        None => return vec![],
    };

    // Language-specific query patterns for functions, classes, methods, interfaces
    let query_patterns = match lang {
        SourceLanguage::Python => vec![
            ("function_definition", "(function_definition name: (identifier) @name) @def"),
            ("class_definition", "(class_definition name: (identifier) @name) @def"),
            ("decorated_definition", "(decorated_definition definition: (function_definition name: (identifier) @name)) @def"),
        ],
        SourceLanguage::JavaScript | SourceLanguage::TypeScript | SourceLanguage::Tsx => vec![
            ("function_declaration", "(function_declaration name: (identifier) @name) @def"),
            ("class_declaration", "(class_declaration name: (identifier) @name) @def"),
            ("method_definition", "(method_definition name: (property_identifier) @name) @def"),
            ("arrow_function", "(variable_declarator name: (identifier) @name value: (arrow_function)) @def"),
        ],
        SourceLanguage::Rust => vec![
            ("function_item", "(function_item name: (identifier) @name) @def"),
            ("struct_item", "(struct_item name: (type_identifier) @name) @def"),
            ("impl_item", "(impl_item trait: (type_identifier) @name) @def"),
            ("trait_item", "(trait_item name: (type_identifier) @name) @def"),
        ],
        SourceLanguage::Go => vec![
            ("function_declaration", "(function_declaration name: (identifier) @name) @def"),
            ("method_declaration", "(method_declaration name: (field_identifier) @name) @def"),
            ("type_declaration", "(type_declaration (type_spec name: (type_identifier) @name)) @def"),
        ],
        SourceLanguage::Java => vec![
            ("method_declaration", "(method_declaration name: (identifier) @name) @def"),
            ("class_declaration", "(class_declaration name: (identifier) @name) @def"),
            ("interface_declaration", "(interface_declaration name: (identifier) @name) @def"),
        ],
        // ... similar patterns for C++, Kotlin, Swift, Ruby, PHP, C#, Scala
        _ => vec![],
    };

    let mut symbols = Vec::new();
    let root = tree.root_node();

    for (kind, pattern) in &query_patterns {
        if let Ok(query) = Query::new(&grammar, pattern) {
            let mut cursor = QueryCursor::new();
            let matches = cursor.matches(&query, root, source.as_bytes());
            for m in matches {
                for capture in m.captures {
                    let node = capture.node;
                    let name = node.utf8_text(source.as_bytes()).unwrap_or("");
                    if !name.is_empty() {
                        symbols.push(SymbolDef {
                            name: name.to_string(),
                            kind: kind.to_string(),
                            file_path: file_path.to_string(),
                            line_start: node.start_position().row as u32,
                            line_end: node.end_position().row as u32,
                            column_start: node.start_position().column as u32,
                            column_end: node.end_position().column as u32,
                            parent_name: None,
                            doc_text: None,
                            is_public: true,
                        });
                    }
                }
            }
        }
    }

    symbols
}
```

### Task 5: FFI Exports

```rust
// src/ffi.rs — PyO3 module exports

use pyo3::prelude::*;
use crate::ast_engine::AstEngine;
use crate::symbol_index;

#[pymodule]
fn cortex_code_intel(m: &Bound<'_, PyModule>) -> PyResult<()> {
    m.add_class::<AstEngine>()?;
    m.add_function(wrap_pyfunction!(parse_file_ffi, m)?)?;
    m.add_function(wrap_pyfunction!(parse_batch_ffi, m)?)?;
    m.add_function(wrap_pyfunction!(extract_symbols_ffi, m)?)?;
    m.add_function(wrap_pyfunction!(detect_language_ffi, m)?)?;
    Ok(())
}

#[pyfunction]
fn parse_file_ffi(file_path: &str, source: &str) -> PyResult<String> {
    let result = AstEngine::parse_file(file_path, source)?;
    serde_json::to_string(&result)
        .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e.to_string()))
}

#[pyfunction]
fn parse_batch_ffi(files: Vec<(String, String)>) -> PyResult<String> {
    let mut results = Vec::new();
    for (path, source) in files {
        let result = AstEngine::parse_file(&path, &source)
            .unwrap_or_else(|_| AstResult {
                language: "unknown".into(),
                root: AstNode { id: 0, kind: "error".into(), text: "".into(),
                    start_line: 0, end_line: 0, start_column: 0, end_column: 0,
                    children: vec![], is_named: false },
                node_count: 0,
                error: Some("Parse failed".into()),
            });
        results.push(result);
    }
    serde_json::to_string(&results)
        .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e.to_string()))
}

#[pyfunction]
fn extract_symbols_ffi(source: &str, language: &str, file_path: &str) -> PyResult<String> {
    let lang = config::SourceLanguage::from_extension(
        file_path.rsplit('.').next().unwrap_or("")
    ).unwrap_or(config::SourceLanguage::Python);

    let symbols = symbol_index::extract_symbols(source, lang, file_path);
    serde_json::to_string(&symbols)
        .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e.to_string()))
}

#[pyfunction]
fn detect_language_ffi(file_path: &str) -> Option<String> {
    let ext = file_path.rsplit('.').next()?;
    config::SourceLanguage::from_extension(ext)
        .map(|l| format!("{:?}", l).to_lowercase())
}
```

### Task 6: Python Service Layer

```python
"""Python bridge to the Rust code-intel crate."""
import json
import hashlib
from pathlib import Path
from typing import Optional

try:
    from cortex_code_intel import (
        parse_file_ffi, parse_batch_ffi, extract_symbols_ffi, detect_language_ffi
    )
    RUST_EXTENSION_AVAILABLE = True
except ImportError:
    RUST_EXTENSION_AVAILABLE = False


class AstService:
    """High-level AST service agnostic to Rust extension availability."""

    SUPPORTED_EXTENSIONS = {
        ".py", ".js", ".jsx", ".ts", ".tsx", ".rs", ".go", ".java",
        ".cpp", ".cc", ".cxx", ".hpp", ".kt", ".kts", ".swift",
        ".rb", ".php", ".cs", ".scala", ".sc", ".ex", ".exs",
        ".lua", ".sql",
    }

    @classmethod
    def is_supported(cls, file_path: str) -> bool:
        ext = Path(file_path).suffix.lower()
        return ext in cls.SUPPORTED_EXTENSIONS

    @classmethod
    def parse(cls, file_path: str, source: str) -> dict:
        """Parse a source file into AST JSON."""
        if not RUST_EXTENSION_AVAILABLE:
            return {"error": "Rust extension not available", "file_path": file_path}
        result = parse_file_ffi(file_path, source)
        return json.loads(result)

    @classmethod
    def parse_batch(cls, files: list[tuple[str, str]]) -> list[dict]:
        """Parse multiple files in one FFI call."""
        if not RUST_EXTENSION_AVAILABLE:
            return [{"error": "Rust extension not available"} for _ in files]
        result = parse_batch_ffi(files)
        return json.loads(result)

    @classmethod
    def extract_symbols(cls, source: str, file_path: str) -> list[dict]:
        """Extract symbol definitions from source."""
        if not RUST_EXTENSION_AVAILABLE:
            return cls._extract_symbols_fallback(source, file_path)
        result = extract_symbols_ffi(source, "", file_path)
        return json.loads(result)

    @classmethod
    def detect_language(cls, file_path: str) -> Optional[str]:
        if not RUST_EXTENSION_AVAILABLE:
            ext = Path(file_path).suffix.lower()
            return {"py": "python", "js": "javascript", "ts": "typescript"}.get(ext)
        return detect_language_ffi(file_path)

    @classmethod
    def _extract_symbols_fallback(cls, source: str, file_path: str) -> list[dict]:
        """Regex-based fallback when Rust extension unavailable."""
        import re
        symbols = []
        ext = Path(file_path).suffix.lower()

        if ext == ".py":
            patterns = [
                ("function", r"^def (\w+)\s*\("),
                ("class", r"^class (\w+)"),
                ("async_function", r"^async def (\w+)\s*\("),
            ]
        elif ext in (".js", ".ts", ".jsx", ".tsx"):
            patterns = [
                ("function", r"(?:function|const)\s+(\w+)\s*(?:=|\(|:)"),
                ("class", r"^class (\w+)"),
                ("arrow_function", r"const (\w+)\s*=\s*(?:async\s*)?\(.*\)\s*=>"),
            ]
        else:
            patterns = [
                ("function", r"(?:pub\s+)?(?:fn|func|def|function)\s+(\w+)"),
                ("class", r"(?:pub\s+)?(?:class|struct|trait)\s+(\w+)"),
            ]

        for line_num, line in enumerate(source.split("\n"), 1):
            for kind, pattern in patterns:
                m = re.match(pattern, line.strip())
                if m:
                    symbols.append({
                        "name": m.group(1),
                        "kind": kind,
                        "file_path": file_path,
                        "line_start": line_num,
                        "line_end": line_num,
                        "is_public": not line.strip().startswith("_"),
                    })

        return symbols
```

### Task 8: Integration test

```python
"""Test multi-language AST parsing."""
import pytest
from backend.app.services.developer.ast_service import AstService


@pytest.mark.skipif(not AstService.RUST_EXTENSION_AVAILABLE,
                    reason="Rust extension not built")
class TestMultiLangAst:
    def test_python_parse(self):
        result = AstService.parse("test.py", "def hello(x):\n    return x + 1\n")
        assert result["language"] == "python"
        assert result["node_count"] > 0

    def test_javascript_parse(self):
        result = AstService.parse("test.js", "function hello(x) { return x + 1; }")
        assert result["language"] == "javascript"

    def test_rust_parse(self):
        result = AstService.parse("test.rs", "fn hello(x: i32) -> i32 { x + 1 }")
        assert result["language"] == "rust"

    def test_parse_batch(self):
        files = [
            ("a.py", "def foo(): pass"),
            ("b.js", "function bar() {}"),
        ]
        results = AstService.parse_batch(files)
        assert len(results) == 2

    def test_symbol_extraction(self):
        symbols = AstService.extract_symbols(
            "def hello(): pass\nclass World: pass\n",
            "test.py"
        )
        names = [s["name"] for s in symbols]
        assert "hello" in names
        assert "World" in names

    def test_unsupported_language(self):
        result = AstService.parse("file.xyz", "some content")
        assert "error" in result

    def test_detect_language(self):
        assert AstService.detect_language("main.py") == "python"
        assert AstService.detect_language("app.ts") == "typescript"
        assert AstService.detect_language("unknown.xyz") is None
```
