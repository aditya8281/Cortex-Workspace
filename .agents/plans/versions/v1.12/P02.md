### Task 1: Implement CodeUnderstandingService

**Architecture:** The code understanding service parses source code using AST (Python) and regex-based analysis (JS/TS/Rust). It extracts functions, classes, imports, exports, decorators, and type annotations. Symbol resolution tracks where functions are defined and called. The service supports multiple languages via pluggable parsers.

**File:** `backend/app/services/developer/code_understanding.py`
```python
import ast
import os
import re
import hashlib
import logging
from typing import Dict, List, Any, Optional

logger = logging.getLogger(__name__)


class CodeUnderstandingService:
    """AST-based code analysis with symbol resolution and multi-language support."""

    EXT_MAP = {
        ".py": "python", ".js": "javascript", ".ts": "typescript",
        ".jsx": "javascript", ".tsx": "typescript",
        ".rs": "rust", ".go": "go", ".java": "java",
    }

    async def analyze_file(self, file_path: str) -> Dict[str, Any]:
        """Analyze a code file for structure, symbols, and metrics."""
        try:
            with open(file_path, "r") as f:
                content = f.read()
            file_hash = hashlib.sha256(content.encode()).hexdigest()
            language = self._detect_language(file_path)

            if language == "python":
                result = await self._analyze_python(content, file_path)
            elif language in ("javascript", "typescript"):
                result = await self._analyze_javascript(content, file_path)
            else:
                result = {"file": file_path, "language": language, "status": "analysis_pending"}

            result["file_hash"] = file_hash
            result["language"] = language
            return result
        except Exception as e:
            return {"error": str(e), "file": file_path}

    def _detect_language(self, file_path: str) -> str:
        ext = os.path.splitext(file_path)[1]
        return self.EXT_MAP.get(ext, "unknown")

    async def _analyze_python(self, content: str, file_path: str) -> Dict[str, Any]:
        """Deep Python AST analysis with symbol resolution."""
        try:
            tree = ast.parse(content)

            functions = []
            classes = []
            imports = []
            decorators = []

            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    decorators.extend([self._get_decorator_name(d) for d in node.decorator_list])
                    functions.append({
                        "name": node.name,
                        "line": node.lineno,
                        "end_line": getattr(node, "end_lineno", node.lineno),
                        "args": [arg.arg for arg in node.args.args],
                        "has_docstring": ast.get_docstring(node) is not None,
                        "is_async": isinstance(node, ast.AsyncFunctionDef),
                        "is_method": False,  # Set during class analysis
                        "decorators": [self._get_decorator_name(d) for d in node.decorator_list],
                        "complexity": self._calc_complexity(node),
                    })
                elif isinstance(node, ast.ClassDef):
                    methods = []
                    for child in ast.iter_child_nodes(node):
                        if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                            methods.append(child.name)
                            # Mark as method in functions list
                            for f in functions:
                                if f["name"] == child.name and f["line"] == child.lineno:
                                    f["is_method"] = True
                    classes.append({
                        "name": node.name,
                        "line": node.lineno,
                        "methods": methods,
                        "method_count": len(methods),
                        "has_docstring": ast.get_docstring(node) is not None,
                        "bases": [self._get_name(b) for b in node.bases],
                        "decorators": [self._get_decorator_name(d) for d in node.decorator_list],
                    })
                elif isinstance(node, ast.Import):
                    for alias in node.names:
                        imports.append({"module": alias.name, "alias": alias.asname})
                elif isinstance(node, ast.ImportFrom):
                    module = node.module or ""
                    for alias in node.names:
                        imports.append({
                            "module": f"{module}.{alias.name}",
                            "alias": alias.asname,
                            "level": node.level,
                        })

            lines = content.split("\n")
            return {
                "file": file_path,
                "language": "python",
                "functions": functions,
                "classes": classes,
                "imports": imports,
                "decorators": list(set(decorators)),
                "lines": len(lines),
                "blank_lines": sum(1 for l in lines if l.strip() == ""),
                "function_count": len(functions),
                "class_count": len(classes),
                "import_count": len(imports),
                "async_count": sum(1 for f in functions if f.get("is_async")),
                "avg_complexity": (
                    sum(f.get("complexity", 1) for f in functions) / max(len(functions), 1)
                ),
            }
        except SyntaxError as e:
            return {"file": file_path, "language": "python", "error": f"Syntax error: {e}"}

    def _calc_complexity(self, node: ast.AST) -> int:
        """Calculate cyclomatic complexity for a function AST node."""
        complexity = 1
        for child in ast.walk(node):
            if isinstance(child, (ast.If, ast.While, ast.For, ast.ExceptHandler)):
                complexity += 1
            elif isinstance(child, ast.BoolOp):
                complexity += len(child.values) - 1
            elif isinstance(child, ast.comprehension):
                complexity += 1
        return complexity

    def _get_decorator_name(self, node: ast.expr) -> str:
        if isinstance(node, ast.Name):
            return node.id
        elif isinstance(node, ast.Attribute):
            return f"{self._get_value(node.value)}.{node.attr}"
        return str(node)

    def _get_name(self, node: ast.expr) -> str:
        if isinstance(node, ast.Name):
            return node.id
        elif isinstance(node, ast.Attribute):
            return f"{self._get_name(node.value)}.{node.attr}"
        return "unknown"

    def _get_value(self, node: ast.expr) -> str:
        if isinstance(node, ast.Name):
            return node.id
        elif isinstance(node, ast.Attribute):
            return f"{self._get_value(node.value)}.{node.attr}"
        return "?"

    async def _analyze_javascript(self, content: str, file_path: str) -> Dict[str, Any]:
        """Regex-based JavaScript/TypeScript analysis."""
        functions = re.findall(
            r"(?:function\s+(\w+)|(?:const|let|var)\s+(\w+)\s*=\s*(?:async\s+)?\()",
            content,
        )
        classes = re.findall(r"class\s+(\w+)", content)
        imports = re.findall(r'import\s+.*?from\s+["\'](.+?)["\']', content)
        exports = re.findall(
            r"export\s+(?:default\s+)?(?:function|class|const|let|var)\s+(\w+)", content
        )
        flat_functions = [f[0] or f[1] for f in functions if any(f)]

        return {
            "file": file_path,
            "functions": flat_functions,
            "classes": classes,
            "imports": imports,
            "exports": exports,
            "lines": len(content.split("\n")),
            "function_count": len(flat_functions),
            "class_count": len(classes),
        }

    async def analyze_directory(self, dir_path: str, max_files: int = 200) -> Dict[str, Any]:
        """Analyze all code files in a directory with depth limit."""
        results = []
        file_count = 0

        for root, _dirs, files in os.walk(dir_path):
            # Skip hidden dirs and common non-source dirs
            _dirs[:] = [d for d in _dirs if not d.startswith(".") and d not in ("node_modules", "__pycache__", "venv", ".git")]
            for fname in files:
                if file_count >= max_files:
                    break
                ext = os.path.splitext(fname)[1]
                if ext in self.EXT_MAP:
                    fpath = os.path.join(root, fname)
                    result = await self.analyze_file(fpath)
                    results.append(result)
                    file_count += 1

        total_functions = sum(r.get("function_count", 0) for r in results)
        total_classes = sum(r.get("class_count", 0) for r in results)
        total_lines = sum(r.get("lines", 0) for r in results)

        return {
            "directory": dir_path,
            "files_analyzed": len(results),
            "total_functions": total_functions,
            "total_classes": total_classes,
            "total_lines": total_lines,
            "languages": list(set(r.get("language", "unknown") for r in results)),
            "file_details": results,
        }

    async def resolve_symbols(self, content: str, language: str = "python") -> Dict[str, List[str]]:
        """Resolve symbol definitions and references."""
        if language == "python":
            return await self._resolve_python_symbols(content)
        return {"defined": [], "referenced": []}

    async def _resolve_python_symbols(self, content: str) -> Dict[str, List[str]]:
        """Resolve Python symbol definitions and references via AST."""
        try:
            tree = ast.parse(content)
            defined = []
            referenced = []

            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    defined.append(node.name)
                elif isinstance(node, ast.Name):
                    if isinstance(node.ctx, ast.Load):
                        referenced.append(node.id)

            return {
                "defined": list(set(defined)),
                "referenced": list(set(referenced)),
            }
        except SyntaxError:
            return {"defined": [], "referenced": []}
```

### Task 2: Implement CodeReviewService

**Architecture:** The code review service scans files and diffs for quality issues, security vulnerabilities, and best practice violations. It uses pattern matching for common issues (long lines, TODOs, debug statements, broad exceptions) and regex for security patterns (hardcoded secrets, SQL injection, eval/exec). The review score is computed as `100 - Σ(severity_weight * count)`.

**File:** `backend/app/services/developer/code_review.py`
```python
import re
import logging
from typing import Dict, List, Any

logger = logging.getLogger(__name__)


class CodeReviewService:
    """Reviews code for quality, security, and best practices."""

    PATTERN_RULES = [
        {"pattern": r".{121,}", "severity": "warning", "rule": "line-length", "message": "Line exceeds 120 chars"},
        {"pattern": r"\bTODO\b", "severity": "info", "rule": "todo-comment", "message": "TODO comment — consider addressing"},
        {"pattern": r"\bFIXME\b", "severity": "warning", "rule": "fixme-comment", "message": "FIXME comment — needs attention"},
        {"pattern": r'\bprint\s*\(', "severity": "info", "rule": "debug-statement", "message": "print() — remove before production"},
        {"pattern": r'console\.log\(', "severity": "info", "rule": "debug-statement", "message": "console.log() — remove before production"},
        {"pattern": r"^except\s*:", "severity": "warning", "rule": "bare-except", "message": "Bare except — catch specific exceptions"},
        {"pattern": r"^except Exception\s*:", "severity": "info", "rule": "broad-except", "message": "Broad Exception — catch specific exceptions"},
        {"pattern": r'(?<![a-zA-Z_])(\d{4,})(?![a-zA-Z_])', "severity": "info", "rule": "magic-number", "message": "Magic number — use named constant"},
    ]

    SECURITY_PATTERNS = [
        (r'(?i)password\s*=\s*["\'][^"\']{4,}["\']', "critical", "hardcoded_secret", "Hardcoded password"),
        (r'(?i)api[_-]?key\s*=\s*["\'][^"\']{4,}["\']', "critical", "hardcoded_secret", "Hardcoded API key"),
        (r'(?i)secret\s*=\s*["\'][^"\']{4,}["\']', "critical", "hardcoded_secret", "Hardcoded secret"),
        (r'(?i)token\s*=\s*["\'][A-Za-z0-9_\-\.]{20,}["\']', "critical", "hardcoded_secret", "Hardcoded token"),
        (r'\.execute\s*\(\s*f["\']', "high", "injection", "SQL injection via f-string"),
        (r'\.execute\s*\(\s*["\'].*%\s', "high", "injection", "SQL injection via string formatting"),
        (r'(?i)verify\s*=\s*False', "medium", "insecure_practice", "SSL verification disabled"),
        (r'(?i)debug\s*=\s*True', "medium", "insecure_practice", "Debug mode enabled"),
        (r'(?i)eval\s*\(', "medium", "insecure_practice", "eval() — potential code injection"),
        (r'(?i)exec\s*\(', "medium", "insecure_practice", "exec() — potential code injection"),
        (r'(?i)pickle\.loads?\s*\(', "medium", "insecure_practice", "Unsafe deserialization with pickle"),
    ]

    async def review_file(self, file_path: str) -> Dict[str, Any]:
        """Review a code file for quality and security issues."""
        try:
            with open(file_path, "r") as f:
                content = f.read()

            findings: List[Dict[str, Any]] = []
            suggestions: List[Dict[str, Any]] = []
            lines = content.split("\n")

            # Pattern-based quality rules
            for i, line in enumerate(lines, 1):
                for rule in self.PATTERN_RULES:
                    # Skip magic number for common HTTP codes
                    if rule["rule"] == "magic-number":
                        matches = re.findall(rule["pattern"], line)
                        matches = [m for m in matches if m not in ("200", "404", "500", "1000")]
                        if matches:
                            for match in matches:
                                findings.append({
                                    "line": i, "severity": rule["severity"],
                                    "rule": rule["rule"],
                                    "message": f"{rule['message']} ({match})",
                                })
                    elif re.search(rule["pattern"], line):
                        if rule["rule"] in ("todo-comment", "fixme-comment"):
                            suggestions.append({"line": i, "message": rule["message"], "rule": rule["rule"]})
                        else:
                            findings.append({
                                "line": i, "severity": rule["severity"],
                                "rule": rule["rule"], "message": rule["message"],
                            })

            # Security patterns
            for i, line in enumerate(lines, 1):
                for pattern, severity, category, message in self.SECURITY_PATTERNS:
                    if re.search(pattern, line):
                        findings.append({
                            "line": i, "severity": severity, "rule": category,
                            "message": message, "code": line.strip()[:80],
                        })

            # Calculate score
            severity_weights = {"critical": 15, "warning": 5, "info": 1}
            deduction = sum(severity_weights.get(f["severity"], 1) for f in findings)
            score = max(0, 100 - deduction)

            return {
                "file": file_path,
                "score": score,
                "findings": findings,
                "suggestions": suggestions,
                "lines_reviewed": len(lines),
                "summary": {
                    "critical": sum(1 for f in findings if f["severity"] == "critical"),
                    "warnings": sum(1 for f in findings if f["severity"] == "warning"),
                    "info": sum(1 for f in findings if f["severity"] == "info"),
                    "suggestions_count": len(suggestions),
                },
            }
        except Exception as e:
            return {"error": str(e)}

    async def review_diff(self, diff: str) -> Dict[str, Any]:
        """Review a git diff for security and quality issues."""
        findings: List[Dict[str, str]] = []
        added_lines = []
        removed_lines = []

        for line in diff.split("\n"):
            if line.startswith("+") and not line.startswith("+++"):
                added_content = line[1:]
                added_lines.append(added_content)
                for pattern, severity, category, message in self.SECURITY_PATTERNS:
                    if re.search(pattern, added_content):
                        findings.append({
                            "severity": severity, "category": category,
                            "message": message, "line": line,
                        })
            elif line.startswith("-") and not line.startswith("---"):
                removed_lines.append(line[1:])

        risk_level = "low"
        if any(f["severity"] == "critical" for f in findings):
            risk_level = "critical"
        elif any(f["severity"] == "high" for f in findings):
            risk_level = "high"
        elif findings:
            risk_level = "medium"

        return {
            "findings": findings,
            "lines_added": len(added_lines),
            "lines_removed": len(removed_lines),
            "risk_level": risk_level,
        }
```

### Task 3: Implement DependencyGraphService

**Architecture:** The dependency graph service constructs call graphs and import relationships between files. It tracks which modules import from which, enabling impact analysis when files change. The graph is built by parsing imports (static) and function calls (AST-based). The graph supports forward and reverse lookups.

**File:** `backend/app/services/developer/dependency_graph.py`
```python
import ast
import os
import logging
from typing import Dict, List, Set, Any

logger = logging.getLogger(__name__)


class DependencyGraphService:
    """Constructs and queries module/file dependency graphs."""

    def __init__(self):
        self._graph: Dict[str, Set[str]] = {}  # file -> set of files it depends on
        self._reverse: Dict[str, Set[str]] = {}  # file -> set of files that depend on it

    async def build_graph(self, dir_path: str) -> Dict[str, Any]:
        """Build dependency graph from a directory of Python files."""
        self._graph.clear()
        self._reverse.clear()

        py_files = []
        for root, _dirs, files in os.walk(dir_path):
            _dirs[:] = [d for d in _dirs if not d.startswith(".") and d not in ("__pycache__", "node_modules")]
            for f in files:
                if f.endswith(".py"):
                    py_files.append(os.path.join(root, f))

        # Build module name mapping
        module_map = {}
        for fpath in py_files:
            rel = os.path.relpath(fpath, dir_path)
            module = rel.replace(os.sep, ".").replace(".py", "")
            module_map[module] = fpath

        # Parse imports and build graph
        for fpath in py_files:
            deps = set()
            try:
                with open(fpath) as f:
                    tree = ast.parse(f.read())
                for node in ast.walk(tree):
                    if isinstance(node, ast.ImportFrom) and node.level == 0:
                        module = node.module or ""
                        # Try to find matching file
                        for candidate in module_map:
                            if candidate == module or candidate.startswith(module + "."):
                                deps.add(module_map[candidate])
            except (SyntaxError, FileNotFoundError):
                pass

            self._graph[fpath] = deps
            for dep in deps:
                if dep not in self._reverse:
                    self._reverse[dep] = set()
                self._reverse[dep].add(fpath)

        return {
            "total_files": len(py_files),
            "total_edges": sum(len(deps) for deps in self._graph.values()),
        }

    def get_dependencies(self, file_path: str) -> List[str]:
        """Get files that file_path depends on."""
        return list(self._graph.get(file_path, set()))

    def get_dependents(self, file_path: str) -> List[str]:
        """Get files that depend on file_path (reverse lookup)."""
        return list(self._reverse.get(file_path, set()))

    def get_impact_analysis(self, file_path: str) -> Dict[str, Any]:
        """Analyze the impact of changing a file."""
        direct = self.get_dependents(file_path)
        indirect = set()
        for d in direct:
            indirect.update(self.get_dependents(d))
        indirect.difference_update(direct)
        indirect.discard(file_path)

        return {
            "file": file_path,
            "direct_impact": direct,
            "indirect_impact": list(indirect),
            "total_affected": len(direct) + len(indirect),
            "risk_level": "high" if len(direct) > 5 else "medium" if direct else "low",
        }
```

import pytest
import tempfile
import os
from backend.app.services.developer.code_understanding import CodeUnderstandingService
from backend.app.services.developer.code_review import CodeReviewService

@pytest.mark.asyncio
async def test_analyze_python_file():
    service = CodeUnderstandingService()
    with tempfile.NamedTemporaryFile(mode="w", suffix=".py", delete=False) as f:
