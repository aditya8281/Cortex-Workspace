## Objective

Create database models and Pydantic schemas for all developer intelligence operations — preserving all original models (CodeAnalysis, RepositoryIntelligence, GitAnalysis, RefactoringSuggestion, CodeReviewFinding, CodeGenerationRecord, SkillRegistry, DeveloperAnalytics) and adding new models for LSP sessions, debug sessions, agent coding sessions, symbol index, and AST cache.

## New Models (Added in Expansion)

### Task 1: LspServerConfig Model

```python
"""LSP server configuration — tracks active language servers per project."""
from sqlalchemy import Column, Integer, String, DateTime, JSON, Boolean, ForeignKey
from datetime import datetime
from backend.app.db.base import Base

class LspServerConfig(Base):
    __tablename__ = "lsp_server_configs"

    id = Column(Integer, primary_key=True, autoincrement=True)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    language = Column(String(50), nullable=False)  # python, typescript, rust, etc.
    server_command = Column(String(500), nullable=False)  # pylsp, typescript-language-server, etc.
    server_args = Column(JSON, default=list)
    env_vars = Column(JSON, default=dict)
    auto_start = Column(Boolean, default=True)
    status = Column(String(20), default="stopped")  # stopped, running, crashed
    project_path = Column(String(1000))
    last_started_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)

class LspSession(Base):
    """Active LSP session — one per project+language pair."""
    __tablename__ = "lsp_sessions"

    id = Column(Integer, primary_key=True, autoincrement=True)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    config_id = Column(Integer, ForeignKey("lsp_server_configs.id"))
    language = Column(String(50), nullable=False)
    project_path = Column(String(1000), nullable=False)
    process_id = Column(Integer)
    status = Column(String(20), default="running")
    last_health_check = Column(DateTime)
    started_at = Column(DateTime, default=datetime.utcnow)
    crashed_at = Column(DateTime)
```

```python
class AgentCodingSession(Base):
    """Tracks an agent-in-the-loop coding interaction."""
    __tablename__ = "agent_coding_sessions"

    id = Column(Integer, primary_key=True, autoincrement=True)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)

    target_file = Column(String(1000))
    task_description = Column(String(2000))
    current_iteration = Column(Integer, default=0)
    max_iterations = Column(Integer, default=5)
    status = Column(String(20), default="active")  # active, completed, failed, user_checkpoint
    last_action = Column(String(100))
    last_diff = Column(JSON)
    test_results = Column(JSON)  # {pass_count, fail_count, output}
    review_findings = Column(JSON, default=list)
    user_approval_required = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    completed_at = Column(DateTime)
```

### Task 4: SymbolIndex Model

```python
class SymbolIndex(Base):
    """Cross-file symbol index for code navigation."""
    __tablename__ = "symbol_index"

    id = Column(Integer, primary_key=True, autoincrement=True)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    symbol_name = Column(String(500), nullable=False, index=True)
    symbol_kind = Column(String(50))  # function, class, variable, method, interface, type
    file_path = Column(String(1000), nullable=False)
    line_start = Column(Integer)
    line_end = Column(Integer)
    column_start = Column(Integer)
    column_end = Column(Integer)
    language = Column(String(50))
    parent_symbol = Column(String(500))  # parent class/module
    is_definition = Column(Boolean, default=True)  # false = reference
    references_count = Column(Integer, default=0)
    doc_text = Column(String(2000))
    is_public = Column(Boolean, default=True)
    updated_at = Column(DateTime, default=datetime.utcnow)

    __table_args__ = (
        Index("ix_symbol_name_file", "symbol_name", "file_path"),
        Index("ix_symbol_owner_kind", "owner_id", "symbol_kind"),
    )
```

```python
class AstCache(Base):
    """Cached AST parse results per file+language."""
    __tablename__ = "ast_cache"

    id = Column(Integer, primary_key=True, autoincrement=True)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    file_path = Column(String(1000), nullable=False)
    file_hash = Column(String(64), nullable=False, index=True)

    ast_json = Column(JSON)  # serialized tree-sitter AST
    node_count = Column(Integer)
    symbol_count = Column(Integer)
    parsed_at = Column(DateTime, default=datetime.utcnow)

    __table_args__ = (
        Index("ix_ast_file_hash", "owner_id", "file_hash"),
    )
```

```python
class EditorActionLog(Base):
```
