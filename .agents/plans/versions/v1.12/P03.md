## Objective

Build an LSP client manager that starts, monitors, and communicates with language servers for diagnostics, completion, hover info, go-to-definition, and find-references. Supports 8+ LSP servers: pylsp/ruff, typescript-language-server, rust-analyzer, gopls, eclipse-jdtls, clangd, solargraph, and more.

---

## Architecture

```
Agent Tools
    │
    ▼
LspService (Python)
    │
    ├── SessionManager ───── start/stop/monitor LSP processes
    │       │
    │       ├── pylsp (Python)
    │       ├── typescript-language-server (JS/TS)
    │       ├── rust-analyzer (Rust)
    │       ├── gopls (Go)
    │       ├── eclipse-jdtls (Java)
    │       ├── clangd (C++)
    │       ├── sourcekit-lsp (Swift)
    │       ├── solargraph (Ruby)
    │       └── ... (extensible via config)
    │
    ├── LanguageClient ───── JSON-RPC 2.0 over stdio
    │       ├── initialize / initialized
    │       ├── textDocument/completion
    │       ├── textDocument/hover
    │       ├── textDocument/definition
    │       ├── textDocument/references
    │       ├── textDocument/documentSymbol
    │       ├── textDocument/formatting
    │       └── textDocument/codeAction
    │
    └── DiagnosticCollector ── store & forward diagnostics
            ├── textDocument/publishDiagnostics
            └── on-diagnostic callback → agent tools
```

### Task 1: JSON-RPC 2.0 Client

```python
"""JSON-RPC 2.0 client for LSP communication over stdio."""
from __future__ import annotations

import json
import uuid
import asyncio
import logging
from typing import Any, Callable, Coroutine

logger = logging.getLogger(__name__)


class JsonRpcClient:
    """Low-level JSON-RPC 2.0 client over stdio transport."""

    def __init__(self):
        self._process: asyncio.subprocess.Process | None = None
        self._pending: dict[str, asyncio.Future] = {}
        self._reader_task: asyncio.Task | None = None
        self._on_notification: Callable | None = None
        self._buffer = b""

    async def start(self, command: str, args: list[str] = None,
                    env: dict[str, str] = None,
                    on_notification: Callable | None = None) -> None:
        """Start the LSP server process."""
        self._on_notification = on_notification
        self._process = await asyncio.create_subprocess_exec(
            command, *(args or []),
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
            env=env,
        )
        self._reader_task = asyncio.create_task(self._read_loop())

    async def send_request(self, method: str, params: dict[str, Any] = None) -> dict[str, Any]:
        """Send a JSON-RPC request and wait for response."""
        msg_id = str(uuid.uuid4())[:8]
        future = asyncio.get_event_loop().create_future()
        self._pending[msg_id] = future

        message = {
            "jsonrpc": "2.0",
            "id": msg_id,
            "method": method,
            "params": params or {},
        }
        await self._send(message)

        try:
            return await asyncio.wait_for(future, timeout=30.0)
        except asyncio.TimeoutError:
            self._pending.pop(msg_id, None)
            raise TimeoutError(f"LSP request {method} timed out")

    async def send_notification(self, method: str, params: dict[str, Any] = None) -> None:
        """Send a JSON-RPC notification (no response expected)."""
        message = {
            "jsonrpc": "2.0",
            "method": method,
            "params": params or {},
        }
        await self._send(message)

    async def close(self) -> None:
        """Shutdown the LSP server."""
        try:
            await self.send_request("shutdown")
            await self.send_notification("exit")
        except Exception:
            pass
        if self._reader_task:
            self._reader_task.cancel()
        if self._process:
            self._process.kill()

    async def _send(self, message: dict) -> None:
        if not self._process or not self._process.stdin:
            raise RuntimeError("LSP not connected")
        body = json.dumps(message)
        header = f"Content-Length: {len(body)}\r\n\r\n"
        self._process.stdin.write(header.encode() + body.encode())
        await self._process.stdin.drain()

    async def _read_loop(self) -> None:
        """Read LSP responses from stdout."""
        while self._process and self._process.stdout:
            try:
                line = await self._process.stdout.readline()
                if not line:
                    break
                self._buffer += line

                # Parse Content-Length header
                if b"Content-Length:" in self._buffer:
                    content_length = int(self._buffer.split(b"\r\n")[0].split(b":")[1].strip())
                    # Wait for full body
                    while len(self._buffer) < content_length + 4:  # \r\n\r\n
                        chunk = await self._process.stdout.read(content_length + 4 - len(self._buffer))
                        if not chunk:
                            break
                        self._buffer += chunk
                    # Extract body after headers
                    body_start = self._buffer.find(b"\r\n\r\n") + 4
                    if body_start > 4 and len(self._buffer) >= body_start + content_length:
                        body = self._buffer[body_start:body_start + content_length]
                        self._buffer = self._buffer[body_start + content_length:]
                        await self._handle_message(json.loads(body))
            except Exception:
                logger.exception("LSP read error")
                break

    async def _handle_message(self, msg: dict) -> None:
        if "id" in msg:
            # Response to a request
            future = self._pending.pop(msg["id"], None)
            if future and not future.done():
                if "error" in msg:
                    future.set_exception(RuntimeError(msg["error"]))
                else:
                    future.set_result(msg.get("result", {}))
        else:
            # Notification (e.g., diagnostics)
            if self._on_notification:
                await self._on_notification(msg)
```

```python
class LspSessionManager:
    """Manages LSP server lifecycle and client connections."""

    MAX_ACTIVE_SESSIONS = 5
    IDLE_TIMEOUT_SECONDS = 900  # 15 min

    def __init__(self):
        self._sessions: dict[str, JsonRpcClient] = {}  # key: "{project}:{language}"
        self._session_info: dict[str, dict] = {}
        self._diagnostics: dict[str, list[dict]] = {}  # key: file_path

    async def get_or_start_session(self, project_path: str, language: str) -> JsonRpcClient:
        """Get existing session or start a new LSP server."""
        key = f"{project_path}:{language}"

        if key in self._sessions:
            client = self._sessions[key]
            # Health check
            try:
                await client.send_request("textDocument/codeAction", {"textDocument": {"uri": ""}, "range": {"start": {"line": 0, "character": 0}, "end": {"line": 0, "character": 0}}, "context": {"diagnostics": []}})
                return client
            except Exception:
                # Session dead, restart
                await self._stop_session(key)

        return await self._start_session(project_path, language, key)

    async def _start_session(self, project_path: str, language: str, key: str) -> JsonRpcClient:
        """Start an LSP server for the given language."""
        server_config = LSP_SERVERS.get(language)
        if not server_config:
            raise ValueError(f"No LSP server configured for {language}")

        # Check if command exists
        command = server_config["command"]
        if not self._command_exists(command):
            logger.warning(f"LSP server {command} not found on PATH")
            raise FileNotFoundError(f"{command} not found on PATH")

        # Enforce max sessions
        if len(self._sessions) >= self.MAX_ACTIVE_SESSIONS:
            # Evict oldest idle session
            oldest = min(self._session_info.items(),
                        key=lambda x: x[1].get("last_used", datetime.min))
            await self._stop_session(oldest[0])

        # Start the server
        client = JsonRpcClient()
        await client.start(command, server_config.get("args"), on_notification=self._on_notification)

        # Initialize
        workspace_uri = Path(project_path).as_uri()
        await client.send_request("initialize", {
            "processId": None,
            "rootUri": workspace_uri,
            "rootPath": project_path,
            "capabilities": {
                "textDocument": {
                    "completion": {"dynamicRegistration": True},
                    "hover": {"dynamicRegistration": True},
                    "definition": {"dynamicRegistration": True},
                    "references": {"dynamicRegistration": True},
                    "formatting": {"dynamicRegistration": True},
                    "codeAction": {"dynamicRegistration": True},
                    "diagnostics": {"dynamicRegistration": True},
                },
            },
        })
        await client.send_notification("initialized")

        self._sessions[key] = client
        self._session_info[key] = {
            "language": language,
            "project_path": project_path,
            "last_used": datetime.utcnow(),
            "started_at": datetime.utcnow(),
        }

        logger.info(f"LSP session started for {language} at {project_path}")
        return client

    async def _stop_session(self, key: str) -> None:
        """Stop an LSP session."""
        client = self._sessions.pop(key, None)
        self._session_info.pop(key, None)
        if client:
            try:
                await client.close()
            except Exception:
                pass

    async def stop_all(self) -> None:
        """Stop all LSP sessions."""
        for key in list(self._sessions.keys()):
            await self._stop_session(key)

    async def _on_notification(self, msg: dict) -> None:
        """Handle LSP notifications (primarily diagnostics)."""
        method = msg.get("method")
        params = msg.get("params", {})

        if method == "textDocument/publishDiagnostics":
            uri = params.get("uri", "")
            diagnostics = params.get("diagnostics", [])
            file_path = self._uri_to_path(uri)
            self._diagnostics[file_path] = diagnostics
            logger.debug(f"Diagnostics received for {file_path}: {len(diagnostics)} issues")

    def get_diagnostics(self, file_path: str) -> list[dict]:
        """Get cached diagnostics for a file."""
        return self._diagnostics.get(file_path, [])

    def _uri_to_path(self, uri: str) -> str:
        """Convert file:// URI to path."""
        if uri.startswith("file://"):
            from urllib.parse import unquote
            return unquote(uri[7:])
        return uri

    @staticmethod
    def _command_exists(command: str) -> bool:
        """Check if a command is available on PATH."""
        import shutil
        return shutil.which(command) is not None

    def get_language_for_file(self, file_path: str) -> str | None:
        """Detect language from file extension."""
        ext = Path(file_path).suffix.lower()
        ext_map = {
            ".py": "python", ".js": "typescript", ".ts": "typescript",
            ".jsx": "typescript", ".tsx": "typescript",
            ".rs": "rust", ".go": "go", ".java": "java",
            ".cpp": "cpp", ".cc": "cpp", ".cxx": "cpp", ".hpp": "cpp",
            ".kt": "kotlin", ".kts": "kotlin",
            ".swift": "swift", ".rb": "ruby", ".php": "php",
            ".cs": "csharp", ".scala": "scala", ".sc": "scala",
        }
        return ext_map.get(ext)
```

### Task 3: LSP Service — High-Level API

```python
"""High-level LSP service wrapping the session manager."""
from __future__ import annotations

from pathlib import Path


class LspService:
    """High-level API for agent tools to interact with LSP."""

    def __init__(self, session_manager: LspSessionManager):
        self._manager = session_manager

    async def get_diagnostics(self, file_path: str) -> list[dict]:
        """Get LSP diagnostics for a file."""
        cached = self._manager.get_diagnostics(file_path)
        if cached:
            return cached

        # Request fresh diagnostics by opening the file
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return []

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        # Open document to trigger diagnostics
        await client.send_notification("textDocument/didOpen", {
            "textDocument": {"uri": uri, "languageId": language, "version": 1, "text": ""}
        })

        return self._manager.get_diagnostics(file_path)

    async def get_completions(self, file_path: str, line: int, column: int) -> list[dict]:
        """Get completion items at a position."""
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return []

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        result = await client.send_request("textDocument/completion", {
            "textDocument": {"uri": uri},
            "position": {"line": line, "character": column},
        })
        return result.get("items", result if isinstance(result, list) else [])

    async def get_hover(self, file_path: str, line: int, column: int) -> dict | None:
        """Get hover/signature info at a position."""
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return None

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        return await client.send_request("textDocument/hover", {
            "textDocument": {"uri": uri},
            "position": {"line": line, "character": column},
        })

    async def go_to_definition(self, file_path: str, line: int, column: int) -> list[dict]:
        """Go to definition of symbol at position."""
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return []

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        result = await client.send_request("textDocument/definition", {
            "textDocument": {"uri": uri},
            "position": {"line": line, "character": column},
        })
        if isinstance(result, dict):
            return [result]
        return result if isinstance(result, list) else []

    async def find_references(self, file_path: str, line: int, column: int) -> list[dict]:
        """Find all references to symbol at position."""
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return []

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        result = await client.send_request("textDocument/references", {
            "textDocument": {"uri": uri},
            "position": {"line": line, "character": column},
        })
        return result if isinstance(result, list) else []

    async def format_file(self, file_path: str) -> list[dict] | None:
        """Get format edits for a file."""
        project_path = self._find_project_root(file_path)
        language = self._manager.get_language_for_file(file_path)
        if not language or not project_path:
            return None

        client = await self._manager.get_or_start_session(project_path, language)
        uri = Path(file_path).as_uri()

        return await client.send_request("textDocument/formatting", {
            "textDocument": {"uri": uri},
            "options": {"tabSize": 4, "insertSpaces": True},
        })

    @staticmethod
    def _find_project_root(file_path: str) -> str | None:
        """Find project root by looking for .git, pyproject.toml, etc."""
        path = Path(file_path).absolute()
        for parent in path.parents:
            if (parent / ".git").exists():
                return str(parent)
            if (parent / "pyproject.toml").exists():
                return str(parent)
            if (parent / "package.json").exists():
                return str(parent)
            if (parent / "Cargo.toml").exists():
                return str(parent)
            if (parent / "go.mod").exists():
                return str(parent)
        return str(path.parent)
```
