### Task 1: Implement DocumentationGenerationService

**Architecture:** The documentation generation service analyzes Python files for missing docstrings, generates suggested docstrings based on function signatures, produces module-level documentation, and generates README content for repositories. Coverage reports track documentation completeness. The `generate_readme` method detects project structure (requirements.txt, package.json, Makefile) and generates appropriate sections.

**File:** `backend/app/services/developer/documentation.py`
```python
import ast
import os
from typing import Dict, List, Any


class DocumentationGenerationService:
    """Generates documentation and analyzes documentation coverage."""

    async def generate_docstring_report(self, file_path: str) -> Dict[str, Any]:
        """Analyze a Python file for missing docstrings with suggestions."""
        try:
            with open(file_path, "r") as f:
                content = f.read()

            tree = ast.parse(content)
            documented = []
            missing = []

            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    docstring = ast.get_docstring(node)
                    entry = {
                        "name": node.name,
                        "type": type(node).__name__,
                        "line": node.lineno,
                    }
                    if docstring:
                        entry["docstring"] = docstring
                        documented.append(entry)
                    else:
                        entry["suggested_docstring"] = self._generate_suggested_docstring(node)
                        missing.append(entry)

            total = len(documented) + len(missing)
            coverage = (len(documented) / max(1, total)) * 100

            return {
                "file": file_path,
                "documented": documented,
                "missing": missing,
                "coverage": round(coverage, 1),
                "total_definitions": total,
                "module_docstring": ast.get_docstring(tree) is not None,
            }
        except Exception as e:
            return {"error": str(e)}

    def _generate_suggested_docstring(self, node: Any) -> str:
        """Generate a suggested docstring for a function/class node."""
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            args = [arg.arg for arg in node.args.args if arg.arg != "self"]
            args_str = ", ".join(args)
            if node.returns:
                return f"TODO: Document {node.name}({args_str}). Returns value."
            return f"TODO: Document {node.name}({args_str})."
        elif isinstance(node, ast.ClassDef):
            methods = [c.name for c in ast.iter_child_nodes(node) if isinstance(c, (ast.FunctionDef, ast.AsyncFunctionDef))]
            return f"TODO: Document class {node.name}. Methods: {', '.join(methods[:5])}."
        return "TODO: Add documentation."

    async def generate_module_doc(self, file_path: str) -> Dict[str, Any]:
        """Generate module-level documentation with structure analysis."""
        try:
            with open(file_path, "r") as f:
                content = f.read()

            tree = ast.parse(content)
            module_docstring = ast.get_docstring(tree)

            functions = [
                node.name for node in ast.iter_child_nodes(tree)
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))
            ]
            classes = [
                node.name for node in ast.iter_child_nodes(tree)
                if isinstance(node, ast.ClassDef)
            ]
            imports = []
            for node in ast.iter_child_nodes(tree):
                if isinstance(node, ast.Import):
                    for alias in node.names:
                        imports.append(alias.name)
                elif isinstance(node, ast.ImportFrom):
                    imports.append(node.module or "")

            suggested = f'"""\nModule: {os.path.basename(file_path)}\n\n'
            if not module_docstring:
                suggested += "TODO: Add module description.\n\n"
            suggested += f"Functions: {len(functions)} — {', '.join(functions[:10])}\n"
            suggested += f"Classes: {len(classes)} — {', '.join(classes[:10])}\n"
            suggested += f"Imports: {len(imports)}\n\"\"\""

            return {
                "file": file_path,
                "existing_docstring": module_docstring,
                "suggested_docstring": suggested,
                "functions": functions,
                "classes": classes,
                "imports": imports,
                "has_existing": module_docstring is not None,
            }
        except Exception as e:
            return {"error": str(e)}

    async def generate_readme(self, repo_path: str) -> str:
        """Generate README content for a repository based on project structure."""
        project_name = os.path.basename(repo_path)

        has_requirements = os.path.exists(os.path.join(repo_path, "requirements.txt"))
        has_package_json = os.path.exists(os.path.join(repo_path, "package.json"))
        has_makefile = os.path.exists(os.path.join(repo_path, "Makefile"))
        has_cargo = os.path.exists(os.path.join(repo_path, "Cargo.toml"))
        has_pyproject = os.path.exists(os.path.join(repo_path, "pyproject.toml"))

        # Install section
        install_section = ""
        if has_pyproject:
            install_section = "```bash\npip install -e .\n```"
        elif has_requirements:
            install_section = "```bash\npip install -r requirements.txt\n```"
        elif has_package_json:
            install_section = "```bash\nnpm install\n```"
        elif has_cargo:
            install_section = "```bash\ncargo build --release\n```"
        else:
            install_section = "TODO: Add installation instructions"

        # Run section
        run_section = ""
        if has_makefile:
            run_section = "```bash\nmake help\n```"
        elif has_package_json:
            run_section = "```bash\nnpm start\n```"
        else:
            run_section = "TODO: Add run instructions"

        # Test section
        test_section = ""
        if has_makefile:
            test_section = "```bash\nmake test\n```"
        elif has_package_json:
            test_section = "```bash\nnpm test\n```"
        elif has_requirements:
            test_section = "```bash\npytest\n```"
        elif has_cargo:
            test_section = "```bash\ncargo test\n```"
        else:
            test_section = "TODO: Add test instructions"

        return f"""# {project_name}

## Description

TODO: Add project description

## Installation

{install_section}

## Usage

{run_section}

## Testing

{test_section}

## Contributing

TODO: Add contributing guidelines

## License

TODO: Add license information
"""
```

### Task 2: Implement TestGenerationService

**Architecture:** The test generation service produces pytest-compatible test scaffolds from Python source files. It infers test arguments from function signatures, generates class-based test setup for class methods, and supports async test generation. The scaffold includes Arrange/Act/Assert comments for test structure guidance.

**File:** `backend/app/services/developer/test_generation.py`
```python
import ast
import os
from typing import Dict, List, Any


class TestGenerationService:
    """Generates test scaffolds and test functions from Python source code."""

    async def generate_test_scaffold(self, file_path: str) -> Dict[str, Any]:
        """Generate a complete test scaffold for a Python file."""
        try:
            with open(file_path, "r") as f:
                content = f.read()

            tree = ast.parse(content)
            file_name = os.path.basename(file_path)
            module_name = file_name.replace(".py", "")

            functions = []
            classes_with_methods = []

            for node in ast.iter_child_nodes(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    if not node.name.startswith("_"):
                        args = [arg.arg for arg in node.args.args if arg.arg != "self"]
                        functions.append({
                            "name": node.name,
                            "args": args,
                            "is_async": isinstance(node, ast.AsyncFunctionDef),
                            "defaults": len(node.args.defaults),
                        })
                elif isinstance(node, ast.ClassDef):
                    methods = []
                    for child in ast.iter_child_nodes(node):
                        if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                            if not child.name.startswith("_") or child.name == "__init__":
                                child_args = [a.arg for a in child.args.args if a.arg != "self"]
                                methods.append({
                                    "name": child.name,
                                    "args": child_args,
                                    "is_async": isinstance(child, ast.AsyncFunctionDef),
                                })
                    classes_with_methods.append({"name": node.name, "methods": methods})

            # Build test file content
            lines = ['"""Auto-generated test scaffold."""', "import pytest", ""]

            # Import path
            import_path = file_path.replace("/", ".").replace(".py", "")
            if import_path.startswith("."):
                import_path = import_path[1:]
            lines.append(f"from {import_path} import *")
            lines.append("")

            # Function tests
            for func in functions:
                prefix = "async " if func["is_async"] else ""
                lines.append(f"{prefix}def test_{func['name']}():")
                if func["args"]:
                    arg_str = ", ".join(f"{a}=None" for a in func["args"])
                    lines.append(f"    # Arrange — set up: {arg_str}")
                lines.append(f"    # Act")
                lines.append(f"    # TODO: Call {func['name']}")
                lines.append(f"    # Assert")
                lines.append(f"    assert True  # TODO: Add assertions")
                lines.append("")

            # Class tests
            for cls in classes_with_methods:
                lines.append(f"class Test{cls['name']}:")
                lines.append(f"    def setup_method(self):")
                lines.append(f'        """Set up test fixtures."""')
                lines.append(f"        self.instance = {cls['name']}()")
                lines.append("")
                for method in cls["methods"]:
                    if method["name"] == "__init__":
                        continue
                    prefix = "async " if method["is_async"] else ""
                    lines.append(f"    {prefix}def test_{method['name']}(self):")
                    lines.append(f"        # Arrange")
                    lines.append(f"        # Act")
                    lines.append(f"        # TODO: Call self.instance.{method['name']}()")
                    lines.append(f"        # Assert")
                    lines.append(f"        assert True  # TODO: Add assertions")
                    lines.append("")

            if not functions and not classes_with_methods:
                lines.append("# TODO: Add tests for this module")

            scaffold = "\n".join(lines)
            test_file_name = f"test_{module_name}.py"

            return {
                "source_file": file_path,
                "test_file": test_file_name,
                "scaffold": scaffold,
                "functions_tested": [f["name"] for f in functions],
                "classes_tested": [c["name"] for c in classes_with_methods],
                "total_tests_generated": len(functions) + sum(
                    len(c["methods"]) - (1 if any(m["name"] == "__init__" for m in c["methods"]) else 0)
                    for c in classes_with_methods
                ),
            }
        except Exception as e:
            return {"error": str(e)}

    async def generate_test_for_function(
        self, function_name: str, args: List[str], is_async: bool = False
    ) -> str:
        """Generate a single test function with Arrange/Act/Assert."""
        prefix = "async " if is_async else ""
        lines = [f"{prefix}def test_{function_name}():", '    """Test for ' + function_name + '."""']
        if args:
            lines.append("    # Arrange")
            for arg in args:
                lines.append(f"    {arg} = None  # TODO: Set up test value")
        lines.append("    # Act")
        lines.append(f"    # TODO: Call {function_name}()")
        lines.append("    # Assert")
        lines.append("    # TODO: Add assertions")
        lines.append("    assert True")
        return "\n".join(lines)
```

### Task 3: Implement RefactoringService

**Architecture:** The refactoring service detects long functions, duplicate code, deeply nested blocks, too many parameters, and other code smells. It integrates with the dependency graph service to assess impact of proposed refactorings. The `suggest_extract_function` method analyzes variable assignments and references to infer parameters. All suggestions include the dependency graph for impact assessment.

**File:** `backend/app/services/developer/refactoring.py`
```python
import ast
import os
import re
import logging
from typing import Dict, List, Any

logger = logging.getLogger(__name__)


class RefactoringService:
    """Detects refactoring opportunities with impact assessment."""

    def __init__(self, dependency_graph=None):
        self.dep_graph = dependency_graph

    async def analyze_file(self, file_path: str) -> Dict[str, Any]:
        """Analyze file for refactoring opportunities."""
        try:
            with open(file_path, "r") as f:
                content = f.read()

            suggestions: List[Dict[str, Any]] = []
            lines = content.split("\n")

            # Check for long functions
            func_start = None
            func_name = None
            func_indent = 0
            for i, line in enumerate(lines, 1):
                stripped = line.strip()
                if stripped.startswith("def ") or stripped.startswith("async def "):
                    func_start = i
                    func_name = stripped.split("(")[0].replace("def ", "").replace("async def ", "")
                    func_indent = len(line) - len(line.lstrip())
                elif func_start and stripped and not stripped.startswith("#"):
                    current_indent = len(line) - len(line.lstrip())
                    if current_indent <= func_indent and stripped:
                        func_len = i - func_start
                        if func_len > 50:
                            suggestions.append({
                                "type": "long_function",
                                "severity": "medium",
                                "line": func_start,
                                "message": f"Function '{func_name}' is {func_len} lines — consider splitting",
                                "refactoring": "Extract smaller functions from this method",
                            })
                        func_start = None

            # Check for duplicate code
            seen_lines: Dict[str, int] = {}
            for i, line in enumerate(lines, 1):
                stripped = line.strip()
                if len(stripped) > 30 and not stripped.startswith("#"):
                    if stripped in seen_lines:
                        suggestions.append({
                            "type": "duplicate_code",
                            "severity": "medium",
                            "line": i,
                            "duplicate_of": seen_lines[stripped],
                            "message": f"Duplicate at line {i} matches line {seen_lines[stripped]}",
                            "refactoring": "Extract duplicated logic into a shared function",
                        })
                    else:
                        seen_lines[stripped] = i

            # Check for too many parameters
            try:
                tree = ast.parse(content)
                for node in ast.walk(tree):
                    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        param_count = len(node.args.args)
                        if param_count > 7:
                            suggestions.append({
                                "type": "too_many_parameters",
                                "severity": "medium",
                                "line": node.lineno,
                                "message": f"'{node.name}' has {param_count} params — use a config object",
                                "refactoring": "Group related parameters into a dataclass or dict",
                            })
            except SyntaxError:
                pass

            # Check for deeply nested code
            for i, line in enumerate(lines, 1):
                if line.strip():
                    depth = (len(line) - len(line.lstrip())) // 4
                    if depth > 4:
                        suggestions.append({
                            "type": "deep_nesting",
                            "severity": "low",
                            "line": i,
                            "message": f"Deep nesting (depth {depth}) — use early returns or extract",
                            "refactoring": "Use guard clauses or extract nested blocks",
                        })

            # Add dependency impact if available
            impact = None
            if self.dep_graph:
                try:
                    impact = self.dep_graph.get_impact_analysis(file_path)
                except Exception:
                    pass

            return {
                "file": file_path,
                "suggestions": suggestions,
                "total_lines": len(lines),
                "total_suggestions": len(suggestions),
                "impact_analysis": impact,
                "by_type": {
                    t: sum(1 for s in suggestions if s["type"] == t)
                    for t in set(s["type"] for s in suggestions)
                } if suggestions else {},
            }
        except Exception as e:
            return {"error": str(e)}

    async def suggest_extract_function(
        self, file_path: str, start_line: int, end_line: int, name: str
    ) -> Dict[str, Any]:
        """Suggest extracting lines into a new function with parameter inference."""
        try:
            with open(file_path, "r") as f:
                all_lines = f.readlines()

            extracted = all_lines[start_line - 1: end_line]
            extracted_code = "".join(extracted)

            # Analyze variable assignments and references
            assigns = set()
            uses = set()
            keywords = {
                "if", "else", "elif", "for", "while", "return", "def", "class",
                "import", "from", "True", "False", "None", "and", "or", "not",
                "in", "is", "try", "except", "finally", "with", "as", "raise",
                "yield", "lambda", "pass", "break", "continue", "del", "global",
                "nonlocal", "assert", "print",
            }
            for line in extracted:
                match = re.match(r"\s+(\w+)\s*=", line)
                if match:
                    assigns.add(match.group(1))
                for word in re.findall(r"\b(\w+)\b", line):
                    if word not in assigns and word not in keywords and not word.isdigit():
                        uses.add(word)

            params = sorted(uses - assigns)

            # Build new function with type hints placeholder
            new_function = f"def {name}({', '.join(params)}):\n"
            new_function += '    """TODO: Document this function."""\n'
            for line in extracted:
                new_function += "    " + line
            new_function += f"\n    return None  # TODO: Add return value\n"

            # Calculate impact
            impact = None
            if self.dep_graph:
                try:
                    impact = self.dep_graph.get_impact_analysis(file_path)
                except Exception:
                    pass

            return {
                "original_lines": f"{start_line}-{end_line}",
                "new_function_name": name,
                "parameters": params,
                "new_function_code": new_function,
                "call_replacement": f"{name}({', '.join(params)})",
                "impact_analysis": impact,
            }
        except Exception as e:
            return {"error": str(e)}
```

```python
@pytest.mark.asyncio
async def test_docstring_report():
    service = DocumentationGenerationService()
```
