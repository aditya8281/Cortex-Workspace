### Task 1: Token Counting

**Objective:** Count input/output tokens per LLM request, track cumulative usage per user and per session.

**File:** `backend/app/agents/token_counter.py` (new or enhance existing)

**Architecture:**

```python
"""Token counting and usage tracking.

Design:
- tiktoken for accurate token counting (with fallback to estimation)
- Per-request tracking: input_tokens, output_tokens, total_tokens
- Per-user cumulative tracking (for billing/limits)
- Per-session tracking (for context management)
- Lightweight: < 1ms overhead per count
"""

import logging
from typing import Optional
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)

# Approximate chars per token for estimation fallback
APPROX_CHARS_PER_TOKEN = 4


@dataclass
class TokenCount:
    """Token count for a single request."""
    input_tokens: int = 0
    output_tokens: int = 0
    total_tokens: int = 0
    model: str = ""
    context_window: int = 0  # Max tokens for this model
    context_usage_pct: float = 0.0  # input_tokens / context_window


@dataclass
class CumulativeUsage:
    """Cumulative token usage over time."""
    total_input_tokens: int = 0
    total_output_tokens: int = 0
    total_requests: int = 0
    request_count: int = 0  # Same as total_requests
    avg_input_tokens: float = 0.0
    avg_output_tokens: float = 0.0


class TokenCounter:
    """Count tokens for LLM requests.

    Supports:
    - tiktoken for accurate counting (when available)
    - Character-based estimation as fallback
    - Per-request and cumulative tracking
    """

    # Model context window sizes
    CONTEXT_WINDOWS = {
        "gpt-4": 8192,
        "gpt-4-turbo": 128000,
        "gpt-4o": 128000,
        "gpt-4o-mini": 128000,
        "claude-3-opus": 200000,
        "claude-3-sonnet": 200000,
        "claude-3-haiku": 200000,
        "claude-3.5-sonnet": 200000,
        "llama3": 8192,
        "mistral": 32768,
        "default": 8192,
    }

    def __init__(self, model: str = "default"):
        self.model = model
        self.max_context_tokens = self.CONTEXT_WINDOWS.get(model, 8192)
        self._tokenizer = None
        self._cumulative = CumulativeUsage()
        self._session_usage: dict[str, CumulativeUsage] = {}

        # Try to load tiktoken
        try:
            import tiktoken
            self._tokenizer = tiktoken.encoding_for_model(model)
        except (ImportError, KeyError):
            logger.info("tiktoken not available for %s, using estimation", model)

    async def count_tokens(self, messages: list[dict]) -> int:
        """Count total tokens in a list of messages.

        Args:
            messages: List of message dicts with 'role' and 'content'

        Returns:
            Total token count
        """
        total = 0
        for msg in messages:
            content = msg.get("content", "")
            if isinstance(content, str):
                total += self._count_text_tokens(content)
            elif isinstance(content, list):
                # Multi-part messages
                for part in content:
                    if isinstance(part, dict) and "text" in part:
                        total += self._count_text_tokens(part["text"])
            # Add overhead per message (role, separators)
            total += 4

        return total

    def _count_text_tokens(self, text: str) -> int:
        """Count tokens in a text string."""
        if self._tokenizer:
            try:
                return len(self._tokenizer.encode(text))
            except Exception:
                pass

        # Fallback: character-based estimation
        return max(1, len(text) // APPROX_CHARS_PER_TOKEN)

    async def count_request(
        self,
        input_messages: list[dict],
        output_text: str,
        model: Optional[str] = None,
    ) -> TokenCount:
        """Count tokens for a full request (input + output).

        Returns:
            TokenCount with breakdown and context usage
        """
        model = model or self.model
        context_window = self.CONTEXT_WINDOWS.get(model, 8192)

        input_tokens = await self.count_tokens(input_messages)
        output_tokens = self._count_text_tokens(output_text)
        total = input_tokens + output_tokens

        count = TokenCount(
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            total_tokens=total,
            model=model,
            context_window=context_window,
            context_usage_pct=input_tokens / context_window if context_window > 0 else 0,
        )

        # Update cumulative
        self._cumulative.total_input_tokens += input_tokens
        self._cumulative.total_output_tokens += output_tokens
        self._cumulative.total_requests += 1
        self._cumulative.request_count = self._cumulative.total_requests
        self._cumulative.avg_input_tokens = (
            self._cumulative.total_input_tokens / self._cumulative.total_requests
        )
        self._cumulative.avg_output_tokens = (
            self._cumulative.total_output_tokens / self._cumulative.total_requests
        )

        return count

    def get_cumulative_usage(self) -> CumulativeUsage:
        """Get cumulative token usage."""
        return self._cumulative

    def get_context_usage(self, messages: list[dict]) -> float:
        """Get current context window usage as a percentage."""
        # Synchronous version for quick checks
        total = 0
        for msg in messages:
            content = str(msg.get("content", ""))
            total += max(1, len(content) // APPROX_CHARS_PER_TOKEN)
            total += 4  # per-message overhead
        return total / self.max_context_tokens if self.max_context_tokens > 0 else 0

    def set_model(self, model: str):
        """Change the model (updates context window)."""
        self.model = model
        self.max_context_tokens = self.CONTEXT_WINDOWS.get(model, 8192)
        # Reset tokenizer
        try:
            import tiktoken
            self._tokenizer = tiktoken.encoding_for_model(model)
        except (ImportError, KeyError):
            self._tokenizer = None
```

### Task 2: TPS Tracking

**Objective:** Measure tokens per second for streaming responses. Display real-time TPS to the user.

**File:** `backend/app/agents/tps_tracker.py` (new)

**Architecture:**

```python
"""Tokens per second (TPS) tracking for streaming responses.

Measures:
- Instantaneous TPS: tokens delivered in the last second
- Average TPS: total tokens / total time
- Peak TPS: maximum instantaneous TPS observed
"""

import time
import logging
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)


@dataclass
class TPSMeasurement:
    """TPS measurement for a streaming response."""
    start_time: float = 0
    end_time: float = 0
    total_tokens: int = 0
    total_duration: float = 0
    avg_tps: float = 0
    peak_tps: float = 0
    first_token_latency_ms: float = 0  # Time to first token
    _last_token_time: float = 0
    _window_tokens: int = 0
    _window_start: float = 0


class TPSTracker:
    """Track tokens per second for streaming responses.

    Usage:
        tracker = TPSTracker()
        tracker.start()

        # For each streamed token:
        tracker.on_token()

        # When stream ends:
        measurement = tracker.finish()
        print(f"Average TPS: {measurement.avg_tps:.1f}")
    """

    def __init__(self):
        self._measurement: TPSMeasurement = TPSMeasurement()
        self._instantaneous_window: list[tuple[float, int]] = []  # (timestamp, token_count)

    def start(self):
        """Start a new TPS measurement."""
        now = time.time()
        self._measurement = TPSMeasurement(start_time=now, _window_start=now)
        self._instantaneous_window.clear()

    def on_token(self, count: int = 1):
        """Called when a token (or token batch) is streamed.

        Args:
            count: Number of tokens delivered (default 1)
        """
        now = time.time()
        m = self._measurement

        m.total_tokens += count
        m._last_token_time = now

        # First token latency
        if m.total_tokens == count:
            m.first_token_latency_ms = (now - m.start_time) * 1000

        # Track for instantaneous TPS
        self._instantaneous_window.append((now, count))

        # Clean old entries (keep last 1 second)
        cutoff = now - 1.0
        self._instantaneous_window = [
            (t, c) for t, c in self._instantaneous_window if t >= cutoff
        ]

    def get_instantaneous_tps(self) -> float:
        """Get current TPS (tokens in last second)."""
        now = time.time()
        cutoff = now - 1.0
        recent_tokens = sum(c for t, c in self._instantaneous_window if t >= cutoff)
        return float(recent_tokens)

    def finish(self) -> TPSMeasurement:
        """Finish measurement and return results."""
        m = self._measurement
        m.end_time = time.time()
        m.total_duration = m.end_time - m.start_time

        if m.total_duration > 0:
            m.avg_tps = m.total_tokens / m.total_duration

        # Peak TPS from instantaneous window history
        if self._instantaneous_window:
            # Calculate peak from 1-second sliding windows
            times = [t for t, _ in self._instantaneous_window]
            if times:
                max_tokens = 0
                for i, (t, _) in enumerate(self._instantaneous_window):
                    window_start = t - 1.0
                    window_tokens = sum(c for wt, c in self._instantaneous_window if wt >= window_start)
                    max_tokens = max(max_tokens, window_tokens)
                m.peak_tps = float(max_tokens)

        return m

    def reset(self):
        """Reset tracker for reuse."""
        self._measurement = TPSMeasurement()
        self._instantaneous_window.clear()
```

### Task 3: Context Usage Percentage

**Objective:** Track how much of the context window is filled at each point in the conversation. Display to user as a progress bar.

**File:** Integrated into `token_counter.py` (Task 1)

**Architecture:**

```python
# Context usage tracking (add to TokenCounter)

@dataclass
class ContextUsageSnapshot:
    """Snapshot of context usage at a point in time."""
    timestamp: float
    tokens_used: int
    tokens_max: int
    usage_pct: float
    message_count: int
    compaction_triggered: bool


class ContextUsageTracker:
    """Track context window fill level over time.

    Provides:
    - Current usage percentage
    - History of usage snapshots
    - Prediction of when compaction will be needed
    - Triggers compaction at 85%
    """

    COMPACTION_THRESHOLD = 0.85

    def __init__(self, token_counter: TokenCounter):
        self.token_counter = token_counter
        self._history: list[ContextUsageSnapshot] = []
        self._compaction_count = 0

    async def snapshot(self, messages: list[dict]) -> ContextUsageSnapshot:
        """Take a snapshot of current context usage."""
        tokens_used = await self.token_counter.count_tokens(messages)
        tokens_max = self.token_counter.max_context_tokens
        usage_pct = tokens_used / tokens_max if tokens_max > 0 else 0

        snapshot = ContextUsageSnapshot(
            timestamp=time.time(),
            tokens_used=tokens_used,
            tokens_max=tokens_max,
            usage_pct=usage_pct,
            message_count=len(messages),
            compaction_triggered=False,
        )

        self._history.append(snapshot)

        # Check if compaction needed
        if usage_pct >= self.COMPACTION_THRESHOLD:
            snapshot.compaction_triggered = True
            self._compaction_count += 1

        return snapshot

    def get_current_usage(self) -> Optional[ContextUsageSnapshot]:
        """Get the most recent snapshot."""
        return self._history[-1] if self._history else None

    def get_usage_history(self) -> list[ContextUsageSnapshot]:
        """Get all snapshots."""
        return self._history

    def predict_compaction_turns(self, messages: list[dict], avg_tokens_per_turn: int = 500) -> int:
        """Predict how many turns until compaction is needed."""
        if not self._history:
            return -1

        current = self._history[-1]
        remaining = current.tokens_max - current.tokens_used
        return remaining // avg_tokens_per_turn if avg_tokens_per_turn > 0 else -1

    def get_stats(self) -> dict:
        """Return context usage statistics."""
        return {
            "current_usage_pct": self._history[-1].usage_pct if self._history else 0,
            "compaction_count": self._compaction_count,
            "snapshots_taken": len(self._history),
            "max_usage_reached": max((s.usage_pct for s in self._history), default=0),
        }
```

### Task 4: Tool Execution Timing

**Objective:** Measure per-tool latency, track success/failure rates, and provide per-tool performance profiles.

**File:** `backend/app/agents/tools/timing.py` (new)

**Architecture:**

```python
"""Tool execution timing and metrics.

Tracks per-tool:
- Execution count
- Success/failure rate
- Latency (avg, p50, p95, p99)
- Output size
- Error types
"""

import time
import logging
from dataclasses import dataclass, field
from typing import Optional

logger = logging.getLogger(__name__)


@dataclass
class ToolTimingRecord:
    """Single tool execution timing record."""
    tool_name: str
    start_time: float
    end_time: float = 0
    duration_ms: float = 0
    success: bool = True
    output_size_bytes: int = 0
    error_type: Optional[str] = None
    is_mcp: bool = False
    mcp_server: Optional[str] = None


class ToolTimingTracker:
    """Track tool execution timing and metrics.

    Usage:
        tracker = ToolTimingTracker()

        # Before tool execution:
        record = tracker.start("search_memory")

        # After tool execution:
        record.end_time = time.time()
        record.success = True
        tracker.finish(record)
    """

    def __init__(self):
        self._records: dict[str, list[ToolTimingRecord]] = {}
        self._active: dict[str, ToolTimingRecord] = {}

    def start(self, tool_name: str, is_mcp: bool = False, mcp_server: str = None) -> ToolTimingRecord:
        """Start timing a tool execution."""
        record = ToolTimingRecord(
            tool_name=tool_name,
            start_time=time.time(),
            is_mcp=is_mcp,
            mcp_server=mcp_server,
        )
        self._active[tool_name] = record
        return record

    def finish(self, record: ToolTimingRecord) -> ToolTimingRecord:
        """Finish timing and store the record."""
        record.end_time = time.time()
        record.duration_ms = (record.end_time - record.start_time) * 1000

        if record.tool_name not in self._records:
            self._records[record.tool_name] = []
        self._records[record.tool_name].append(record)

        # Clean from active
        self._active.pop(record.tool_name, None)

        return record

    def get_tool_stats(self, tool_name: str) -> dict:
        """Get statistics for a specific tool."""
        records = self._records.get(tool_name, [])
        if not records:
            return {"tool": tool_name, "executions": 0}

        durations = [r.duration_ms for r in records]
        successes = sum(1 for r in records if r.success)
        failures = len(records) - successes

        # Calculate percentiles
        sorted_durations = sorted(durations)
        n = len(sorted_durations)

        return {
            "tool": tool_name,
            "executions": n,
            "successes": successes,
            "failures": failures,
            "success_rate": successes / n if n > 0 else 0,
            "avg_ms": sum(durations) / n,
            "p50_ms": sorted_durations[n // 2] if n > 0 else 0,
            "p95_ms": sorted_durations[int(n * 0.95)] if n > 0 else 0,
            "p99_ms": sorted_durations[int(n * 0.99)] if n > 0 else 0,
            "min_ms": min(durations) if durations else 0,
            "max_ms": max(durations) if durations else 0,
        }

    def get_all_stats(self) -> dict:
        """Get statistics for all tools."""
        return {
            tool_name: self.get_tool_stats(tool_name)
            for tool_name in self._records
        }

    def get_slow_tools(self, threshold_ms: float = 1000) -> list[dict]:
        """Get tools with average latency above threshold."""
        all_stats = self.get_all_stats()
        return [
            stats for stats in all_stats.values()
            if stats.get("avg_ms", 0) > threshold_ms
        ]

    def reset(self):
        """Reset all records."""
        self._records.clear()
        self._active.clear()
```

### Task 5: Structured Logging

**Objective:** Implement JSON structured logs for all agent operations. Every agent event produces a structured log entry.

**File:** `backend/app/core/structured_logging.py` (new)

**Architecture:**

```python
"""Structured logging for agent operations.

Every agent event produces a JSON log entry with:
- timestamp
- event_type
- component (agent_loop, compactor, tool_executor, etc.)
- duration_ms
- user_id
- run_id
- metadata (event-specific fields)

Logs are:
- JSON formatted for machine parsing
- Written to stdout for container environments
- Optionally written to file for local debugging
"""

import json
import logging
import sys
import time
from datetime import datetime, timezone
from typing import Any, Optional
from dataclasses import dataclass, asdict


@dataclass
class AgentLogEntry:
    """Structured log entry for agent operations."""
    timestamp: str
    level: str
    event_type: str
    component: str
    message: str
    duration_ms: Optional[float] = None
    user_id: Optional[int] = None
    run_id: Optional[str] = None
    iteration: Optional[int] = None
    tool_name: Optional[str] = None
    token_count: Optional[int] = None
    context_usage_pct: Optional[float] = None
    metadata: Optional[dict] = None

    def to_json(self) -> str:
        """Serialize to JSON string."""
        d = asdict(self)
        # Remove None values for cleaner output
        return json.dumps({k: v for k, v in d.items() if v is not None})


class StructuredLogger:
    """Structured logger for agent operations.

    Usage:
        logger = StructuredLogger("agent_loop")

        # Log an event
        logger.info("llm_call_started", user_id=1, iteration=3, metadata={"model": "gpt-4o"})

        # Log with timing
        with logger.timed("tool_execution", tool_name="search_memory"):
            result = await execute_tool("search_memory", args)
    """

    def __init__(self, component: str, log_file: Optional[str] = None):
        self.component = component
        self._logger = logging.getLogger(f"cortex.{component}")

        # Set up JSON handler
        handler = logging.StreamHandler(sys.stdout)
        handler.setFormatter(logging.Formatter("%(message)s"))
        self._logger.addHandler(handler)
        self._logger.setLevel(logging.DEBUG)

        # Optional file handler
        if log_file:
            file_handler = logging.FileHandler(log_file)
            file_handler.setFormatter(logging.Formatter("%(message)s"))
            self._logger.addHandler(file_handler)

    def _log(
        self,
        level: str,
        event_type: str,
        message: str,
        **kwargs,
    ):
        """Log a structured event."""
        entry = AgentLogEntry(
            timestamp=datetime.now(timezone.utc).isoformat(),
            level=level,
            event_type=event_type,
            component=self.component,
            message=message,
            **{k: v for k, v in kwargs.items() if v is not None},
        )
        self._logger.log(
            getattr(logging, level.upper(), logging.INFO),
            entry.to_json(),
        )

    def debug(self, event_type: str, message: str, **kwargs):
        self._log("debug", event_type, message, **kwargs)

    def info(self, event_type: str, message: str, **kwargs):
        self._log("info", event_type, message, **kwargs)

    def warning(self, event_type: str, message: str, **kwargs):
        self._log("warning", event_type, message, **kwargs)

    def error(self, event_type: str, message: str, **kwargs):
        self._log("error", event_type, message, **kwargs)

    def timed(self, event_type: str, **kwargs):
        """Context manager for timing an operation."""
        return _TimedContext(self, event_type, **kwargs)


class _TimedContext:
    """Context manager that logs duration."""

    def __init__(self, logger: StructuredLogger, event_type: str, **kwargs):
        self.logger = logger
        self.event_type = event_type
        self.kwargs = kwargs
        self.start_time = 0

    def __enter__(self):
        self.start_time = time.time()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        duration_ms = (time.time() - self.start_time) * 1000
        if exc_type:
            self.logger.error(
                self.event_type,
                f"{self.event_type} failed",
                duration_ms=duration_ms,
                metadata={"error": str(exc_val), **self.kwargs},
            )
        else:
            self.logger.info(
                self.event_type,
                f"{self.event_type} completed",
                duration_ms=duration_ms,
                **self.kwargs,
            )
        return False  # Don't suppress exceptions
```

```python
"""Performance baseline capture and comparison.

Captures baseline metrics:
- Agent loop latency (first token, total)
- Tool execution latency (per tool)
- TPS (average and peak)
- Context compaction time
- MCP tool call round-trip
- Memory usage

Stores baselines in database for comparison over time.
"""

import time
import logging
import psutil
from typing import Optional
from dataclasses import dataclass

logger = logging.getLogger(__name__)


@dataclass
class BaselineMetrics:
    """Complete performance baseline snapshot."""
    # Agent loop metrics
    first_token_latency_ms: float = 0
    avg_loop_duration_ms: float = 0
    avg_iterations: float = 0

    # Tool metrics
    avg_tool_latency_ms: float = 0
    p95_tool_latency_ms: float = 0

    # TPS metrics
    avg_tps: float = 0
    peak_tps: float = 0

    # Compaction metrics
    avg_compaction_time_ms: float = 0

    # MCP metrics

class PerformanceBaseline:
    """Capture and store performance baselines.

    Usage:
        baseline = PerformanceBaseline(token_counter, tps_tracker, tool_timing)

        # After a series of operations:
        metrics = await baseline.capture()
        await baseline.store(metrics)

        # Compare with previous baseline:
        comparison = await baseline.compare(metrics)
    """

    def __init__(
        self,
        token_counter=None,
        tps_tracker=None,
        tool_timing=None,
        context_tracker=None,
    ):
        self.token_counter = token_counter
        self.tps_tracker = tps_tracker
        self.tool_timing = tool_timing
        self.context_tracker = context_tracker

    async def capture(self) -> BaselineMetrics:
        """Capture current performance metrics."""
        metrics = BaselineMetrics(
            captured_at=time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        )

        # Tool metrics
        if self.tool_timing:
            all_stats = self.tool_timing.get_all_stats()
            if all_stats:
                latencies = [s.get("avg_ms", 0) for s in all_stats.values()]
                p95_latencies = [s.get("p95_ms", 0) for s in all_stats.values()]
                metrics.avg_tool_latency_ms = sum(latencies) / len(latencies) if latencies else 0
                metrics.p95_tool_latency_ms = max(p95_latencies) if p95_latencies else 0

        # TPS metrics
        if self.tps_tracker:
            measurement = self.tps_tracker.finish()
            metrics.avg_tps = measurement.avg_tps
            metrics.peak_tps = measurement.peak_tps

        # Context metrics
        if self.context_tracker:
            stats = self.context_tracker.get_stats()
            metrics.avg_compaction_time_ms = 0  # Would come from compaction timing

        # Resource metrics
        try:
            process = psutil.Process()
            metrics.memory_rss_mb = process.memory_info().rss / 1024 / 1024
            metrics.cpu_percent = process.cpu_percent(interval=0.1)
        except Exception:
            pass

        return metrics

        """Store baseline metrics in database."""
```
