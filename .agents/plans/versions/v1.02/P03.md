## Implementation Tasks

### Task 1: Streaming Agent Loop Architecture

**Objective:** Rewrite the agent loop as a single async generator that streams responses via SSE, parses tool calls, and accumulates responses across multiple LLM rounds.

**Constitution:** Section 10.1 — One streaming agent loop handles all user interactions. Planning is a tool call, not a separate agent. No Planner→Executor separation.

**File:** `backend/app/agents/loop.py` (rewrite existing)

**Architecture:**

```python
"""Streaming agent loop — the central nervous system of CORTEX.

Single async generator that:
1. Receives user message
2. Classifies intent (P03-T4)
3. Checks context usage (triggers compaction if needed, P03-T2)
4. Builds tool policy (P03-T8)
5. Calls LLM with tools
6. Parses tool calls from LLM response
7. Executes tools (with sandbox, P05-T5)
8. Feeds tool results back to LLM
9. Repeats until: LLM produces final answer, stall detected, or max iterations
10. Optionally verifies completion (P03-T6)
11. Streams all of the above via SSE

Max iterations: 25 (configurable).
All state in conversation history (no separate planner state).
"""

import asyncio
import json
import logging
import time
from dataclasses import dataclass, field
from typing import AsyncGenerator, Any, Optional

logger = logging.getLogger(__name__)


@dataclass
class AgentState:
    """Mutable state for a single agent loop execution."""
    messages: list[dict[str, str]] = field(default_factory=list)
    tool_calls_this_turn: int = 0
    total_tool_calls: int = 0
    total_tokens_used: int = 0
    iteration: int = 0
    max_iterations: int = 25
    start_time: float = field(default_factory=time.time)
    compacted: bool = False
    original_message_count: int = 0


@dataclass
class StreamEvent:
    """Events emitted by the agent loop for SSE streaming."""
    event_type: str  # "text", "tool_call", "tool_result", "thinking", "done", "error"
    data: dict[str, Any]
    timestamp: float = field(default_factory=time.time)


async def agent_loop(
    user_message: str,
    conversation_history: list[dict[str, str]],
    tools: list[dict[str, Any]],
    tool_executor: Any,  # Callable[[str, dict], Awaitable[Any]]
    llm_chat: Any,  # Callable for LLM chat
    token_counter: Any,  # For context usage tracking
    event_bus: Any,  # For lifecycle events
    user_id: int,
    tool_policy: Optional[Any] = None,  # Per-turn tool policy (P03-T8)
    intent_classifier: Any = None,  # (P03-T4)
    compactor: Any = None,  # (P03-T2)
    stall_detector: Any = None,  # (P03-T5)
    verifier: Any = None,  # (P03-T6)
    model: str = "default",
) -> AsyncGenerator[StreamEvent, None]:
    """The single streaming agent loop. Everything the agent does flows through here.

    Yields StreamEvents for SSE streaming to the client.
    """
    state = AgentState(
        messages=list(conversation_history),
        original_message_count=len(conversation_history),
    )

    # Step 1: Classify intent
    if intent_classifier:
        intent = await intent_classifier.classify(user_message, conversation_history)
        yield StreamEvent(event_type="thinking", data={"step": "intent", "intent": intent})

        if intent == "casual":
            # Handle casual message without full agent loop
            response = await intent_classifier.handle_casual(user_message)
            yield StreamEvent(event_type="text", data={"content": response})
            yield StreamEvent(event_type="done", data={"reason": "casual_intent"})
            return
        elif intent == "greeting":
            response = await intent_classifier.handle_greeting(user_message)
            yield StreamEvent(event_type="text", data={"content": response})
            yield StreamEvent(event_type="done", data={"reason": "greeting_intent"})
            return

    # Add user message to state
    state.messages.append({"role": "user", "content": user_message})

    # Step 2: Check context usage and compact if needed
    if compactor and token_counter:
        token_count = await token_counter.count_tokens(state.messages)
        context_usage = token_count / token_counter.max_context_tokens

        if context_usage > 0.85:
            yield StreamEvent(event_type="thinking", data={
                "step": "compaction",
                "usage": f"{context_usage:.1%}",
            })
            summary = await compactor.compact(
                state.messages[:-1],  # Don't compact the current message
                llm_chat=llm_chat,
                model=model,
            )
            state.messages = [
                {"role": "system", "content": f"[Context Summary]\n{summary}"},
                state.messages[-1],  # Keep current user message
            ]
            state.compacted = True
            state.total_tokens_used = await token_counter.count_tokens(state.messages)

    # Step 3: Agent loop — iterate until done
    while state.iteration < state.max_iterations:
        state.iteration += 1

        # Build per-turn tool policy
        active_tools = tools
        if tool_policy:
            active_tools = await tool_policy.compose(
                tools=tools,
                conversation_history=state.messages,
                iteration=state.iteration,
            )

        # Step 3a: Call LLM
        yield StreamEvent(event_type="thinking", data={
            "step": "llm_call",
            "iteration": state.iteration,
            "tools_available": len(active_tools),
        })

        llm_start = time.time()
        response = await llm_chat(
            messages=state.messages,
            tools=active_tools,
            model=model,
            stream=True,
        )
        llm_duration = time.time() - llm_start

        # Step 3b: Parse LLM response
        content = ""
        tool_calls = []

        async for chunk in response:
            if chunk.get("type") == "text":
                content += chunk["content"]
                yield StreamEvent(event_type="text", data={"content": chunk["content"]})
            elif chunk.get("type") == "tool_call":
                tool_calls.append(chunk)

        # Add assistant response to history
        assistant_msg: dict[str, Any] = {"role": "assistant", "content": content}
        if tool_calls:
            assistant_msg["tool_calls"] = tool_calls
        state.messages.append(assistant_msg)

        # Step 3c: If no tool calls, we're done
        if not tool_calls:
            break

        # Step 3d: Check for stalls
        if stall_detector:
            stall = await stall_detector.check(
                tool_calls=tool_calls,
                history=state.messages,
                iteration=state.iteration,
            )
            if stall.is_stalled:
                yield StreamEvent(event_type="thinking", data={
                    "step": "stall_detected",
                    "reason": stall.reason,
                })
                # Force a final answer from the LLM
                state.messages.append({
                    "role": "user",
                    "content": "Please provide your final answer now. Do not use any more tools."
                })
                continue  # One more LLM call to get final answer

        # Step 3e: Execute tool calls
        for tool_call in tool_calls:
            tool_name = tool_call["function"]["name"]
            tool_args = json.loads(tool_call["function"]["arguments"])

            yield StreamEvent(event_type="tool_call", data={
                "tool": tool_name,
                "args": tool_args,
                "call_id": tool_call.get("id"),
            })

            # Execute with sandbox (P05-T5)
            try:
                result = await tool_executor(tool_name, tool_args)
                result_str = json.dumps(result) if isinstance(result, dict) else str(result)
            except Exception as e:
                result_str = f"Error: {str(e)}"
                logger.error("Tool %s failed: %s", tool_name, e)

            state.total_tool_calls += 1

            yield StreamEvent(event_type="tool_result", data={
                "tool": tool_name,
                "result": result_str[:500],  # Truncate for streaming
                "call_id": tool_call.get("id"),
            })

            # Add tool result to history
            state.messages.append({
                "role": "tool",
                "tool_call_id": tool_call.get("id"),
                "content": result_str,
            })

    # Step 4: Completion verification (optional)
    if verifier and state.iteration > 1:
        yield StreamEvent(event_type="thinking", data={"step": "verification"})
        verification = await verifier.verify(
            original_message=user_message,
            conversation=state.messages,
            llm_chat=llm_chat,
            model=model,
        )
        yield StreamEvent(event_type="thinking", data={
            "step": "verification_result",
            "confidence": verification.confidence,
            "complete": verification.is_complete,
        })

    # Step 5: Publish lifecycle event
    if event_bus:
        from app.core.events import Event, EventType
        await event_bus.publish(Event(
            type=EventType.TASK_COMPLETED,
            source="agent_loop",
            data={
                "iterations": state.iteration,
                "tool_calls": state.total_tool_calls,
                "tokens": state.total_tokens_used,
                "compacted": state.compacted,
                "duration": time.time() - state.start_time,
            },
            user_id=user_id,
        ))

    yield StreamEvent(event_type="done", data={
        "iterations": state.iteration,
        "tool_calls": state.total_tool_calls,
        "duration": time.time() - state.start_time,
    })
```

### Task 2: Context Compaction Service

**Objective:** Implement auto-compaction at 85% context window with structured Goal/Done/State/Pending summaries.

**Constitution:** Section 10.4 — Context is managed. Auto-compaction preserves task state across long conversations.

**File:** `backend/app/agents/compactor.py` (enhance existing)

**Architecture:**

```python
"""Context compaction service.

Triggers: when token count exceeds 85% of context window.
Output: Structured summary with Goal/Done/State/Pending sections.
Preservation: Current user message is never compacted.
"""

import logging
from typing import Optional, Any
from .token_counter import TokenCounter

logger = logging.getLogger(__name__)

COMPACTION_PROMPT = (
    "You are a context compaction assistant. Your job is to summarize the following "
    "conversation between a user and an AI assistant. Produce a concise, structured summary "
    "with these sections:\n\n"
    "GOAL: What the user originally wanted\n"
    "DONE: What has been accomplished so far\n"
    "STATE: Current state of the task\n"
    "PENDING: What remains to be done\n\n"
    "Be concise. Preserve all important facts, file paths, errors, and decisions. "
    "Drop pleasantries and redundant exchanges. "
    "Keep the summary under 500 tokens."
)


@dataclass
class CompactionResult:
    """Result of a compaction operation."""
    summary: str
    tokens_before: int
    tokens_after: int
    reduction_ratio: float  # (before - after) / before
    sections: dict[str, str]  # Parsed GOAL/DONE/STATE/PENDING


class ContextCompactor:
    """Auto-compaction at 85% context window utilization.

    Design:
    - Triggers at 85% of context window (configurable)
    - Produces structured Goal/Done/State/Pending summary
    - Never compacts the current user message
    - Falls back to simple truncation if LLM unavailable
    - Tracks compaction metrics for observability
    """

    THRESHOLD = 0.85  # 85% context window utilization

    def __init__(
        self,
        token_counter: TokenCounter,
        llm_chat: Any = None,
        threshold: float = 0.85,
    ):
        self.token_counter = token_counter
        self.llm_chat = llm_chat
        self.threshold = threshold
        self._compaction_count = 0
        self._total_tokens_saved = 0

    async def should_compact(self, messages: list[dict]) -> bool:
        """Check if compaction is needed."""
        token_count = await self.token_counter.count_tokens(messages)
        max_tokens = self.token_counter.max_context_tokens
        return (token_count / max_tokens) >= self.threshold

    async def compact(
        self,
        messages: list[dict],
        llm_chat: Any = None,
        model: str = "default",
    ) -> str:
        """Compact conversation history into a structured summary.

        Args:
            messages: Full conversation history (excluding current user message)
            llm_chat: Optional LLM chat function (overrides instance default)
            model: Model to use for compaction

        Returns:
            Structured summary string
        """
        chat_fn = llm_chat or self.llm_chat

        tokens_before = await self.token_counter.count_tokens(messages)

        if chat_fn is None:
            summary = self._fallback_compact(messages)
        else:
            try:
                history_text = self._format_history(messages)
                response = await chat_fn(
                    messages=[
                        {"role": "system", "content": COMPACTION_PROMPT},
                        {"role": "user", "content": f"Summarize this conversation:\n\n{history_text}"},
                    ],
                    model=model,
                )
                summary = response.get("content", "") if isinstance(response, dict) else str(response)
            except Exception as e:
                logger.error("LLM compaction failed, using fallback: %s", e)
                summary = self._fallback_compact(messages)

        tokens_after = await self.token_counter.count_tokens([
            {"role": "system", "content": summary}
        ])

        self._compaction_count += 1
        self._total_tokens_saved += (tokens_before - tokens_after)

        result = CompactionResult(
            summary=summary,
            tokens_before=tokens_before,
            tokens_after=tokens_after,
            reduction_ratio=(tokens_before - tokens_after) / max(tokens_before, 1),
            sections=self._parse_sections(summary),
        )

        logger.info(
            "Context compacted: %d → %d tokens (%.1f%% reduction)",
            tokens_before, tokens_after, result.reduction_ratio * 100,
        )

        return summary

    def _format_history(self, messages: list[dict]) -> str:
        """Format messages for compaction input."""
        lines = []
        for msg in messages:
            role = msg.get("role", "unknown")
            content = msg.get("content", "")
            if isinstance(content, str):
                lines.append(f"[{role}]: {content[:1000]}")
            elif isinstance(content, list):
                # Handle multi-part messages
                text_parts = [p.get("text", "") for p in content if isinstance(p, dict)]
                lines.append(f"[{role}]: {' '.join(text_parts)[:1000]}")
        return "\n".join(lines)

    def _fallback_compact(self, messages: list[dict]) -> str:
        """Simple truncation fallback when LLM is unavailable."""
        recent = messages[-10:]  # Keep last 10 messages
        return (
            "[Context compacted — LLM unavailable, using truncation]\n"
            + "\n".join(
                f"[{m.get('role', '?')}]: {str(m.get('content', ''))[:200]}"
                for m in recent
            )
        )

    def _parse_sections(self, summary: str) -> dict[str, str]:
        """Parse Goal/Done/State/Pending sections from summary."""
        sections = {"goal": "", "done": "", "state": "", "pending": ""}
        current_section = None

        for line in summary.split("\n"):
            line_upper = line.upper().strip()
            if line_upper.startswith("GOAL:"):
                current_section = "goal"
                sections["goal"] = line.split(":", 1)[1].strip()
            elif line_upper.startswith("DONE:"):
                current_section = "done"
                sections["done"] = line.split(":", 1)[1].strip()
            elif line_upper.startswith("STATE:"):
                current_section = "state"
                sections["state"] = line.split(":", 1)[1].strip()
            elif line_upper.startswith("PENDING:"):
                current_section = "pending"
                sections["pending"] = line.split(":", 1)[1].strip()
            elif current_section and line.strip():
                sections[current_section] += " " + line.strip()

        return sections

    def get_stats(self) -> dict:
        """Return compaction statistics."""
        return {
            "compaction_count": self._compaction_count,
            "total_tokens_saved": self._total_tokens_saved,
            "threshold": self.threshold,
        }
```

### Task 3: Tool Schemas (OpenAI-Compatible)

**File:** `backend/app/agents/tools/schemas.py` (enhance existing)

**Architecture:**

```python
"""Tool schema generation and management.

Generates OpenAI-compatible function tool schemas from:
1. @tool decorator with explicit schema
2. Python type hints + docstrings (auto-generation)
3. Manual schema definitions for complex tools

Format: OpenAI function-calling schema (compatible with most LLMs).
"""

import inspect
import json
from typing import Any, Callable, Optional
from dataclasses import dataclass


@dataclass
class ToolSchema:
    """OpenAI-compatible tool schema."""
    name: str
    description: str
    parameters: dict  # JSON Schema format
    required: list[str]
    category: str = "general"  # For domain mapping (P05-T3)

    def to_openai_format(self) -> dict:
        """Convert to OpenAI function-calling format."""
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": {
                    "type": "object",
                    "properties": self.parameters,
                    "required": self.required,
                },
            },
        }


class ToolSchemaGenerator:
    """Generate OpenAI-compatible schemas for all registered tools."""

    # Python type to JSON Schema type mapping
    TYPE_MAP = {
        str: "string",
        int: "integer",
        float: "number",
        bool: "boolean",
        list: "array",
        dict: "object",
    }

    def from_function(self, func: Callable, name: Optional[str] = None) -> ToolSchema:
        """Generate schema from a Python function's type hints and docstring."""
        sig = inspect.signature(func)
        doc = inspect.getdoc(func) or ""
        func_name = name or func.__name__

        properties = {}
        required = []

        for param_name, param in sig.parameters.items():
            if param_name in ("self", "cls"):
                continue

            param_type = self.TYPE_MAP.get(param.annotation, "string")
            param_desc = self._extract_param_doc(doc, param_name)

            prop: dict[str, Any] = {"type": param_type}
            if param_desc:
                prop["description"] = param_desc
            if param.default is not inspect.Parameter.empty:
                prop["default"] = param.default
            else:
                required.append(param_name)

            properties[param_name] = prop

        return ToolSchema(
            name=func_name,
            description=self._extract_description(doc),
            parameters=properties,
            required=required,
        )

    def from_dict(self, schema_dict: dict) -> ToolSchema:
        """Create schema from manual dictionary definition."""
        func = schema_dict.get("function", {})
        return ToolSchema(
            name=func.get("name", "unknown"),
            description=func.get("description", ""),
            parameters=func.get("parameters", {}).get("properties", {}),
            required=func.get("parameters", {}).get("required", []),
        )

    def _extract_description(self, docstring: str) -> str:
        """Extract first line of docstring as description."""
        if not docstring:
            return ""
        lines = docstring.strip().split("\n")
        return lines[0].strip()

    def _extract_param_doc(self, docstring: str, param_name: str) -> str:
        """Extract parameter description from docstring."""
        for line in docstring.split("\n"):
            if f"Args:" in line or f"Parameters:" in line:
                continue
            if f"{param_name}:" in line or f"{param_name} (" in line:
                return line.split(":", 1)[-1].strip() if ":" in line else ""
        return ""
```

### Task 4: Intent Classification

**Objective:** Classify user messages before entering the agent loop: casual (skip loop), greeting (simple response), simple (single-turn), complex (full agent loop).

**Constitution:** Section 10.6 — Intent is classified. The agent knows what kind of request it's handling before committing resources.

**File:** `backend/app/agents/intent.py` (new or enhance existing)

**Architecture:**

```python
"""Intent classification for user messages.

Classifies messages into:
- casual: Small talk, simple questions (skip agent loop, use direct LLM call)
- greeting: Hello, goodbye (simple response, no tools needed)
- simple: Single-step request (one tool call max)
- complex: Multi-step request (full agent loop)
- continuation: Continuing a previous task (resume context)

Classification is fast (< 50ms) using pattern matching + lightweight LLM call.
"""

import re
import logging
from typing import Optional
from dataclasses import dataclass

logger = logging.getLogger(__name__)


@dataclass
class IntentClassification:
    """Result of intent classification."""
    intent: str  # "casual", "greeting", "simple", "complex", "continuation"
    confidence: float  # 0.0 - 1.0
    reasoning: str  # Why this classification was chosen


# Pattern-based pre-classification (fast, no LLM call)
GREETING_PATTERNS = [
    r"^(hi|hello|hey|howdy|greetings|good\s*(morning|afternoon|evening))\b",
    r"^(bye|goodbye|see\s*ya|talk\s*later|good\s*night)\b",
    r"^(thanks|thank\s*you|thx|ty)\b",
]

CASUAL_PATTERNS = [
    r"^(how\s*are\s*you|what'?s\s*up|how'?s\s*it\s*going)\b",
    r"^(what\s*do\s*you\s*think|in\s*your\s*opinion)\b",
    r"^(tell\s*me\s*a\s*joke|something\s*funny)\b",
    r"^(what'?s\s+the\s+meaning\s+of\s+life)\b",
]

COMPLEX_INDICATORS = [
    r"\b(implement|build|create|refactor|migrate|deploy)\b",
    r"\b(multiple|several|all|every|each)\b",
    r"\b(step\s*by\s*step|plan|strategy|architecture)\b",
    r"\b(analyze|compare|evaluate|assess)\b.*\b(and|then|also)\b",
]


class IntentClassifier:
    """Classify user intent before entering the agent loop.

    Two-phase classification:
    1. Pattern matching (instant, ~0ms)
    2. LLM classification (if patterns inconclusive, ~100ms)
    """

    def __init__(self, llm_chat=None):
        self.llm_chat = llm_chat

    async def classify(
        self,
        message: str,
        conversation_history: list[dict],
    ) -> str:
        """Classify user message intent. Returns intent string."""
        # Phase 1: Pattern matching
        intent = self._pattern_classify(message)
        if intent:
            return intent

        # Phase 2: Check for continuation
        if self._is_continuation(message, conversation_history):
            return "continuation"

        # Phase 3: LLM classification (if available)
        if self.llm_chat:
            intent = await self._llm_classify(message, conversation_history)
            if intent:
                return intent

        # Default: treat as complex (full agent loop)
        return "complex"

    def _pattern_classify(self, message: str) -> Optional[str]:
        """Fast pattern-based classification."""
        msg_lower = message.strip().lower()

        for pattern in GREETING_PATTERNS:
            if re.match(pattern, msg_lower):
                return "greeting"

        for pattern in CASUAL_PATTERNS:
            if re.match(pattern, msg_lower):
                return "casual"

        for pattern in COMPLEX_INDICATORS:
            if re.search(pattern, msg_lower):
                return "complex"

        return None

    def _is_continuation(self, message: str, history: list[dict]) -> bool:
        """Check if message is continuing a previous task."""
        if not history:
            return False

        # Check for pronouns referencing previous context
        continuation_words = ["it", "that", "this", "those", "them", "the same", "also", "and then"]
        msg_lower = message.lower()
        return any(word in msg_lower for word in continuation_words) and len(history) > 2

    async def _llm_classify(self, message: str, history: list[dict]) -> Optional[str]:
        """LLM-based classification for ambiguous messages."""
        try:
            # Use a very small/fast model for classification
            prompt = (
                "Classify this user message into exactly one category:\n"
                "- casual: small talk, opinions, jokes\n"
                "- simple: single-step request answerable in one response\n"
                "- complex: multi-step task requiring tools or planning\n\n"
                f"Message: {message}\n\n"
                "Category:"
            )
            response = await self.llm_chat(
                messages=[{"role": "user", "content": prompt}],
                max_tokens=10,
                temperature=0.0,
            )
            content = response.get("content", "").strip().lower() if isinstance(response, dict) else str(response).strip().lower()
            for intent in ["casual", "simple", "complex"]:
                if intent in content:
                    return intent
        except Exception as e:
            logger.warning("LLM intent classification failed: %s", e)
        return None

    async def handle_casual(self, message: str) -> str:
        """Handle casual message with direct LLM call (no tools)."""
        if self.llm_chat:
            response = await self.llm_chat(
                messages=[{"role": "user", "content": message}],
                max_tokens=200,
            )
            return response.get("content", "I'm not sure how to respond to that.") if isinstance(response, dict) else str(response)
        return "I'm here to help with tasks and questions. What can I do for you?"

    async def handle_greeting(self, message: str) -> str:
        """Handle greeting with a friendly response."""
        msg_lower = message.lower()
        if "bye" in msg_lower or "goodbye" in msg_lower:
            return "Goodbye! Feel free to come back anytime."
        if "good morning" in msg_lower:
            return "Good morning! How can I help you today?"
        if "good afternoon" in msg_lower:
            return "Good afternoon! What can I help you with?"
        if "good evening" in msg_lower:
            return "Good evening! How can I assist you?"
        return "Hello! How can I help you today?"
```

### Task 5: Stall/Runaway Detection

**Objective:** Implement loop-breaker that detects repeated identical tool calls, excessive iterations, and timeout conditions. Forces an answer instead of looping forever.

**Constitution:** Section 10.7 — Stalls are detected. The loop-breaker detects repeated identical tool calls and forces an answer. No more agent stuck in infinite loops. Stall detection is the primary loop control, not max iterations.

**File:** `backend/app/agents/stall.py` (enhance existing)

**Architecture:**

```python
"""Stall and runaway detection for the agent loop.

Detects:
1. Repeated identical tool calls (same tool + same args)
2. Repeated identical LLM responses (same content)
3. Excessive total iterations (hard limit)
4. Timeout (wall clock time exceeded)

When a stall is detected, the loop-breaker forces the LLM to produce a final answer.
"""

import time
import logging
from dataclasses import dataclass, field
from collections import Counter

logger = logging.getLogger(__name__)


@dataclass
class StallDetection:
    """Result of stall check."""
    is_stalled: bool
    reason: str
    stall_type: str = ""  # "repeated_tool", "repeated_response", "timeout", "max_iterations"
    iteration: int = 0
    details: dict = field(default_factory=dict)


class StallDetector:
    """Detect agent loop stalls and runaway conditions.

    Thresholds:
    - Repeated tool call: same tool+args 3 times in a row → stall
    - Repeated response: same LLM output 2 times in a row → stall
    - Timeout: > 300 seconds wall clock → stall
    - Max iterations: > 25 iterations → stall (hard limit)
    """

    REPEATED_TOOL_THRESHOLD = 3  # Same tool+args 3 times
    REPEATED_RESPONSE_THRESHOLD = 2  # Same response 2 times
    TIMEOUT_SECONDS = 300  # 5 minutes wall clock
    MAX_ITERATIONS = 25  # Hard limit

    def __init__(
        self,
        repeated_tool_threshold: int = 3,
        repeated_response_threshold: int = 2,
        timeout_seconds: int = 300,
        max_iterations: int = 25,
    ):
        self.repeated_tool_threshold = repeated_tool_threshold
        self.repeated_response_threshold = repeated_response_threshold
        self.timeout_seconds = timeout_seconds
        self.max_iterations = max_iterations
        self._tool_history: list[str] = []
        self._response_history: list[str] = []

    async def check(
        self,
        tool_calls: list[dict],
        history: list[dict],
        iteration: int,
        start_time: float = None,
    ) -> StallDetection:
        """Check for stall conditions. Called after each agent loop iteration."""

        # Check 1: Max iterations
        if iteration >= self.max_iterations:
            return StallDetection(
                is_stalled=True,
                reason=f"Reached maximum iterations ({self.max_iterations})",
                stall_type="max_iterations",
                iteration=iteration,
            )

        # Check 2: Timeout
        if start_time:
            elapsed = time.time() - start_time
            if elapsed > self.timeout_seconds:
                return StallDetection(
                    is_stalled=True,
                    reason=f"Timeout after {elapsed:.0f}s (limit: {self.timeout_seconds}s)",
                    stall_type="timeout",
                    iteration=iteration,
                    details={"elapsed_seconds": elapsed},
                )

        # Check 3: Repeated identical tool calls
        for tool_call in tool_calls:
            call_key = self._tool_call_key(tool_call)
            self._tool_history.append(call_key)

            # Check last N calls
            if len(self._tool_history) >= self.repeated_tool_threshold:
                recent = self._tool_history[-self.repeated_tool_threshold:]
                if len(set(recent)) == 1:
                    return StallDetection(
                        is_stalled=True,
                        reason=f"Repeated identical tool call {recent[0]} x{self.repeated_tool_threshold}",
                        stall_type="repeated_tool",
                        iteration=iteration,
                        details={"tool_key": recent[0], "count": self.repeated_tool_threshold},
                    )

        # Check 4: Repeated identical LLM responses
        if history and history[-1].get("role") == "assistant":
            last_content = history[-1].get("content", "")
            if last_content:
                self._response_history.append(last_content)
                if len(self._response_history) >= self.repeated_response_threshold:
                    recent_responses = self._response_history[-self.repeated_response_threshold:]
                    if len(set(recent_responses)) == 1:
                        return StallDetection(
                            is_stalled=True,
                            reason=f"Repeated identical LLM response x{self.repeated_response_threshold}",
                            stall_type="repeated_response",
                            iteration=iteration,
                        )

        return StallDetection(is_stalled=False, reason="", iteration=iteration)

    def _tool_call_key(self, tool_call: dict) -> str:
        """Generate a unique key for a tool call."""
        name = tool_call.get("function", {}).get("name", "")
        args = tool_call.get("function", {}).get("arguments", "")
        return f"{name}:{args}"

    def reset(self):
        """Reset history for a new conversation."""
        self._tool_history.clear()
        self._response_history.clear()
```

### Task 6: Completion Verification

**Architecture:**

```python
"""Completion verification via fresh-context subagent.

Design:
- Fresh context = separate LLM call with NO prior conversation history
- Only receives: original user message + final agent response
- Produces: confidence score (0.0-1.0) + completion assessment
- Does NOT affect the main agent loop output
- Used for observability and optional re-run trigger
"""

import logging
from dataclasses import dataclass
from typing import Any

logger = logging.getLogger(__name__)


@dataclass
class VerificationResult:
    """Result of completion verification."""
    is_complete: bool
    confidence: float  # 0.0 - 1.0
    assessment: str  # Human-readable assessment
    missing: list[str]  # What's missing (if incomplete)


VERIFICATION_PROMPT = (
    "You are a task completion verifier. You will receive:\n"
    "1. The original user message (what they asked for)\n"
    "2. The assistant's final response (what was delivered)\n\n"
    "Assess whether the task was completed satisfactorily.\n"
    "Respond in JSON format:\n"
    '{{"confidence": 0.0-1.0, "complete": true/false, '
    '"assessment": "brief explanation", "missing": ["list of missing items"]}}\n\n'
    "Be honest. A confidence of 0.7 means there's a 30% chance something was missed."
)


class CompletionVerifier:
    """Fresh-context completion verifier.

    Does NOT use the conversation history. Only sees:
    1. Original user message
    2. Final agent response

    This prevents the verifier from being biased by the agent's reasoning.
    """

    def __init__(self, llm_chat: Any = None):
        self.llm_chat = llm_chat
        self._verification_count = 0
        self._total_confidence = 0.0

    async def verify(
        self,
        original_message: str,
        conversation: list[dict],
        llm_chat: Any = None,
        model: str = "default",
    ) -> VerificationResult:
        """Verify task completion with fresh context.

        Args:
            original_message: The user's original request
            conversation: Full conversation (used only to extract final response)
            llm_chat: LLM chat function
            model: Model to use for verification

        Returns:
            VerificationResult with confidence and assessment
        """
        chat_fn = llm_chat or self.llm_chat
        if not chat_fn:
            return VerificationResult(
                is_complete=True,
                confidence=0.5,
                assessment="Verification skipped (no LLM available)",
                missing=[],
            )

        # Extract the final assistant response
        final_response = ""
        for msg in reversed(conversation):
            if msg.get("role") == "assistant" and msg.get("content"):
                final_response = msg["content"]
                break

        if not final_response:
            return VerificationResult(
                is_complete=False,
                confidence=0.0,
                assessment="No assistant response found",
                missing=["No response"],
            )

        try:
            prompt = (
                f"{VERIFICATION_PROMPT}\n\n"
                f"--- USER MESSAGE ---\n{original_message}\n\n"
                f"--- ASSISTANT RESPONSE ---\n{final_response[:2000]}"
            )
            response = await chat_fn(
                messages=[{"role": "user", "content": prompt}],
                model=model,
                temperature=0.0,
                max_tokens=300,
            )

            content = response.get("content", "") if isinstance(response, dict) else str(response)

            # Parse JSON response
            import json
            result = json.loads(content)

            self._verification_count += 1
            self._total_confidence += result.get("confidence", 0.5)

            return VerificationResult(
                is_complete=result.get("complete", False),
                confidence=result.get("confidence", 0.5),
                assessment=result.get("assessment", ""),
                missing=result.get("missing", []),
            )

        except Exception as e:
            logger.error("Verification failed: %s", e)
            return VerificationResult(
                is_complete=True,
                confidence=0.3,
                assessment=f"Verification error: {str(e)}",
                missing=[],
            )

    def get_stats(self) -> dict:
        """Return verification statistics."""
        return {
            "verification_count": self._verification_count,
            "average_confidence": (
                self._total_confidence / self._verification_count
                if self._verification_count > 0 else 0.0
            ),
        }
```

### Task 7: Detached Agent Runs

**Objective:** Implement server-side persistence for agent runs with replay buffer, surviving daemon restarts and supporting multi-device access.

**Constitution:** Section 10.9 — Runs are persistent. Agent runs survive daemon restart. Server-side persistence with replay buffer. PID tracking for long-running tasks. Orphan detection and cleanup.

```python
"""Detached agent runs with server-side persistence.

Design (reference architecture pattern):
- Agent runs are stored server-side, not client-side
- SSE clients subscribe to a run by ID
- When client disconnects (tab close, refresh), the run continues
- Reconnecting with the same run_id resumes the stream
- Replay buffer stores all StreamEvents for replay
- 180-second grace period before run is considered abandoned
"""

import json
import time
import logging
import uuid
from dataclasses import dataclass, field
from typing import Optional, AsyncGenerator
from enum import Enum

logger = logging.getLogger(__name__)


class RunStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    PAUSED = "paused"  # Client disconnected, waiting for reconnect
    COMPLETED = "completed"
    FAILED = "failed"
    ABANDONED = "abandoned"  # No reconnect within grace period


@dataclass
class AgentRun:
    """Persistent agent run state."""
    run_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    user_id: int = 0
    status: RunStatus = RunStatus.PENDING
    conversation_id: Optional[int] = None
    user_message: str = ""
    created_at: float = field(default_factory=time.time)
    started_at: Optional[float] = None
    completed_at: Optional[float] = None
    last_event_at: Optional[float] = None
    subscribers: set = field(default_factory=set)  # SSE client IDs
    event_buffer: list = field(default_factory=list)  # Replay buffer
    error: Optional[str] = None
    result: Optional[str] = None
    iteration: int = 0
    tool_calls: int = 0

    GRACE_PERIOD = 180  # seconds before run is abandoned

    @property
    def is_alive(self) -> bool:
        return self.status in (RunStatus.PENDING, RunStatus.RUNNING, RunStatus.PAUSED)

    @property
    def should_abandon(self) -> bool:
        if not self.last_event_at:
            return False
        return (
            self.status == RunStatus.PAUSED
            and (time.time() - self.last_event_at) > self.GRACE_PERIOD
        )


class RunStore:
    """In-memory store for agent runs. Extend with PostgreSQL for production."""

    def __init__(self):
        self._runs: dict[str, AgentRun] = {}

    async def create_run(self, user_id: int, message: str, conversation_id: int = None) -> AgentRun:
        run = AgentRun(
            user_id=user_id,
            user_message=message,
            conversation_id=conversation_id,
        )
        self._runs[run.run_id] = run
        return run

    async def get_run(self, run_id: str) -> Optional[AgentRun]:
        return self._runs.get(run_id)

    async def update_run(self, run: AgentRun):
        self._runs[run.run_id] = run

    async def list_user_runs(self, user_id: int) -> list[AgentRun]:
        return [r for r in self._runs.values() if r.user_id == user_id]

    async def cleanup_abandoned(self) -> int:
        """Remove runs that exceeded grace period. Returns count cleaned."""
        abandoned = []
        for run_id, run in self._runs.items():
            if run.should_abandon:
                run.status = RunStatus.ABANDONED
                abandoned.append(run_id)
        for run_id in abandoned:
            del self._runs[run_id]
        return len(abandoned)


class DetachedRunManager:
    """Manage detached agent runs with replay and subscriber fan-out.

    Flow:
    1. Client sends message → create_run() → start agent loop
    2. Agent loop yields StreamEvents → store in replay buffer → fan out to subscribers
    3. Client disconnects → subscriber removed, run continues (status: PAUSED)
    4. Client reconnects with run_id → replay buffered events → resume subscription
    5. Run completes → final events buffered → subscribers notified
    6. Grace period expires → run cleaned up
    """

    def __init__(self, run_store: RunStore):
        self.run_store = run_store
        self._active_loops: dict[str, asyncio.Task] = {}

    async def start_run(
        self,
        run: AgentRun,
        agent_loop_fn,  # The agent_loop async generator
        **kwargs,
    ) -> str:
        """Start an agent run. Returns run_id for client tracking."""
        run.status = RunStatus.RUNNING
        run.started_at = time.time()

        # Start the agent loop as a background task
        task = asyncio.create_task(
            self._run_loop(run, agent_loop_fn, **kwargs)
        )
        self._active_loops[run.run_id] = task

        return run.run_id

    async def _run_loop(self, run: AgentRun, agent_loop_fn, **kwargs):
        """Execute agent loop and fan out events to subscribers."""
        try:
            async for event in agent_loop_fn(**kwargs):
                # Store in replay buffer
                run.event_buffer.append({
                    "event_type": event.event_type,
                    "data": event.data,
                    "timestamp": event.timestamp,
                })
                run.last_event_at = time.time()

                # Fan out to active subscribers
                for subscriber_id in list(run.subscribers):
                    await self._send_to_subscriber(subscriber_id, event)

                # Update run state
                if event.event_type == "done":
                    run.status = RunStatus.COMPLETED
                    run.completed_at = time.time()
                elif event.event_type == "error":
                    run.status = RunStatus.FAILED
                    run.error = event.data.get("message", "Unknown error")

        except asyncio.CancelledError:
            run.status = RunStatus.PAUSED
            logger.info("Run %s paused (client disconnected)", run.run_id)
        except Exception as e:
            run.status = RunStatus.FAILED
            run.error = str(e)
            logger.error("Run %s failed: %s", run.run_id, e)

    async def subscribe(self, run_id: str, subscriber_id: str) -> AsyncGenerator:

            raise ValueError(f"Run {run_id} not found")
```

### Task 8: Per-Turn Tool Policy

**Objective:** Implement dynamic tool composition per turn: allow/deny/ask (require approval) for each tool, plan mode restriction, and MCP tool gating.

**Constitution:** Section 10.3 — Tool policy is per-turn. Every turn, the agent can allow, deny, or require approval for specific tools. This replaces HMAC approval tokens. Policy is composable and configurable.

**File:** `backend/app/agents/tools/policy.py` (enhance existing)

**Architecture:**

```python
"""Per-turn tool policy engine.

Design:
- Policy is evaluated fresh each iteration of the agent loop
- Three modes per tool: "allow", "deny", "ask" (require user approval)
- Plan mode: only read-only tools allowed (no writes, no executions)
- MCP tools gated separately from built-in tools
- Policy composable: default policy + per-turn overrides + context overrides
"""

import logging
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional

logger = logging.getLogger(__name__)


class ToolAccess(str, Enum):
    ALLOW = "allow"
    DENY = "deny"
    ASK = "ask"  # Require user approval before execution


@dataclass
class ToolPolicyRule:
    """Single policy rule for a tool."""
    tool_name: str
    access: ToolAccess
    reason: str = ""
    requires_approval: bool = False


@dataclass
class PolicyContext:
    """Context for policy evaluation."""
    iteration: int
    is_plan_mode: bool = False
    user_role: str = "user"  # "user", "admin", "system"
    conversation_history: list = field(default_factory=list)
    mcp_tools: list = field(default_factory=list)  # MCP tool names


class ToolPolicy:
    """Per-turn tool policy engine.

    Default policy:
    - Built-in read tools: ALLOW (search_memory, read_file, etc.)
    - Built-in write tools: ALLOW (create_memory, write_file, etc.)
    - Dangerous tools: ASK (execute_command, delete_file, etc.)
    - MCP tools: DENY by default, ALLOW if explicitly enabled

    Plan mode:
    - Only read-only tools allowed
    - Write tools DENIED
    - Execute tools DENIED
    """

    # Default tool access rules
    DEFAULT_RULES: dict[str, ToolAccess] = {
        "search_memory": ToolAccess.ALLOW,
        "create_memory": ToolAccess.ALLOW,
        "read_file": ToolAccess.ALLOW,
        "write_file": ToolAccess.ALLOW,
        "plan_task": ToolAccess.ALLOW,
        "web_search": ToolAccess.ALLOW,
        "execute_command": ToolAccess.ASK,
        "delete_file": ToolAccess.ASK,
    }

    # Tools that are read-only (allowed in plan mode)
    READ_ONLY_TOOLS = {"search_memory", "read_file", "web_search", "plan_task"}

    # Tools that are always denied in plan mode
    PLAN_MODE_DENIED = {"write_file", "execute_command", "delete_file", "create_memory"}

    def __init__(self):
        self._custom_rules: dict[str, ToolPolicyRule] = {}
        self._approval_pending: dict[str, bool] = {}

    async def compose(
        self,
        tools: list[dict],
        conversation_history: list[dict],
        iteration: int,
        context: Optional[PolicyContext] = None,
    ) -> list[dict]:
        """Compose the allowed tool list for this turn.

        Returns filtered list of tools the agent can use.
        """
        ctx = context or PolicyContext(iteration=iteration)
        allowed_tools = []

        for tool in tools:
            tool_name = tool.get("function", {}).get("name", "")
            access = self._evaluate_tool(tool_name, ctx)

            if access == ToolAccess.ALLOW:
                allowed_tools.append(tool)
            elif access == ToolAccess.ASK:
                # Check if already approved
                if self._approval_pending.get(tool_name, False):
                    allowed_tools.append(tool)
                else:
                    logger.info("Tool %s requires approval (not yet approved)", tool_name)
            else:  # DENY
                logger.debug("Tool %s denied by policy", tool_name)

        return allowed_tools

    def _evaluate_tool(self, tool_name: str, context: PolicyContext) -> ToolAccess:
        """Evaluate access for a single tool."""
        # Check custom rules first
        if tool_name in self._custom_rules:
            return self._custom_rules[tool_name].access

        # Plan mode override
        if context.is_plan_mode:
            if tool_name in self.PLAN_MODE_DENIED:
                return ToolAccess.DENY
            if tool_name in self.READ_ONLY_TOOLS:
                return ToolAccess.ALLOW
            return ToolAccess.DENY  # Default deny in plan mode

        # MCP tools: deny by default
        if tool_name in context.mcp_tools:
            return ToolAccess.DENY  # MCP tools must be explicitly enabled

        # Default rules
        return self.DEFAULT_RULES.get(tool_name, ToolAccess.ALLOW)

    def set_rule(self, tool_name: str, access: ToolAccess, reason: str = ""):
        """Set a custom rule for a tool."""
        self._custom_rules[tool_name] = ToolPolicyRule(
            tool_name=tool_name,
            access=access,
            reason=reason,
        )

    def approve_tool(self, tool_name: str):
        """Approve a tool that requires approval."""
        self._approval_pending[tool_name] = True

    def revoke_approval(self, tool_name: str):
        """Revoke approval for a tool."""
        self._approval_pending.pop(tool_name, None)

    def reset(self):
        """Reset all custom rules and approvals."""
        self._custom_rules.clear()
        self._approval_pending.clear()
```

- MCP tools denied by default
