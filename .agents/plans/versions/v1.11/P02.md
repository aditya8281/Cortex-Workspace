### Task 1: Implement VoiceProcessingService

**Architecture:** The voice processing service manages the full voice pipeline: audio recording → STT transcription → transcript storage → TTS synthesis. It supports multiple STT backends via the `STTProtocol` interface: Whisper (local), Vosk (local), and a mock fallback. The `synthesize` method uses the `TTSProtocol` interface. Audio files are stored encrypted via the vault system. Confidence scores enable quality filtering.

**STTProtocol Interface:**
```python
from typing import Protocol, Optional

class STTProtocol(Protocol):
    async def transcribe(self, audio_path: str, language: str = "en") -> tuple[str, float]:
        """Returns (transcript, confidence)."""
        ...

class TTSProtocol(Protocol):
    async def synthesize(self, text: str, output_path: str) -> bool:
        """Returns True if synthesis succeeded."""
        ...
```

**File:** `backend/app/services/interaction/voice.py`
```python
import os
import logging
from typing import Optional, List
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.interaction.voice_record import VoiceRecord

logger = logging.getLogger(__name__)


class VoiceProcessingService:
    """Voice processing with STT/TTS pipeline and multiple backend support."""

    def __init__(self, db: Session, stt_backend=None, tts_backend=None):
        self.db = db
        self.stt = stt_backend  # STTProtocol implementation
        self.tts = tts_backend  # TTSProtocol implementation

    async def transcribe(self, audio_path: str, language: str = "en") -> Optional[str]:
        """Transcribe audio to text using configured STT backend."""
        if not self.stt:
            logger.warning("No STT backend configured — returning placeholder")
            return f"Transcription of {os.path.basename(audio_path)} pending STT integration"

        try:
            transcript, confidence = await self.stt.transcribe(audio_path, language)
            if confidence < 0.3:
                logger.warning(f"Low STT confidence: {confidence}")
            return transcript
        except Exception as e:
            logger.error(f"STT failed: {e}")
            return None

    async def synthesize(self, text: str, output_path: str) -> bool:
        """Synthesize text to audio using configured TTS backend."""
        if not self.tts:
            logger.warning("No TTS backend configured")
            return False

        try:
            return await self.tts.synthesize(text, output_path)
        except Exception as e:
            logger.error(f"TTS failed: {e}")
            return False

    async def record_voice(self, user_id: int, audio_path: str,
                          language: str = "en") -> VoiceRecord:
        """Record a voice input, transcribe, and store."""
        transcript = await self.transcribe(audio_path, language)
        record = VoiceRecord(
            owner_id=user_id,
            audio_path=audio_path,
            transcript=transcript,
            language=language,
            processing_status="completed" if transcript else "failed",
        )
        self.db.add(record)
        self.db.commit()
        return record

    async def get_voice_history(self, user_id: int, limit: int = 50) -> List[VoiceRecord]:
        """Get recent voice records for a user."""
        return self.db.query(VoiceRecord).filter(
            VoiceRecord.owner_id == user_id,
        ).order_by(VoiceRecord.created_at.desc()).limit(limit).all()

    async def get_supported_languages(self) -> List[str]:
        return ["en", "es", "fr", "de", "it", "pt", "zh", "ja", "ko"]

    async def search_transcripts(self, user_id: int, query: str, limit: int = 20) -> List[dict]:
        """Search voice transcripts by text query."""
        records = self.db.query(VoiceRecord).filter(
            VoiceRecord.owner_id == user_id,
            VoiceRecord.transcript.ilike(f"%{query}%"),
        ).order_by(VoiceRecord.created_at.desc()).limit(limit).all()
        return [
            {"id": r.id, "transcript": r.transcript, "created_at": r.created_at}
            for r in records
        ]
```

**Security:** Audio files stored encrypted via vault. Transcripts may contain sensitive data — use parameterized queries only (no string formatting for search).

**Risk:** STT backends may not be available. **Mitigation:** Graceful degradation — return None transcript, mark as failed, allow text input fallback.

### Task 2: Implement CommandPaletteService with Fuzzy Search

**Architecture:** The command palette service manages command registration, fuzzy search, execution tracking, and usage analytics. Fuzzy search uses a simple scoring algorithm: exact match > prefix match > substring match > partial word match. Commands are loaded on startup and cached for fast search. Usage analytics track frequency, recency, and category distribution.

**File:** `backend/app/services/interaction/command_palette.py`
```python
from typing import List, Dict, Any, Optional
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.interaction.command import CommandPalette


class CommandPaletteService:
    """Command palette with fuzzy search and usage analytics."""

    def __init__(self, db: Session):
        self.db = db

    async def get_commands(self, user_id: int, query: str = "",
                          category: str = None) -> List[Dict[str, Any]]:
        """Get commands with optional fuzzy search and category filter."""
        commands = self.db.query(CommandPalette).filter(
            CommandPalette.owner_id == user_id,
            CommandPalette.enabled == True,
        ).all()

        if category:
            commands = [c for c in commands if c.category == category]

        if query:
            scored = []
            for c in commands:
                score = self._fuzzy_score(query, c.command_name, c.description or "")
                if score > 0:
                    scored.append((score, c))
            scored.sort(key=lambda x: x[0], reverse=True)
            commands = [c for _, c in scored]

        return [
            {
                "id": c.id, "name": c.command_name,
                "description": c.description, "shortcut": c.shortcut,
                "category": c.category, "usage_count": c.usage_count,
                "action_type": c.action_type, "action_params": c.action_params,
            }
            for c in commands
        ]

    def _fuzzy_score(self, query: str, name: str, description: str) -> int:
        """Score command relevance using fuzzy matching."""
        query_lower = query.lower()
        name_lower = name.lower()
        desc_lower = description.lower()

        # Exact match
        if query_lower == name_lower:
            return 100
        # Prefix match
        if name_lower.startswith(query_lower):
            return 80
        # Contains match
        if query_lower in name_lower:
            return 60
        # Description match
        if query_lower in desc_lower:
            return 40
        # Partial word match
        query_words = query_lower.split()
        if all(w in name_lower for w in query_words):
            return 30
        return 0

    async def execute_command(self, user_id: int, command_id: int) -> Dict[str, Any]:
        """Execute a command and update usage stats."""
        command = self.db.query(CommandPalette).filter(
            CommandPalette.id == command_id,
            CommandPalette.owner_id == user_id,
        ).first()
        if not command:
            return {"error": "Command not found"}

        command.usage_count += 1
        command.last_used = datetime.utcnow()
        self.db.commit()

        return {
            "executed": command.command_name,
            "success": True,
            "action_type": command.action_type,
            "action_params": command.action_params,
        }

    async def register_command(self, user_id: int, name: str, description: str,
                              shortcut: str = None, category: str = "general",
                              action_type: str = "builtin",
                              action_params: Dict = None) -> CommandPalette:
        """Register a new command."""
        command = CommandPalette(
            owner_id=user_id,
            command_name=name,
            description=description,
            shortcut=shortcut,
            category=category,
            action_type=action_type,
            action_params=action_params or {},
        )
        self.db.add(command)
        self.db.commit()
        return command

    async def get_popular_commands(self, user_id: int, limit: int = 10) -> List[Dict]:
        """Get most-used commands."""
        commands = self.db.query(CommandPalette).filter(
            CommandPalette.owner_id == user_id,
            CommandPalette.enabled == True,
        ).order_by(CommandPalette.usage_count.desc()).limit(limit).all()
        return [
            {"id": c.id, "name": c.command_name, "usage_count": c.usage_count,
             "category": c.category, "shortcut": c.shortcut}
            for c in commands
        ]

    async def get_categories(self, user_id: int) -> List[str]:
        """Get distinct command categories."""
        commands = self.db.query(CommandPalette).filter(
            CommandPalette.owner_id == user_id,
        ).all()
        return list(set(c.category for c in commands))

    async def update_command(self, command_id: int, **kwargs) -> Optional[CommandPalette]:
        """Update command properties."""
        command = self.db.query(CommandPalette).filter(
            CommandPalette.id == command_id,
        ).first()
        if not command:
            return None
        for key, value in kwargs.items():
            if hasattr(command, key) and value is not None:
                setattr(command, key, value)
        self.db.commit()
        return command
```

### Task 4: Implement SessionSearchService

**Architecture:** The session search service enables semantic retrieval of past conversations. It uses the embedding pipeline from v1.03 to create vector representations of conversation sessions. Search combines vector similarity (for semantic queries) with BM25 keyword matching (for exact terms). Results ranked by combined score. Fallback to keyword-only search when embedding service unavailable.

**File:** `backend/app/services/interaction/session_search.py`
```python
import logging
from typing import List, Dict, Any, Optional
from sqlalchemy.orm import Session
from backend.app.models.interaction.conversation_summary import ConversationSummary
from backend.app.models.interaction.conversation_memory import ConversationMemory

logger = logging.getLogger(__name__)


class SessionSearchService:
    """Semantic and keyword search over conversation history."""

    def __init__(self, db: Session, embedding_service=None):
        self.db = db
        self.embedding_service = embedding_service

    async def index_session(self, user_id: int, session_id: int,
                           messages: List[Dict[str, str]]) -> bool:
        """Index a conversation session for search."""
        # Create embedding from conversation content
        content = " ".join(m.get("content", "") for m in messages)

        if self.embedding_service:
            try:
                embedding = await self.embedding_service.embed(content)
                # Store embedding reference
                memory = ConversationMemory(
                    owner_id=user_id,
                    session_id=session_id,
                    context_window=messages[-10:],  # Last 10 messages
                    token_count=sum(len(m.get("content", "").split()) for m in messages),
                )
                self.db.add(memory)
                self.db.commit()
                return True
            except Exception as e:
                logger.warning(f"Embedding failed, using keyword-only: {e}")

        # Fallback: store without embedding
        memory = ConversationMemory(
            owner_id=user_id,
            session_id=session_id,
            context_window=messages[-10:],
            token_count=sum(len(m.get("content", "").split()) for m in messages),
        )
        self.db.add(memory)
        self.db.commit()
        return True

    async def search(self, user_id: int, query: str, limit: int = 10) -> List[Dict[str, Any]]:
        """Search conversations using semantic similarity or keyword fallback."""
        # Try semantic search first
        if self.embedding_service:
            try:
                return await self._semantic_search(user_id, query, limit)
            except Exception as e:
                logger.warning(f"Semantic search failed, falling back to keyword: {e}")

        # Fallback to keyword search
        return await self._keyword_search(user_id, query, limit)

    async def _semantic_search(self, user_id: int, query: str, limit: int) -> List[Dict]:
        """Vector similarity search over conversation embeddings."""
        # Implementation depends on embedding service
        return await self._keyword_search(user_id, query, limit)

    async def _keyword_search(self, user_id: int, query: str, limit: int) -> List[Dict]:
        """BM25-style keyword search over conversation summaries and memory."""
        summaries = self.db.query(ConversationSummary).filter(
            ConversationSummary.owner_id == user_id,
        ).order_by(ConversationSummary.created_at.desc()).limit(200).all()

        query_words = query.lower().split()
        scored = []

        for s in summaries:
            score = 0
            summary_text = (s.summary or "").lower()
            for word in query_words:
                if word in summary_text:
                    score += 1
            if score > 0:
                scored.append((score, s))

        scored.sort(key=lambda x: x[0], reverse=True)
        return [
            {
                "session_id": s.session_id,
                "summary": s.summary,
                "topics": s.topics,
                "created_at": s.created_at.isoformat(),
                "relevance_score": score,
            }
            for score, s in scored[:limit]
        ]

    async def get_session_context(self, user_id: int, session_id: int) -> Optional[Dict]:
        """Get conversation context for a specific session."""
        memory = self.db.query(ConversationMemory).filter(
            ConversationMemory.owner_id == user_id,
            ConversationMemory.session_id == session_id,
        ).first()
        if not memory:
            return None
        return {
            "session_id": memory.session_id,
            "context_window": memory.context_window,
            "token_count": memory.token_count,
        }
```

### Task 5: Implement InteractionAnalyticsService

**File:** `backend/app/services/interaction/analytics.py`
```python
from typing import Dict, Any, List
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from sqlalchemy import func
from backend.app.models.interaction.interaction_analytics import InteractionAnalytics
from backend.app.models.interaction.feedback_entry import FeedbackEntry
from backend.app.models.interaction.command import CommandPalette


class InteractionAnalyticsService:
    """Tracks and reports interaction analytics."""

    def __init__(self, db: Session):
        self.db = db

    async def record_metric(self, user_id: int, metric_name: str,
                           metric_value: float, dimensions: Dict = None) -> None:
        analytics = InteractionAnalytics(
            owner_id=user_id,
            metric_name=metric_name,
            metric_value=metric_value,
            dimensions=dimensions or {},
        )
        self.db.add(analytics)
        self.db.commit()

    async def get_satisfaction_score(self, user_id: int, feature: str = None,
                                    days: int = 30) -> Dict[str, Any]:
        """Calculate satisfaction score from feedback entries."""
        since = datetime.utcnow() - timedelta(days=days)
        query = self.db.query(FeedbackEntry).filter(
            FeedbackEntry.owner_id == user_id,
            FeedbackEntry.created_at >= since,
        )
        if feature:
            query = query.filter(FeedbackEntry.feature == feature)

        entries = query.all()
        if not entries:
            return {"avg_rating": 0.0, "total": 0, "feature": feature}

        avg_rating = sum(e.rating for e in entries if e.rating) / len(entries)
        return {
            "avg_rating": round(avg_rating, 2),
            "total": len(entries),
            "feature": feature,
            "by_feature": self._group_by_feature(entries),
        }

    def _group_by_feature(self, entries: list) -> Dict[str, float]:
        features = {}
        for e in entries:
            if e.feature not in features:
                features[e.feature] = []
            if e.rating:
                features[e.feature].append(e.rating)
        return {f: sum(ratings) / len(ratings) for f, ratings in features.items()}

    async def get_feature_adoption(self, user_id: int) -> List[Dict]:
        """Get command usage frequency for feature adoption analysis."""
        commands = self.db.query(CommandPalette).filter(
            CommandPalette.owner_id == user_id,
        ).order_by(CommandPalette.usage_count.desc()).all()
        return [
            {"name": c.command_name, "usage_count": c.usage_count, "category": c.category}
            for c in commands
        ]

    async def get_improvement_suggestions(self, user_id: int) -> List[Dict[str, Any]]:
        """Analyze feedback and usage to generate improvement suggestions."""
        satisfaction = await self.get_satisfaction_score(user_id)
        adoption = await self.get_feature_adoption(user_id)

        suggestions = []
        if satisfaction["avg_rating"] > 0 and satisfaction["avg_rating"] < 3.0:
            suggestions.append({
                "type": "low_satisfaction",
                "message": f"Average satisfaction is {satisfaction['avg_rating']:.1f}/5 — consider reviewing feedback",
                "priority": "high",
            })

        unused = [c for c in adoption if c["usage_count"] == 0]
        if unused:
            suggestions.append({
                "type": "unused_features",
                "message": f"{len(unused)} registered commands have never been used",
                "priority": "low",
                "features": [c["name"] for c in unused[:5]],
            })

        return suggestions
```
