### Task 1: Implement AmbientIntelligenceService

**Architecture:** The ambient intelligence service runs as a daemon background process, continuously monitoring user context at configurable intervals. It uses asyncio tasks for non-blocking operation. The service has lifecycle management: start → running → stopped. Health monitoring tracks cycle count and last activity time. CPU usage is capped via adaptive polling frequency.

**File:** `backend/app/services/interaction/ambient.py`
```python
import asyncio
import logging
from typing import Dict, Any, Callable, List

logger = logging.getLogger(__name__)


class AmbientIntelligenceService:
    """Background intelligence that monitors user context and surfaces insights."""

    def __init__(self):
        self._running = False
        self._task = None
        self._handlers: List[Callable] = []
        self._state: Dict[str, Any] = {
            "status": "idle",
            "cycles": 0,
            "last_cycle": None,
            "errors": 0,
        }
        self._poll_interval = 60  # seconds
        self._min_interval = 10
        self._max_interval = 300

    async def start(self):
        """Start the ambient intelligence loop."""
        if self._running:
            return
        self._running = True
        self._state["status"] = "active"
        self._task = asyncio.create_task(self._ambient_loop())
        logger.info("Ambient intelligence started")

    async def stop(self):
        """Stop the ambient intelligence loop gracefully."""
        self._running = False
        self._state["status"] = "stopping"
        if self._task and not self._task.done():
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
        self._state["status"] = "idle"
        logger.info("Ambient intelligence stopped")

    async def _ambient_loop(self):
        """Main ambient processing loop with adaptive frequency."""
        while self._running:
            try:
                self._state["cycles"] += 1
                self._state["last_cycle"] = asyncio.get_event_loop().time()

                # Run all registered handlers
                for handler in self._handlers:
                    try:
                        await handler()
                    except Exception as e:
                        logger.error(f"Ambient handler error: {e}")
                        self._state["errors"] += 1

                # Adaptive polling: increase interval on low activity
                await asyncio.sleep(self._poll_interval)

            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"Ambient loop error: {e}")
                self._state["errors"] += 1
                await asyncio.sleep(self._poll_interval)

    async def register_handler(self, handler: Callable) -> bool:
        """Register a handler for ambient processing."""
        self._handlers.append(handler)
        return True

    async def unregister_handler(self, handler: Callable) -> bool:
        if handler in self._handlers:
            self._handlers.remove(handler)
            return True
        return False

    async def get_ambient_state(self) -> Dict[str, Any]:
        return {
            "running": self._running,
            "status": self._state["status"],
            "cycles": self._state["cycles"],
            "handlers": len(self._handlers),
            "errors": self._state["errors"],
            "poll_interval": self._poll_interval,
        }

    def set_poll_interval(self, interval: int):
        """Set polling interval with bounds checking."""
        self._poll_interval = max(self._min_interval, min(self._max_interval, interval))
```

### Task 2: Implement SummarizationService

**Architecture:** The summarization service compresses conversations and activity logs into structured summaries. For LLM-backed summarization, it sends messages to the configured LLM provider. For non-LLM mode, it uses extractive summarization: sentence scoring by position, length, and keyword density. The `compression_ratio` metric tracks quality. Key points are extracted from the top-scoring sentences.

**File:** `backend/app/services/interaction/summarization.py`
```python
from typing import List, Dict, Any
from sqlalchemy.orm import Session
from backend.app.models.interaction.conversation_summary import ConversationSummary


class SummarizationService:
    """Summarizes conversations and activities with quality metrics."""

    def __init__(self, db: Session, llm_service=None):
        self.db = db
        self.llm = llm_service

    async def summarize_conversation(self, user_id: int, messages: List[Dict],
                                    session_id: int = None) -> ConversationSummary:
        """Summarize a list of conversation messages."""
        all_content = " ".join(m.get("content", "") for m in messages)
        word_count = len(all_content.split())

        if self.llm:
            summary_text = await self._llm_summarize(all_content)
        else:
            summary_text = self._extractive_summarize(all_content)

        # Extract key points
        sentences = [s.strip() for s in all_content.split(".") if len(s.strip()) > 20]
        key_points = sentences[:5]

        # Extract topics
        topics = self._extract_topics(all_content)

        summary_word_count = len(summary_text.split())
        compression_ratio = word_count / max(summary_word_count, 1)

        summary = ConversationSummary(
            owner_id=user_id,
            session_id=session_id,
            summary=summary_text,
            key_points=key_points,
            topics=", ".join(topics),
            word_count_original=word_count,
            compression_ratio=round(compression_ratio, 2),
        )
        self.db.add(summary)
        self.db.commit()
        return summary

    def _extractive_summarize(self, text: str) -> str:
        """Extractive summarization: select top sentences by score."""
        sentences = [s.strip() for s in text.split(".") if len(s.strip()) > 10]
        if not sentences:
            return text[:500]

        # Score sentences
        scored = []
        for i, s in enumerate(sentences):
            score = 0
            # Position bonus (first and last sentences)
            if i == 0 or i == len(sentences) - 1:
                score += 3
            # Length bonus
            if 20 < len(s) < 200:
                score += 2
            # Keyword density
            words = s.lower().split()
            if len(words) > 5:
                score += 1
            scored.append((score, s))

        scored.sort(key=lambda x: x[0], reverse=True)
        top_sentences = [s for _, s in scored[:3]]
        return ". ".join(top_sentences) + "."

    def _extract_topics(self, text: str) -> List[str]:
        """Extract key topics from text using word frequency."""
        stop_words = {"the", "a", "an", "is", "are", "was", "were", "be", "been",
                      "being", "have", "has", "had", "do", "does", "did", "will",
                      "would", "could", "should", "may", "might", "can", "shall",
                      "i", "you", "he", "she", "it", "we", "they", "me", "him",
                      "her", "us", "them", "my", "your", "his", "its", "our", "their"}
        words = text.lower().split()
        word_freq = {}
        for w in words:
            w = w.strip(".,!?;:'\"")
            if len(w) > 3 and w not in stop_words:
                word_freq[w] = word_freq.get(w, 0) + 1

        sorted_words = sorted(word_freq.items(), key=lambda x: x[1], reverse=True)
        return [w for w, _ in sorted_words[:5]]

    async def _llm_summarize(self, text: str) -> str:
        """LLM-backed summarization (when available)."""
        try:
            result = await self.llm.complete(
                f"Summarize the following conversation concisely:\n\n{text[:4000]}"
            )
            return result
        except Exception:
            return self._extractive_summarize(text)

    async def summarize_activity(self, user_id: int, activities: List[Dict]) -> str:
        """Summarize a list of user activities."""
        if not activities:
            return "No recent activity."
        activity_types = {}
        for a in activities:
            atype = a.get("type", "unknown")
            activity_types[atype] = activity_types.get(atype, 0) + 1
        parts = [f"{count} {atype} activities" for atype, count in activity_types.items()]
        return f"Recent activity: {', '.join(parts)}"

    async def get_summaries(self, user_id: int, limit: int = 20) -> List[Dict]:
        summaries = self.db.query(ConversationSummary).filter(
            ConversationSummary.owner_id == user_id,
        ).order_by(ConversationSummary.created_at.desc()).limit(limit).all()
        return [
            {
                "id": s.id, "summary": s.summary, "topics": s.topics,
                "compression_ratio": s.compression_ratio,
                "created_at": s.created_at.isoformat(),
            }
            for s in summaries
        ]

    async def delete_summary(self, summary_id: int) -> bool:
        summary = self.db.query(ConversationSummary).filter(
            ConversationSummary.id == summary_id,
        ).first()
        if summary:
            self.db.delete(summary)
            self.db.commit()
            return True
        return False
```

### Task 3: Implement FeedbackCollectionService

**Architecture:** The feedback collection service captures per-feature ratings and comments. It computes rolling satisfaction scores, identifies improvement areas, and generates actionable suggestions. The service integrates with the learning engine (v1.09) for feedback-driven improvement.

**File:** `backend/app/services/interaction/feedback.py`
```python
from typing import Dict, List, Any
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from backend.app.models.interaction.feedback_entry import FeedbackEntry


class FeedbackCollectionService:
    """Collects, analyzes, and aggregates user feedback per feature."""

    def __init__(self, db: Session):
        self.db = db

    async def record_feedback(self, user_id: int, feature: str, rating: int,
                             comment: str = None) -> FeedbackEntry:
        feedback = FeedbackEntry(
            owner_id=user_id,
            feature=feature,
            rating=rating,
            comment=comment,
        )
        self.db.add(feedback)
        self.db.commit()
        return feedback

    async def get_feedback_summary(self, user_id: int, days: int = 30) -> Dict[str, Any]:
        since = datetime.utcnow() - timedelta(days=days)
        entries = self.db.query(FeedbackEntry).filter(
            FeedbackEntry.owner_id == user_id,
            FeedbackEntry.created_at >= since,
        ).all()

        if not entries:
            return {"total": 0, "avg_rating": 0.0, "by_feature": {}}

        by_feature = {}
        for e in entries:
            if e.feature not in by_feature:
                by_feature[e.feature] = []
            if e.rating:
                by_feature[e.feature].append(e.rating)

        return {
            "total": len(entries),
            "avg_rating": sum(e.rating for e in entries if e.rating) / len([e for e in entries if e.rating]),
            "by_feature": {
                f: sum(ratings) / len(ratings)
                for f, ratings in by_feature.items()
            },
        }

    async def get_feedback_for_feature(self, user_id: int, feature: str) -> List[Dict]:
        entries = self.db.query(FeedbackEntry).filter(
            FeedbackEntry.owner_id == user_id,
            FeedbackEntry.feature == feature,
        ).order_by(FeedbackEntry.created_at.desc()).limit(100).all()
        return [
            {"id": e.id, "rating": e.rating, "comment": e.comment, "created_at": e.created_at}
            for e in entries
        ]

    async def get_improvement_areas(self, user_id: int) -> List[Dict[str, Any]]:
        summary = await self.get_feedback_summary(user_id)
        areas = []
        for feature, avg_rating in summary.get("by_feature", {}).items():
            if avg_rating < 3.0:
                areas.append({
                    "feature": feature,
                    "avg_rating": avg_rating,
                    "suggestion": f"Low satisfaction ({avg_rating:.1f}/5) — review {feature} feedback",
                    "priority": "high" if avg_rating < 2.0 else "medium",
                })
        return areas
```

### Task 4: Implement MultiDeviceSyncService

**Architecture:** Multi-device sync tracks device registration, entity versioning, and conflict resolution. Each synced entity has a version number and checksum. Conflicts detected when two devices modify the same entity between syncs. Resolution strategy: last-writer-wins for simple fields, user-prompted merge for true conflicts. The service uses the `device_registry` and `sync_state` models from P01.

**File:** `backend/app/services/interaction/multi_device_sync.py`
```python
from typing import Dict, List, Any, Optional
from datetime import datetime
from sqlalchemy.orm import Session
from backend.app.models.interaction.device_registry import DeviceRegistry
from backend.app.models.interaction.sync_state import SyncState


class MultiDeviceSyncService:
    """Multi-device synchronization with conflict resolution."""

    def __init__(self, db: Session):
        self.db = db

    async def register_device(self, user_id: int, device_name: str,
                             device_type: str, capabilities: List[str] = None) -> DeviceRegistry:
        device = DeviceRegistry(
            owner_id=user_id,
            device_name=device_name,
            device_type=device_type,
            capabilities=capabilities or [],
            last_seen=datetime.utcnow(),
        )
        self.db.add(device)
        self.db.commit()
        return device

    async def get_devices(self, user_id: int) -> List[Dict]:
        devices = self.db.query(DeviceRegistry).filter(
            DeviceRegistry.owner_id == user_id,
        ).all()
        return [
            {"id": d.id, "name": d.device_name, "type": d.device_type,
             "last_seen": d.last_seen, "capabilities": d.capabilities}
            for d in devices
        ]

    async def sync_entity(self, user_id: int, device_id: int, entity_type: str,
                         entity_id: int, version: int, checksum: str,
                         data: Any = None) -> Dict[str, Any]:
        """Sync an entity from a device. Detects and resolves conflicts."""
        existing = self.db.query(SyncState).filter(
            SyncState.entity_type == entity_type,
            SyncState.entity_id == entity_id,
        ).first()

        if existing:
            if existing.version >= version:
                # Conflict: remote version is older or equal
                if existing.checksum != checksum:
                    existing.conflict_count += 1
                    self.db.commit()
                    return {
                        "status": "conflict",
                        "local_version": existing.version,
                        "remote_version": version,
                        "resolution": "local_wins",
                    }
            else:
                # Update: remote version is newer
                existing.version = version
                existing.checksum = checksum
                existing.device_id = device_id
                existing.last_synced = datetime.utcnow()
                self.db.commit()
                return {"status": "updated", "version": version}
        else:
            # New entity
            sync = SyncState(
                owner_id=user_id,
                device_id=device_id,
                entity_type=entity_type,
                entity_id=entity_id,
                version=version,
                checksum=checksum,
                last_synced=datetime.utcnow(),
            )
            self.db.add(sync)
            self.db.commit()
            return {"status": "created", "version": version}

    async def get_pending_syncs(self, user_id: int, device_id: int) -> List[Dict]:
        """Get entities that need syncing for a device."""
        pending = self.db.query(SyncState).filter(
            SyncState.owner_id == user_id,
            SyncState.device_id != device_id,
        ).all()
        return [
            {"entity_type": s.entity_type, "entity_id": s.entity_id,
             "version": s.version, "checksum": s.checksum}
            for s in pending
        ]

    async def get_sync_stats(self, user_id: int) -> Dict[str, Any]:
        all_syncs = self.db.query(SyncState).filter(
            SyncState.owner_id == user_id,
        ).all()
        total_conflicts = sum(s.conflict_count for s in all_syncs)
        return {
            "total_entities": len(all_syncs),
            "total_conflicts": total_conflicts,
            "devices": len(await self.get_devices(user_id)),
        }
```
