### Task 1: Create Interaction API Routers

**Architecture:** Interaction endpoints are grouped into sub-routers: `/api/v1/interaction/voice`, `/api/v1/interaction/commands`, `/api/v1/interaction/proactive`, `/api/v1/interaction/summarization`, `/api/v1/interaction/ambient`, `/api/v1/interaction/feedback`, `/api/v1/interaction/sync`. All endpoints use `get_db()` and `get_current_user` dependency injection. Response models defined on all decorators. Ownership enforced via `owner_id` checks.

**File:** `backend/app/api/v1/interaction/voice.py`
```python
from fastapi import APIRouter, Depends, UploadFile, File, HTTPException
from sqlalchemy.orm import Session
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.interaction.voice import VoiceProcessingService
from backend.app.schemas.interaction.voice import VoiceRecordResponse

router = APIRouter()


@router.post("/transcribe", response_model=VoiceRecordResponse)
async def transcribe_voice(
    file: UploadFile = File(...),
    language: str = "en",
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Transcribe uploaded audio file."""
    file_path = f"/tmp/voice_{current_user.id}_{file.filename}"
    with open(file_path, "wb") as f:
        content = await file.read()
        f.write(content)
    service = VoiceProcessingService(db)
    record = await service.record_voice(current_user.id, file_path, language)
    return record


@router.get("/history", response_model=list[VoiceRecordResponse])
async def get_voice_history(
    limit: int = 50,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Get voice record history."""
    service = VoiceProcessingService(db)
    return await service.get_voice_history(current_user.id, limit)


@router.get("/search")
async def search_transcripts(
    query: str = "",
    limit: int = 20,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    """Search voice transcripts."""
    service = VoiceProcessingService(db)
    return await service.search_transcripts(current_user.id, query, limit)
```

### Task 4: Create Frontend API Client

**File:** `frontend/features/interaction/api.ts`
```typescript
import { apiClient } from '@/shared/api/client';

export const interactionApi = {
  // Voice
  transcribeVoice: (file: File, language: string = 'en') => {
    const formData = new FormData();
    formData.append('file', file);
    return apiClient.post(`/interaction/voice/transcribe?language=${language}`, formData);
  },
  getVoiceHistory: (limit: number = 50) =>
    apiClient.get(`/interaction/voice/history?limit=${limit}`),
  searchTranscripts: (query: string) =>
    apiClient.get(`/interaction/voice/search?query=${encodeURIComponent(query)}`),

  // Commands
  getCommands: (query: string = '', category?: string) =>
    apiClient.get(`/interaction/commands/commands?query=${query}${category ? `&category=${category}` : ''}`),
  registerCommand: (name: string, description: string, shortcut?: string, category: string = 'general') =>
    apiClient.post('/interaction/commands/commands', { command_name: name, description, shortcut, category }),
  executeCommand: (id: number) =>
    apiClient.post(`/interaction/commands/commands/${id}/execute`),
  getPopularCommands: (limit: number = 10) =>
    apiClient.get(`/interaction/commands/commands/popular?limit=${limit}`),

  // Proactive Suggestions
  getSuggestions: (minConfidence: number = 0.5) =>
    apiClient.get(`/interaction/proactive/suggestions?min_confidence=${minConfidence}`),
  acceptSuggestion: (id: number) =>
    apiClient.post(`/interaction/proactive/suggestions/${id}/accept`),
  dismissSuggestion: (id: number) =>
    apiClient.post(`/interaction/proactive/suggestions/${id}/dismiss`),

  // Summarization
  getSummaries: (limit: number = 20) =>
    apiClient.get(`/interaction/summarization/summaries?limit=${limit}`),
  deleteSummary: (id: number) =>
    apiClient.delete(`/interaction/summarization/summaries/${id}`),

  // Ambient
  getAmbientState: () =>
    apiClient.get('/interaction/ambient/state'),
  startAmbient: () =>
    apiClient.post('/interaction/ambient/start'),
  stopAmbient: () =>
    apiClient.post('/interaction/ambient/stop'),

  // Feedback
  submitFeedback: (feature: string, rating: number, comment?: string) =>
    apiClient.post('/interaction/feedback/', { feature, rating, comment }),
  getFeedbackSummary: (days: number = 30) =>
    apiClient.get(`/interaction/feedback/summary?days=${days}`),

  // Sync
  registerDevice: (name: string, type: string, capabilities: string[] = []) =>
    apiClient.post('/interaction/sync/devices', { device_name: name, device_type: type, capabilities }),
  getDevices: () =>
    apiClient.get('/interaction/sync/devices'),
  syncEntity: (deviceId: number, entityType: string, entityId: number, version: number, checksum: string) =>
    apiClient.post('/interaction/sync/sync', { device_id: deviceId, entity_type: entityType, entity_id: entityId, version, checksum }),
  getSyncStats: () =>
    apiClient.get('/interaction/sync/stats'),
};
```
