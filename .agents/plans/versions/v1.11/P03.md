### Task 1: Implement ProactiveAssistanceService

**Architecture:** The proactive assistance service generates suggestions based on user context, activity patterns, and time-of-day. Suggestions have confidence scores that determine display priority. The dismissal feedback loop learns from user accept/dismiss patterns to improve future suggestion quality. Expired suggestions are automatically cleaned up. The `context` field captures the trigger event for analytics.

**File:** `backend/app/services/interaction/proactive.py`
```python
from typing import List, Dict, Any, Optional
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from backend.app.models.interaction.proactive_suggestion import ProactiveSuggestion


class ProactiveAssistanceService:
    """Proactive suggestion generation with confidence thresholding and feedback loops."""

    def __init__(self, db: Session, confidence_threshold: float = 0.5):
        self.db = db
        self.confidence_threshold = confidence_threshold

    async def suggest(self, user_id: int, suggestion_type: str, title: str,
                     description: str, context: str, confidence: float = 0.8,
                     ttl_hours: int = 24) -> ProactiveSuggestion:
        """Create a proactive suggestion."""
        suggestion = ProactiveSuggestion(
            owner_id=user_id,
            suggestion_type=suggestion_type,
            title=title,
            description=description,
            context=context,
            confidence=confidence,
            expires_at=datetime.utcnow() + timedelta(hours=ttl_hours),
        )
        self.db.add(suggestion)
        self.db.commit()
        return suggestion

    async def get_suggestions(self, user_id: int, limit: int = 10,
                             min_confidence: float = None) -> List[Dict[str, Any]]:
        """Get active suggestions above confidence threshold."""
        threshold = min_confidence or self.confidence_threshold
        suggestions = self.db.query(ProactiveSuggestion).filter(
            ProactiveSuggestion.owner_id == user_id,
            ProactiveSuggestion.accepted.is_(None),
            ProactiveSuggestion.confidence >= threshold,
            ProactiveSuggestion.expires_at > datetime.utcnow(),
        ).order_by(ProactiveSuggestion.confidence.desc()).limit(limit).all()
        return [
            {
                "id": s.id, "type": s.suggestion_type,
                "title": s.title, "description": s.description,
                "confidence": s.confidence, "context": s.context,
            }
            for s in suggestions
        ]

    async def accept_suggestion(self, suggestion_id: int) -> bool:
        """Accept a suggestion — boosts confidence for similar future suggestions."""
        suggestion = self.db.query(ProactiveSuggestion).filter(
            ProactiveSuggestion.id == suggestion_id,
        ).first()
        if not suggestion:
            return False
        suggestion.accepted = 1
        suggestion.responded_at = datetime.utcnow()
        self.db.commit()
        return True

    async def dismiss_suggestion(self, suggestion_id: int) -> bool:
        """Dismiss a suggestion — reduces confidence for similar future suggestions."""
        suggestion = self.db.query(ProactiveSuggestion).filter(
            ProactiveSuggestion.id == suggestion_id,
        ).first()
        if not suggestion:
            return False
        suggestion.accepted = 0
        suggestion.responded_at = datetime.utcnow()
        self.db.commit()
        return True

    async def cleanup_expired(self) -> int:
        """Remove expired suggestions."""
        expired = self.db.query(ProactiveSuggestion).filter(
            ProactiveSuggestion.expires_at <= datetime.utcnow(),
            ProactiveSuggestion.accepted.is_(None),
        ).all()
        count = len(expired)
        for s in expired:
            self.db.delete(s)
        self.db.commit()
        return count

    async def get_acceptance_rate(self, user_id: int, suggestion_type: str = None) -> Dict[str, Any]:
        """Calculate suggestion acceptance rate for feedback loop analytics."""
        query = self.db.query(ProactiveSuggestion).filter(
            ProactiveSuggestion.owner_id == user_id,
            ProactiveSuggestion.accepted.isnot(None),
        )
        if suggestion_type:
            query = query.filter(ProactiveSuggestion.suggestion_type == suggestion_type)

        responded = query.all()
        if not responded:
            return {"total": 0, "accepted": 0, "dismissed": 0, "rate": 0.0}

        accepted = sum(1 for s in responded if s.accepted == 1)
        return {
            "total": len(responded),
            "accepted": accepted,
            "dismissed": len(responded) - accepted,
            "rate": accepted / len(responded),
        }
```

### Task 2: Implement ContextualSuggestionsService

**Architecture:** The contextual service analyzes the user's current environment (active application, recent files, time of day, current task) and generates relevant suggestions. It integrates with the awareness engine (v1.08) for activity data. Suggestions are categorized by context type for filtering and analytics.

**File:** `backend/app/services/interaction/contextual.py`
```python
from typing import Dict, List, Any
from datetime import datetime


class ContextualSuggestionsService:
    """Generates contextual suggestions based on user's current environment."""

    async def analyze_context(self, context: Dict[str, Any]) -> List[Dict[str, Any]]:
        """Analyze current context and return relevant suggestions."""
        suggestions = []

        # Application context
        active_app = context.get("active_app", "")
        if active_app:
            suggestions.append({
                "type": "app_context",
                "title": f"You're working in {active_app}",
                "description": f"I can help with {active_app}-related tasks",
                "confidence": 0.7,
            })

        # File context
        recent_files = context.get("recent_files", [])
        if recent_files:
            file_types = set()
            for f in recent_files:
                ext = f.rsplit(".", 1)[-1] if "." in f else "unknown"
                file_types.add(ext)
            suggestions.append({
                "type": "file_context",
                "title": f"{len(recent_files)} recently modified files ({', '.join(file_types)})",
                "description": "I can help organize, review, or analyze these files",
                "confidence": 0.6,
            })

        # Time context
        hour = datetime.now().hour
        if 5 <= hour < 12:
            suggestions.append({
                "type": "time_context",
                "title": "Good morning",
                "description": "Would you like a daily briefing?",
                "confidence": 0.5,
            })
        elif 12 <= hour < 17:
            suggestions.append({
                "type": "time_context",
                "title": "Afternoon check-in",
                "description": "Here's a progress summary of your work today",
                "confidence": 0.5,
            })
        elif 17 <= hour < 22:
            suggestions.append({
                "type": "time_context",
                "title": "Evening review",
                "description": "Would you like to review today's accomplishments?",
                "confidence": 0.5,
            })
        else:
            suggestions.append({
                "type": "time_context",
                "title": "Late night reminder",
                "description": "Consider taking a break — I can save your progress",
                "confidence": 0.4,
            })

        # Task context
        current_task = context.get("current_task", "")
        if current_task:
            suggestions.append({
                "type": "task_context",
                "title": f"Task: {current_task}",
                "description": "I can help break this down or check progress",
                "confidence": 0.6,
            })

        # Calendar context
        upcoming_meetings = context.get("upcoming_meetings", [])
        if upcoming_meetings:
            suggestions.append({
                "type": "calendar_context",
                "title": f"{len(upcoming_meetings)} upcoming meetings",
                "description": "I can help you prepare or summarize previous notes",
                "confidence": 0.6,
            })

        return suggestions

    async def build_context_from_activity(self, activity: Dict[str, Any]) -> Dict[str, Any]:
        """Build a context dict from raw activity data."""
        return {
            "active_app": activity.get("active_app", ""),
            "recent_files": activity.get("recent_files", []),
            "current_task": activity.get("current_task", ""),
            "upcoming_meetings": activity.get("upcoming_meetings", []),
            "timestamp": datetime.utcnow().isoformat(),
        }

    async def get_suggestion_priority(self, suggestion: Dict) -> int:
        """Calculate display priority for a suggestion (lower = higher priority)."""
        type_priorities = {
            "calendar_context": 1,
            "task_context": 2,
            "app_context": 3,
            "file_context": 4,
            "time_context": 5,
        }
        return type_priorities.get(suggestion.get("type", ""), 10)
```

### Task 3: Implement MultiModalService

**Architecture:** The multi-modal service routes input from different modalities (text, voice, image, file) to appropriate processors. Each processor implements the `ModalityProcessorProtocol`. The service handles modality detection, validation, and error handling. Future modalities can be added by implementing the protocol.

**File:** `backend/app/services/interaction/multimodal.py`
```python
from typing import Dict, Any, Optional, Protocol


class ModalityProcessorProtocol(Protocol):
    async def process(self, data: Any) -> Dict[str, Any]: ...


class MultiModalService:
    """Routes and processes input from multiple modalities."""

    def __init__(self):
        self._processors = {}

    def register_processor(self, modality: str, processor: ModalityProcessorProtocol):
        self._processors[modality] = processor

    async def process_input(self, input_type: str, data: Any) -> Dict[str, Any]:
        processor = self._processors.get(input_type)
        if processor:
            return await processor.process(data)
        # Fallback processing
        fallbacks = {
            "text": self._process_text,
            "voice": self._process_voice,
            "image": self._process_image,
            "file": self._process_file,
        }
        fallback = fallbacks.get(input_type)
        if fallback:
            return await fallback(data)
        return {"error": f"Unknown input type: {input_type}"}

    async def _process_text(self, text: str) -> Dict[str, Any]:
        return {"type": "text", "processed": True, "content": text}

    async def _process_voice(self, audio_data: bytes) -> Dict[str, Any]:
        return {"type": "voice", "processed": True, "transcript": "pending STT"}

    async def _process_image(self, image_data: bytes) -> Dict[str, Any]:
        return {"type": "image", "processed": True, "description": "pending vision model"}

    async def _process_file(self, file_path: str) -> Dict[str, Any]:
        return {"type": "file", "processed": True, "path": file_path}

    async def get_supported_modalities(self) -> list:
        return ["text", "voice", "image", "file"]
```

### Task 4: Implement NotificationIntegrationService

**Architecture:** Notifications are stored in the database with read/unread tracking. The service supports priority levels (low, normal, high, critical) with corresponding display behavior. Notifications can be linked to specific features for click-through navigation. Batch operations support bulk mark-read and clear.

**File:** `backend/app/services/interaction/notifications.py`
```python
from typing import Dict, List, Any
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from backend.app.models.interaction.interaction_analytics import InteractionAnalytics


class NotificationIntegrationService:
    """Manages interaction notifications with priority and batch operations."""

    def __init__(self, db: Session):
        self.db = db
        self._notifications: List[Dict[str, Any]] = []

    async def send_notification(self, title: str, body: str, priority: str = "normal",
                               feature: str = None) -> Dict[str, Any]:
        notification = {
            "title": title,
            "body": body,
            "priority": priority,
            "feature": feature,
            "timestamp": datetime.utcnow().isoformat(),
            "read": False,
        }
        self._notifications.append(notification)
        return notification

    async def get_notifications(self, unread_only: bool = True, limit: int = 50) -> List[Dict]:
        notifications = self._notifications
        if unread_only:
            notifications = [n for n in notifications if not n["read"]]
        return notifications[-limit:]

    async def mark_read(self, index: int) -> bool:
        if 0 <= index < len(self._notifications):
            self._notifications[index]["read"] = True
            return True
        return False

    async def mark_all_read(self) -> int:
        count = sum(1 for n in self._notifications if not n["read"])
        for n in self._notifications:
            n["read"] = True
        return count

    async def clear_all(self) -> int:
        count = len(self._notifications)
        self._notifications.clear()
        return count

    async def get_unread_count(self) -> int:
        return sum(1 for n in self._notifications if not n["read"])
```

### Task 5: Implement ResponseFormattingService

**Architecture:** Response formatting adapts content for different output modes: text (plain), markdown (rich), voice (stripped of formatting), notification (truncated), and JSON (structured). The service is stateless and can be used by any other service that needs output adaptation.

**File:** `backend/app/services/interaction/formatting.py`
```python
from typing import Dict, Any, List
import re


class ResponseFormattingService:
    """Formats responses for different output modes."""

    async def format_response(self, content: str, mode: str = "text",
                             max_length: int = None) -> Dict[str, Any]:
        formatters = {
            "text": self._format_text,
            "markdown": self._format_markdown,
            "voice": self._format_voice,
            "notification": self._format_notification,
            "json": self._format_json,
        }
        formatter = formatters.get(mode, self._format_text)
        result = await formatter(content)
        if max_length and len(result.get("content", "")) > max_length:
            result["content"] = result["content"][:max_length] + "..."
            result["truncated"] = True
        return result

    async def _format_text(self, content: str) -> Dict[str, Any]:
        return {"type": "text", "content": content}

    async def _format_markdown(self, content: str) -> Dict[str, Any]:
        return {"type": "markdown", "content": content}

    async def _format_voice(self, content: str) -> Dict[str, Any]:
        plain = re.sub(r'[*_`#\[\]()]', '', content)
        plain = re.sub(r'\n+', '. ', plain)
        return {"type": "voice", "content": plain}

    async def _format_notification(self, content: str) -> Dict[str, Any]:
        truncated = content[:200] + "..." if len(content) > 200 else content
        return {"type": "notification", "content": truncated}

    async def _format_json(self, content: str) -> Dict[str, Any]:
        return {"type": "json", "content": content, "structured": True}

    async def get_supported_modes(self) -> List[str]:
        return ["text", "markdown", "voice", "notification", "json"]
```
