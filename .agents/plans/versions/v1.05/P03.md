### Task 1: Implement Audit Logging Service

**Architecture:** The audit service is the trust backbone. Every significant action across the system flows through this service. Key properties:
- **Immutable:** Once written, audit records are never updated or deleted (append-only)
- **Complete:** All CRUD operations, auth events, consent changes, and data exports must be logged
- **Performance:** Logging is non-blocking; uses async write with background flush
- **Queryable:** Composite indexes enable fast queries by user, resource, time range, and action

**Implementation:**
```python
# backend/app/services/privacy/audit.py

import time
from datetime import datetime
from typing import Optional, Dict, Any, List
from sqlalchemy.orm import Session
from sqlalchemy import and_
from backend.app.models.privacy.audit_log import AuditLog


class AuditLoggingService:
    """Immutable audit logging service.
    
    All audit writes are append-only. The service provides:
    - Structured logging with full context (user, action, resource, IP, timing)
    - Query by user, resource, action, time range
    - Performance statistics
    - Log retention support (called by scheduled job)
    
    Usage:
        service = AuditLoggingService(db)
        await service.log(user_id=1, action="create", resource_type="memory", resource_id=42)
    """
    
    def __init__(self, db: Session):
        self.db = db
    
    async def log(self, user_id: int, action: str, resource_type: str,
                  resource_id: Optional[int] = None,
                  details: Optional[Dict[str, Any]] = None,
                  ip_address: Optional[str] = None,
                  user_agent: Optional[str] = None,
                  session_id: Optional[str] = None,
                  success: bool = True,
                  error_message: Optional[str] = None,
                  duration_ms: Optional[int] = None) -> AuditLog:
        """Create an immutable audit log entry.
        
        Args:
            user_id: ID of the acting user
            action: Action performed: create, read, update, delete, login, logout,
                    consent_grant, consent_revoke, export, access_check, agent_action
            resource_type: Domain: memory, file, settings, agent, workflow, consent, audit
            resource_id: ID of the affected resource (NULL for non-resource actions)
            details: Structured context (old_value, new_value, metadata)
            ip_address: Client IP address
            user_agent: Client user agent
            session_id: Session identifier
            success: Whether the action succeeded
            error_message: Error details if action failed
            duration_ms: Action duration in milliseconds
        
        Returns:
            Created AuditLog record
        """
        log_entry = AuditLog(
            user_id=user_id,
            action=action,
            resource_type=resource_type,
            resource_id=resource_id,
            details=details,
            ip_address=ip_address,
            user_agent=user_agent,
            session_id=session_id,
            timestamp=datetime.utcnow(),
            success=1 if success else 0,
            error_message=error_message,
            duration_ms=duration_ms,
        )
        self.db.add(log_entry)
        self.db.commit()
        self.db.refresh(log_entry)
        return log_entry
    
    async def get_user_logs(self, user_id: int, limit: int = 100,
                           offset: int = 0,
                           action_filter: Optional[str] = None,
                           resource_type_filter: Optional[str] = None,
                           start_time: Optional[datetime] = None,
                           end_time: Optional[datetime] = None) -> List[AuditLog]:
        """Query audit logs for a user with filtering and pagination.
        
        Uses composite index on (user_id, timestamp) for efficient queries.
        """
        query = self.db.query(AuditLog).filter(AuditLog.user_id == user_id)
        
        if action_filter:
            query = query.filter(AuditLog.action == action_filter)
        if resource_type_filter:
            query = query.filter(AuditLog.resource_type == resource_type_filter)
        if start_time:
            query = query.filter(AuditLog.timestamp >= start_time)
        if end_time:
            query = query.filter(AuditLog.timestamp <= end_time)
        
        return query.order_by(AuditLog.timestamp.desc()).offset(offset).limit(limit).all()
    
    async def get_resource_logs(self, resource_type: str, resource_id: int) -> List[AuditLog]:
        """Get complete audit history for a specific resource."""
        return self.db.query(AuditLog).filter(
            AuditLog.resource_type == resource_type,
            AuditLog.resource_id == resource_id,
        ).order_by(AuditLog.timestamp.desc()).all()
    
    async def get_action_stats(self, user_id: int, days: int = 30) -> Dict[str, Any]:
        """Get action statistics for a user over the given period."""
        from datetime import timedelta
        since = datetime.utcnow() - timedelta(days=days)
        
        logs = self.db.query(AuditLog).filter(
            AuditLog.user_id == user_id,
            AuditLog.timestamp >= since,
        ).all()
        
        action_counts: Dict[str, int] = {}
        success_count = 0
        failure_count = 0
        
        for log in logs:
            action_counts[log.action] = action_counts.get(log.action, 0) + 1
            if log.success:
                success_count += 1
            else:
                failure_count += 1
        
        return {
            "total_actions": len(logs),
            "action_breakdown": action_counts,
            "success_count": success_count,
            "failure_count": failure_count,
            "success_rate": (success_count / len(logs) * 100) if logs else 0,
            "period_days": days,
        }
    
    async def get_recent_activity(self, user_id: int, limit: int = 20) -> List[AuditLog]:
        """Get most recent activity for dashboard display."""
        return self.db.query(AuditLog).filter(
            AuditLog.user_id == user_id,
        ).order_by(AuditLog.timestamp.desc()).limit(limit).all()
    
    async def count_user_logs(self, user_id: int) -> int:
        """Count total audit logs for a user."""
        return self.db.query(AuditLog).filter(
            AuditLog.user_id == user_id,
        ).count()
    
    async def delete_old_logs(self, retention_days: int = 90) -> int:
        """Delete audit logs older than retention period.
        
        WARNING: This is the ONLY legitimate deletion path for audit logs.
        All other code must treat audit_logs as append-only.
        """
        from datetime import timedelta
        cutoff = datetime.utcnow() - timedelta(days=retention_days)
        deleted = self.db.query(AuditLog).filter(
            AuditLog.timestamp < cutoff,
        ).delete()
        self.db.commit()
        return deleted
```

**Performance Considerations:**
- Composite index on `(user_id, timestamp)` supports the most common query pattern
- `get_user_logs` uses pagination (offset/limit) to avoid loading entire log history
- `get_action_stats` aggregates in Python for accuracy; for high-volume scenarios, consider DB-level GROUP BY
- `delete_old_logs` should run as a scheduled daily job during low-traffic hours

**Risk Notes:**
- Audit log deletion is irreversible — only retention policy should trigger it
- High write volume: consider batching inserts in high-throughput scenarios

### Task 2: Implement Transparency Service

**Architecture:** The transparency service provides explainable decisions. When Cortex makes an automated decision (e.g., which memories to retrieve, how to score a hypothesis, which tools to use), this service generates a human-readable explanation of:
- What factors were considered
- What alternatives existed
- What the confidence level is
- What the audit trail is

This enables users to understand and verify Cortex's reasoning.

**Implementation:**
```python
# backend/app/services/privacy/transparency.py

from typing import Dict, Any, List, Optional
from dataclasses import dataclass, field, asdict


@dataclass
class DecisionFactor:
    """A single factor in a decision."""
    key: str
    value: Any
    weight: float = 1.0
    description: Optional[str] = None


@dataclass
class DecisionExplanation:
    """Structured explanation of an automated decision."""
    decision_type: str
    reasoning: str
    factors: List[DecisionFactor]
    confidence: float
    alternatives_considered: List[str]
    recommendation: str
    risk_level: str
    audit_trail: List[str] = field(default_factory=list)


class TransparencyService:
    """Provides explainable decisions and actions.
    
    Every automated decision Cortex makes should be explainable through
    this service. The service generates structured explanations that
    can be rendered in the UI or returned via API.
    
    Usage:
        service = TransparencyService()
        explanation = await service.explain_decision(
            "memory_retrieval",
            {"query": "user's preference", "results": 5, "confidence": 0.85}
        )
    """
    
    DECISION_TEMPLATES = {
        "memory_retrieval": {
            "description": "Retrieved relevant memories based on query similarity",
            "factors": ["query_similarity", "recency", "relevance_score", "access_frequency"],
            "alternatives": ["semantic_search", "keyword_match", "graph_traversal"],
        },
        "hypothesis_scoring": {
            "description": "Scored hypothesis based on evidence balance and novelty",
            "factors": ["evidence_count", "evidence_quality", "novelty", "historical_accuracy"],
            "alternatives": ["simple_majority_vote", "weighted_average", "bayesian_update"],
        },
        "tool_selection": {
            "description": "Selected tool based on task requirements and past performance",
            "factors": ["tool_capability", "past_success_rate", "resource_requirements", "user_preferences"],
            "alternatives": ["alternative_tool_a", "alternative_tool_b", "manual_intervention"],
        },
        "error_analysis": {
            "description": "Analyzed error to determine root cause and resolution",
            "factors": ["error_type", "stack_trace", "historical_patterns", "context"],
            "alternatives": ["retry", "escalate", "skip", "alternative_approach"],
        },
        "access_decision": {
            "description": "Evaluated access request against policies and consent",
            "factors": ["user_role", "resource_owner", "consent_state", "policy_match"],
            "alternatives": ["deny", "request_consent", "escalate_to_admin"],
        },
    }
    
    async def explain_decision(self, decision_type: str, 
                               context: Dict[str, Any]) -> Dict[str, Any]:
        """Generate a structured explanation for an automated decision.
        
        Args:
            decision_type: Type of decision (memory_retrieval, hypothesis_scoring, etc.)
            context: Decision context with relevant data
        
        Returns:
            Structured explanation dict for API response
        """
        template = self.DECISION_TEMPLATES.get(decision_type, {})
        
        factors = self._extract_factors(decision_type, context)
        confidence = context.get("confidence", 0.5)
        alternatives = template.get("alternatives", [])
        
        explanation = DecisionExplanation(
            decision_type=decision_type,
            reasoning=self._generate_reasoning(decision_type, context, template),
            factors=factors,
            confidence=confidence,
            alternatives_considered=alternatives,
            recommendation=self._generate_recommendation(confidence),
            risk_level=self._assess_risk_level(confidence, context),
            audit_trail=context.get("audit_trail", []),
        )
        
        return asdict(explanation)
    
    def _extract_factors(self, decision_type: str, context: Dict[str, Any]) -> List[DecisionFactor]:
        """Extract decision factors from context."""
        template = self.DECISION_TEMPLATES.get(decision_type, {})
        factor_keys = template.get("factors", [])
        
        factors = []
        for key in factor_keys:
            value = context.get(key, "N/A")
            weight = context.get(f"{key}_weight", 1.0)
            factors.append(DecisionFactor(
                key=key,
                value=value,
                weight=weight,
                description=f"Weight of {key} in decision: {weight}",
            ))
        
        # Add any extra context factors not in template
        for key, value in context.items():
            if key not in factor_keys and not key.endswith("_weight") and key not in ("confidence", "audit_trail"):
                factors.append(DecisionFactor(
                    key=key, value=value, weight=0.5,
                    description=f"Additional context: {key}",
                ))
        
        return factors
    
    def _generate_reasoning(self, decision_type: str, context: Dict[str, Any],
                           template: Dict[str, Any]) -> str:
        """Generate human-readable reasoning."""
        description = template.get("description", f"Decision of type '{decision_type}'")
        factor_count = len(template.get("factors", []))
        confidence = context.get("confidence", 0.5)
        
        return (
            f"{description}. "
            f"Evaluated {factor_count} factors with overall confidence of {confidence:.0%}. "
            f"Context keys: {', '.join(list(context.keys())[:5])}."
        )
    
    def _generate_recommendation(self, confidence: float) -> str:
        """Generate recommendation based on confidence."""
        if confidence > 0.8:
            return "Proceed with high confidence. Decision is well-supported by evidence."
        elif confidence > 0.6:
            return "Proceed with verification. Decision is supported but could benefit from additional evidence."
        elif confidence > 0.4:
            return "Verify with user before proceeding. Confidence is moderate."
        else:
            return "Seek additional information or user input. Confidence is low."
    
    def _assess_risk_level(self, confidence: float, context: Dict[str, Any]) -> str:
        """Assess risk level of the decision."""
        # Low confidence = high risk
        if confidence < 0.4:
            return "high"
        elif confidence < 0.7:
            return "medium"
        else:
            return "low"
```

### Task 3: Implement Data Export Service

**Architecture:** Data export enables GDPR Article 20 (data portability). Users can request a complete export of all their data. The service:
1. Creates an export request (status: pending)
2. Gathers data from all domains (memory, files, settings, agents, workflows)
3. Serializes to requested format (JSON or CSV)
4. Computes SHA-256 checksum for integrity verification
5. Stores the file and updates status to complete
6. Provides download URL with automatic expiry

**Implementation:**
```python
# backend/app/services/privacy/export.py

import json
import csv
import hashlib
import os
import io
from datetime import datetime, timedelta
from typing import Optional, Dict, Any, List
from sqlalchemy.orm import Session
from backend.app.models.privacy.data_export import DataExport


class DataExportService:
    """Complete data export service for data portability.
    
    Supports:
    - Full export (all user data)
    - Partial export (specific data types)
    - JSON and CSV formats
    - SHA-256 integrity verification
    - Automatic file expiry and cleanup
    
    Export flow:
    1. Create export request → pending
    2. Gather data from all domains → processing
    3. Serialize and compute checksum → complete
    4. User downloads file → expires after 7 days
    """
    
    EXPORT_DIR = "/tmp/cortex_exports"
    EXPIRY_DAYS = 7
    MAX_EXPORT_SIZE_MB = 100
    
    def __init__(self, db: Session):
        self.db = db
    
    async def create_export(self, user_id: int, export_type: str = "full",
                           data_types: Optional[List[str]] = None,
                           format: str = "json") -> DataExport:
        """Create a data export request."""
        os.makedirs(self.EXPORT_DIR, exist_ok=True)
        
        export = DataExport(
            user_id=user_id,
            export_type=export_type,
            data_types=data_types,
            format=format,
            status="pending",
            created_at=datetime.utcnow(),
            expires_at=datetime.utcnow() + timedelta(days=self.EXPIRY_DAYS),
        )
        self.db.add(export)
        self.db.commit()
        self.db.refresh(export)
        return export
    
    async def process_export(self, export_id: int) -> DataExport:
        """Process a data export: gather, serialize, verify, store."""
        export = self.db.query(DataExport).filter(DataExport.id == export_id).first()
        if not export:
            raise ValueError(f"Export {export_id} not found")
        
        export.status = "processing"
        export.started_at = datetime.utcnow()
        self.db.commit()
        
        try:
            # Gather user data from all domains
            user_data = await self._gather_user_data(export.user_id, export.data_types)
            
            # Serialize
            if export.format == "json":
                content = json.dumps(user_data, indent=2, default=str, ensure_ascii=False)
                file_ext = "json"
            elif export.format == "csv":
                content = self._to_csv(user_data)
                file_ext = "csv"
            else:
                raise ValueError(f"Unsupported format: {export.format}")
            
            # Check size limit
            content_bytes = content.encode("utf-8")
            size_mb = len(content_bytes) / (1024 * 1024)
            if size_mb > self.MAX_EXPORT_SIZE_MB:
                raise ValueError(f"Export exceeds {self.MAX_EXPORT_SIZE_MB}MB limit")
            
            # Compute checksum
            checksum = hashlib.sha256(content_bytes).hexdigest()
            
            # Write to file
            file_path = os.path.join(
                self.EXPORT_DIR,
                f"cortex_export_{export.user_id}_{export.id}.{file_ext}"
            )
            with open(file_path, "wb") as f:
                f.write(content_bytes)
            
            # Update export record
            export.file_path = file_path
            export.file_size_bytes = len(content_bytes)
            export.checksum_sha256 = checksum
            export.status = "complete"
            export.completed_at = datetime.utcnow()
            self.db.commit()
            self.db.refresh(export)
            
            return export
            
        except Exception as e:
            export.status = "failed"
            export.error_message = str(e)
            self.db.commit()
            self.db.refresh(export)
            return export
    
    async def _gather_user_data(self, user_id: int,
                               data_types: Optional[List[str]] = None) -> Dict[str, Any]:
        """Gather all user data from domain services."""
        all_types = ["memories", "files", "settings", "agents", "workflows",
                     "consents", "audit_summary"]
        
        types_to_gather = data_types or all_types
        
        data: Dict[str, Any] = {
            "export_metadata": {
                "user_id": user_id,
                "export_date": datetime.utcnow().isoformat(),
                "format_version": "1.0",
                "data_types_included": types_to_gather,
            },
        }
        
        # Gather from each domain (stub implementations — real services in later phases)
        for dtype in types_to_gather:
            data[dtype] = await self._gather_domain_data(user_id, dtype)
        
        return data
    
    async def _gather_domain_data(self, user_id: int, domain: str) -> Any:
        """Gather data from a specific domain."""
        # In production, this queries actual domain services
        # For now, return structured stubs
        domain_stubs = {
            "memories": {"count": 0, "items": []},
            "files": {"count": 0, "items": []},
            "settings": {},
            "agents": {"count": 0, "items": []},
            "workflows": {"count": 0, "items": []},
            "consents": {"count": 0, "items": []},
            "audit_summary": {"total_actions": 0},
        }
        return domain_stubs.get(domain, {})
    
    def _to_csv(self, data: Dict[str, Any]) -> str:
        """Convert nested data to CSV format."""
        output = io.StringIO()
        
        # Write metadata section
        writer = csv.writer(output)
        writer.writerow(["=== Cortex Data Export ==="])
        writer.writerow([])
        
        for section, content in data.items():
            writer.writerow([f"--- {section} ---"])
            if isinstance(content, dict):
                for key, value in content.items():
                    if isinstance(value, (list, dict)):
                        writer.writerow([key, json.dumps(value, default=str)])
                    else:
                        writer.writerow([key, value])
            else:
                writer.writerow([content])
            writer.writerow([])
        
        return output.getvalue()
    
    async def verify_export(self, export_id: int) -> Dict[str, Any]:
        """Verify export integrity by recomputing checksum."""
        export = self.db.query(DataExport).filter(DataExport.id == export_id).first()
        if not export:
            raise ValueError(f"Export {export_id} not found")
        
        if not export.file_path or not os.path.exists(export.file_path):
            return {"valid": False, "reason": "Export file not found"}
        
        with open(export.file_path, "rb") as f:
            computed = hashlib.sha256(f.read()).hexdigest()
        
        return {
            "valid": computed == export.checksum_sha256,
            "stored_checksum": export.checksum_sha256,
            "computed_checksum": computed,
            "file_size_bytes": export.file_size_bytes,
        }
    
    async def cleanup_expired_exports(self) -> int:
        """Delete export files that have expired."""
        cutoff = datetime.utcnow()
        expired = self.db.query(DataExport).filter(
            DataExport.expires_at != None,
            DataExport.expires_at < cutoff,
            DataExport.file_path != None,
        ).all()
        
        deleted_count = 0
        for export in expired:
            if os.path.exists(export.file_path):
                os.remove(export.file_path)
            export.file_path = None
            deleted_count += 1
        
        self.db.commit()
        return deleted_count
```

### Task 4: Implement Data Deletion Service

**Architecture:** Supports data sovereignty (P6). Users can request deletion of specific data categories or all data. The service:
1. Creates a deletion request (status: pending)
2. Computes SHA-256 hash of data to be deleted (for proof)
3. Deletes the data from all applicable domains
4. Records the deletion proof hash
5. Emits deletion event for audit trail

**Implementation:**
```python
# backend/app/services/privacy/deletion.py

import hashlib
import json
from datetime import datetime
from typing import Optional, Dict, Any, List
from sqlalchemy.orm import Session
from backend.app.models.privacy.data_deletion import DataDeletionRequest


class DataDeletionService:
    """Data deletion with cryptographic proof.
    
    Supports:
    - Full deletion (all user data)
    - Partial deletion (specific data types)
    - SHA-256 deletion manifest for audit proof
    - Deletion event emission for system-wide coordination
    """
    
    def __init__(self, db: Session):
        self.db = db
    
    async def create_deletion_request(self, user_id: int,
                                     deletion_type: str = "full",
                                     data_types: Optional[List[str]] = None) -> DataDeletionRequest:
        """Create a data deletion request."""
        request = DataDeletionRequest(
            user_id=user_id,
            deletion_type=deletion_type,
            data_types=data_types,
            status="pending",
            created_at=datetime.utcnow(),
        )
        self.db.add(request)
        self.db.commit()
        self.db.refresh(request)
        return request
    
    async def process_deletion(self, request_id: int) -> DataDeletionRequest:
        """Process a deletion request with cryptographic proof."""
        request = self.db.query(DataDeletionRequest).filter(
            DataDeletionRequest.id == request_id
        ).first()
        if not request:
            raise ValueError(f"Deletion request {request_id} not found")
        
        request.status = "processing"
        self.db.commit()
        
        try:
            # Gather data to be deleted
            data_to_delete = await self._gather_data_for_deletion(
                request.user_id, request.data_types
            )
            
            # Compute deletion manifest hash BEFORE deletion
            manifest = {
                "user_id": request.user_id,
                "deletion_type": request.deletion_type,
                "data_types": request.data_types,
                "items": data_to_delete,
                "timestamp": datetime.utcnow().isoformat(),
            }
            manifest_hash = hashlib.sha256(
                json.dumps(manifest, sort_keys=True, default=str).encode()
            ).hexdigest()
            
            # Perform deletion (stub — real implementation deletes from each domain)
            items_deleted = await self._execute_deletion(
                request.user_id, request.data_types
            )
            
            # Record proof
            request.deletion_proof_hash = manifest_hash
            request.items_deleted_count = items_deleted
            request.status = "complete"
            request.completed_at = datetime.utcnow()
            self.db.commit()
            self.db.refresh(request)
            
            return request
            
        except Exception as e:
            request.status = "failed"
            request.error_message = str(e)
            self.db.commit()
            self.db.refresh(request)
            return request
    
    async def _gather_data_for_deletion(self, user_id: int,
                                       data_types: Optional[List[str]] = None) -> Dict[str, Any]:
        """Gather information about data to be deleted."""
        all_types = ["memories", "files", "settings", "agents", "workflows"]
        types_to_delete = data_types or all_types
        
        result = {}
        for dtype in types_to_delete:
            result[dtype] = {"status": "identified"}
        return result
    
    async def _execute_deletion(self, user_id: int,
                               data_types: Optional[List[str]] = None) -> int:
        """Execute the actual deletion. Returns count of items deleted."""
        # In production, this calls each domain service's delete method
        # For now, return 0 (no-op until domain services exist)
        return 0
    
    async def verify_deletion(self, request_id: int) -> Dict[str, Any]:
        """Verify deletion proof by checking manifest hash."""
        request = self.db.query(DataDeletionRequest).filter(
            DataDeletionRequest.id == request_id
        ).first()
        if not request:
            raise ValueError(f"Deletion request {request_id} not found")
        
        return {
            "request_id": request.id,
```

## Security Considerations

1. **Audit log immutability:** Only `delete_old_logs` (retention policy) may delete from `audit_logs`. All other code must use append-only pattern.
2. **Export file security:** Export files stored in `/tmp` with restrictive permissions. Auto-deleted after 7 days.
3. **Deletion proof integrity:** SHA-256 manifest hash is computed BEFORE deletion and stored permanently. Cannot be tampered with.
4. **Export size limit:** 100MB cap prevents resource exhaustion from massive data exports.
5. **Consent before export:** Export service should verify `export` consent before processing.

## Cross-Domain Integration

- **Memory domain (v1.03):** Export gathers memories; deletion deletes memories; audit logs all memory operations
- **File awareness domain (v1.04):** Export gathers file metadata; deletion removes file references
- **Event system (v1.02):** Export completion emits `export.complete`; deletion emits `data.deleted` with proof hash
- **Agent system (v1.01):** Agent actions logged to audit; agent execution history included in exports
