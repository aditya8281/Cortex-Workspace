### Task 1: Create Privacy API Router Registration

**Architecture:** Privacy API is a sub-router mounted at `/api/v1/privacy/`. It includes sub-routers for audit, consent, export, transparency, and access control. All endpoints require authentication via JWT.

```python
# backend/app/api/v1/privacy/__init__.py

from fastapi import APIRouter
from backend.app.api.v1.privacy import audit, consent, export, transparency, access_control

router = APIRouter()
router.include_router(audit.router, prefix="/audit", tags=["Privacy - Audit Logging"])
router.include_router(consent.router, prefix="/consent", tags=["Privacy - Consent Management"])
router.include_router(export.router, prefix="/export", tags=["Privacy - Data Export"])
router.include_router(transparency.router, prefix="/transparency", tags=["Privacy - Transparency"])
router.include_router(access_control.router, prefix="/access", tags=["Privacy - Access Control"])
```

The main API router must include the privacy router:

```python
# In backend/app/api/v1/__init__.py (modify existing)
from backend.app.api.v1.privacy import router as privacy_router
api_router.include_router(privacy_router, prefix="/privacy", tags=["Privacy"])
```

**Performance Considerations:** All endpoints use `response_model=` on decorators for automatic response validation and Swagger schema generation.

**Risk Note:** Route ordering matters — specific routes before parameterized routes. Privacy routes must not conflict with existing routes.

### Task 2: Create Audit API Endpoints

```python
# backend/app/api/v1/privacy/audit.py

from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session
from typing import Optional
from datetime import datetime
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.privacy.audit import AuditLoggingService
from backend.app.schemas.privacy.audit import AuditLogResponse, AuditLogListResponse

router = APIRouter()


@router.get("/logs", response_model=AuditLogListResponse)
async def get_audit_logs(
    limit: int = Query(100, le=1000, description="Max logs to return"),
    offset: int = Query(0, ge=0, description="Offset for pagination"),
    action: Optional[str] = Query(None, description="Filter by action type"),
    resource_type: Optional[str] = Query(None, description="Filter by resource type"),
    start_time: Optional[datetime] = Query(None, description="Start of time range"),
    end_time: Optional[datetime] = Query(None, description="End of time range"),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Get audit logs for the current user with filtering and pagination.

    Uses composite index on (user_id, timestamp) for efficient queries.
    All parameters are optional — defaults return most recent logs.
    """
    service = AuditLoggingService(db)
    logs = await service.get_user_logs(
        user_id=current_user.id,
        limit=limit,
        offset=offset,
        action_filter=action,
        resource_type_filter=resource_type,
        start_time=start_time,
        end_time=end_time,
    )
    total = await service.count_user_logs(current_user.id)
    return AuditLogListResponse(
        items=[AuditLogResponse.model_validate(log) for log in logs],
        total=total,
        page=(offset // limit) + 1,
        page_size=limit,
    )


@router.get("/logs/resource/{resource_type}/{resource_id}", response_model=list[AuditLogResponse])
async def get_resource_audit_logs(
    resource_type: str,
    resource_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Get audit logs for a specific resource."""
    service = AuditLoggingService(db)
    logs = await service.get_resource_logs(resource_type, resource_id)
    return [AuditLogResponse.model_validate(log) for log in logs]


@router.get("/stats")
async def get_audit_stats(
    days: int = Query(30, ge=1, le=365, description="Stats period in days"),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Get action statistics for the current user."""
    service = AuditLoggingService(db)
    return await service.get_action_stats(current_user.id, days=days)


@router.get("/recent", response_model=list[AuditLogResponse])
async def get_recent_activity(
    limit: int = Query(20, le=100),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Get most recent activity for dashboard display."""
    service = AuditLoggingService(db)
    logs = await service.get_recent_activity(current_user.id, limit=limit)
    return [AuditLogResponse.model_validate(log) for log in logs]
```

**Security Considerations:** All endpoints enforce ownership: `user_id=current_user.id` filter on all queries. Users cannot view other users' audit logs.

```python
# backend/app/api/v1/privacy/consent.py

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import Optional

from backend.app.auth.dependencies import get_current_user
from backend.app.services.privacy.access_control import AccessControlService
from backend.app.schemas.privacy.consent import ConsentCreate, ConsentResponse

router = APIRouter()


@router.get("/", response_model=list[ConsentResponse])
async def list_consents(
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """List all consents for the current user."""
    service = AccessControlService(db)
    consents = await service.get_active_consents(current_user.id)
    return [ConsentResponse.model_validate(c) for c in consents]


@router.post("/grant", response_model=ConsentResponse)
async def grant_consent(
    body: ConsentCreate,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Grant consent for a data category."""
    service = AccessControlService(db)
    consent = await service.grant_consent(
        user_id=current_user.id,
        consent_type=body.consent_type,
        scope=body.scope,
        context=body.context,
        expires_at=body.expires_at,
    )
    return ConsentResponse.model_validate(consent)


@router.post("/revoke")
async def revoke_consent(
    consent_type: str = Query(..., description="Consent type to revoke"),
    reason: Optional[str] = Query(None, description="Revocation reason"),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Revoke active consent."""
    service = AccessControlService(db)
    success = await service.revoke_consent(
        user_id=current_user.id,
        consent_type=consent_type,
        reason=reason,
    )
    return {"success": success, "consent_type": consent_type}


@router.get("/check")
async def check_consent(
    consent_type: str = Query(..., description="Consent type to check"),
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),

    """Check if user has active consent for a type."""
    service = AccessControlService(db)
    has_consent = await service.check_consent(current_user.id, consent_type)
    return {"consent_type": consent_type, "granted": has_consent}
```

```python
# backend/app/api/v1/privacy/export.py

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import Optional, List
from backend.app.db.session import get_db
from backend.app.auth.dependencies import get_current_user
from backend.app.services.privacy.export import DataExportService
from backend.app.schemas.privacy.export import ExportCreate, ExportResponse

router = APIRouter()


@router.post("/create", response_model=ExportResponse)
async def create_export(
    body: ExportCreate,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Create a data export request."""
    service = DataExportService(db)
    export = await service.create_export(
        user_id=current_user.id,
        export_type=body.export_type,
        data_types=body.data_types,
        format=body.format,
    )
    return ExportResponse.model_validate(export)


@router.post("/{export_id}/process", response_model=ExportResponse)
async def process_export(
    export_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Process a data export (gather, serialize, store)."""
    service = DataExportService(db)
    export = await service.process_export(export_id)
    if export.user_id != current_user.id:
        raise HTTPException(status_code=404, detail="Export not found")
    return ExportResponse.model_validate(export)


@router.get("/{export_id}/verify")
async def verify_export(
    export_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """Verify export integrity via checksum."""
    service = DataExportService(db)
    result = await service.verify_export(export_id)
    return result
```

```typescript
import { apiClient } from '@/shared/api/client';

export interface AuditLog {
  id: number;
  user_id: number;
  action: string;

  details: Record<string, any> | null;

  success: number;
  error_message: string | null;
  duration_ms: number | null;
}

export interface AuditLogListResponse {
  items: AuditLog[];
  total: number;
  page: number;
  page_size: number;
}

export interface Consent {
  id: number;
  user_id: number;
  consent_type: string;
  granted: number;
  scope: string | null;
  context: Record<string, any> | null;
  created_at: string;
  expires_at: string | null;
  revoked_at: string | null;
  revoked_reason: string | null;
  version: number;
}

export interface DataExport {
  id: number;
  user_id: number;
  export_type: string;
  data_types: string[] | null;
  format: string;
  status: string;
  file_path: string | null;
  file_size_bytes: number | null;
  checksum_sha256: string | null;
  created_at: string;
  completed_at: string | null;
}

export interface DecisionExplanation {
  decision_type: string;
  reasoning: string;
  factors: Array<{ key: string; value: any; weight: number; description: string | null }>;
  confidence: number;
  alternatives_considered: string[];
  recommendation: string;
  risk_level: string;
}

export interface AccessCheckResult {
  allowed: boolean;
  resource_type: string;
  action: string;
}

export const privacyApi = {
  // --- Audit ---
  getAuditLogs: (params?: { limit?: number; offset?: number; action?: string; resource_type?: string }) => {
    const query = new URLSearchParams();
    if (params?.limit) query.set('limit', String(params.limit));
    if (params?.offset) query.set('offset', String(params.offset));
    if (params?.action) query.set('action', params.action);
    if (params?.resource_type) query.set('resource_type', params.resource_type);
    return apiClient.get<AuditLogListResponse>(`/privacy/audit/logs?${query.toString()}`);
  },

  getAuditStats: (days: number = 30) =>
    apiClient.get(`/privacy/audit/stats?days=${days}`),

  getRecentActivity: (limit: number = 20) =>
    apiClient.get<AuditLog[]>(`/privacy/audit/recent?limit=${limit}`),

  getResourceLogs: (resourceType: string, resourceId: number) =>
    apiClient.get<AuditLog[]>(`/privacy/audit/logs/resource/${resourceType}/${resourceId}`),

  // --- Consent ---
  listConsents: () =>
    apiClient.get<Consent[]>('/privacy/consent/'),

  grantConsent: (consentType: string, scope?: string, expiresAt?: string) =>
    apiClient.post<Consent>('/privacy/consent/grant', {
      consent_type: consentType,
      scope: scope || null,
      expires_at: expiresAt || null,
    }),

  revokeConsent: (consentType: string, reason?: string) =>
    apiClient.post(`/privacy/consent/revoke?consent_type=${consentType}${reason ? `&reason=${encodeURIComponent(reason)}` : ''}`),

  checkConsent: (consentType: string) =>
    apiClient.get<{ consent_type: string; granted: boolean }>(`/privacy/consent/check?consent_type=${consentType}`),

  // --- Export ---
  createExport: (exportType: string = 'full', format: string = 'json', dataTypes?: string[]) =>
    apiClient.post<DataExport>('/privacy/export/create', {
      export_type: exportType,
      format,
      data_types: dataTypes || null,
    }),

  processExport: (exportId: number) =>
    apiClient.post<DataExport>(`/privacy/export/${exportId}/process`),

  verifyExport: (exportId: number) =>
    apiClient.get(`/privacy/export/${exportId}/verify`),

  // --- Transparency ---
  explainDecision: (decisionType: string, context: Record<string, any>) =>
    apiClient.post<DecisionExplanation>('/privacy/transparency/explain', {
      decision_type: decisionType,
      context,
    }),

  // --- Access Control ---
  checkAccess: (resourceType: string, action: string) =>
    apiClient.get<AccessCheckResult>(`/privacy/access/check?resource_type=${resourceType}&action=${action}`),

  getMyRoles: () =>
    apiClient.get<Array<{ id: number; name: string; description: string }>>('/privacy/access/roles'),

  getMyPermissions: () =>
    apiClient.get<Array<{ resource_type: string; action: string; description: string }>>('/privacy/access/permissions'),
};
```

```python
import pytest
from backend.app.services.privacy.audit import AuditLoggingService
from backend.app.services.privacy.access_control import AccessControlService
from backend.app.services.privacy.export import DataExportService
from backend.app.models.privacy.role import Role, Permission


@pytest.mark.asyncio
async def test_full_privacy_flow(db_session):
    """Integration test: grant consent → access data → audit log → export."""
    # 1. Grant consent
    ac_service = AccessControlService(db_session)
    consent = await ac_service.grant_consent(1, "memory_read", scope="all")
    assert consent.granted == 1

    # 2. Check consent
    has_consent = await ac_service.check_consent(1, "memory_read")
    assert has_consent is True

    # 3. Log the access
    audit_service = AuditLoggingService(db_session)
    log = await audit_service.log(
        user_id=1, action="read", resource_type="memory",
        details={"consent_type": "memory_read", "consent_verified": True}
    )
    assert log.action == "read"

    # 4. Create export
    export_service = DataExportService(db_session)
    export = await export_service.create_export(user_id=1, export_type="full")
    assert export.status == "pending"

    # 5. Verify audit trail
    logs = await audit_service.get_user_logs(1)
    assert len(logs) >= 1
    assert logs[0].action == "read"


@pytest.mark.asyncio
async def test_rbac_permission_flow(db_session):
    """Integration test: create role → assign → check access → deny without role."""
    ac_service = AccessControlService(db_session)

    # Create role with permissions
    role = Role(name="test_editor")
    perm = Permission(resource_type="file", action="write")
    role.permissions.append(perm)
    db_session.add(role)
    db_session.commit()

    # Initially no access
    assert await ac_service.check_permission(1, "file", "write") is False

    # Assign role
    await ac_service.assign_role(1, "test_editor")

    # Now has access
    assert await ac_service.check_permission(1, "file", "write") is True

    # Remove role
    await ac_service.remove_role(1, "test_editor")
    assert await ac_service.check_permission(1, "file", "write") is False


@pytest.mark.asyncio
async def test_consent_gates_data_access(db_session):
    """Integration test: consent required before data access."""
    ac_service = AccessControlService(db_session)

    # No consent → no access
    has_consent = await ac_service.check_consent(1, "file_read")
    assert has_consent is False

    # Grant consent
    await ac_service.grant_consent(1, "file_read", scope="user_files")

    # Now has access
    has_consent = await ac_service.check_consent(1, "file_read")
    assert has_consent is True

    # Revoke
    await ac_service.revoke_consent(1, "file_read")
    has_consent = await ac_service.check_consent(1, "file_read")
    assert has_consent is False
```

## Security Considerations

1. **Authentication required:** All endpoints use `Depends(get_current_user)` — unauthenticated requests get 401.
2. **Ownership enforcement:** Audit log queries filter by `user_id=current_user.id`. Users cannot view others' logs.
3. **Input validation:** Pydantic schemas validate all inputs. Query parameters have bounds (`le=1000`, `ge=0`).
4. **Export ownership:** Process endpoint checks `export.user_id != current_user.id` before processing.
5. **No sensitive data in responses:** Audit logs include IP and user agent but no passwords or tokens.

## Cross-Domain Integration

- **Memory domain (v1.03):** Memory CRUD calls `check_consent` before operations; memory access logged to audit
- **File awareness domain (v1.04):** File operations check consent; file access logged to audit; file data included in export
- **Event system (v1.02):** Consent changes emit events; export completion emits event; audit entries trigger logging events
- **Agent system (v1.01):** Agent actions go through access control; agent output included in exports
