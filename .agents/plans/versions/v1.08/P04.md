### Task 2: Implement Stub Calendar Provider

**File:** `backend/app/services/awareness/calendar_providers/stub.py`
**Purpose:** In-memory calendar provider for development and testing. Stores events in a list. Satisfies CalendarProvider protocol.

```python
# backend/app/services/awareness/calendar_providers/stub.py

from typing import Dict, List
from datetime import datetime, timedelta


class StubCalendarProvider:
    """In-memory calendar provider for development/testing.
    
    Satisfies CalendarProvider protocol. No external dependencies.
    """
    
    def __init__(self):
        self._events: List[Dict] = []
        self._next_id = 1
    
    async def get_events(self, start: datetime, end: datetime) -> List[Dict]:
        return [
            e for e in self._events
            if start <= datetime.fromisoformat(e["start"]) <= end
        ]
    
    async def create_event(self, title: str, start: datetime, end: datetime,
                           description: str = "", location: str = "") -> Dict:
        event = {
            "id": str(self._next_id),
            "title": title,
            "start": start.isoformat(),
            "end": end.isoformat(),
            "description": description,
            "location": location,
            "created": datetime.utcnow().isoformat()
        }
        self._events.append(event)
        self._next_id += 1
        return event
    
    async def delete_event(self, event_id: str) -> bool:
        before = len(self._events)
        self._events = [e for e in self._events if e["id"] != event_id]
        return len(self._events) < before
    
    async def get_upcoming(self, hours: int = 24) -> List[Dict]:
        now = datetime.utcnow()
        end = now + timedelta(hours=hours)
        return await self.get_events(now, end)
```

### Task 3: Implement Stub Email Provider

**File:** `backend/app/services/awareness/email_providers/stub.py`
**Purpose:** In-memory email provider for development and testing.

```python
# backend/app/services/awareness/email_providers/stub.py

from typing import Dict, List, Optional
from datetime import datetime


class StubEmailProvider:
    """In-memory email provider for development/testing.
    
    Satisfies EmailProvider protocol. No external dependencies.
    """
    
    def __init__(self):
        self._emails: List[Dict] = []
        self._next_id = 1
    
    async def get_recent(self, limit: int = 20) -> List[Dict]:
        return self._emails[-limit:]
    
    async def search(self, query: str) -> List[Dict]:
        q = query.lower()
        return [
            e for e in self._emails
            if q in e.get("subject", "").lower() or
               q in e.get("preview", "").lower() or
               q in e.get("sender", "").lower()
        ]
    
    async def get_unread_count(self) -> int:
        return sum(1 for e in self._emails if not e.get("read", False))
    
    async def get_email(self, email_id: str) -> Optional[Dict]:
        for e in self._emails:
            if str(e["id"]) == email_id:
                return e
        return None
    
    async def record_email(self, subject: str, sender: str,
                           preview: str = "") -> Dict:
        email = {
            "id": str(self._next_id),
            "subject": subject,
            "sender": sender,
            "preview": preview,
            "timestamp": datetime.utcnow().isoformat(),
            "read": False
        }
        self._emails.append(email)
        self._next_id += 1
        return email
```

### Task 4: Implement Calendar Awareness Service

**File:** `backend/app/services/awareness/calendar.py`
**Purpose:** Calendar awareness service using pluggable provider. Defaults to stub provider. Can be configured to use CalDAV or Google Calendar in production.

```python
# backend/app/services/awareness/calendar.py

from typing import Dict, List, Optional
from datetime import datetime, timedelta
from backend.app.services.awareness.protocols import CalendarProvider


class CalendarAwarenessService:
    """Monitors calendar events via pluggable provider.
    
    Default provider: StubCalendarProvider (development).
    Production providers: CalDAVProvider, GoogleCalendarProvider.
    
    Architecture ref: 1.5 (Plugin Boundaries) — Protocol-based providers.
    1.3 (Daemon-Background polling via arq for calendar sync.
    """
    
    def __init__(self, provider: CalendarProvider = None):
        if provider is None:
            from backend.app.services.awareness.calendar_providers.stub import StubCalendarProvider
            provider = StubCalendarProvider()
        self._provider = provider
    
    async def get_upcoming(self, hours: int = 24) -> List[Dict]:
        """Get upcoming events within N hours."""
        return await self._provider.get_upcoming(hours)
    
    async def get_events(self, start: datetime, end: datetime) -> List[Dict]:
        """Get events in time range."""
        return await self._provider.get_events(start, end)
    
    async def add_event(self, title: str, start: datetime, end: datetime,
                        description: str = "", location: str = "") -> Dict:
        """Add a calendar event."""
        return await self._provider.create_event(title, start, end, description, location)
    
    async def delete_event(self, event_id: str) -> bool:
        """Delete a calendar event."""
        return await self._provider.delete_event(event_id)
    
    async def get_next_event(self) -> Optional[Dict]:
        """Get the next upcoming event."""
        events = await self._provider.get_upcoming(hours=168)  # Next 7 days
        return events[0] if events else None
    
    async def check_time_conflicts(self, start: datetime, end: datetime) -> List[Dict]:
        """Check for time conflicts in the given range."""
        existing = await self._provider.get_events(start, end)
        return existing
```

```python
# backend/app/services/awareness/email.py

from typing import Dict, List, Optional
from backend.app.services.awareness.protocols import EmailProvider


class EmailAwarenessService:
    """Monitors email activity via pluggable provider.
    
    Default provider: StubEmailProvider (development).
    Production providers: IMAPProvider, ExchangeProvider, GmailProvider.
    
    Architecture ref: 1.5 (Plugin Boundaries) — Protocol-based providers.
    """
    
    def __init__(self, provider: EmailProvider = None):
        if provider is None:
            from backend.app.services.awareness.email_providers.stub import StubEmailProvider
            provider = StubEmailProvider()
        self._provider = provider
    
    async def get_recent_emails(self, limit: int = 20) -> List[Dict]:
        """Get recent emails."""
        return await self._provider.get_recent(limit)
    
    async def get_unread_count(self) -> int:
        """Get unread email count."""
        return await self._provider.get_unread_count()
    
    async def search_emails(self, query: str) -> List[Dict]:
        """Search emails."""
        return await self._provider.search(query)
    
    async def get_email(self, email_id: str) -> Optional[Dict]:
        """Get a specific email."""
        return await self._provider.get_email(email_id)
    
    async def get_email_summary(self) -> Dict:
        """Get email activity summary."""
        recent = await self._provider.get_recent(limit=50)
        unread = await self._provider.get_unread_count()
        
        # Group by sender
        senders = {}
        for email in recent:
            sender = email.get("sender", "unknown")
            senders[sender] = senders.get(sender, 0) + 1
        
        return {
            "total_recent": len(recent),
            "unread_count": unread,
            "top_senders": dict(sorted(senders.items(), key=lambda x: x[1], reverse=True)[:10])
        }
```

## Security Considerations

1. **Provider credentials** — Never stored in code. Use environment variables or vault.
2. **Email content** — May contain sensitive data. Store summaries only, not full content.
3. **Calendar access** — Read-only by default. Write operations require explicit user consent.
4. **API rate limits** — Respect provider API rate limits. Implement exponential backoff.

---

## Performance Considerations

1. **Provider latency** — External APIs (CalDAV, IMAP) have 100-500ms latency. Cache results for 5 minutes.
2. **Email search** — Full-text search on large mailboxes is slow. Use provider-native search when available.
3. **Polling frequency** — Calendar polled every 5 minutes. Email polled every 2 minutes.
4. **Background sync** — Use arq jobs for background calendar/email sync. Don't block API requests.
