### Task 1: Implement System Resource Awareness Service

**File:** `backend/app/services/awareness/system.py`
**Purpose:** Monitor system resources: CPU, memory, disk, network. Uses `psutil` for cross-platform resource monitoring.

```python
# backend/app/services/awareness/system.py

import platform
from typing import Dict
from backend.app.services.awareness.subprocess_util import SafeSubprocess


class SystemResourceService:
    """Monitors system resource usage.
    
    Features:
    - CPU usage (overall and per-core)
    - Memory usage (total, used, available)
    - Disk usage (total, used, free per mount)
    - Network status (connectivity check)
    
    Architecture ref: 1.3 (Daemon-Background monitoring.
    1.7 (Incremental Safety) — resource limits prevent runaway monitoring.
    """
    
    async def get_cpu_usage(self) -> Dict:
        """Get CPU usage statistics."""
        # Try psutil first (if installed)
        try:
            import psutil
            return {
                "percent": psutil.cpu_percent(interval=0.1),
                "count": psutil.cpu_count(),
                "per_core": psutil.cpu_percent(interval=0.1, percpu=True)
            }
        except ImportError:
            pass
        
        # Fallback: read /proc/stat on Linux
        if platform.system() == "Linux":
            result = await SafeSubprocess.run(["cat", "/proc/stat"])
            if result:
                lines = result.split("\n")
                cpu_line = lines[0] if lines else ""
                parts = cpu_line.split()
                if len(parts) >= 5:
                    user, nice, system, idle = int(parts[1]), int(parts[2]), int(parts[3]), int(parts[4])
                    total = user + nice + system + idle
                    used = user + system
                    return {"percent": round(used / total * 100, 1) if total > 0 else 0, "count": "unknown"}
        
        return {"percent": 0, "count": "unknown", "status": "monitoring unavailable"}
    
    async def get_memory_usage(self) -> Dict:
        """Get memory usage statistics."""
        try:
            import psutil
            mem = psutil.virtual_memory()
            return {
                "total_gb": round(mem.total / (1024**3), 2),
                "used_gb": round(mem.used / (1024**3), 2),
                "available_gb": round(mem.available / (1024**3), 2),
                "percent": mem.percent
            }
        except ImportError:
            pass
        
        # Fallback: read /proc/meminfo on Linux
        if platform.system() == "Linux":
            result = await SafeSubprocess.run(["cat", "/proc/meminfo"])
            if result:
                info = {}
                for line in result.split("\n"):
                    parts = line.split(":")
                    if len(parts) == 2:
                        key = parts[0].strip()
                        val = parts[1].strip().split()[0]  # Get kB value
                        info[key] = int(val) if val.isdigit() else 0
                
                total = info.get("MemTotal", 0)
                available = info.get("MemAvailable", 0)
                used = total - available
                return {
                    "total_gb": round(total / (1024**2), 2),
                    "used_gb": round(used / (1024**2), 2),
                    "available_gb": round(available / (1024**2), 2),
                    "percent": round(used / total * 100, 1) if total > 0 else 0
                }
        
        return {"total_gb": 0, "used_gb": 0, "percent": 0, "status": "monitoring unavailable"}
    
    async def get_disk_usage(self) -> Dict:
        """Get disk usage for primary mount."""
        try:
            import psutil
            disk = psutil.disk_usage("/")
            return {
                "total_gb": round(disk.total / (1024**3), 2),
                "used_gb": round(disk.used / (1024**3), 2),
                "free_gb": round(disk.free / (1024**3), 2),
                "percent": round(disk.percent, 1)
            }
        except ImportError:
            pass
        
        # Fallback: df command
        result = await SafeSubprocess.run(["df", "-h", "/"])
        if result:
            lines = result.split("\n")
            if len(lines) >= 2:
                parts = lines[1].split()
                if len(parts) >= 5:
                    return {
                        "total_gb": parts[1],
                        "used_gb": parts[2],
                        "free_gb": parts[3],
                        "percent": float(parts[4].replace("%", ""))
                    }
        
        return {"total_gb": 0, "used_gb": 0, "free_gb": 0, "percent": 0}
    
    async def get_network_status(self) -> Dict:
        """Check network connectivity."""
        # Try ping
        result = await SafeSubprocess.run(
            ["ping", "-c", "1", "-W", "2", "8.8.8.8"],
            timeout=3.0
        )
        connected = result is not None and "1 received" in result
        
        return {
            "connected": connected,
            "ping_target": "8.8.8.8"
        }
    
    async def get_system_summary(self) -> Dict:
        """Get complete system resource summary."""
        cpu = await self.get_cpu_usage()
        memory = await self.get_memory_usage()
        disk = await self.get_disk_usage()
        network = await self.get_network_status()
        
        return {
            "cpu": cpu,
            "memory": memory,
            "disk": disk,
            "network": network,
            "platform": platform.system(),
            "python_version": platform.python_version()
        }
```

## Security Considerations

1. **Authentication required** — All endpoints use `Depends(get_current_user)`.
2. **Data sensitivity** — Clipboard may contain passwords. Log warning but don't store.
3. **System info exposure** — System resource endpoints reveal hardware details. Rate limit to 1/second.
4. **File system access** — Workspace endpoints are read-only. No write operations.

---

## Performance Considerations

1. **Response time** — Awareness endpoints should respond in <500ms. Cache where possible.
2. **Polling frequency** — Frontend should poll awareness data every 5-30 seconds depending on type.
3. **Background sync** — Heavy operations (workspace walk, file stats) run in background.
4. **Response size** — Limit all list responses. Don't return 1000 files in one response.
