### Task 1: Implement Browser Awareness Service

**File:** `backend/app/services/awareness/browser.py`
**Purpose:** Monitor browser activity via Chrome DevTools Protocol (CDP) and Firefox remote debugging. Connects to browsers that expose debugging ports. Graceful fallback when connection refused.

```python
# backend/app/services/awareness/browser.py

import json
from typing import Dict, List, Optional
from backend.app.services.awareness.subprocess_util import SafeSubprocess


class BrowserAwarenessService:
    """Monitors browser activity via debugging protocols.
    
    Supported browsers:
    - Chrome/Chromium: Chrome DevTools Protocol (CDP) on port 9222
    - Firefox: Remote debugging protocol on port 6000
    
    Both require the browser to be launched with debugging enabled:
    - Chrome: google-chrome --remote-debugging-port=9222
    - Firefox: about:config → devtools.debugger.remote-enabled=true
    
    Architecture ref: 1.2 (Graceful Degradation) — connection refused → empty tabs.
    """
    
    CDP_PORT = 9222
    FIREFOX_PORT = 6000
    CONNECTION_TIMEOUT = 3.0  # seconds
    
    async def get_browser_tabs(self) -> List[Dict]:
        """Get open tabs from all available browsers."""
        tabs = []
        
        chrome_tabs = await self._get_chrome_tabs()
        tabs.extend([{**t, "browser": "chrome"} for t in chrome_tabs])
        
        firefox_tabs = await self._get_firefox_tabs()
        tabs.extend([{**t, "browser": "firefox"} for t in firefox_tabs])
        
        return tabs
    
    async def get_current_page(self) -> Optional[Dict]:
        """Get the currently active browser page."""
        tabs = await self.get_browser_tabs()
        # Return the first tab (most recently active)
        return tabs[0] if tabs else None
    
    async def get_browser_status(self) -> Dict:
        """Get status of browser connections."""
        chrome_ok = await self._check_chrome_connection()
        firefox_ok = await self._check_firefox_connection()
        
        return {
            "chrome": {"available": chrome_ok, "port": self.CDP_PORT},
            "firefox": {"available": firefox_ok, "port": self.FIREFOX_PORT},
        }
    
    async def _get_chrome_tabs(self) -> List[Dict]:
        """Get Chrome tabs via CDP HTTP endpoint."""
        if not await self._check_chrome_connection():
            return []
        
        result = await SafeSubprocess.run(
            ["curl", "-s", f"http://localhost:{self.CDP_PORT}/json"],
            timeout=self.CONNECTION_TIMEOUT
        )
        
        if not result:
            return []
        
        try:
            targets = json.loads(result)
            return [
                {
                    "id": t.get("id", ""),
                    "title": t.get("title", "Untitled"),
                    "url": t.get("url", ""),
                    "type": t.get("type", "page"),
                    "active": t.get("type") == "page"
                }
                for t in targets if t.get("type") == "page"
            ]
        except (json.JSONDecodeError, KeyError):
            return []
    
    async def _get_firefox_tabs(self) -> List[Dict]:
        """Get Firefox tabs via remote debugging."""
        if not await self._check_firefox_connection():
            return []
        
        # Firefox remote debugging uses a JSON endpoint
        result = await SafeSubprocess.run(
            ["curl", "-s", f"http://localhost:{self.FIREFOX_PORT}/json/list"],
            timeout=self.CONNECTION_TIMEOUT
        )
        
        if not result:
            return []
        
        try:
            targets = json.loads(result)
            return [
                {
                    "id": t.get("id", ""),
                    "title": t.get("title", "Untitled"),
                    "url": t.get("url", ""),
                    "type": t.get("type", "page"),
                    "active": t.get("type") == "page"
                }
                for t in targets if t.get("type") == "page"
            ]
        except (json.JSONDecodeError, KeyError):
            return []
    
    async def _check_chrome_connection(self) -> bool:
        """Check if Chrome CDP is accessible."""
        result = await SafeSubprocess.run(
            ["curl", "-s", "-o", "/dev/null", "-w", "%{http_code}",
             f"http://localhost:{self.CDP_PORT}/json/version"],
            timeout=self.CONNECTION_TIMEOUT
        )
        return result == "200"
    
    async def _check_firefox_connection(self) -> bool:
        """Check if Firefox remote debugging is accessible."""
        result = await SafeSubprocess.run(
            ["curl", "-s", "-o", "/dev/null", "-w", "%{http_code}",
             f"http://localhost:{self.FIREFOX_PORT}/json/version"],
            timeout=self.CONNECTION_TIMEOUT
        )
        return result == "200"
```

### Task 2: Implement Clipboard Awareness Service

**File:** `backend/app/services/awareness/clipboard.py`
**Purpose:** Read, write, and watch clipboard content. Platform-specific commands: `xclip` on Linux, `pbpaste`/`pbcopy` on macOS. Polling-based clipboard watching with proper asyncio sleep.

```python
# backend/app/services/awareness/clipboard.py

import asyncio
from typing import Optional, Callable, Awaitable
from backend.app.services.awareness.platform import PlatformDetector, OSType
from backend.app.services.awareness.subprocess_util import SafeSubprocess


class ClipboardAwarenessService:
    """Monitors and manages clipboard content.
    
    Platform support:
    - Linux: xclip (X11) or wl-clipboard (Wayland)
    - macOS: pbpaste / pbcopy
    
    Features:
    - Read clipboard content
    - Write to clipboard
    - Watch for changes (polling-based)
    
    Architecture ref: 1.1 (Local-First) — OS-native, no cloud dependency.
    1.3 (Daemon-First) — polling via asyncio.sleep, not blocking.
    """
    
    MAX_CLIPBOARD_SIZE = 10000  # Max characters to return
    DEFAULT_POLL_INTERVAL = 1.0  # seconds
    
    def __init__(self):
        self._platform = PlatformDetector()
        self._last_content: Optional[str] = None
    
    async def get_clipboard(self) -> Optional[str]:
        """Get current clipboard content."""
        content = None
        
        if self._platform.os_type == OSType.LINUX:
            content = await self._read_linux_clipboard()
        elif self._platform.os_type == OSType.DARWIN:
            content = await self._read_macos_clipboard()
        
        if content and len(content) > self.MAX_CLIPBOARD_SIZE:
            content = content[:self.MAX_CLIPBOARD_SIZE] + "... [truncated]"
        
        self._last_content = content
        return content
    
    async def set_clipboard(self, text: str) -> bool:
        """Set clipboard content."""
        if self._platform.os_type == OSType.LINUX:
            return await self._write_linux_clipboard(text)
        elif self._platform.os_type == OSType.DARWIN:
            return await self._write_macos_clipboard(text)
        return False
    
    async def watch_clipboard(self, callback: Callable[[Optional[str]], Awaitable[None]],
                               interval: float = None) -> None:
        """Watch clipboard for changes. Calls callback when content changes.
        
        Runs indefinitely until cancelled. Yields control via asyncio.sleep.
        """
        interval = interval or self.DEFAULT_POLL_INTERVAL
        self._last_content = await self.get_clipboard()
        
        while True:
            current = await self.get_clipboard()
            if current != self._last_content:
                await callback(current)
                self._last_content = current
            await asyncio.sleep(interval)
    
    async def _read_linux_clipboard(self) -> Optional[str]:
        """Read clipboard via xclip."""
        if self._platform.tool_available("xclip"):
            return await SafeSubprocess.run(
                ["xclip", "-selection", "clipboard", "-o"]
            )
        # Try wl-paste for Wayland
        if self._platform.tool_available("wl-paste"):
            return await SafeSubprocess.run(["wl-paste"])
        return None
    
    async def _write_linux_clipboard(self, text: str) -> bool:
        """Write clipboard via xclip."""
        if self._platform.tool_available("xclip"):
            return await SafeSubprocess.run_popen(
                ["xclip", "-selection", "clipboard"], input_data=text
            )
        if self._platform.tool_available("wl-copy"):
            return await SafeSubprocess.run_popen(
                ["wl-copy"], input_data=text
            )
        return False
    
    async def _read_macos_clipboard(self) -> Optional[str]:
        """Read clipboard via pbpaste."""
        return await SafeSubprocess.run(["pbpaste"])
    
    async def _write_macos_clipboard(self, text: str) -> bool:
        """Write clipboard via pbcopy."""
        return await SafeSubprocess.run_popen(["pbcopy"], input_data=text)
```

## Security Considerations

1. **Browser data privacy** — All tab data is local-only. No external transmission.
2. **Clipboard sensitivity** — Clipboard may contain passwords. Return content as-is but log warning if content looks like a password pattern.
3. **Connection security** — CDP connections are localhost-only. No remote connections.
4. **User opt-in** — Browser monitoring requires explicit opt-in. Default to disabled.

---

## Performance Considerations

1. **CDP connection** — Each CDP request has ~100ms latency. Cache tab list for 5 seconds.
2. **Clipboard polling** — 1-second interval is sufficient. Don't poll faster.
3. **Curl overhead** — Each curl call has ~50ms overhead. Consider using `aiohttp` for persistent connections in future.
