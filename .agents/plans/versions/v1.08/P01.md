### Task 1: Implement Platform Detection Utility

**File:** `backend/app/services/awareness/platform.py`
**Purpose:** Centralized platform detection and tool availability checking. All awareness services use this to determine OS-specific command paths and availability.

```python
# backend/app/services/awareness/platform.py

import platform
import shutil
from typing import Optional
from enum import Enum


class OSType(Enum):
    LINUX = "linux"
    DARWIN = "darwin"
    WINDOWS = "windows"
    UNKNOWN = "unknown"


class PlatformDetector:
    """Detects OS type and tool availability for awareness services."""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._os_type = None
            cls._instance._tool_cache = {}
        return cls._instance
    
    @property
    def os_type(self) -> OSType:
        if self._os_type is None:
            system = platform.system().lower()
            if system == "linux":
                self._os_type = OSType.LINUX
            elif system == "darwin":
                self._os_type = OSType.DARWIN
            elif system == "windows":
                self._os_type = OSType.WINDOWS
            else:
                self._os_type = OSType.UNKNOWN
        return self._os_type
    
    def tool_available(self, tool_name: str) -> bool:
        """Check if a CLI tool is available on the system."""
        if tool_name not in self._tool_cache:
            self._tool_cache[tool_name] = shutil.which(tool_name) is not None
        return self._tool_cache[tool_name]
    
    def get_desktop_env(self) -> str:
        """Get Linux desktop environment name."""
        import os
        return os.environ.get("XDG_CURRENT_DESKTOP", "unknown")
    
    def get_display_server(self) -> str:
        """Get display server type (X11, Wayland)."""
        import os
        if os.environ.get("WAYLAND_DISPLAY"):
            return "wayland"
        elif os.environ.get("DISPLAY"):
            return "x11"
        return "unknown"
```

### Task 2: Implement Safe Subprocess Utility

**File:** `backend/app/services/awareness/subprocess_util.py`
**Purpose:** Safe subprocess execution with timeout, error handling, and platform-specific command building. All awareness services use this instead of direct subprocess calls.

```python
# backend/app/services/awareness/subprocess_util.py

import subprocess
import asyncio
from typing import Optional, List
from backend.app.services.awareness.platform import PlatformDetector


class SafeSubprocess:
    """Safe subprocess execution with timeout and error handling."""
    
    DEFAULT_TIMEOUT = 5.0  # seconds
    
    @staticmethod
    async def run(cmd: List[str], timeout: float = None, 
                  input_data: Optional[str] = None) -> Optional[str]:
        """Run a subprocess with timeout and error handling.
        
        Returns stdout on success, None on any failure.
        Never raises exceptions — all errors are caught and logged.
        """
        timeout = timeout or SafeSubprocess.DEFAULT_TIMEOUT
        
        try:
            kwargs = {
                "capture_output": True,
                "text": True,
                "timeout": timeout,
            }
            if input_data:
                kwargs["input"] = input_data
            
            result = subprocess.run(cmd, **kwargs)
            
            if result.returncode == 0:
                return result.stdout.strip()
            else:
                return None
                
        except subprocess.TimeoutExpired:
            return None
        except FileNotFoundError:
            return None
        except PermissionError:
            return None
        except Exception:
            return None
    
    @staticmethod
    async def run_popen(cmd: List[str], input_data: Optional[str] = None) -> bool:
        """Run a subprocess via Popen for write operations (e.g., clipboard)."""
        try:
            process = subprocess.Popen(
                cmd, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE
            )
            stdout, stderr = process.communicate(
                input=input_data.encode() if input_data else None,
                timeout=5
            )
            return process.returncode == 0
        except (subprocess.TimeoutExpired, FileNotFoundError, PermissionError, Exception):
            return False
```

### Task 3: Implement Desktop Awareness Service

**File:** `backend/app/services/awareness/desktop.py`
**Purpose:** Detect desktop state: open windows, focused application, window count, desktop environment. Uses `xdotool` on Linux (X11), `osascript` on macOS. Graceful fallback when tools unavailable.

```python
# backend/app/services/awareness/desktop.py

from typing import Dict, List, Optional
from backend.app.services.awareness.platform import PlatformDetector, OSType
from backend.app.services.awareness.subprocess_util import SafeSubprocess


class DesktopAwarenessService:
    """Monitors desktop state — windows, focus, layout.
    
    Platform support:
    - Linux (X11): xdotool for window list, focus, name
    - macOS: osascript + System Events for frontmost app
    - Fallback: returns {"status": "monitoring unavailable"}
    
    Architecture ref: 1.3 (Daemon-First) — runs as background polling task.
    1.2 (Graceful Degradation) — all failures return safe defaults.
    """
    
    def __init__(self):
        self._platform = PlatformDetector()
    
    async def get_desktop_info(self) -> Dict:
        """Get current desktop information."""
        if self._platform.os_type == OSType.LINUX:
            return await self._get_linux_desktop()
        elif self._platform.os_type == OSType.DARWIN:
            return await self._get_macos_desktop()
        else:
            return {"os": self._platform.os_type.value, "status": "unsupported"}
    
    async def get_window_list(self) -> List[Dict]:
        """Get list of open windows."""
        if self._platform.os_type == OSType.LINUX:
            return await self._get_linux_windows()
        elif self._platform.os_type == OSType.DARWIN:
            return await self._get_macos_windows()
        return []
    
    async def get_focused_window(self) -> Dict:
        """Get currently focused window/application."""
        if self._platform.os_type == OSType.LINUX:
            name = await SafeSubprocess.run(["xdotool", "getactivewindow", "getwindowname"])
            return {"name": name or "unknown", "os": "linux"}
        elif self._platform.os_type == OSType.DARWIN:
            name = await SafeSubprocess.run([
                "osascript", "-e",
                'tell application "System Events" to get name of first application process whose frontmost is true'
            ])
            return {"name": name or "unknown", "os": "darwin"}
        return {"name": "unknown", "os": "unsupported"}
    
    async def _get_linux_desktop(self) -> Dict:
        """Get Linux desktop info via xdotool."""
        if not self._platform.tool_available("xdotool"):
            return {"os": "linux", "status": "xdotool not installed"}
        
        windows = await self._get_linux_windows()
        focused = await self.get_focused_window()
        
        return {
            "os": "linux",
            "window_count": len(windows),
            "focused_window": focused.get("name", "unknown"),
            "desktop_env": self._platform.get_desktop_env(),
            "display_server": self._platform.get_display_server(),
            "windows": windows[:10]  # Limit for response size
        }
    
    async def _get_macos_desktop(self) -> Dict:
        """Get macOS desktop info via osascript."""
        focused = await self.get_focused_window()
        
        return {
            "os": "darwin",
            "focused_app": focused.get("name", "unknown"),
            "status": "active"
        }
    
    async def _get_linux_windows(self) -> List[Dict]:
        """Get Linux window list via xdotool."""
        if not self._platform.tool_available("xdotool"):
            return []
        
        window_ids = await SafeSubprocess.run(["xdotool", "search", "--name", ""])
        if not window_ids:
            return []
        
        windows = []
        for wid in window_ids.split("\n")[:20]:  # Limit to 20
            wid = wid.strip()
            if not wid:
                continue
            name = await SafeSubprocess.run(["xdotool", "getwindowname", wid])
            windows.append({"id": wid, "name": name or "unknown"})
        
        return windows
    
    async def _get_macos_windows(self) -> List[Dict]:
        """Get macOS window list via osascript."""
        result = await SafeSubprocess.run([
            "osascript", "-e",
            'tell application "System Events" to get name of every application process whose visible is true'
        ])
        if not result:
            return []
        
        return [{"name": name.strip()} for name in result.split(", ") if name.strip()]
```

### Task 4: Implement Terminal Awareness Service

**File:** `backend/app/services/awareness/terminal.py`
**Purpose:** Monitor terminal sessions (active shells) and retrieve recent command history. Reads shell history files directly. Supports bash, zsh, and fish.

```python
# backend/app/services/awareness/terminal.py

import os
from typing import Dict, List, Optional
from backend.app.services.awareness.subprocess_util import SafeSubprocess


class TerminalAwarenessService:
    """Monitors terminal sessions and command history.
    
    Features:
    - Active shell process detection (bash, zsh, fish)
    - Recent command history from shell history files
    - Working directory detection per shell
    
    Architecture ref: 1.4 (Separation of Concerns) — terminal service
    is separate from desktop service.
    """
    
    HISTORY_LIMIT = 50  # Max history entries to return
    SHELL_PATTERNS = ["bash", "zsh", "fish", "sh", "tmux", "screen"]
    
    def __init__(self):
        self._history_files = {
            "bash": os.path.expanduser("~/.bash_history"),
            "zsh": os.path.expanduser("~/.zsh_history"),
            "fish": os.path.expanduser("~/.local/share/fish/fish_history"),
        }
    
    async def get_terminal_sessions(self) -> List[Dict]:
        """Get active terminal shell sessions."""
        result = await SafeSubprocess.run(["ps", "aux"])
        if not result:
            return []
        
        sessions = []
        for line in result.split("\n"):
            parts = line.split()
            if len(parts) < 11:
                continue
            
            command = parts[10]
            is_shell = any(shell in command.lower() for shell in self.SHELL_PATTERNS)
            
            if is_shell:
                sessions.append({
                    "pid": parts[1],
                    "user": parts[0],
                    "cpu": parts[2],
                    "memory": parts[3],
                    "command": command,
                    "start_time": parts[8] if len(parts) > 8 else "unknown"
                })
        
        return sessions
    
    async def get_recent_commands(self, shell: str = "bash", 
                                   limit: int = 20) -> List[str]:
        """Get recent commands from shell history."""
        limit = min(limit, self.HISTORY_LIMIT)
        
        history_file = self._history_files.get(shell)
        if not history_file or not os.path.exists(history_file):
            return []
        
        try:
            with open(history_file, 'r', errors='replace') as f:
                lines = f.readlines()
                commands = [line.strip() for line in lines if line.strip()]
                return commands[-limit:]  # Most recent
        except (OSError, PermissionError):
            return []
    
    async def get_all_history(self, limit: int = 50) -> Dict[str, List[str]]:
        """Get recent history from all available shells."""
        result = {}
        for shell in self._history_files:
            history = await self.get_recent_commands(shell, limit)
            if history:
                result[shell] = history
        return result
    
    async def get_working_directories(self) -> List[Dict]:
        """Get working directories of active shell sessions."""
        result = await SafeSubprocess.run(["ps", "aux"])
        if not result:
            return []
        
        dirs = []
        for line in result.split("\n"):
            parts = line.split()
            if len(parts) < 11:
                continue
            command = parts[10].lower()
            if any(shell in command for shell in self.SHELL_PATTERNS):
                # Try to get CWD via /proc (Linux only)
                pid = parts[1]
                cwd = await SafeSubprocess.run(["readlink", f"/proc/{pid}/cwd"])
                if cwd:
                    dirs.append({"pid": pid, "cwd": cwd, "shell": command})
        
        return dirs
```

## Security Considerations

1. **Subprocess injection** — All subprocess commands are hardcoded strings. No user input passed to subprocess.
2. **History file access** — Read-only access to shell history. No write operations. Permission errors caught gracefully.
3. **Process information** — Only return user-owned processes. Filter out system/kernel processes.
4. **Timeout enforcement** — All subprocess calls have 5-second timeout. Prevents hanging.
5. **Data minimization** — Return only necessary fields. No full process memory dumps.

---

## Performance Considerations

1. **Subprocess overhead** — Each subprocess call has ~50ms overhead. Batch where possible.
2. **History file size** — Limit to last 50 entries. Don't read entire 10MB history files.
3. **Polling frequency** — Desktop state polled every 5 seconds. Terminal history polled every 30 seconds.
4. **Caching** — Cache tool availability checks. Cache desktop info for 10 seconds.
