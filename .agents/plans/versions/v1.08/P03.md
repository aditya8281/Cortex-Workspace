### Task 1: Implement Workspace Awareness Service

**File:** `backend/app/services/awareness/workspace.py`
**Purpose:** Monitor workspace organization: directory structure, recently modified files, file type distribution. Filters hidden directories and common non-project directories.

```python
# backend/app/services/awareness/workspace.py

import os
from typing import Dict, List, Optional
from datetime import datetime


class WorkspaceAwarenessService:
    """Monitors workspace organization and file activity.
    
    Features:
    - Directory structure mapping (depth-limited)
    - Recently modified files (sorted by mtime)
    - File type distribution
    - Project detection (package.json, pyproject.toml, etc.)
    
    Architecture ref: 1.4 (Separation of Concerns) — workspace service
    is separate from terminal/desktop services.
    """
    
    MAX_DEPTH = 3           # Max directory nesting to traverse
    MAX_FILES = 1000        # Max files to return
    SKIP_DIRS = {
        ".git", "node_modules", "__pycache__", ".venv", "venv",
        ".tox", ".mypy_cache", ".pytest_cache", "dist", "build",
        ".next", ".nuxt", "coverage", ".cortex"
    }
    PROJECT_MARKERS = [
        "package.json", "pyproject.toml", "Cargo.toml", "go.mod",
        "Makefile", "Dockerfile", "docker-compose.yml", ".git"
    ]
    
    async def get_workspace_structure(self, path: str, 
                                       max_depth: int = None) -> Dict:
        """Get workspace directory structure."""
        max_depth = max_depth or self.MAX_DEPTH
        
        structure = {
            "name": os.path.basename(path) or path,
            "type": "directory",
            "path": path,
            "children": []
        }
        
        try:
            for item in sorted(os.listdir(path)):
                if item.startswith(".") and item not in (".git", ".env"):
                    continue
                if item in self.SKIP_DIRS:
                    continue
                
                item_path = os.path.join(path, item)
                try:
                    stat = os.stat(item_path)
                    child = {
                        "name": item,
                        "type": "directory" if os.path.isdir(item_path) else "file",
                        "size": stat.st_size if os.path.isfile(item_path) else 0,
                        "modified": datetime.fromtimestamp(stat.st_mtime).isoformat()
                    }
                    
                    if os.path.isdir(item_path) and max_depth > 0:
                        child["children"] = []
                        # Don't recurse into skip dirs
                        if item not in self.SKIP_DIRS:
                            sub = await self.get_workspace_structure(item_path, max_depth - 1)
                            child["children"] = sub.get("children", [])
                    
                    structure["children"].append(child)
                except (OSError, PermissionError):
                    continue
        except (OSError, PermissionError):
            structure["error"] = "Cannot access directory"
        
        return structure
    
    async def get_recent_files(self, workspace_path: str, 
                                limit: int = 20) -> List[Dict]:
        """Get recently modified files in workspace."""
        files = []
        
        for root, dirs, filenames in os.walk(workspace_path):
            # Filter skip directories
            dirs[:] = [d for d in dirs if d not in self.SKIP_DIRS and not d.startswith(".")]
            
            for filename in filenames:
                filepath = os.path.join(root, filename)
                try:
                    stat = os.stat(filepath)
                    files.append({
                        "path": filepath,
                        "name": filename,
                        "size": stat.st_size,
                        "modified": datetime.fromtimestamp(stat.st_mtime).isoformat(),
                        "extension": os.path.splitext(filename)[1],
                        "relative_path": os.path.relpath(filepath, workspace_path)
                    })
                except (OSError, PermissionError):
                    continue
        
        files.sort(key=lambda x: x["modified"], reverse=True)
        return files[:min(limit, self.MAX_FILES)]
    
    async def get_file_type_distribution(self, workspace_path: str) -> Dict:
        """Get file type distribution in workspace."""
        type_counts = {}
        
        for root, dirs, filenames in os.walk(workspace_path):
            dirs[:] = [d for d in dirs if d not in self.SKIP_DIRS and not d.startswith(".")]
            for filename in filenames:
                ext = os.path.splitext(filename)[1] or "(no extension)"
                type_counts[ext] = type_counts.get(ext, 0) + 1
        
        return dict(sorted(type_counts.items(), key=lambda x: x[1], reverse=True))
    
    async def detect_project_type(self, workspace_path: str) -> Optional[str]:
        """Detect project type from marker files."""
        for marker in self.PROJECT_MARKERS:
            if os.path.exists(os.path.join(workspace_path, marker)):
                if marker == "package.json":
                    return "node"
                elif marker == "pyproject.toml":
                    return "python"
                elif marker == "Cargo.toml":
                    return "rust"
                elif marker == "go.mod":
                    return "go"
                elif marker in ("Dockerfile", "docker-compose.yml"):
                    return "docker"
                return "generic"
        return None
```

### Task 2: Implement Notification Awareness Service

**File:** `backend/app/services/awareness/notifications.py`
**Purpose:** Record, retrieve, and manage system notifications. Event-driven recording with read/unread tracking.

```python
# backend/app/services/awareness/notifications.py

from typing import Dict, List, Optional
from datetime import datetime


class NotificationAwarenessService:
    """Monitors and manages system notifications.
    
    Features:
    - Record notifications from various sources
    - Retrieve recent notifications
    - Mark as read/unread
    - Count unread notifications
    - Filter by source
    
    Architecture ref: 1.3 (Daemon-All notifications stored in memory.
    For persistence, would use PostgreSQL in production.
    """
    
    MAX_NOTIFICATIONS = 500  # Max notifications to keep in memory
    
    def __init__(self):
        self._notifications: List[Dict] = []
    
    async def record_notification(self, title: str, body: str,
                                   source: str = "system",
                                   priority: str = "normal") -> Dict:
        """Record a notification."""
        notification = {
            "id": len(self._notifications) + 1,
            "title": title,
            "body": body,
            "source": source,
            "priority": priority,
            "timestamp": datetime.utcnow().isoformat(),
            "read": False
        }
        
        self._notifications.append(notification)
        
        # Trim if over limit
        if len(self._notifications) > self.MAX_NOTIFICATIONS:
            self._notifications = self._notifications[-self.MAX_NOTIFICATIONS:]
        
        return notification
    
    async def get_notifications(self, limit: int = 50, 
                                 source: Optional[str] = None,
                                 unread_only: bool = False) -> List[Dict]:
        """Get recent notifications with optional filtering."""
        filtered = self._notifications
        
        if source:
            filtered = [n for n in filtered if n["source"] == source]
        if unread_only:
            filtered = [n for n in filtered if not n["read"]]
        
        return filtered[-limit:]
    
    async def mark_read(self, notification_id: int) -> bool:
        """Mark a notification as read."""
        for notif in self._notifications:
            if notif["id"] == notification_id:
                notif["read"] = True
                return True
        return False
    
    async def mark_all_read(self) -> int:
        """Mark all notifications as read. Returns count marked."""
        count = 0
        for notif in self._notifications:
            if notnotif["read"]:
                notif["read"] = True
                count += 1
        return count
    
    async def get_unread_count(self, source: Optional[str] = None) -> int:
        """Get count of unread notifications."""
        filtered = self._notifications
        if source:
            filtered = [n for n in filtered if n["source"] == source]
        return sum(1 for n in filtered if not n["read"])
    
    async def clear_old(self, days: int = 7) -> int:
        """Clear notifications older than N days."""
        cutoff = datetime.utcnow().timestamp() - (days * 86400)
        before = len(self._notifications)
        self._notifications = [
            n for n in self._notifications
            if datetime.fromisoformat(n["timestamp"]).timestamp() > cutoff
        ]
        return before - len(self._notifications)
```

## Security Considerations

1. **File system access** — Read-only. No write operations on workspace files.
2. **Hidden directories** — Skip `.git`, `.env`, and other sensitive directories.
3. **Permission errors** — Catch and skip inaccessible files/directories.
4. **Notification content** — May contain sensitive data. Store in memory only, not persisted.

---

## Performance Considerations

1. **Directory walking** — `os.walk` is efficient but can be slow on large workspaces. Limit depth and file count.
2. **File stats** — Each `os.stat` call has ~1ms overhead. For 1000 files, that's ~1 second.
3. **Notification storage** — In-memory list is fast but limited. For production, use PostgreSQL.
4. **Caching** — Cache workspace structure for 30 seconds. Invalidate on file change events.
